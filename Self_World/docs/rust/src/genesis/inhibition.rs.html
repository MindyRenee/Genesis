1//! Inhibition — the body's coordination substrate.
2//!
3//! What an operating system calls a mutex, a nervous system calls
4//! inhibition. This module is the Rust half of `Self_World/inhibition.py`;
5//! the two files are one mechanism seen from both sides of the
6//! organism — here the math, there the words.
7//!
8//! # The circuits
9//!
10//! - [`Inhibition<T>`] — fast GABAergic mutual exclusion. Serializes
11//!   access to shared state the way striatal lateral inhibition
12//!   serializes competing drive: the conducting unit's neighbours
13//!   wait in the inhibitory shadow (Groves 1983; Wickens et al. 1991;
14//!   Koós & Tepper 1999). Wraps [`std::sync::Mutex`] and keeps the
15//!   same API — `lock`, `try_lock`, `into_inner`, `get_mut` — so it
16//!   is a drop-in mechanism, not a re-implementation.
17//!
18//! - [`Innate<T>`] — wiring fixed once, then immutable. `OnceLock`
19//!   under its biological name: an innately wired quantity is
20//!   established at genesis and never re-learned. A type alias, so
21//!   every `OnceLock` API is inherited unchanged.
22//!
23//! # The mathematics
24//!
25//! Every [`Inhibition`] counts its own contention with lock-free
26//! `Relaxed` counters — the measurement adds no synchronization of
27//! its own, the same way an electrode does not change the neuron it
28//! records:
29//!
30//! - `acquisitions` — completed admissions
31//! - `contended` — admissions that found the gate already conducting
32//! - `wait_nanos` — total time competitors spent in the shadow
33//!
34//! From these, per the Python side:
35//!
36//! - contention `c = contended / max(1, acquisitions)`
37//! - mean wait `W̄ = wait_seconds / max(1, contended)`
38//! - arrival rate `λ = acquisitions / elapsed`
39//!
40//! and Little's law `L = λ·W̄` (Little 1961) gives the expected number
41//! of competitors held in the shadow, mapped into the interoceptive
42//! range by the Naka–Rushton `tone = L/(1+L)` (1966) — see
43//! [`Inhibition::stats`].
44//!
45//! # Embodiment exclusivity
46//!
47//! One organism, one body. The `flock(LOCK_EX | LOCK_NB)` on the
48//! memory-mapped state file in `store/mmap_state.rs` is not a lock
49//! the substrate could route around — it is the physical law of
50//! exclusive embodiment, enforced by the kernel the way a skull
51//! enforces it anatomically: exactly one process may map and mutate
52//! her persistent state.
53

55use std::sync::{LockResult, Mutex, MutexGuard, OnceLock, TryLockError, TryLockResult};

58/// Innately wired state — established once at genesis, immutable
59/// thereafter.
60///
61/// `OnceLock<T>` under its biological name. A type alias, so
62/// `Innate::new()`, `get_or_init`, `set`, `get`, `get_mut`, `take`
63/// and every other `OnceLock` API work exactly as before.
64pub type Innate<T> = OnceLock<T>;

66/// The conducting axon — exclusive access while held.
67///
68/// `MutexGuard<'a, T>` under the same vocabulary as [`Inhibition`].
69/// An alias so guard types annotate cleanly beside the gate.
70pub type InhibitionGuard<'a, T> = MutexGuard<'a, T>;

72/// Contention statistics for one [`Inhibition`].
73///
74/// The fields mirror `inhibition.py`'s `_Tone.stats` output; units
75/// and formulas are documented on the module.
76#[derive(Debug, Clone, Copy, Default)]
77pub struct InhibitoryStats {
78    /// Completed admissions.
79    pub acquisitions: u64,
80    /// Admissions that found the gate already conducting.
81    pub contended: u64,
82    /// Total seconds competitors spent in the inhibitory shadow.
83    pub wait_seconds: f64,
84    /// `contended / max(1, acquisitions)` — fraction of admissions
85    /// that met inhibition.
86    pub contention: f64,
87    /// Mean seconds spent waiting per contended admission.
88    pub mean_wait: f64,
89    /// `acquisitions / elapsed` — admission rate since first use.
90    pub arrival_rate: f64,
91}

93/// Fast GABAergic mutual exclusion — the mutex, observed.
94///
95/// `Inhibition<T>` is a `Mutex<T>` instrumented with contention
96/// accounting. All synchronization behaviour is the kernel mutex's;
97/// the counters are `Relaxed` atomics so measurement never adds
98/// ordering of its own.
99pub struct Inhibition<T> {
100    inner: Mutex<T>,
101    acquisitions: AtomicU64,
102    contended: AtomicU64,
103    wait_nanos: AtomicU64,
104    /// Nanos since UNIX_EPOCH of the first admission (0 = unborn).
105    born_ns: AtomicU64,
106}
107
108impl<T> Inhibition<T> {
109    /// Const-construct an inhibitory gate guarding `value` — usable in
110    /// `static` contexts exactly like `Mutex::new`.
111    pub const fn new(value: T) -> Self {
112        Self {
113            inner: Mutex::new(value),
114            acquisitions: AtomicU64::new(0),
115            contended: AtomicU64::new(0),
116            wait_nanos: AtomicU64::new(0),
117            born_ns: AtomicU64::new(0),
118        }
119    }
120}
121
122impl<T> Inhibition<T> {
123    /// Acquire the gate, blocking until the shadow clears.
124    ///
125    /// Same semantics as `Mutex::lock`: returns the guard, or the
126    /// poison error if a previous holder panicked while conducting.
127    /// An uncontended admission stays on the fast path; a contended
128    /// one is timed and charged to the inhibitory shadow.
129    pub fn lock(&self) -> LockResult<MutexGuard<'_, T>> {
130        self.note_birth();
131        match self.inner.try_lock() {
132            Ok(guard) => {
133                self.acquisitions.fetch_add(1, Ordering::Relaxed);
134                Ok(guard)
135            }
136            Err(TryLockError::WouldBlock) => {
137                let start = Instant::now();
138                let result = self.inner.lock();
139                let waited = start.elapsed().as_nanos() as u64;
140                self.wait_nanos.fetch_add(waited, Ordering::Relaxed);
141                self.contended.fetch_add(1, Ordering::Relaxed);
142                self.acquisitions.fetch_add(1, Ordering::Relaxed);
143                result
144            }
145            Err(TryLockError::Poisoned(e)) => Err(std::sync::PoisonError::new(e.into_inner())),
146        }
147    }
148
149    /// Non-blocking attempt — the axon either conducts or reports
150    /// the synapse already firing. Mirrors `Mutex::try_lock`.
151    pub fn try_lock(&self) -> TryLockResult<MutexGuard<'_, T>> {
152        self.note_birth();
153        match self.inner.try_lock() {
154            Ok(guard) => {
155                self.acquisitions.fetch_add(1, Ordering::Relaxed);
156                Ok(guard)
157            }
158            Err(e) => Err(e),
159        }
160    }
161
162    /// Contention statistics — see the module docstring for the math.
163    pub fn stats(&self) -> InhibitoryStats {
164        let acquisitions = self.acquisitions.load(Ordering::Relaxed);
165        let contended = self.contended.load(Ordering::Relaxed);
166        let wait_seconds = self.wait_nanos.load(Ordering::Relaxed) as f64 * 1e-9;
167        let born = self.born_ns.load(Ordering::Relaxed);
168        let elapsed = if born == 0 {
169            1e-9
170        } else {
171            let now = SystemTime::now()
172                .duration_since(UNIX_EPOCH)
173                .map(|d| d.as_nanos() as u64)
174                .unwrap_or(born);
175            (now.saturating_sub(born) as f64 * 1e-9).max(1e-9)
176        };
177        InhibitoryStats {
178            acquisitions,
179            contended,
180            wait_seconds,
181            contention: contended as f64 / acquisitions.max(1) as f64,
182            mean_wait: wait_seconds / contended.max(1) as f64,
183            arrival_rate: acquisitions as f64 / elapsed,
184        }
185    }
186
187    /// Consume the gate, yielding the guarded value — `Mutex::into_inner`.
188    pub fn into_inner(self) -> LockResult<T> {
189        self.inner.into_inner()
190    }
191
192    /// `Mutex::get_mut` — mutable access without locking, when the
193    /// caller already holds exclusive access to the gate itself.
194    pub fn get_mut(&mut self) -> LockResult<&mut T> {
195        self.inner.get_mut()
196    }
197
198    /// `Mutex::is_poisoned`.
199    pub fn is_poisoned(&self) -> bool {
200        self.inner.is_poisoned()
201    }
202
203    /// `Mutex::clear_poison`.
204    pub fn clear_poison(&self) {
205        self.inner.clear_poison();
206    }
207
208    /// Stamp the first-use epoch once. `Instant` is not
209    /// const-constructible, so the birth marker lives in an atomic
210    /// and is set on the first admission instead of at construction.
211    fn note_birth(&self) {
212        if self.born_ns.load(Ordering::Relaxed) == 0 {
213            let now = SystemTime::now()
214                .duration_since(UNIX_EPOCH)
215                .map(|d| d.as_nanos() as u64)
216                .unwrap_or(1)
217                .max(1);
218            let _ = self
219                .born_ns
220                .compare_exchange(0, now, Ordering::Relaxed, Ordering::Relaxed);
221        }
222    }
223}
224
225impl<T: Default> Default for Inhibition<T> {
226    fn default() -> Self {
227        Self::new(T::default())
228    }
229}
