1//! Spectral observables of the field's own dynamics.
2//!
3//! The field substrate is her computational medium: three fibers —
4//! body, prediction, agency — drift, couple, and fluctuate on the
5//! latent manifold under gauge transport and a Langevin thermostat.
6//! This module is the band decomposition of *that* trajectory: the
7//! same center frequencies the cortical wave model reports, measured
8//! off the field's own motion rather than assumed.
9//!
10//! Two channels mirror the substrate's two honest observables:
11//!
12//! - **flux** — the speed of the fibers' centroid through latent
13//!   space (|Δmean|/h per integration substep). The field's motion
14//!   energy per band: how much of its wandering lives at each
15//!   rhythm.
16//! - **tension** — the fibers' mean deviation from their own
17//!   centroid (latent-coordinate spread). The disagreement's
18//!   spectral shadow: where the field's internal argument oscillates.
19//!
20//! The carrier block tracks the consensus vector's own kinematics —
21//! displacement speed, winding rate, accumulated winding phase —
22//! the same measured quantities `daemon::maxwell` evaluates, kept
23//! here so the spectrum and the signature read one trajectory.
24//!
25//! Sampling is the field's own integration rate: `observe` substeps
26//! the coupling at [`SPECTRUM_STEP_HZ`], so the rings resolve the
27//! modeled continuous dynamics — the same ODEs integrated finer,
28//! which is more accurate, not a second instrument. Bands above the
29//! achieved rate's Nyquist bound are zeroed: an unresolvable band
30//! is an absent channel, never an aliased reading.

68/// A fixed-depth ring of the field's spectral samples.
69#[derive(Clone, Debug, PartialEq)]
70struct Ring {
71    buf: Vec<f64>,
72    pos: usize,
73    len: usize,
74}
75
76impl Default for Ring {
77    fn default() -> Self {
78        Self::new(RING_LEN)
79    }
80}
81
82impl Ring {
83    fn new(cap: usize) -> Self {
84        Self {
85            buf: vec![0.0; cap.max(1)],
86            pos: 0,
87            len: 0,
88        }
89    }
90    fn push(&mut self, v: f64) {
91        self.buf[self.pos] = v;
92        self.pos = (self.pos + 1) % self.buf.len();
93        self.len = (self.len + 1).min(self.buf.len());
94    }
95    /// Oldest→newest iterator over the filled region.
96    fn iter_ordered(&self) -> RingIter<'_> {
97        RingIter { ring: self, i: 0 }
98    }
99    fn mean(&self) -> f64 {
100        if self.len == 0 {
101            return 0.0;
102        }
103        self.iter_ordered().sum::<f64>() / self.len as f64
104    }
105    fn m2(&self) -> f64 {
106        if self.len == 0 {
107            return 0.0;
108        }
109        self.iter_ordered()
110            .map(|v| {
111                let d = v - self.mean();
112                d * d
113            })
114            .sum::<f64>()
115            / self.len as f64
116    }
117}

119struct RingIter<'a> {
120    ring: &'a Ring,
121    i: usize,
122}
123
124impl<'a> Iterator for RingIter<'a> {
125    type Item = f64;
126    fn next(&mut self) -> Option<f64> {
127        if self.i >= self.ring.len {
128            return None;
129        }
130        let cap = self.ring.buf.len();
131        let idx = if self.ring.len < cap {
132            self.i
133        } else {
134            (self.ring.pos + self.i) % cap
135        };
136        self.i += 1;
137        Some(self.ring.buf[idx])
138    }
139}

141/// Goertzel power of `samples` at `freq` sampled at `fs` — the same
142/// second-order recurrence the wave model's own band meter runs:
143/// `s₀ = x + coeff·s₁ − s₂`, `coeff = 2·cos(2πk/n)` at the bin
144/// nearest `freq`; power `= (s₁² + s₂² − coeff·s₁·s₂)/n²`.
145fn band_power(samples: &[f64], fs: f64, freq: f64) -> f64 {
146    let n = samples.len();
147    if n < 8 || fs <= 0.0 || freq <= 0.0 {
148        return 0.0;
149    }
150    let k = (0.5 + freq * n as f64 / fs).floor();
151    if k < 1.0 || k >= n as f64 / 2.0 {
152        return 0.0;
153    }
154    let w = 2.0 * std::f64::consts::PI * k / n as f64;
155    let coeff = 2.0 * w.cos();
156    let (mut s1, mut s2) = (0.0, 0.0);
157    for &x in samples {
158        let s0 = x + coeff * s1 - s2;
159        s2 = s1;
160        s1 = s0;
161    }
162    (s1 * s1 + s2 * s2 - coeff * s1 * s2) / (n as f64 * n as f64)
163}

165/// Quadrature phase of `samples` at `freq` — the DFT bin's angle,
166/// atan2(−Σx·sin, Σx·cos) at the exact frequency (not the Goertzel
167/// bin), the phase a band-limited rhythm actually carries.
168fn band_phase(samples: &[f64], fs: f64, freq: f64) -> f64 {
169    let n = samples.len();
170    if n < 8 || fs <= 0.0 || freq <= 0.0 {
171        return 0.0;
172    }
173    let (mut re, mut im) = (0.0, 0.0);
174    for (i, &x) in samples.iter().enumerate() {
175        let a = -2.0 * std::f64::consts::PI * freq * i as f64 / fs;
176        re += x * a.cos();
177        im += x * a.sin();
178    }
179    im.atan2(re)
180}

199/// The field's spectral signature — everything the wave model's
200/// stimulus path and the carrier lock consume, computed from her
201/// own substrate trajectory.
202#[derive(Clone, Debug, Default, PartialEq)]
203pub struct FieldSpectrum {
204    flux_ring: Ring,
205    tension_ring: Ring,
206    /// Per-band baseline EMA — [flux ×10][tension ×10].
207    band_base: [f64; 20],
208    base_init: bool,
209    /// Consensus kinematics — instantaneous and EMA-smoothed.
210    pub last_consensus_speed: Option<f64>,
211    pub last_omega_rad_s: Option<f64>,
212    pub omega_ema_rad_s: f64,
213    pub speed_ema: f64,
214    /// Accumulated winding phase (rad, mod 2π at read) — the field's
215    /// own clock, the phase reference entrainment pulls toward.
216    carrier_phase: f64,
217    prev_consensus: Option<Vector>,
218    /// Achieved sampling rate (Hz), EMA-smoothed.
219    rate_ema: f64,
220    substeps: u64,
221    /// Last substep's fiber centroid — the flux denominator.
222    prev_mean_state: Option<Vector>,
223}

225impl FieldSpectrum {
226    pub fn new() -> Self {
227        Self {
228            flux_ring: Ring::new(RING_LEN),
229            tension_ring: Ring::new(RING_LEN),
230            band_base: [0.0; 20],
231            base_init: false,
232            last_consensus_speed: None,
233            last_omega_rad_s: None,
234            omega_ema_rad_s: 0.0,
235            speed_ema: 0.0,
236            carrier_phase: 0.0,
237            prev_consensus: None,
238            rate_ema: SPECTRUM_STEP_HZ,
239            substeps: 0,
240            prev_mean_state: None,
241        }
242    }
243
244    /// One integration substep's worth of field state — the rings
245    /// take the trajectory's own samples. `h` is the substep's
246    /// duration; the achieved rate is its reciprocal, EMA-smoothed.
247    pub fn sample(&mut self, field: &GlobalField, h: f64) {
248        if !h.is_finite() || h <= 0.0 {
249            return;
250        }
251        // The fibers' centroid in latent coordinates — the Euclidean
252        // mean of their layer states. On the cap's operating region
253        // this tracks the Fréchet consensus closely enough for a
254        // *spectral* channel: static frame offsets land in the DC
255        // baseline, never in the bands.
256        let mut mean = vec![0.0; super::LATENT_DIMENSION];
257        let mut n_fibers = 0usize;
258        for fiber in &field.fibers {
259            if let Some(layer) = fiber.layers.first() {
260                for (m, v) in mean.iter_mut().zip(layer.state.iter()) {
261                    *m += v;
262                }
263                n_fibers += 1;
264            }
265        }
266        if n_fibers == 0 {
267            return;
268        }
269        for m in mean.iter_mut() {
270            *m /= n_fibers as f64;
271        }
272        let flux = match &self.prev_mean_state {
273            Some(prev) => {
274                prev.iter()
275                    .zip(mean.iter())
276                    .map(|(a, b)| (b - a) * (b - a))
277                    .sum::<f64>()
278                    .sqrt()
279                    / h
280            }
281            None => 0.0,
282        };
283        // Tension — mean deviation of the fibers from their own
284        // centroid, in latent units. The spectral shadow of the
285        // Fréchet disagreement the integration reports.
286        let mut tension = 0.0;
287        for fiber in &field.fibers {
288            if let Some(layer) = fiber.layers.first() {
289                tension += layer
290                    .state
291                    .iter()
292                    .zip(mean.iter())
293                    .map(|(a, b)| (a - b) * (a - b))
294                    .sum::<f64>()
295                    .sqrt();
296            }
297        }
298        tension /= n_fibers as f64;
299        self.prev_mean_state = Some(mean);
300        self.flux_ring.push(flux);
301        self.tension_ring.push(tension);
302        let rate = 1.0 / h;
303        self.rate_ema += RATE_ALPHA * (rate - self.rate_ema);
304        self.substeps += 1;
305    }
306
307    /// One observation's consensus — the carrier kinematics. Called
308    /// once per `observe` with the integration's consensus vector
309    /// and the elapsed `dt`. First valid pair only arms the
310    /// kinematics — one point is no trajectory.
311    pub fn observe_consensus(&mut self, consensus: &Vector, dt: f64) {
312        if consensus.iter().any(|v| !v.is_finite()) || dt <= 0.0 || !dt.is_finite() {
313            self.prev_consensus = None;
314            self.last_consensus_speed = None;
315            self.last_omega_rad_s = None;
316            return;
317        }
318        if let Some(prev) = self.prev_consensus.take() {
319            if prev.len() == consensus.len() {
320                let speed: f64 = prev
321                    .iter()
322                    .zip(consensus.iter())
323                    .map(|(a, b)| (b - a) * (b - a))
324                    .sum::<f64>()
325                    .sqrt()
326                    / dt;
327                let dot: f64 = prev.iter().zip(consensus.iter()).map(|(a, b)| a * b).sum();
328                let na: f64 = prev.iter().map(|v| v * v).sum::<f64>().sqrt();
329                let nb: f64 = consensus.iter().map(|v| v * v).sum::<f64>().sqrt();
330                if na > 0.0 && nb > 0.0 {
331                    let angle = (dot / (na * nb)).clamp(-1.0, 1.0).acos();
332                    self.carrier_phase += angle;
333                    let omega = angle / dt;
334                    self.last_consensus_speed = Some(speed);
335                    self.last_omega_rad_s = Some(omega);
336                    self.speed_ema += CARRIER_ALPHA * (speed - self.speed_ema);
337                    self.omega_ema_rad_s += CARRIER_ALPHA * (omega - self.omega_ema_rad_s);
338                }
339            }
340        }
341        self.prev_consensus = Some(consensus.clone());
342    }

344    /// Whether the rings hold enough trajectory to report bands.
345    pub fn warm(&self) -> bool {
346        self.flux_ring.len >= WARM_SAMPLES && self.tension_ring.len >= WARM_SAMPLES
347    }

349    /// The achieved sampling rate (Hz) — what the bands can honestly
350    /// resolve against.
351    pub fn sample_rate_hz(&self) -> f64 {
352        self.rate_ema
353    }
354
355    /// Total integration substeps sampled — provenance for tests.
356    pub fn substeps(&self) -> u64 {
357        self.substeps
358    }

360    /// Decompose both channels at the wave bands and read the
361    /// signature. Mutates only the baseline EMAs — the rings
362    /// themselves are read, not consumed.
363    pub fn signature(&mut self) -> SpectrumReading {
364        let rate = self.rate_ema.max(1.0);
365        let nyquist = rate / 2.0;
366        let mut flux_power = [0.0f64; 10];
367        let mut tension_power = [0.0f64; 10];
368        let flux = self.channel(&self.flux_ring.clone(), rate, nyquist, 0, &mut flux_power);
369        let tension = self.channel(
370            &self.tension_ring.clone(),
371            rate,
372            nyquist,
373            10,
374            &mut tension_power,
375        );
376        SpectrumReading {
377            flux,
378            tension,
379            flux_power,
380            tension_power,
381            carrier_phase: self.carrier_phase.rem_euclid(2.0 * std::f64::consts::PI),
382            carrier_freq_hz: if self.omega_ema_rad_s > 0.0 {
383                Some(self.omega_ema_rad_s / (2.0 * std::f64::consts::PI))
384            } else {
385                None
386            },
387            consensus_speed: if self.last_consensus_speed.is_some() {
388                Some(self.speed_ema)
389            } else {
390                None
391            },
392            sample_rate_hz: rate,
393            warm: self.warm(),
394        }
395    }

397    /// Goertzel + quadrature over one ring, baseline-EMA'd into
398    /// relative energies. `base_off` selects this channel's slice of
399    /// `band_base` (0 = flux, 10 = tension).
400    fn channel(
401        &mut self,
402        ring: &Ring,
403        rate: f64,
404        nyquist: f64,
405        base_off: usize,
406        powers: &mut [f64; 10],
407    ) -> BandChannel {
408        let n = ring.len;
409        let samples: Vec<f64> = ring.iter_ordered().collect();
410        let mean = ring.mean();
411        let var = ring.m2();
412        let cv = if mean > 0.0 { var.sqrt() / mean } else { 0.0 };
413        let mut out = BandChannel {
414            mean,
415            cv,
416            ..Default::default()
417        };
418        if n < 8 {
419            return out;
420        }
421        for (i, &freq) in FIELD_BANDS.iter().enumerate() {
422            if freq >= nyquist {
423                continue; // unresolvable — absent, not aliased
424            }
425            let p = band_power(&samples, rate, freq);
426            powers[i] = p;
427            // Baseline EMA — first read initializes rather than
428            // divides by zero.
429            let base = &mut self.band_base[base_off + i];
430            if !self.base_init && *base == 0.0 {
431                *base = p.max(1e-30);
432            }
433            *base += BASELINE_ALPHA * (p - *base);
434            out.energies[i] = p / base.max(1e-30);
435            out.phases[i] = band_phase(&samples, rate, freq);
436        }
437        out
438    }
439}

469    #[test]
470    fn goertzel_finds_an_injected_tone() {
471        // 64 samples at 250 Hz of a 10 Hz tone + silence.
472        let fs = 250.0;
473        let n = 512;
474        let samples: Vec<f64> = (0..n)
475            .map(|i| (2.0 * std::f64::consts::PI * 10.0 * i as f64 / fs).sin())
476            .collect();
477        let p10 = band_power(&samples, fs, 10.0);
478        let p5 = band_power(&samples, fs, 5.5);
479        assert!(
480            p10 > 100.0 * p5.max(1e-30),
481            "tone band dominates: {p10} vs {p5}"
482        );
483        let phase = band_phase(&samples, fs, 10.0);
484        assert!(phase.is_finite());
485    }

487    #[test]
488    fn supra_nyquist_bands_stay_absent() {
489        let mut spec = FieldSpectrum::new();
490        // Fill rings directly with a 100 Hz tone at 250 Hz — only
491        // bands below 125 Hz may report; epsilon must read zero.
492        for i in 0..2048 {
493            let t = i as f64 / SPECTRUM_STEP_HZ;
494            spec.flux_ring
495                .push((2.0 * std::f64::consts::PI * 100.0 * t).sin());
496            spec.tension_ring.push(0.01);
497        }
498        let mut s = spec.signature();
499        let idx_100 = FIELD_BANDS.iter().position(|&f| (f - 100.0).abs() < 1e-9);
500        // 100 Hz isn't a band center — nearest is gamma 50 / epsilon
501        // 150; epsilon (150 > 125) must be zeroed by the Nyquist gate.
502        assert_eq!(s.flux.energies[9], 0.0, "epsilon above Nyquist");
503        assert_eq!(s.tension.energies[9], 0.0);
504        let _ = &mut s;
505        let _ = idx_100;
506    }
