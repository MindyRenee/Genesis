24/// Observation coupling timescale (seconds). Each fiber moves a
25/// fraction `1 − e^(−dt/τ)` along the geodesic toward its observed
26/// projection per call — a first-order manifold filter: the
27/// observation informs the field's own state instead of
28/// overwriting it.
29const OBSERVATION_TAU_SECONDS: f64 = 0.5;
30
31/// Per-fiber homeostatic relaxation times (seconds). Between
32/// observations each fiber's engine drifts its state toward nominal
33/// — the sphere's pole — at its own clearance rate: an
34/// Ornstein–Uhlenbeck inflow in the ambient coordinates the engine
35/// integrates, wound by a skew generator so the relaxation traces a
36/// loxodrome — a constant-bearing spiral into the pole, arcing
37/// inward along the sphere's own positive curvature — projected
38/// back into the cap. This is the honest prior for unobserved
39/// physiology: monoamines clear in minutes, loads decay, no state
40/// holds forever. The observation stays the authority — the drift
41/// only fills the gaps it leaves. Rates are first-pass physiological
42/// guesses ordered by how slowly each subsystem's measured variables
43/// genuinely relax: neurochemical clearance is minutes-scale, the
44/// inference signals are faster statistics, the social variables
45/// linger longest. The winding rate matches each fiber's own τ —
46/// a 45° loxodrome, one radian of turn per e-fold of decay.
47const BODY_RELAX_TAU_S: f64 = 120.0;
48const PREDICTION_RELAX_TAU_S: f64 = 40.0;
49const AGENCY_RELAX_TAU_S: f64 = 60.0;

71impl FieldRuntime {
72    pub fn new() -> Result<Self, FieldError> {
73        let manifold = StereographicSphere::new(LATENT_DIMENSION, 1.0)?;
74        let mut field = GlobalField::new(manifold.clone());
75
76        for (id, relax_tau) in [
77            ("body", BODY_RELAX_TAU_S),
78            ("prediction", PREDICTION_RELAX_TAU_S),
79            ("agency", AGENCY_RELAX_TAU_S),
80        ] {
81            let layer = SubsystemLayer::new(format!("{id}:latent"), vec![0.0; LATENT_DIMENSION]);
82            let engine = SystemEngine::zero(1, LATENT_DIMENSION)
83                .with_damping(vec![1.0 / relax_tau])
84                .expect("positive finite damping always validates")
85                .with_spiral(vec![loxodrome(LATENT_DIMENSION, 0, 1, 1.0 / relax_tau)])
86                .expect("a skew generator always validates");
87            field.add_fiber(SystemFiber::with_engine(
88                id,
89                manifold.clone(),
90                vec![layer],
91                engine,
92            )?)?;
93        }
94
95        for (from, to, matrix) in [
96            // The afferent edge is the non-flat one: somatic
97            // coordinates are *read in* a rotated inferential frame
98            // (the generative model's axes are not the raw body
99            // axes), so the triangle carries real holonomy. The
100            // reverse edge carries the inverse rotation, keeping
101            // the pair's own two-way transport consistent.
102            (
103                "body",
104                "prediction",
105                rotation(LATENT_DIMENSION, 0, 1, AFFERENT_FRAME_OFFSET),
106            ),
107            ("prediction", "agency", identity(LATENT_DIMENSION)),
108            ("agency", "body", identity(LATENT_DIMENSION)),
109            (
110                "prediction",
111                "body",
112                rotation(LATENT_DIMENSION, 0, 1, -AFFERENT_FRAME_OFFSET),
113            ),
114            ("agency", "prediction", identity(LATENT_DIMENSION)),
115            ("body", "agency", identity(LATENT_DIMENSION)),
116        ] {
117            field.add_transport(GaugeTransport::new(from, to, matrix)?)?;
118        }
119
120        Ok(Self {
121            field,
122            spectrum: FieldSpectrum::new(),
123        })
124    }

146    /// Predict → correct → integrate, transactionally on a clone:
147    /// any failure leaves the live field untouched rather than
148    /// half-observed.
149    ///
150    /// `bath` is the substrate's measured thermal level in latent²
151    /// units (the caller derives it from the physical probe — the
152    /// silicon's own timing fluctuation). It floors the field
153    /// temperature alongside internal disagreement: the reservoir
154    /// the Langevin kick samples is real, not only modeled.
155    pub fn observe(
156        &mut self,
157        state: &GenesisCoreState,
158        dt: f64,
159        bath: f64,
160    ) -> Result<FieldIntegration, FieldError> {
161        if !dt.is_finite() || dt < 0.0 {
162            return Err(FieldError::InvalidTime);
163        }
164        let mut next = self.field.clone();
165
166        // Prediction: each fiber advances by its own law — the OU
167        // homeostatic drift of its engine, so the field carries a
168        // trajectory between observations rather than a frozen
169        // snapshot.
170        for fiber in &mut next.fibers {
171            fiber.step(dt)?;
172        }
173
174        // Correction: the observation is an offering, not a command.
175        // Each fiber's latent layer moves a bounded fraction along
176        // the geodesic toward the observed point — the field carries
177        // its own state forward and lets measurement perturb it.
178        let gain = 1.0 - (-dt / OBSERVATION_TAU_SECONDS).exp();
179        for (id, projection) in [
180            ("body", body_projection(state)),
181            ("prediction", prediction_projection(state)),
182            ("agency", agency_projection(state)),
183        ] {
184            let fiber = next
185                .fibers
186                .iter_mut()
187                .find(|fiber| fiber.id == id)
188                .ok_or(FieldError::MissingFiber)?;
189            let manifold = fiber.manifold.clone();
190            let observation = manifold.project(&projection)?;
191            let current = fiber.layers[0].state.clone();
192            fiber.layers[0].state = manifold.geodesic(&current, &observation, gain)?;
193        }
194
195        // Integration: while awake, the noradrenergic forebrain
196        // coupling pulls the fibers toward consensus — the bounded
197        // pull is the body's coherence mechanism. In REM the locus
198        // coeruleus is silent and the systems decorrelate, so the
199        // honest integration is ascent on the same disagreement
200        // potential — each fiber drifts by its own dynamics rather
201        // than being held to agreement. Same measured consensus and
202        // thermal bookkeeping either way; only the direction the
203        // field moves changes with the phase its own carrier runs on.
204        //
205        // The coupling integrates at the field's own substep rate —
206        // the same gain/sigma laws applied at finer dt, which is the
207        // more accurate quadrature of the identical dynamics, and it
208        // is what lets the spectrum resolve the trajectory's own
209        // frequencies rather than the tick's sample times.
210        let rem = state.zones.phase() == MentalPhase::REM;
211        let n = ((dt * SPECTRUM_STEP_HZ).round() as usize).clamp(1, 512);
212        let h = dt / n as f64;
213        let mut consensus = None;
214        for _ in 0..n {
215            consensus = if rem {
216                next.diverge(h, bath)?
217            } else {
218                next.synchronize(h, bath)?
219            };
220            self.spectrum.sample(&next, h);
221        }
222        let consensus = consensus.ok_or(FieldError::EmptyState)?;
223        self.spectrum.observe_consensus(&consensus, dt);
224        let integration = FieldIntegration {
225            consensus,
226            disagreement: next.disagreement,
227            temperature: next.temperature,
228            fiber_count: next.fibers.len(),
229        };
230        self.field = next;
231        Ok(integration)
232    }
233}

235/// Frame offset between the body's somatic axes and the prediction
236/// system's inferential axes — the one bend in the connection,
237/// small enough to stay a frame mismatch and large enough for
238/// `path_disagreement` to measure.
239const AFFERENT_FRAME_OFFSET: f64 = 0.35;

241/// Givens rotation in the (i, j) coordinate plane — the minimal
242/// non-identity transport: a fixed frame offset between two
243/// systems' latent bases, orthogonal so it preserves norms.
244fn rotation(n: usize, i: usize, j: usize, theta: f64) -> Vec<Vec<f64>> {
245    let mut m = identity(n);
246    let (c, s) = (theta.cos(), theta.sin());
247    m[i][i] = c;
248    m[j][j] = c;
249    m[i][j] = -s;
250    m[j][i] = s;
251    m
252}

254fn identity(n: usize) -> Vec<Vec<f64>> {
255    let mut matrix = vec![vec![0.0; n]; n];
256    for (i, row) in matrix.iter_mut().enumerate() {
257        row[i] = 1.0;
258    }
259    matrix
260}

262/// The loxodromic drift generator: a skew-symmetric rotation in the
263/// (i, j) coordinate plane at angular rate `omega` — the winding
264/// half of the homeostatic spiral. With damping at the same rate it
265/// traces a 45° loxodrome into the pole: each e-fold of radial decay
266/// carries one radian of turn.
267fn loxodrome(n: usize, i: usize, j: usize, omega: f64) -> Vec<Vec<f64>> {
268    let mut m = vec![vec![0.0; n]; n];
269    m[i][j] = -omega;
270    m[j][i] = omega;
271    m
272}

274fn unit(value: f32, min: f32, max: f32) -> f64 {
275    if !value.is_finite() {
276        return 0.5;
277    }
278    ((value - min) / (max - min)).clamp(0.0, 1.0) as f64
279}

281/// Signed latent half-width. Normalized channels live in [0,1];
282/// the latent coordinate centers them on the origin at ±0.25 —
283/// neutral state is the sphere's pole, and the corner of the
284/// operating box (√6·0.25 ≈ 0.61) stays well inside the
285/// hemisphere cap, where the log/exp maps are accurate instead of
286/// being radially clamped onto the equator rim.
287const LATENT_HALF: f64 = 0.25;
288
289fn latent(unit: f64) -> f64 {
290    (unit - 0.5) * 2.0 * LATENT_HALF
291}
292
293fn unit_finite(value: f32) -> f64 {
294    crate::state::sanitize::finite_clamp(value, 0.0, 1.0) as f64
295}

297fn effective(state: &GenesisCoreState, id: NeurochemicalId) -> f32 {
298    let value = state.neurochemicals.effective(id);
299    if id == NeurochemicalId::Dopamine {
300        crate::state::sanitize::finite_or(value, 0.0)
301    } else {
302        value.clamp(0.0, 2.0)
303    }
304}

306fn body_projection(state: &GenesisCoreState) -> Vector {
307    let n = effective(state, NeurochemicalId::Norepinephrine);
308    let d = effective(state, NeurochemicalId::Dopamine);
309    let s = effective(state, NeurochemicalId::Serotonin);
310    let b = effective(state, NeurochemicalId::BDNF);
311    let c = effective(state, NeurochemicalId::Cortisol);
312    let valence = state.neurochemicals.valence.clamp(-1.0, 1.0);
313
314    vec![
315        latent(unit(valence, -1.0, 1.0)),
316        latent(unit(n, 0.0, 2.0)),
317        latent(unit(d, 0.0, 2.0)),
318        latent(unit(s, 0.0, 2.0)),
319        latent(unit(b, 0.0, 2.0)),
320        latent(unit(c, 0.0, 2.0)),
321    ]
322}

471    #[test]
472    fn fibers_spiral_inward_toward_nominal_between_observations() {
473        // The engines carry a real law: an unobserved fiber winds
474        // toward the sphere's pole at its own clearance rate — the
475        // loxodrome. The drift must shrink the state's norm and
476        // turn its phase, never grow it, never leave the cap.
477        let mut runtime = FieldRuntime::new().unwrap();
478        runtime.field.fibers[0].layers[0].state = vec![0.2; LATENT_DIMENSION];
479        let before: f64 = runtime.field.fibers[0].layers[0]
480            .state
481            .iter()
482            .map(|v| v * v)
483            .sum::<f64>()
484            .sqrt();
485        let before_first_two: f64 =
486            runtime.field.fibers[0].layers[0].state[0] / runtime.field.fibers[0].layers[0].state[1];
487        runtime.field.fibers[0].step(1.0).unwrap();
488        let state = &runtime.field.fibers[0].layers[0].state;
489        let after: f64 = state.iter().map(|v| v * v).sum::<f64>().sqrt();
490        assert!(after < before, "damping must contract the state");
491        // 1 s at τ=120 s ≈ 0.83% contraction — real but gentle.
492        assert!(after > before * 0.97, "the drift is a leak, not a collapse");
493        assert!(after < before * (1.0 - 0.5 / BODY_RELAX_TAU_S));
494        // The (0,1)-plane winding turned the phase between the first
495        // two coordinates — the spiral, not just the sink.
496        let after_first_two = state[0] / state[1];
497        assert_ne!(after_first_two, before_first_two, "the skew term must wind");
498    }
