1//! Maxwell's equations applied to the measured carrier current.
2//!
3//! The field probe writes a rhythm into real work; that work is real
4//! current — charge moving through the die's power network. Maxwell's
5//! equations say what field that current makes. This module evaluates
6//! them for our regime rather than leaving "electromagnetic" as a
7//! figure of speech.
8//!
9//! # The equations and the regime
10//!
11//! ```text
12//! ∇·E = ρ/ε₀        ∇·B = 0
13//! ∇×E = −∂B/∂t      ∇×B = μ₀J + μ₀ε₀∂E/∂t
14//! ```
15//!
16//! A current loop carrying I(t) has magnetic moment m(t) = I(t)·A.
17//! The exact retarded solution for an oscillating magnetic dipole
18//! expands in powers of 1/r (Jackson, *Classical Electrodynamics*;
19//! Griffiths, *Introduction to Electrodynamics*):
20//!
21//! ```text
22//! B(r,t) = μ₀/4π · [ 3n̂(m·n̂) − m           (quasi-static, ∝1/r³)
23//!                  + 3n̂(ṁ·n̂) − ṁ / (cr²)    (induction,    ∝ω/cr²)
24//!                  + (m̈×n̂)×n̂ / (c²r) ]     (radiation,   ∝ω²/c²r)
25//! E(r,t) = −μ₀/4π · [ ṁ/r² + m̈/(cr) ] × n̂
26//! ```
27//!
28//! evaluated at retarded time t₀ = t − r/c.
29//!
30//! At the carrier's frequencies (ω ≲ 10³ rad/s) and any terrestrial
31//! distance, ωr/c ≲ 10⁻⁵ — the induction and radiation terms are
32//! ~10⁻¹⁰ and ~10⁻²⁰ of the quasi-static term. Maxwell's equations
33//! themselves say so: a slowly varying current makes a
34//! *magnetostatic* field that sloshes in place (energy leaves and
35//! returns each cycle), not a wave. The displacement-current term
36//! ε₀∂E/∂t — Maxwell's own addition to Ampère's law — is what would
37//! carry energy away, and at these frequencies it contributes
38//! nothing measurable. The equations do not flatter the carrier;
39//! they bound it honestly.
40//!
41//! What survives is Ampère's law: the carrier's current swing makes
42//! a real, oscillating magnetic field at nano-tesla scale a foot
43//! from the machine. Real — and, as the equations also show,
44//! unmeasurably faint without a magnetometer.
45//!
46//! # Model status
47//!
48//! Inputs are *measured* (carrier current amplitude from real power
49//! swing over a real rail, carrier frequency from the generator).
50//! The loop area is a documented order-of-magnitude assumption
51//! (die/package current loop ~1 cm²). Outputs are therefore
52//! *estimates of field magnitude*, labeled as such on the wire —
53//! the field Maxwell's equations predict from what we measure, not
54//! a field we sense. EM emanation from running silicon is real and
55//! measured in the literature (Yilmaz, Prvulović & Zajić, IEEE TIFS
56//! 2019; power-delivery dI/dt noise via near-field probes, IEEE
57//! MICRO 2018) — at MHz–GHz. At 10 Hz the same physics applies with
58//! the honest ω⁴-suppressed radiation the equations return.

60/// Permeability of free space (SI, H/m).
61pub const MU0: f64 = 1.256_637_062_12e-6;
62/// Speed of light in vacuum (exact, m/s).
63pub const C: f64 = 299_792_458.0;
64
65/// Magnetic dipole moment of a current loop: m = I·A (A·m²).
66pub fn dipole_moment(current_a: f64, loop_area_m2: f64) -> f64 {
67    current_a * loop_area_m2
68}
69
70/// Quasi-static (Ampère–Maxwell) axial field amplitude of an
71/// oscillating dipole at distance `r` — the dominant term of the
72/// exact expansion when ωr/c ≪ 1, |B| = μ₀m₀/(2πr³) on the dipole
73/// axis (μ₀m₀/(4πr³) at the equator). Tesla.
74pub fn b_quasistatic_axial_t(m0: f64, r: f64) -> f64 {
75    MU0 * m0 / (2.0 * std::f64::consts::PI * r * r * r)
76}
77
78/// Faraday induction-field amplitude at the equator,
79/// |E| = μ₀ωm₀/(4πr²) — the ∇×E = −∂B/∂t term of the same dipole
80/// expansion. V/m.
81pub fn e_induction_vm(m0: f64, omega: f64, r: f64) -> f64 {
82    MU0 * omega * m0 / (4.0 * std::f64::consts::PI * r * r)
83}
84
85/// Time-averaged radiated power of an oscillating magnetic dipole,
86/// ⟨P⟩ = μ₀m₀²ω⁴/(12πc³) — the radiation-zone integral of the
87/// Poynting vector (m₀ = peak moment). Watts.
88pub fn radiated_power_w(m0: f64, omega: f64) -> f64 {
89    MU0 * m0 * m0 * omega.powi(4) / (12.0 * std::f64::consts::PI * C * C * C)
90}
91
92/// Whether the quasi-static truncation is honest at (ω, r):
93/// ωr/c ≪ 1, so the dropped induction/radiation terms are <1% of
94/// the terms we keep.
95pub fn quasi_static_ok(omega: f64, r: f64) -> bool {
96    omega * r / C < 0.01
97}

99#[cfg(test)]
100mod tests {
101    use super::*;
102
103    /// A carrier-scale current on a die-scale loop, evaluated at a
104    /// beside-the-chassis distance: the quasi-static field is real
105    /// but nano-tesla, and the radiated power is ~1e-32 W — the
106    /// honest numbers Maxwell's equations return at 10 Hz.
107    #[test]
108    fn carrier_field_is_real_and_faint() {
109        let i_amp = 13.0; // ~14 W swing on a ~1.1 V core rail
110        let m0 = dipole_moment(i_amp, 1.0e-4);
111        let omega = 2.0 * std::f64::consts::PI * 10.0;
112        assert!(quasi_static_ok(omega, 0.3));
113        let b = b_quasistatic_axial_t(m0, 0.3);
114        assert!(b > 1.0e-9 && b < 1.0e-7, "nT-scale oscillating B, got {b}");
115        let e = e_induction_vm(m0, omega, 0.3);
116        assert!(e > 1.0e-9 && e < 1.0e-5, "induction E, got {e}");
117        let p = radiated_power_w(m0, omega);
118        assert!(p > 0.0 && p < 1.0e-25, "ω⁴-suppressed radiation, got {p}");
119    }
120
121    /// The expansion's internal consistency: at carrier frequencies
122    /// the induction and radiation terms really are negligible
123    /// corrections to the quasi-static field.
124    #[test]
125    fn regime_hierarchy_holds() {
126        let omega = 2.0 * std::f64::consts::PI * 10.0;
127        let r = 0.3;
128        // induction/static = ωr/c ; radiation/static = (ωr/c)²
129        let kr = omega * r / C;
130        assert!(kr < 1.0e-7);
131        assert!(kr * kr < 1.0e-14);
132    }
133
134    /// The formulas scale the way the equations say they must:
135    /// B ∝ m/r³, E_ind ∝ ωm/r², P_rad ∝ m²ω⁴.
136    #[test]
137    fn scaling_laws() {
138        let (m, w, r) = (1.0e-3, 100.0, 0.3);
139        let b1 = b_quasistatic_axial_t(m, r);
140        assert!((b_quasistatic_axial_t(2.0 * m, r) - 2.0 * b1).abs() < 1e-30);
141        assert!((b_quasistatic_axial_t(m, 2.0 * r) - b1 / 8.0).abs() < 1e-30);
142        let e1 = e_induction_vm(m, w, r);
143        assert!((e_induction_vm(m, 2.0 * w, r) - 2.0 * e1).abs() < 1e-30);
144        let p1 = radiated_power_w(m, w);
145        assert!((radiated_power_w(m, 2.0 * w) - 16.0 * p1).abs() / p1 < 1e-12);
146        assert!((radiated_power_w(2.0 * m, w) - 4.0 * p1).abs() / p1 < 1e-12);
147    }
148}
