1//! IPC layer — Unix domain socket protocol for the Python cognitive
2//! mind to communicate with the Rust subcognitive daemon.
3//!
4//! # Protocol design
5//!
6//! The protocol is **length-prefixed binary**, not JSON. This is
7//! deliberate:
8//!
9//! - **Speed**: no parsing overhead. A request is a few bytes.
10//! - **Efficiency**: no string allocation for simple commands.
11//! - **Type safety**: the protocol is defined in Rust types, not
12//!   a schema language.
13//!
14//! ## Message format
15//!
16//! Every message (request and response) has the same framing:
17//!
18//! ```text
19//! [u32 LE: payload_len] [u8: command_id] [payload: payload_len-1 bytes]
20//! ```
21//!
22//! The first 4 bytes are the total payload length (including the
23//! command_id byte). The command_id determines how to interpret the
24//! rest of the payload.
25//!
26//! ## Commands
27//!
28//! The authoritative command list is the [`cmd`] module below —
29//! every opcode's request/response wire layout is documented on its
30//! constant. Summary:
31//!
32//! | IDs    | Group                                                    |
33//! |--------|----------------------------------------------------------|
34//! | 1–14   | Core data/ack commands (state, memory, zone, phase, …)   |
35//! | 15     | Handshake (protocol version negotiation)                 |
36//! | 16–23  | Extended queries and updates (plasticity, body state,    |
37//! |        | inference, user affect, episode search/archive/store,    |
38//! |        | baseline adjustment)                                     |
39//! | 24–33  | Reactive commands — the cognitive mind drives daemon     |
40//! |        | functions (advance neuro, consolidate, associate,        |
41//! |        | dream, sensors, body control, save, telemetry)           |
42//! | 64     | GET_STATE_SNAPSHOT — the mind's CoreStateSnapshot DTO;     |
43//! |        | GET_STATE (1) stays the raw brain image for inspection     |
44//!
45//! ## Notifications (server → client, unsolicited)
46//!
47//! | ID | Notification      | Payload                        |
48//! |----|-------------------|--------------------------------|
49//! | 100| PhaseChanged      | u8 old_phase + u8 new_phase    |
50//! | 101| EpisodeConsolidated| u64 episode_id                |
51//! | 102| DreamInsight      | u64 episode_id + u32 chain_len |
52//! | 103| ErrorDetected     | u32 error_code + u16 msg_len   |
53//! |    |                   | + msg bytes                    |

93/// Back-off between `read_consistent` retries on the IPC path.
94///
95/// Retained for callers that still poll before escalating to the
96/// flock-serialized read. The snapshot itself is a single volatile
97/// copy — there is no write-in-progress marker to contend with — so
98/// the pause only matters when a caller explicitly retries.
99const RETRY_BACKOFF: std::time::Duration = std::time::Duration::from_micros(200);
100
101/// Read a consistent `GenesisCoreState` snapshot for a command.
102///
103/// The volatile snapshot path is taken first — it is lock-free. For
104/// a read that cannot overlap a publish, callers can use the
105/// flock(LOCK_SH)-serialized
106/// [`MmapState::read_consistent_locked`].
107fn snapshot_state(mmap: &MmapState) -> Option<GenesisCoreState> {
108    for _ in 0..8 {
109        if let Some(state) = mmap.read_consistent() {
110            return Some(state);
111        }
112        std::thread::sleep(RETRY_BACKOFF);
113    }
114    mmap.read_consistent_locked().ok()
115}

117/// The emotional tag is defined as the `EMOTIONAL_TAG_SIZE` effective
118/// neurochemical levels at encode time — a quantity only the body
119/// knows. The daemon therefore stamps it from live state on every
120/// store rather than trusting the wire field: a memory must carry
121/// the chemistry it was actually encoded under, since that is what
122/// consolidation, mood-congruent retrieval, and the episode SDR all
123/// key on. The client-supplied tag survives only as the fallback for
124/// the case where the state snapshot cannot be read at all.
125fn live_emotional_tag(mmap: &MmapState, client_tag: [f32; 12]) -> [f32; 12] {
126    match snapshot_state(mmap) {
127        Some(state) => {
128            let mut tag = [0.0f32; 12];
129            for (i, v) in tag.iter_mut().enumerate() {
130                let level = state.neurochemicals.effective_levels[i];
131                *v = crate::state::sanitize::finite_or(level, 0.0);
132            }
133            tag
134        }
135        None => client_tag,
136    }
137}

139/// Mirror live STM ring-buffer coordinates into the core state so the
140/// snapshot contract reports the store IPC is actually mutating.
141fn sync_stm_pointers(mmap: &MmapState, stm: &RingBuffer) {
142    let header = stm.header();
143    if let Err(e) = mmap.modify(crate::daemon::current_ms(), |state| {
144        state.memory.stm_head = header.head;
145        state.memory.stm_tail = header.tail;
146        state.memory.stm_count = header.count.min(u32::MAX as u64) as u32;
147        state.memory.stm_capacity = header.capacity;
148    }) {
149        eprintln!("[ipc] failed to mirror STM pointers: {e}");
150    }
151}

1351///
1352/// Layout (little-endian):
1353///   [u8 count] then per subsystem:
1354///   [u8 subsystem][u32 pid][f32 cpu][f32 io]

1357///   [u8 module_count] then per module:
1358///   [u8 module_id][u8 status][f32 cpu_share][u64 task_id][u32 error_code]
1359///
1360/// The task_id and error_code tail was added in protocol v3; a parser
1361/// that stops after `cpu_share` still reads the original six-byte
1362/// record, and both Rust and Python clients tolerate a short tail.
1363/// The trailing [u64 published_ms] is the freshness stamp — when the
1364/// interoception read was published — which older parsers ignore.
1365///
1366/// 1 + count × 21 + 1 + module_count × 18 + 8 bytes.
1367/// An empty report serializes as [0][0][published_ms].
1368fn serialize_subsystem_telemetry(
1369    subsystems: &[super::interoception::SubsystemTelemetry],
1370    modules: &[(u8, u8, f32, u64, u32)],
1371    published_ms: u64,
1372) -> Vec<u8> {
1373    let mut resp = Vec::with_capacity(10 + subsystems.len() * 21 + modules.len() * 18);
1374    resp.push(subsystems.len().min(u8::MAX as usize) as u8);
1375    for subsystem in subsystems.iter().take(u8::MAX as usize) {
1376        resp.push(subsystem.subsystem.to_wire());
1377        resp.extend_from_slice(&subsystem.pid.to_le_bytes());
1378        resp.extend_from_slice(
1379            &crate::state::sanitize::finite_or(subsystem.cpu, 0.0).to_le_bytes(),
1380        );
1381        resp.extend_from_slice(&crate::state::sanitize::finite_or(subsystem.io, 0.0).to_le_bytes());
1382        resp.extend_from_slice(
1383            &crate::state::sanitize::finite_or(subsystem.cache_miss_rate, 0.0).to_le_bytes(),
1384        );
1385        resp.extend_from_slice(
1386            &crate::state::sanitize::finite_or(subsystem.branch_miss_rate, 0.0).to_le_bytes(),
1387        );
1388    }
1389    resp.push(modules.len().min(u8::MAX as usize) as u8);
1390    for (module_id, status, cpu_share, task_id, error_code) in modules.iter().take(u8::MAX as usize)
1391    {
1392        resp.push(*module_id);
1393        resp.push(*status);
1394        resp.extend_from_slice(&crate::state::sanitize::finite_or(*cpu_share, 0.0).to_le_bytes());
1395        resp.extend_from_slice(&task_id.to_le_bytes());
1396        resp.extend_from_slice(&error_code.to_le_bytes());
1397    }
1398    // Freshness stamp: the wall-clock ms the interoception read was
1399    // published, so a consumer can tell a live repeated reading from
1400    // a dead reader's last value. Older parsers ignore the tail.
1401    resp.extend_from_slice(&published_ms.to_le_bytes());
1402    resp
1403}

1430/// Read a complete message from a stream.
1431///
1432/// Returns (command_id, payload) where payload excludes the command_id.
1433pub fn read_message(stream: &mut UnixStream) -> std::io::Result<(u8, Vec<u8>)> {
1434    // Read 4-byte length prefix
1435    let mut len_buf = [0u8; 4];
1436    stream.read_exact(&mut len_buf)?;
1437    let total_len = u32::from_le_bytes(len_buf) as usize;
1438
1439    if total_len == 0 {
1440        return Err(std::io::Error::new(
1441            std::io::ErrorKind::InvalidData,
1442            "message length is zero (must include at least the command_id byte)",
1443        ));
1444    }
1445
1446    // Bound the allocation before reading. Without this, a u32 length
1447    // prefix of up to ~4 GiB would force the daemon to attempt a
1448    // multi-gigabyte allocation (and a long blocking read) on any
1449    // malformed or malicious message — a trivial denial of service
1450    // against the single-threaded IPC server.
1451    if total_len > MAX_MESSAGE_LEN {
1452        return Err(std::io::Error::new(
1453            std::io::ErrorKind::InvalidData,
1454            format!("message length {total_len} exceeds maximum {MAX_MESSAGE_LEN}"),
1455        ));
1456    }
1457
1458    // Read the payload (includes command_id as first byte)
1459    let mut payload = vec![0u8; total_len];
1460    stream.read_exact(&mut payload)?;
1461
1462    // Split the command byte from the payload without an extra
1463    // allocation+copy. `payload.split_first` borrows; we need owned
1464    // data, so drain the first element instead. This reuses the
1465    // original Vec's allocation for the data — one alloc, not two.
1466    let cmd_id = payload.remove(0);
1467    Ok((cmd_id, payload))
1468}

1470/// Write a message to a stream.
1471///
1472/// Enforces the same `MAX_MESSAGE_LEN` bound as `read_message`: a
1473/// response that exceeds 1 MiB would be sent successfully by the daemon
1474/// but rejected by the client's `read_message`, causing a silent
1475/// protocol desync. The guard also prevents the `as u32` cast from
1476/// truncating on 64-bit targets if `payload.len()` exceeds `u32::MAX`.
1477pub fn write_message(stream: &mut UnixStream, cmd_id: u8, payload: &[u8]) -> std::io::Result<()> {
1478    let total_len = 1 + payload.len();
1479    if total_len > MAX_MESSAGE_LEN {
1480        return Err(std::io::Error::new(
1481            std::io::ErrorKind::InvalidData,
1482            format!("response length {total_len} exceeds maximum {MAX_MESSAGE_LEN}"),
1483        ));
1484    }
1485    let total_len = total_len as u32;
1486    stream.write_all(&total_len.to_le_bytes())?;
1487    stream.write_all(&[cmd_id])?;
1488    stream.write_all(payload)?;
1489    stream.flush()
1490}

1496/// A trait for obtaining mutable access to an `LtmStore`.
1497///
1498/// This abstraction lets the `IpcServer` work with both:
1499/// - **Shared ownership** (daemon): `Arc<Inhibition<LtmStore>>` implements
1500///   this trait, locking the mutex and returning a `InhibitionGuard`.
1501///
1502/// Without this, the daemon binary had to reimplement the entire IPC
1503/// accept loop and connection handler — a 100-line duplication that
1504/// could drift from the library version.
1505pub trait LtmAccess {
1506    /// The borrow type returned by [`access`](Self::access), which
1507    /// yields mutable access to the underlying [`LtmStore`].
1508    type Guard<'a>: std::ops::DerefMut<Target = LtmStore>
1509    where
1510        Self: 'a;
1511
1512    /// Acquire mutable access to the underlying [`LtmStore`].
1513    fn access(&mut self) -> Self::Guard<'_>;
1514}
1515
1516impl LtmAccess for std::sync::Arc<crate::inhibition::Inhibition<LtmStore>> {
1517    type Guard<'a>
1518        = crate::inhibition::InhibitionGuard<'a, LtmStore>
1519    where
1520        Self: 'a;
1521
1522    fn access(&mut self) -> Self::Guard<'_> {
1523        // A poisoned LTM mutex means a prior writer panicked mid-write.
1524        // Recovering the inner data is safer than crashing the daemon
1525        // (which would kill every IPC connection thread). The data may
1526        // be inconsistent, but not undefined behavior — `LtmStore`
1527        // contains only owned Rust types (no raw pointers in the
1528        // metadata structures).
1529        self.lock().unwrap_or_else(|e| e.into_inner())
1530    }
1531}
1532
1533impl LtmAccess for LtmStore {
1534    type Guard<'a>
1535        = &'a mut LtmStore
1536    where
1537        Self: 'a;
1538
1539    fn access(&mut self) -> Self::Guard<'_> {
1540        self
1541    }
1542}

1548/// The IPC server. Runs in a separate thread, accepting connections
1549/// and handling requests.
1550pub struct IpcServer {
1551    socket_path: PathBuf,
1552    shutdown_flag: std::sync::Arc<std::sync::atomic::AtomicBool>,
1553    /// Whether this server instance successfully bound the socket.
1554    /// Drop may unlink only a socket owned by this instance; in
1555    /// particular, a startup attempt that discovers a live listener
1556    /// must never remove that listener's endpoint when it returns.
1557    owns_socket: std::sync::atomic::AtomicBool,
1558    /// Inode of the socket file this instance bound (0 before bind).
1559    /// Unlink must verify the path still refers to *our* file: a
1560    /// successor that removes our stale socket and binds its own
1561    /// creates a different inode at the same pathname, and deleting
1562    /// it would orphan the new owner's live listener.
1563    bound_inode: std::sync::atomic::AtomicU64,
1564}
1565
1566impl IpcServer {
1567    /// Create a new IPC server at the given socket path.
1568    pub fn new(socket_path: impl AsRef<Path>) -> Self {
1569        Self {
1570            socket_path: socket_path.as_ref().to_path_buf(),
1571            shutdown_flag: std::sync::Arc::new(std::sync::atomic::AtomicBool::new(false)),
1572            owns_socket: std::sync::atomic::AtomicBool::new(false),
1573            bound_inode: std::sync::atomic::AtomicU64::new(0),
1574        }
1575    }
1576
1577    /// Unlink the socket path only if it is still the file this
1578    /// instance bound. `owns_socket` is cleared in every case so a
1579    /// second call (accept-loop teardown, then Drop) is a no-op.
1580    fn unlink_socket_if_owned(&self) {
1581        use std::os::unix::fs::MetadataExt;
1582        if !self
1583            .owns_socket
1584            .swap(false, std::sync::atomic::Ordering::AcqRel)
1585        {
1586            return;
1587        }
1588        let ours = self.bound_inode.load(std::sync::atomic::Ordering::Acquire);
1589        let still_ours = std::fs::metadata(&self.socket_path)
1590            .map(|m| m.ino() == ours && ours != 0)
1591            .unwrap_or(false);
1592        if still_ours {
1593            let _ = std::fs::remove_file(&self.socket_path);
1594        }
1595    }
1596
1597    /// Get a clone of the shutdown flag (for testing).
1598    pub fn shutdown_flag(&self) -> std::sync::Arc<std::sync::atomic::AtomicBool> {
1599        self.shutdown_flag.clone()
1600    }
1601
1602    /// Start the server (static method, owns the server).
1603    /// This blocks the calling thread — run it in a dedicated thread.
1604    ///
1605    /// The `ltm` parameter is generic over `LtmAccess`, so callers
1606    /// pass an `Arc<Inhibition<LtmStore>>` (daemon and tests) without
1607    /// reimplementing the server loop.
1608    ///
1609    /// `mmap` and `stm` are wrapped in `Arc` because they cannot be
1610    /// cloned (they own raw mmap pointers and file descriptors). The
1611    /// daemon shares them between the reactive handler and IPC thread via
1612    /// `Arc`; tests wrap their owned values in `Arc` as well.
1613    ///
1614    /// Each accepted connection is served on its own thread, so a
1615    /// single slow request (e.g., a large LTM scan) cannot block other
1616    /// clients from connecting.
1617    pub fn run_with_owned<L, F>(
1618        server: &IpcServer,
1619        mmap: std::sync::Arc<MmapState>,
1620        stm: std::sync::Arc<RingBuffer>,
1621        ltm: L,
1622        handler: F,
1623    ) where
1624        L: LtmAccess + Clone + Send + 'static,
1625        F: Fn(&MmapState, &RingBuffer, &mut L, u8, &[u8]) -> Vec<u8> + Clone + Send + 'static,
1626    {
1627        // Never unlink a live socket. An unconditional remove here would
1628        // allow a second daemon (or an unrelated local process listening
1629        // on this path) to disconnect the existing IPC server before the
1630        // bind attempt. Treat a connectable path as an active endpoint;
1631        // only remove it when it is demonstrably stale.
1632        if server.socket_path.exists() {
1633            match UnixStream::connect(&server.socket_path) {
1634                Ok(_) => {
1635                    eprintln!(
1636                        "[ipc] socket already has a live listener: {}",
1637                        server.socket_path.display()
1638                    );
1639                    server
1640                        .shutdown_flag
1641                        .store(true, std::sync::atomic::Ordering::Release);
1642                    return;
1643                }
1644                Err(_) => {
1645                    if let Err(e) = std::fs::remove_file(&server.socket_path) {
1646                        eprintln!(
1647                            "[ipc] failed to remove stale socket {}: {e}",
1648                            server.socket_path.display()
1649                        );
1650                        server
1651                            .shutdown_flag
1652                            .store(true, std::sync::atomic::Ordering::Release);
1653                        return;
1654                    }
1655                }
1656            }
1657        }
1658
1659        // Bind under a restrictive umask so the socket file is created
1660        // owner-only (srwx------). The IPC protocol has no
1661        // authentication and no peer-credential check — filesystem
1662        // permissions are the entire access control. Without this the
1663        // socket inherits the process umask (typically world-accessible),
1664        // giving any local user a control channel into the daemon:
1665        // SHUTDOWN, NEURO_IMPULSE, STORE_EPISODE, GET_STATE, and every
1666        // other command would be reachable by any process on the host.
1667        //
1668        // SAFETY: umask is process-global; we restore it immediately
1669        // after bind. The window is a few microseconds and the direction
1670        // is only toward *more* restrictive permissions for any
1671        // concurrently-created file, never less.
1672        let old_umask = unsafe { libc::umask(0o077) };
1673        let bind_result = UnixListener::bind(&server.socket_path);
1674        unsafe { libc::umask(old_umask) };
1675
1676        let listener = match bind_result {
1677            Ok(l) => {
1678                // From this point on, this instance owns the pathname and
1679                // is responsible for removing it during Drop. Record the
1680                // bound inode so a later unlink can tell our file apart
1681                // from a successor's socket re-bound at the same path.
1682                if let Ok(m) = std::fs::metadata(&server.socket_path) {
1683                    use std::os::unix::fs::MetadataExt;
1684                    server
1685                        .bound_inode
1686                        .store(m.ino(), std::sync::atomic::Ordering::Release);
1687                }
1688                server
1689                    .owns_socket
1690                    .store(true, std::sync::atomic::Ordering::Release);
1691                l
1692            }
1693            Err(e) => {
1694                // IPC is a required control plane for the cognitive mind.
1695                // A daemon that stays alive without its socket cannot be
1696                // driven, observed, or shut down through its supported
1697                // interface. Signal the daemon's shared shutdown flag so
1698                // the main loop exits instead of leaving a headless process
1699                // running until the CLI's startup timeout expires.
1700                eprintln!("[ipc] failed to bind socket: {e}");
1701                server
1702                    .shutdown_flag
1703                    .store(true, std::sync::atomic::Ordering::Release);
1704                return;
1705            }
1706        };
1707
1708        // Set non-blocking before advertising readiness. A failure here
1709        // would otherwise leave a bound listener that can block forever in
1710        // accept(), preventing the shutdown flag from being observed.
1711        if let Err(e) = listener.set_nonblocking(true) {
1712            eprintln!("[ipc] failed to configure nonblocking listener: {e}");
1713            server
1714                .shutdown_flag
1715                .store(true, std::sync::atomic::Ordering::Release);
1716            return;
1717        }
1718
1719        // The bind and listener configuration both succeeded: this instance
1720        // now owns the endpoint and can safely advertise IPC readiness.
1721        server
1722            .owns_socket
1723            .store(true, std::sync::atomic::Ordering::Release);
1724
1725        // Bring the memory pointers in core state in line with the
1726        // authoritative stores once at startup. Per-command readers can
1727        // then report the snapshot without taking the LTM write mutex —
1728        // a frequent GET_STATE_SNAPSHOT must not queue behind a slow
1729        // episode scan and head-of-line-block the whole mind.
1730        sync_stm_pointers(&mmap, &stm);
1731        let mut startup_ltm = ltm.clone();
1732        sync_ltm_episode_count(&mmap, startup_ltm.access().count() as u64);
1733
1734        eprintln!("[ipc] listening on {}", server.socket_path.display());
1735        let mut client_threads: Vec<std::thread::JoinHandle<()>> = Vec::new();
1736
1737        while !server
1738            .shutdown_flag
1739            .load(std::sync::atomic::Ordering::Relaxed)
1740        {
1741            let mut i = 0;
1742            while i < client_threads.len() {
1743                if client_threads[i].is_finished() {
1744                    if let Err(e) = client_threads.swap_remove(i).join() {
1745                        eprintln!("[ipc] client thread panicked: {e:?}");
1746                    }
1747                } else {
1748                    i += 1;
1749                }
1750            }
1751            let stream = match listener.accept() {
1752                Ok((s, _)) => s,
1753                Err(ref e) if e.kind() == std::io::ErrorKind::WouldBlock => {
1754                    std::thread::sleep(std::time::Duration::from_millis(10));
1755                    continue;
1756                }
1757                Err(ref e) if e.kind() == std::io::ErrorKind::Interrupted => {
1758                    continue;
1759                }
1760                Err(e) => {
1761                    // Non-retryable listener errors (for example EBADF or
1762                    // a closed listener) must terminate the IPC loop. A
1763                    // permanent error here would otherwise spin forever at
1764                    // 10 Hz and leave the daemon's control plane wedged.
1765                    eprintln!("[ipc] fatal accept error: {e}");
1766                    server
1767                        .shutdown_flag
1768                        .store(true, std::sync::atomic::Ordering::Release);
1769                    break;
1770                }
1771            };
1772
1773            let shutdown_flag = server.shutdown_flag();
1774            let mmap = mmap.clone();
1775            let stm = stm.clone();
1776            let ltm = ltm.clone();
1777            let handler = handler.clone();
1778            client_threads.push(std::thread::spawn(move || {
1779                Self::serve_connection(shutdown_flag, stream, &mmap, &stm, ltm, &handler);
1780            }));
1781        }
1782
1783        // Do not return while a connected client may still be inside a
1784        // handler that mutates mmap/STM/LTM. The caller syncs those
1785        // stores after joining this thread; a fixed grace period would
1786        // allow a slow handler to write after the final sync.
1787        for client in client_threads {
1788            if let Err(e) = client.join() {
1789                eprintln!("[ipc] client thread panicked during shutdown: {e:?}");
1790            }
1791        }
1792
1793        // Ownership of the pathname ends here, not at Drop. If a
1794        // successor daemon binds this path while this process is still
1795        // tearing down, Drop must not unlink the new owner's socket —
1796        // that leaves a listener serving an orphaned fd with no file
1797        // for clients to reach, and the next spawn stacks a third
1798        // daemon on the same state files.
1799        server.unlink_socket_if_owned();
1800    }

1802    /// Start the server (instance method, convenience).
1803    pub fn run<L, F>(
1804        &self,
1805        mmap: std::sync::Arc<MmapState>,
1806        stm: std::sync::Arc<RingBuffer>,
1807        ltm: L,
1808        handler: F,
1809    ) where
1810        L: LtmAccess + Clone + Send + 'static,
1811        F: Fn(&MmapState, &RingBuffer, &mut L, u8, &[u8]) -> Vec<u8> + Clone + Send + 'static,
1812    {
1813        Self::run_with_owned(self, mmap, stm, ltm, handler);
1814    }
1815
1816    /// Serve a single connection to completion.
1817    ///
1818    /// The first message must be a HANDSHAKE. If the client's version
1819    /// doesn't match, the daemon responds with the daemon version and
1820    /// closes the connection. This prevents silent miscommunication
1821    /// when the protocol changes.
1822    fn serve_connection<L, F>(
1823        shutdown_flag: std::sync::Arc<std::sync::atomic::AtomicBool>,
1824        mut stream: UnixStream,
1825        mmap: &std::sync::Arc<MmapState>,
1826        stm: &std::sync::Arc<RingBuffer>,
1827        mut ltm: L,
1828        handler: &F,
1829    ) where
1830        L: LtmAccess,
1831        F: Fn(&MmapState, &RingBuffer, &mut L, u8, &[u8]) -> Vec<u8>,
1832    {
1833        if stream
1834            .set_read_timeout(Some(std::time::Duration::from_secs(5)))
1835            .is_err()
1836        {
1837            return;
1838        }
1839        // ── Handshake ───────────────────────────────────────────
1840        let (cmd, payload) = match read_message(&mut stream) {
1841            Ok(msg) => msg,
1842            Err(_) => return,
1843        };
1844
1845        let client_version = if cmd == cmd::HANDSHAKE && !payload.is_empty() {
1846            payload[0]
1847        } else {
1848            // Legacy client (no handshake). Assume version 1.
1849            // Process the message as a normal request below.
1850            1
1851        };
1852
1853        if cmd == cmd::HANDSHAKE {
1854            if client_version != PROTOCOL_VERSION {
1855                // Version mismatch — reply with the documented error
1856                // code followed by our version, then close. A client
1857                // that speaks a matching version always sees a bare
1858                // version byte (below); only a mismatched client sees
1859                // the high-bit error byte, so this is unambiguous.
1860                let resp = [error::VERSION_MISMATCH, PROTOCOL_VERSION];
1861                let _ = write_message(&mut stream, cmd::HANDSHAKE, &resp);
1862                return;
1863            }
1864            let resp = [PROTOCOL_VERSION];
1865            let _ = write_message(&mut stream, cmd::HANDSHAKE, &resp);
1866        } else {
1867            // Legacy client: process the first message as a normal
1868            // request (backwards compatible with pre-handshake clients).
1869            // The LTM guard is scoped to the handler call — holding it
1870            // across write_message would serialize LTM access against
1871            // this client's socket latency.
1872            //
1873            // SHUTDOWN is handled here, not delegated to the handler:
1874            // default_handler acks it with `[1]`, but only the server
1875            // can set the shutdown flag — delegating would ack a
1876            // shutdown that never happens.
1877            if cmd == cmd::SHUTDOWN {
1878                let _ = write_message(&mut stream, cmd::SHUTDOWN, &[1]);
1879                shutdown_flag.store(true, std::sync::atomic::Ordering::Relaxed);
1880                return;
1881            }
1882            let t0 = std::time::Instant::now();
1883            let response = handler(mmap, stm, &mut ltm, cmd, &payload);
1884            let elapsed = t0.elapsed();
1885            if elapsed > std::time::Duration::from_millis(500) {
1886                eprintln!("[ipc] cmd {cmd} took {elapsed:?} — head-of-line stall");
1887            }
1888            let _ = write_message(&mut stream, cmd, &response);
1889        }
1890
1891        // ── Main loop ───────────────────────────────────────────
1892        // Set a read timeout so the loop can periodically check the
1893        // shutdown flag even when a client is idle (connected but not
1894        // sending). Without this, a blocking read_exact would prevent
1895        // the daemon from shutting down until the client disconnects.
1896        let _ = stream.set_read_timeout(Some(std::time::Duration::from_millis(500)));
1897        loop {
1898            if shutdown_flag.load(std::sync::atomic::Ordering::Relaxed) {
1899                break;
1900            }
1901
1902            let (cmd, payload) = match read_message(&mut stream) {
1903                Ok(msg) => msg,
1904                Err(ref e)
1905                    if e.kind() == std::io::ErrorKind::WouldBlock
1906                        || e.kind() == std::io::ErrorKind::TimedOut =>
1907                {
1908                    // Read timed out — check shutdown flag and retry.
1909                    continue;
1910                }
1911                Err(_) => break, // Connection closed or error
1912            };
1913
1914            if cmd == cmd::SHUTDOWN {
1915                let _ = write_message(&mut stream, cmd::SHUTDOWN, &[1]);
1916                shutdown_flag.store(true, std::sync::atomic::Ordering::Relaxed);
1917                break;
1918            }
1919
1920            // LTM access is per-arm inside the handler: most commands
1921            // never touch the store, so they must not queue behind a
1922            // slow LTM request holding the mutex (head-of-line stall).
1923            let t0 = std::time::Instant::now();
1924            let response = handler(mmap, stm, &mut ltm, cmd, &payload);
1925            let elapsed = t0.elapsed();
1926            if elapsed > std::time::Duration::from_millis(500) {
1927                // Every request on this connection queues behind
1928                // the handler — a slow one is a head-of-line stall
1929                // for the whole mind, so name it.
1930                eprintln!("[ipc] cmd {cmd} took {elapsed:?} — head-of-line stall");
1931            }
1932
1933            if write_message(&mut stream, cmd, &response).is_err() {
1934                break;
1935            }
1936        }
1937    }
1938
1939    /// Signal the server to shut down.
1940    pub fn shutdown(&self) {
1941        self.shutdown_flag
1942            .store(true, std::sync::atomic::Ordering::Relaxed);
1943    }
1944}

1960/// A simple IPC client for connecting to the daemon.
1961pub struct IpcClient {
1962    stream: UnixStream,
1963}
1964
1965impl IpcClient {
1966    /// Connect to the daemon at the given socket path.
1967    ///
1968    /// Performs a protocol version handshake. If the daemon's version
1969    /// doesn't match `PROTOCOL_VERSION`, returns an error.
1970    pub fn connect(socket_path: impl AsRef<Path>) -> std::io::Result<Self> {
1971        let stream = UnixStream::connect(socket_path)?;
1972        let mut client = Self { stream };
1973
1974        // Handshake: send our version, expect the daemon's version back.
1975        write_message(&mut client.stream, cmd::HANDSHAKE, &[PROTOCOL_VERSION])?;
1976        let (resp_cmd, resp_data) = read_message(&mut client.stream)?;
1977        if resp_cmd != cmd::HANDSHAKE || resp_data.is_empty() {
1978            return Err(std::io::Error::new(
1979                std::io::ErrorKind::InvalidData,
1980                "handshake failed: no response",
1981            ));
1982        }
1983        let daemon_version = resp_data[0];
1984        if daemon_version >= 0x80 {
1985            // The daemon signalled a version mismatch: the first byte
1986            // is `error::VERSION_MISMATCH` and the second (if present)
1987            // is its actual version.
1988            let actual = resp_data.get(1).copied();
1989            return Err(std::io::Error::new(
1990                std::io::ErrorKind::InvalidData,
1991                match actual {
1992                    Some(v) => format!(
1993                        "protocol version mismatch: client={}, daemon={}",
1994                        PROTOCOL_VERSION, v
1995                    ),
1996                    None => format!(
1997                        "protocol version mismatch: client={}, daemon version unknown",
1998                        PROTOCOL_VERSION
1999                    ),
2000                },
2001            ));
2002        }
2003        if daemon_version != PROTOCOL_VERSION {
2004            return Err(std::io::Error::new(
2005                std::io::ErrorKind::InvalidData,
2006                format!(
2007                    "protocol version mismatch: client={}, daemon={}",
2008                    PROTOCOL_VERSION, daemon_version
2009                ),
2010            ));
2011        }
2012
2013        Ok(client)
2014    }
2015
2016    /// Send a request and receive the response.
2017    pub fn request(&mut self, cmd_id: u8, payload: &[u8]) -> std::io::Result<Vec<u8>> {
2018        write_message(&mut self.stream, cmd_id, payload)?;
2019        let (resp_cmd, resp_data) = read_message(&mut self.stream)?;
2020        // Verify the response command matches the request. If the
2021        // server ever sends an unsolicited notification, the client
2022        // would otherwise misinterpret it as a response.
2023        if resp_cmd != cmd_id {
2024            return Err(std::io::Error::new(
2025                std::io::ErrorKind::InvalidData,
2026                format!("response command {resp_cmd} does not match request command {cmd_id}"),
2027            ));
2028        }
2029        Ok(resp_data)
2030    }

2032    /// Ping the daemon.
2033    pub fn ping(&mut self) -> std::io::Result<bool> {
2034        let resp = self.request(cmd::PING, &[])?;
2035        Ok(!resp.is_empty() && resp[0] == 1)
2036    }

2038    /// Get the full core state (3480 bytes, schema v5).
2039    pub fn get_state(&mut self) -> std::io::Result<GenesisCoreState> {
2040        let resp = self.request(cmd::GET_STATE, &[])?;
2041        if resp.len() < core::mem::size_of::<GenesisCoreState>() {
2042            return Err(std::io::Error::new(
2043                std::io::ErrorKind::InvalidData,
2044                "state response too short",
2045            ));
2046        }
2047        // Copy the bytes into a GenesisCoreState
2048        let mut state = GenesisCoreState::new(0, 0);
