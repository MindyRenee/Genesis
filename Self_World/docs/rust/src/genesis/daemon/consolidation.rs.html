1//! Consolidation engine — promotes short-term memories to long-term
2//! storage based on emotional gating.
3//!
4//! # How human memory consolidation works
5//!
6//! In the human brain, the hippocampus replays recent experiences during
7//! rest and sleep, transferring important ones to the neocortex for
8//! long-term storage. This isn't a bulk copy — it's **selective**:
9//! emotionally significant experiences (high cortisol = threatening,
10//! high dopamine = rewarding) are prioritized. The amygdala tags
11//! memories with emotional significance, and the hippocampus uses those
12//! tags to decide what to consolidate.
13//!
14//! # How Genesis's consolidation works
15//!
16//! Genesis mirrors this architecture:
17//!
18//! 1. **Scan** the STM ring buffer for unconsolidated entries
19//! 2. **Score** each entry using the current emotional gating weights
20//!    from the core state's memory pointers
21//! 3. **Promote** high-scoring entries to LTM
22//! 4. **Mark** consolidated entries in STM (so they're not re-consolidated)
23//! 5. **Update** the core state's memory pointers with new counts
24//!
25//! The emotional gating weights (`encoding_weight`, `consolidation_weight`,
26//! `plasticity_gate`) come from the neurochemical system and control
27//! which memories survive. This is why chronic stress impairs memory
28//! formation — when the plasticity gate is near zero, nothing gets
29//! consolidated, no matter how important it is.
30//!
31//! # What's novel
32//!
33//! Standard AI memory systems either (a) keep everything forever (no
34//! forgetting, no selectivity) or (b) use a fixed retention policy
35//! (FIFO, LRU). Genesis uses **emotionally-modulated consolidation**:
36//! the decision to keep or forget a memory depends on the system's
37//! neurochemical state at consolidation time, not just the memory's
38//! recency or frequency.
39

44/// The result of a consolidation pass.
45#[derive(Clone, Copy, Debug, Default)]
46pub struct ConsolidationResult {
47    /// Number of STM entries scanned.
48    pub scanned: u32,
49    /// Number of entries promoted to LTM.
50    pub promoted: u32,
51    /// Number of entries skipped (already consolidated or below threshold).
52    pub skipped: u32,
53    /// Number of entries that couldn't be consolidated (plasticity gate closed).
54    pub blocked: u32,
55    /// The new LTM episode count after consolidation.
56    pub ltm_count: u32,
57}

59/// The consolidation engine.
60///
61/// This is stateless — it operates on the STM ring buffer and LTM store
62/// passed to it, reading emotional gating weights from the core state.
63pub struct ConsolidationEngine;
64
65impl ConsolidationEngine {
66    /// Run a consolidation pass.
67    ///
68    /// Scans the STM ring buffer for entries that haven't been
69    /// consolidated yet, scores them using the current emotional
70    /// gating weights, and promotes high-scoring entries to LTM.
71    ///
72    /// # Parameters
73    /// - `state`: the core state (read for gating weights, written
74    ///   with updated LTM count)
75    /// - `stm`: the short-term memory ring buffer
76    /// - `ltm`: the long-term memory store
77    /// - `now_ms`: current timestamp
78    ///
79    /// # Consolidation score
80    ///
81    /// Each STM entry has a `salience` score (set at insertion time)
82    /// and an `emotional_tag` (neurochemical snapshot). The
83    /// consolidation score combines this with the *current* gating
84    /// weights:
85    ///
86    /// ```text
87    /// score = entry.salience * state.memory.consolidation_weight
88    /// ```
89    ///
90    /// If `state.memory.plasticity_gate < 0.1`, consolidation is
91    /// blocked entirely (chronic stress state — no new long-term
92    /// memories can form).
93    ///
94    /// If `score < threshold`, the entry is skipped (it will decay
95    /// in the ring buffer as new entries overwrite it).
96    pub fn consolidate(
97        state: &mut GenesisCoreState,
98        stm: &RingBuffer,
99        ltm: &mut LtmStore,
100        now_ms: u64,
101        threshold: f32,
102    ) -> ConsolidationResult {
103        let mut result = ConsolidationResult::default();
104
105        // Check if plasticity gate is closed (chronic stress)
106        if !state.memory.can_form_new_memories() {
107            // Count entries: already-consolidated ones are skipped,
108            // unconsolidated ones are blocked by the closed gate.
109            for entry in stm.iter() {
110                result.scanned += 1;
111                if entry.is_consolidated() {
112                    result.skipped += 1;
113                } else {
114                    result.blocked += 1;
115                }
116            }
117            return result;
118        }
119
120        let consolidation_weight = state.memory.consolidation_weight;
121
122        // Atomically snapshot all entries with their slot indices.
123        // `iter_with_slots()` holds `access_lock` for the entire scan,
124        // preventing a concurrent IPC `push()` from changing `head`/
125        // `tail` and overwriting slots mid-iteration. The snapshot is
126        // owned, so the lock is released before we store to LTM.
127        let entries = stm.iter_with_slots();
128
129        for (slot, entry) in entries {
130            result.scanned += 1;
131
132            // Skip entries that have already been consolidated
133            if entry.is_consolidated() {
134                result.skipped += 1;
135                continue;
136            }
137
138            // Compute consolidation score
139            let score = entry.salience * consolidation_weight;
140
141            if score < threshold {
142                result.skipped += 1;
143                continue;
144            }
145
146            // Build the full emotional tag from the STM entry's tag
147            // (which stores 12 effective levels) and the compact tag
148            // [arousal, valence, cortisol, dopamine] from it. Both the
149            // consolidation path and the direct STORE_EPISODE IPC path
150            // use the same helpers so LTM payloads are identical.
151            let full_tag = expand_emotional_tag(&entry.emotional_tag);
152            let compact_tag = compute_compact_tag(&full_tag);
153
154            // Store in LTM
155            let text = entry.text();
156
157            // An episode with no text is noise, not a memory — its
158            // SimHash degenerates to 0 and it "associates" with every
159            // other empty episode at distance 0. Mark it consolidated
160            // so the empty slot isn't retried every pass until the
161            // ring overwrites it.
162            if text.trim().is_empty() {
163                stm.mark_consolidated_if_match(slot, entry.timestamp);
164                result.skipped += 1;
165                continue;
166            }
167
168            let timestamp = entry.timestamp;
169            let salience = entry.salience;
170            let event_type = entry.event_type;
171            let source_module = entry.source_module;
172
173            match ltm.store(
174                timestamp,
175                salience,
176                full_tag,
177                compact_tag,
178                event_type,
179                source_module,
180                text,
181            ) {
182                Ok(_episode_id) => {
183                    // Mark the STM entry as consolidated so it's not
184                    // re-consolidated on the next tick. Use the
185                    // timestamp-matched variant: between the
186                    // `iter_with_slots()` snapshot (which releases the
187                    // ring-buffer lock) and this mark, a concurrent IPC
188                    // `push()` can wrap the ring buffer and overwrite the
189                    // slot. If we marked the slot unconditionally, we
190                    // would mark the *new* (unconsolidated) entry as
191                    // consolidated, permanently preventing it from being
192                    // promoted. The timestamp match ensures we only mark
193                    // the entry we actually promoted to LTM.
194                    stm.mark_consolidated_if_match(slot, entry.timestamp);
195                    result.promoted += 1;
196                }
197                Err(e) => {
198                    eprintln!("[consolidation] failed to store episode: {e}");
199                    result.skipped += 1;
200                }
201            }
202        }
203
204        // Update the core state's memory pointers
205        let new_ltm_count = ltm.count();
206        state
207            .memory
208            .mark_consolidation(now_ms, new_ltm_count as u64);
209
210        result.ltm_count = new_ltm_count;
211        result
212    }
213}

215/// Compute a compact 4-element emotional tag from the full 12-element tag.
216///
217/// [arousal, valence, cortisol, dopamine] — the four dimensions that
218/// most influence memory retrieval and behaviour.
219///
220/// The arousal and valence formulas here **must match** the corrected
221/// formulas in `NeurochemicalVector::recompute_derived` (see
222/// `state/neurochemical.rs`). They were previously out of sync — the
223/// compact tag used the deprecated equations that the neurobiology
224/// audit corrected (see AGENTS.md). This caused every consolidated
225/// LTM episode to carry inaccurate arousal/valence metadata.
226///
227/// The emotional tag only stores 12 chemicals (the original v2 set).
228/// The 6 chemicals added in v3 (endocannabinoid, vasopressin, CRH,
229/// orexin, epinephrine, melatonin) are not in the tag, so we use
230/// their genetic default values. This means the compact tag's
231/// arousal/valence are approximate for episodes encoded with non-
232/// default levels of the 6 new chemicals. The approximation error
233/// is small for typical operation (the 6 new chemicals are usually
234/// near their defaults) but could be significant during extreme
235/// states (e.g., very high orexin during prolonged wakefulness).
236///
237/// The bistable flip-flop hysteresis in `recompute_derived` requires
238/// the previous arousal value, which is not available from a static
239/// emotional tag. We use the non-hysteretic formula (no ±0.10 bias),
240/// which gives the "neutral" arousal — the value the system would
241/// converge to if it had no memory of its previous state.
242/// Expand a 12-element emotional tag (the STM/IPC wire format) into
243/// the full `LTM_EMOTIONAL_TAG_SIZE`-element tag used by `LtmStore::store`.
244///
245/// The STM ring buffer and the IPC wire format carry 12 effective levels
246/// (the original v2 chemical set). `LtmStore` stores
247/// `LTM_EMOTIONAL_TAG_SIZE` elements (currently also 12, but the store
248/// is the authority on the on-disk width). This helper centralises the
249/// expansion so that both the consolidation engine and the direct
250/// `STORE_EPISODE` IPC path produce identical LTM payloads.
251pub fn expand_emotional_tag(tag: &[f32; 12]) -> [f32; LTM_TAG_SIZE] {
252    let mut full_tag = [0.0f32; LTM_TAG_SIZE];
253    let tag_len = tag.len().min(LTM_TAG_SIZE);
254    full_tag[..tag_len].copy_from_slice(&tag[..tag_len]);
255    full_tag
256}

258/// Compute the 4-element compact tag `[arousal, valence, cortisol, dopamine]`
259/// from a full emotional tag.
260///
261/// This is a thin wrapper around `compute_compact_tag` so that callers
262/// outside this module do not need to know the full-tag width.
263pub fn compact_tag_from_12(tag: &[f32; 12]) -> [f32; 4] {
264    compute_compact_tag(&expand_emotional_tag(tag))
265}
266
267fn compute_compact_tag(full_tag: &[f32; LTM_TAG_SIZE]) -> [f32; 4] {
268    // Indices match NeurochemicalId order:
269    // 0=Dopamine, 1=Serotonin, 2=NE, 3=ACh, 4=GABA, 5=Glutamate,
270    // 6=Cortisol, 7=Oxytocin, 8=Endorphin, 9=Histamine, 10=Adenosine, 11=BDNF
271    let da = full_tag[0];
272    let srt = full_tag[1];
273    let ne = full_tag[2];
274    let ach = full_tag[3];
275    let gaba = full_tag[4];
276    let glu = full_tag[5];
277    let cort = full_tag[6];
278    let oxy = full_tag[7];
279    let end = full_tag[8];
280    let hist = full_tag[9];
281    let adn = full_tag[10];
282
283    // The 6 chemicals added in v3 are not in the 12-chemical tag.
284    // Use their genetic defaults as approximations.
285    let orexin = NeurochemicalId::Orexin.default_baseline();
286    let epi = NeurochemicalId::Epinephrine.default_baseline();
287    let mel = NeurochemicalId::Melatonin.default_baseline();
288
289    // Arousal: matches recompute_derived() exactly, minus the Wilson-Cowan
290    // self-excitation (which requires the previous arousal state, unavailable
291    // from a static emotional tag). Without self-excitation, the sigmoid
292    // gives the "neutral" arousal — the value the system would converge to
293    // if it had no memory of its previous state.
294    //
295    // Arousal promoters: NE 0.20, HIST 0.15, DA 0.15, ACh 0.15,
296    //   orexin 0.20, epinephrine 0.10
297    // Sleep promoters: GABA 0.25, adenosine 0.65, melatonin 0.35
298    let arousal_promoters =
299        ne * 0.20 + hist * 0.15 + da * 0.15 + ach * 0.15 + orexin * 0.20 + epi * 0.10;
300    let sleep_promoters = gaba * 0.25 + adn * 0.65 + mel * 0.35;
301    let net_drive = arousal_promoters - sleep_promoters;
302    // Must match recompute_derived(): arousal uses AROUSAL_GAIN (3.0).
303    // WC_GAIN (2.0) is the consolidation net-drive gain — a different
304    // decision with its own thresholds. Using it here understated
305    // arousal stored in the LTM compact tag relative to the live value.
306    let total_input = net_drive * crate::state::neurochemical::AROUSAL_GAIN; // no self-excitation
307    let arousal = if total_input > 20.0 {
308        1.0
309    } else if total_input < -20.0 {
310        0.0
311    } else {
312        1.0 / (1.0 + (-total_input).exp())
313    };
314    let arousal = crate::state::sanitize::finite_or(arousal, 0.0);
315
316    // Valence: matches recompute_derived() exactly.
317    //
318    // Positive: DA 0.25, SRT 0.25, OXY 0.10, END 0.15, NE 0.10
319    // Negative: CORT -0.35, glutamate excess -0.15
320    // Synergy: DA × SRT × 0.10
321    let glu_excess = (glu - 0.70).max(0.0);
322    let da_srt_synergy = da * srt * 0.10;
323    let valence = da * 0.25 + srt * 0.25 + oxy * 0.10 + end * 0.15 + ne * 0.10
324        - crate::state::neurochemical::cortisol_stress_fraction(cort) * 0.35
325        - glu_excess * 0.15
326        + da_srt_synergy;
327
328    [
329        arousal,
330        crate::state::sanitize::finite_or(valence, 0.0),
331        cort,
332        da,
333    ]
334}

341    /// Verify that compute_compact_tag produces arousal/valence that
342    /// match NeurochemicalVector::recompute_derived's output, when given
343    /// the same chemical levels.
344    ///
345    /// This is the critical sync test: if the formulas drift apart,
346    /// every consolidated LTM episode will carry inaccurate emotional
347    /// metadata. The test creates a NeurochemicalVector with the same
348    /// 12 chemical levels as the tag, calls recompute_derived(), and
349    /// compares the results.
350    ///
351    /// Tolerance for arousal is 0.06: the bistable flip-flop hysteresis
352    /// in recompute_derived adds ±0.10 to net_drive (±0.05 to arousal),
353    /// which compute_compact_tag omits (it has no previous state).
354    /// Valence has no hysteresis, so tolerance is tight (1e-5).
355    #[test]
356    fn test_compact_tag_matches_recompute_derived() {
357        // Set up 12 chemical levels (the emotional tag subset)
358        let levels: [(NeurochemicalId, f32); 12] = [
359            (NeurochemicalId::Dopamine, 0.80),
360            (NeurochemicalId::Serotonin, 0.70),
361            (NeurochemicalId::Norepinephrine, 0.60),
362            (NeurochemicalId::Acetylcholine, 0.50),
363            (NeurochemicalId::GABA, 0.20),
364            (NeurochemicalId::Glutamate, 0.55),
365            (NeurochemicalId::Cortisol, 0.30),
366            (NeurochemicalId::Oxytocin, 0.40),
367            (NeurochemicalId::Endorphin, 0.35),
368            (NeurochemicalId::Histamine, 0.50),
369            (NeurochemicalId::Adenosine, 0.10),
370            (NeurochemicalId::BDNF, 0.45),
371        ];
372
373        // Build the emotional tag (12-element array, NeurochemicalId order)
374        let mut tag = [0.0f32; 12];
375        for (id, level) in &levels {
376            tag[*id as usize] = *level;
377        }
378
379        // Build a NeurochemicalVector with the same levels.
380        // The 6 new chemicals (indices 12-17) stay at their defaults.
381        let mut vec = NeurochemicalVector::new(0);
382        for (id, level) in &levels {
383            if let Some(c) = vec.get_mut(*id) {
384                c.level = *level;
385            }
386        }
387
388        // Compute derived fields (arousal, valence) from the live system
389        vec.recompute_derived();
390
391        // Compute compact tag from the same levels
392        let compact = compute_compact_tag(&tag);
393
394        // Arousal: with bistability, the live arousal depends on its
395        // previous state; the compact tag has no previous state and
396        // reports the non-hysteretic (unstable-branch) value. Tolerance
397        // must cover the full hysteresis gap (~0.4 for WC_SELF=2.05).
398        let arousal_diff = (compact[0] - vec.arousal).abs();
399        assert!(
400            arousal_diff < 0.5,
401            "compact arousal {} should match recompute_derived arousal {} (diff {}, hysteresis tolerance 0.5)",
402            compact[0],
403            vec.arousal,
404            arousal_diff
405        );
406
407        // Valence: exact match (no hysteresis in valence)
408        let valence_diff = (compact[1] - vec.valence).abs();
409        assert!(
410            valence_diff < 1e-5,
411            "compact valence {} should match recompute_derived valence {} (diff {})",
412            compact[1],
413            vec.valence,
414            valence_diff
415        );
416
417        // Cortisol and DA are passed through directly
418        assert!((compact[2] - 0.30).abs() < 1e-5, "cortisol passed through");
419        assert!((compact[3] - 0.80).abs() < 1e-5, "dopamine passed through");
420    }

423    /// result — this guards against accidental reversion.
424    #[test]
425    fn test_compact_tag_differs_from_deprecated_formula() {
426        let mut tag = [0.0f32; 12];
427        tag[0] = 1.0; // DA
428        tag[1] = 1.0; // SRT
429        tag[2] = 1.0; // NE
430        tag[3] = 1.0; // ACh  (was ignored by old arousal formula)
431        tag[5] = 0.0; // Glu  (was ignored by old valence formula)
432        tag[6] = 0.0; // Cort
433        tag[7] = 0.0; // Oxy  (was ignored by old valence formula)
434        tag[8] = 0.0; // Endorphin (was ignored by old valence formula)
435        tag[9] = 1.0; // Hist
436
437        let compact = compute_compact_tag(&tag);
438
439        // Old valence = (DA + SRT) * 0.5 - Cort = 1.0
440        let old_valence = (1.0 + 1.0) * 0.5 - 0.0;
441        let new_valence = compact[1];
442        assert!(
443            (new_valence - old_valence).abs() > 0.01,
444            "corrected valence {} should differ from deprecated {}",
445            new_valence,
446            old_valence
447        );
448
449        // Old arousal = ((NE + Hist)*0.5 - (GABA+Adn)*0.5 + 1.0) * 0.5

451        // epinephrine (0.15 default × 0.10 = 0.015), ACh (0.5 × 0.15 = 0.075),
452        // and DA (0.8 × 0.15 = 0.12) as arousal promoters, plus different

454        // Gain regression: the arousal sigmoid must use AROUSAL_GAIN

464            + tag[0] * 0.15
465            + tag[3] * 0.15
466            + NeurochemicalId::Orexin.default_baseline() * 0.20
467            + NeurochemicalId::Epinephrine.default_baseline() * 0.10;
468        let sleep_promoters =
469            tag[4] * 0.25 + tag[10] * 0.65 + NeurochemicalId::Melatonin.default_baseline() * 0.35;
470        let net_drive = arousal_promoters - sleep_promoters;
471        let with_arousal_gain =

473        let with_wc_gain = 1.0 / (1.0 + (-net_drive * crate::state::neurochemical::WC_GAIN).exp());
474        assert!(
475            (compact[0] - with_arousal_gain).abs() < 1e-5,
476            "compact arousal {} should match the AROUSAL_GAIN scaling {}",
477            compact[0],
478            with_arousal_gain
479        );
480        assert!(
481            (compact[0] - with_wc_gain).abs() > 0.01,

488    /// Verify that compute_compact_tag's arousal matches the non-hysteretic
489    /// branch of recompute_derived when GABA is high. This catches drift in
490    /// the sleep-promoter weights (a previous bug had GABA at 0.30 in
491    /// compute_compact_tag but 0.25 in recompute_derived, masked by the
492    /// loose hysteresis tolerance in test_compact_tag_matches_recompute_derived
493    /// which used a low GABA value).
494    #[test]
495    fn test_compact_tag_gaba_weight_matches_recompute_derived() {
496        let levels: [(NeurochemicalId, f32); 12] = [
497            (NeurochemicalId::Dopamine, 0.50),
498            (NeurochemicalId::Serotonin, 0.50),
499            (NeurochemicalId::Norepinephrine, 0.50),
500            (NeurochemicalId::Acetylcholine, 0.50),
501            (NeurochemicalId::GABA, 0.80), // high GABA — the discriminating case
502            (NeurochemicalId::Glutamate, 0.50),
503            (NeurochemicalId::Cortisol, 0.20),
504            (NeurochemicalId::Oxytocin, 0.40),
505            (NeurochemicalId::Endorphin, 0.35),
506            (NeurochemicalId::Histamine, 0.50),
507            (NeurochemicalId::Adenosine, 0.10),
508            (NeurochemicalId::BDNF, 0.45),
509        ];
510
511        let mut tag = [0.0f32; 12];
512        for (id, level) in &levels {
513            tag[*id as usize] = *level;
514        }
515
516        let mut vec = NeurochemicalVector::new(0);
517        for (id, level) in &levels {
518            if let Some(c) = vec.get_mut(*id) {
519                c.level = *level;
520            }
521        }
522        vec.recompute_derived();
523
524        let compact = compute_compact_tag(&tag);
525
526        // The compact tag has no self-excitation, so it reports the
527        // unstable-branch (neutral) arousal. The live system with its
528        // initial arousal=0.5 starts near the unstable branch and
529        // converges to one of the two stable fixed points. With high
530        // GABA, the live system should converge to low arousal (asleep),
531        // while the compact tag reports the neutral value. The
532        // difference should be the hysteresis gap (~0.4), NOT a weight
533        // drift. We verify the compact tag's arousal is the neutral
534        // (non-hysteretic) value by computing it directly.
535        let gaba = 0.80f32;
536        let adn = 0.10f32;
537        let mel = NeurochemicalId::Melatonin.default_baseline();
538        let sleep_promoters = gaba * 0.25 + adn * 0.65 + mel * 0.35;
539        let ne = 0.50f32;
540        let hist = 0.50f32;
541        let da = 0.50f32;
542        let ach = 0.50f32;
543        let orexin = NeurochemicalId::Orexin.default_baseline();
544        let epi = NeurochemicalId::Epinephrine.default_baseline();
545        let arousal_promoters =
546            ne * 0.20 + hist * 0.15 + da * 0.15 + ach * 0.15 + orexin * 0.20 + epi * 0.10;
547        let net_drive = arousal_promoters - sleep_promoters;
548        let total_input = net_drive * crate::state::neurochemical::AROUSAL_GAIN;
549        let expected_arousal = if total_input > 20.0 {
550            1.0
551        } else if total_input < -20.0 {
552            0.0
553        } else {
554            1.0 / (1.0 + (-total_input).exp())
555        };
556
557        assert!(
558            (compact[0] - expected_arousal).abs() < 1e-5,
559            "compact arousal {} should match direct computation {} (GABA weight 0.25)",
560            compact[0],
561            expected_arousal
562        );
563
564        // The old buggy weight (0.30) would give a different value:
565        let buggy_sleep = gaba * 0.30 + adn * 0.65 + mel * 0.35;
566        let buggy_drive = arousal_promoters - buggy_sleep;
567        let buggy_input = buggy_drive * crate::state::neurochemical::AROUSAL_GAIN;
568        let buggy_arousal = 1.0 / (1.0 + (-buggy_input).exp());
