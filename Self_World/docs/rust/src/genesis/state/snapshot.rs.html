1//! Core-state snapshot — the mind's wire contract for `GET_STATE_SNAPSHOT`.
2//!
3//! ## Why this type exists
4//!
5//! `GET_STATE` returns the raw [`GenesisCoreState`] — the brain's
6//! authoritative memory image, schema-versioned by `header.version`.
7//! That is the right payload for body inspection (the Rust test
8//! harness and forensics read it that way), but it is the wrong
9//! payload for the cognitive mind: every consumer then has to mirror
10//! the brain's byte layout in its own language, and the mirror has no
11//! tripwire. A schema bump does not fail — it returns plausible
12//! numbers read from the wrong fields. That has happened twice
13//! (v3→v4 and v4→v5 offset bugs) and nearly a third time (the
14//! transient v6 layout).
15//!
16//! This struct is the anti-corruption layer between the two: a
17//! fixed-layout DTO that carries exactly what the mind consumes,
18//! mapped field-by-field in [`From<&GenesisCoreState>`]. The brain's
19//! schema can evolve — fields move, blocks insert, versions bump —
20//! and the wire contract only changes when this file intentionally
21//! changes, under its own [`SNAPSHOT_VERSION`]. A renamed or
22//! retyped source field fails to compile here; it cannot silently
23//! re-encode on the wire.
24//!
25//! The payload is self-describing: `magic`, `version`, and
26//! `state_size` lead the struct so a mismatched or foreign buffer
27//! is rejected before a single field offset is trusted.
28//!
29//! ## Layout (232 bytes)
30//!
31//! ```text
32//! offset  field                     type
33//! ------  -----                     ----
34//!    0    magic                     [u8;4]   b"GSS1"
35//!    4    version                   u32      SNAPSHOT_VERSION
36//!    8    state_size                u32      size_of::<Self>()
37//!   12    _pad                      u32      alignment
38//!   16    created_at                u64      ms epoch
39//!   24    last_updated              u64      ms epoch
40//!   32    heartbeat                 u64      tick counter
41//!   40    instance_id               u64
42//!   48    ltm_episode_count         u64
43//!   56    effective_levels          [f32;18] level × sensitivity ×
44//!                                          desensitization ×
45//!                                          internalization, per
46//!                                          NeurochemicalId order
47//!  128    global_tone               f32
48//!  132    arousal                   f32
49//!  136    valence                   f32
50//!  140    plasticity_gate           f32      neurochemical gate
51//!  144    surprise_ema              f32      ┐
52//!  148    free_energy               f32      │
53//!  152    expected_free_energy      f32      │
54//!  156    allostasis_load           f32      │
55//!  160    precision                 f32      │  InferenceSignals,
56//!  164    attunement                f32      │  flattened so the
57//!  168    dyadic_synchrony          f32      │  brain block can
58//!  172    user_valence              f32      │  evolve without
59//!  176    user_arousal              f32      │  moving the wire
60//!  180    user_engagement           f32      │
61//!  184    prediction_error_dopamine f32      │
62//!  188    prediction_error_cortisol f32      │
63//!  192    prediction_error_serotonin f32     │
64//!  196    model_maturity            f32      │
65//!  200    policy_authority          f32      ┘
66//!  204    inference_tick_count      u32
67//!  208    encoding_weight           f32      ┐
68//!  212    consolidation_weight      f32      │  MemoryPointers
69//!  216    retrieval_weight          f32      ┘  gating weights
70//!  220    sleep_pressure            f32      h / H_MAX ∈ [0,1]
71//!  224    stm_count                 u32
72//!  228    cognitive_zone            u8       CognitiveZone
73//!  229    emergent_phase            u8       MentalPhase
74//!  230    _pad_tail                 [u8;2]
75//!  232    TOTAL
76//! ```
77//!
78//! Deliberately absent: the coupling matrix, per-chemical dynamics
79//! (level/baseline/velocity/tonic/phasic/vesicular), the manifest,
80//! ions, zones' task bookkeeping, and every other field the mind does
81//! not read. The snapshot is a projection, not a copy — fields are
82//! added here when a consumer needs them, never because the brain
83//! gained them.

89/// Magic bytes identifying a [`CoreStateSnapshot`] payload. ASCII
90/// `"GSS1"` — distinct from the state file's `"GNSX"` so a raw
91/// `GenesisCoreState` can never be mistaken for a snapshot.
92pub const SNAPSHOT_MAGIC: [u8; 4] = *b"GSS1";
93
94/// Snapshot schema version. Bump when this layout changes in any way
95/// that breaks binary compatibility — independently of the brain's
96/// `SCHEMA_VERSION`, which this type exists to decouple from.
97pub const SNAPSHOT_VERSION: u32 = 1;

99/// The mind's view of the core state — see the module docs for why
100/// this is a separate type from [`GenesisCoreState`].
101#[repr(C)]
102#[derive(Clone, Copy, Debug, Default)]
103pub struct CoreStateSnapshot {
104    /// `SNAPSHOT_MAGIC` — payload identity.
105    pub magic: [u8; 4],
106    /// [`SNAPSHOT_VERSION`].
107    pub version: u32,
108    /// `size_of::<CoreStateSnapshot>()` as sent.
109    pub state_size: u32,
110    /// Alignment padding before the u64 block.
111    pub _pad: u32,
112    /// Milliseconds-timestamp when the state was first created.
113    pub created_at: u64,
114    /// Milliseconds-timestamp of the last write.
115    pub last_updated: u64,
116    /// Monotonic heartbeat counter (incremented each tick).
117    pub heartbeat: u64,
118    /// State instance identifier (creation timestamp in practice).
119    pub instance_id: u64,
120    /// Long-term memory episode count.
121    pub ltm_episode_count: u64,
122    /// Effective neurochemical levels, indexed by `NeurochemicalId`
123    /// discriminant — what the mind actually feels, not raw levels.
124    pub effective_levels: [f32; NEUROCHEMICAL_COUNT],
125    /// Affective tone — weighted mean of effective levels.
126    pub global_tone: f32,
127    /// Arousal axis [0,1].
128    pub arousal: f32,
129    /// Valence axis [-1,+1].
130    pub valence: f32,
131    /// Plasticity gate [0,1] — neurochemical memory-forming capacity.
132    pub plasticity_gate: f32,
133    /// ─── Active-inference projection ────────────────────────────
134    /// Exponential moving average of total surprise. [0,1]
135    pub surprise_ema: f32,
136    /// Current estimated free energy. [0,1]
137    pub free_energy: f32,
138    /// Expected future free energy — anticipatory demand. [0,1]
139    pub expected_free_energy: f32,
140    /// Accumulated allostatic load — persistent regulatory burden.
141    /// [0,1]
142    pub allostasis_load: f32,
143    /// Precision of the generative model. [0,1]
144    pub precision: f32,
145    /// Oxytocin-mediated attunement to the user's affect. [0,1]
146    pub attunement: f32,
147    /// Rolling valence correlation with the user. [-1,1]
148    pub dyadic_synchrony: f32,
149    /// Inferred user valence. [-1,1]
150    pub user_valence: f32,
151    /// Inferred user arousal. [0,1]
152    pub user_arousal: f32,
153    /// Inferred user engagement. [0,1]
154    pub user_engagement: f32,
155    /// Dopamine prediction error. Finite and unbounded.
156    pub prediction_error_dopamine: f32,
157    /// Cortisol prediction error. [-2,2]
158    pub prediction_error_cortisol: f32,
159    /// Serotonin prediction error. [-2,2]
160    pub prediction_error_serotonin: f32,
161    /// Evidence the generative model is predictive and constrained.
162    /// [0,1]
163    pub model_maturity: f32,
164    /// Delegated policy authority. [0,1]
165    pub policy_authority: f32,
166    /// Inference cycles run.
167    pub inference_tick_count: u32,
168    /// ─── Memory gating ──────────────────────────────────────────
169    /// Encoding weight — current ability to form new memories.
170    pub encoding_weight: f32,
171    /// Consolidation weight — current drive to consolidate STM→LTM.
172    pub consolidation_weight: f32,
173    /// Retrieval weight — current ability to recall.
174    pub retrieval_weight: f32,
175    /// Homeostatic sleep pressure `h` as a fraction of its ceiling —
176    /// a discharge gauge: it decays as sleep clears the need.
177    pub sleep_pressure: f32,
178    /// Short-term memory event count.
179    pub stm_count: u32,
180    /// Cognitive task zone — `CognitiveZone` discriminant.
181    pub cognitive_zone: u8,
182    /// Emergent mental phase — `MentalPhase` discriminant.
183    pub emergent_phase: u8,
184    /// Tail padding to the 8-byte boundary.
185    pub _pad_tail: [u8; 2],
186}

188impl CoreStateSnapshot {
189    /// Verify the self-describing header of a received payload.
190    ///
191    /// Call after copying the wire bytes into a snapshot — a mismatch
192    /// means the payload is not a [`SNAPSHOT_VERSION`] snapshot and no
193    /// field offset may be trusted.
194    pub fn verify(&self) -> Result<(), &'static str> {
195        if self.magic != SNAPSHOT_MAGIC {
196            return Err("bad snapshot magic");
197        }
198        if self.version != SNAPSHOT_VERSION {
199            return Err("snapshot version mismatch");
200        }
201        if self.state_size as usize != core::mem::size_of::<Self>() {
202            return Err("snapshot size mismatch");
203        }
204        Ok(())
205    }
206}

208impl From<&GenesisCoreState> for CoreStateSnapshot {
209    /// Project the authoritative state into the mind's wire contract.
210    /// This is the only place that knows both layouts — every field
211    /// is named on both sides, so a brain-schema change either
212    /// compiles (the projection still resolves) or fails loudly here.
213    fn from(state: &GenesisCoreState) -> Self {
214        let neuro = &state.neurochemicals;
215        let signals = &state.inference_signals;
216        let memory = &state.memory;
217        Self {
218            magic: SNAPSHOT_MAGIC,
219            version: SNAPSHOT_VERSION,
220            state_size: core::mem::size_of::<Self>() as u32,
221            _pad: 0,
222            created_at: state.header.created_at,
223            last_updated: state.header.last_updated,
224            heartbeat: state.header.heartbeat,
225            instance_id: state.header.instance_id,
226            ltm_episode_count: memory.ltm_episode_count,
227            effective_levels: neuro.effective_levels,
228            global_tone: neuro.global_tone,
229            arousal: neuro.arousal,
230            valence: neuro.valence,
231            plasticity_gate: neuro.plasticity_gate,
232            surprise_ema: signals.surprise_ema,
233            free_energy: signals.free_energy,
234            expected_free_energy: signals.expected_free_energy,
235            allostasis_load: signals.allostasis_load,
236            precision: signals.precision,
237            attunement: signals.attunement,
238            dyadic_synchrony: signals.dyadic_synchrony,
239            user_valence: signals.user_valence,
240            user_arousal: signals.user_arousal,
241            user_engagement: signals.user_engagement,
242            prediction_error_dopamine: signals.prediction_error_dopamine,
243            prediction_error_cortisol: signals.prediction_error_cortisol,
244            prediction_error_serotonin: signals.prediction_error_serotonin,
245            model_maturity: signals.model_maturity,
246            policy_authority: signals.policy_authority,
247            inference_tick_count: signals.inference_tick_count,
248            encoding_weight: memory.encoding_weight,
249            consolidation_weight: memory.consolidation_weight,
250            retrieval_weight: memory.retrieval_weight,
251            sleep_pressure: (state.sleep_network[4] / sleep_network::H_MAX) as f32,
252            stm_count: memory.stm_count,
253            cognitive_zone: state.zones.cognitive_zone,
254            emergent_phase: state.zones.emergent_phase,
255            _pad_tail: [0; 2],
256        }
257    }
258}
