1//! Sleep-wake regulatory network: wake, NREM, and REM populations with
2//! a homeostatic sleep drive and a circadian oscillator.
3//!
4//! Transcribed from the MATLAB source published with ModelDB entry 244485
5//! (`DinizBehnBooth_SWnetworkModel.m`, `run_DinizBehnBooth_SWnetworkModel.m`).
6//! That implements Gleit, Diniz Behn & Booth, J. Biol. Rhythms 28:339–355
7//! (2013), as revised in Booth, Xique & Diniz Behn, SIAM J. Appl. Dyn. Sys.
8//! 16(2):1089–1112 (2017).
9//!
10//! Populations: `f_lc` (wake, locus coeruleus), `f_vlpo` (NREM / sleep,
11//! ventrolateral preoptic area), `f_r` (REM-on, sublaterodorsal/precoeruleus),
12//! `f_scn` (circadian pacemaker).
13//!
14//! Sleep onset is `f_lc` falling below [`SLEEP_THRESHOLD`]; REM onset is
15//! `f_r` rising above [`REM_THRESHOLD`]. Both are the source's own event
16//! definitions. Time is in minutes.
17
18use std::f64::consts::PI;

20// Synaptic gains between the neurotransmitter pools and the populations.
21const GACH_LC: f64 = 0.8;
22const GGABA_LC: f64 = 1.5;
23const GSCN_LC: f64 = 0.5;
24// The stress axis's own alerting channel onto LC. CRH and elevated
25// glucocorticoids recruit the noradrenergic arousal system even with no
26// sensory event — CRH activates the LC even in the absence of stress,
27// and nocturnal cortisol elevation tracks sleep fragmentation (Buckley
28// & Schatzberg, JCEM 2005; Vgontzas et al.). It is also the waking side
29// of the diurnal rhythm: the pre-awakening cortisol surge accompanies
30// the sleep→wake transition. The input is the cortisol stress fraction
31// (0 at the basal floor, 1 at the reachable peak), so the term is silent
32// at the nadir and ramps through the late-sleep surge. Half a volitional
33// wake push: enough to lighten and fragment sleep under stress, not
34// enough to drag a fresh VLPO down on its own.
35const GCORT_LC: f64 = 0.5;
36const GNE_VLPO: f64 = 1.5;
37const GSCN_VLPO: f64 = 0.1;
38const GACH_R: f64 = 2.2;
39const GNE_R: f64 = 8.0;
40const GGABA_R: f64 = 1.0;
41const GSCN_R: f64 = 0.8;
42
43// Population time constants (min) and saturating firing rates (Hz).
44const TAU_LC: f64 = 23.0;
45const TAU_R: f64 = 1.0;
46const TAU_VLPO: f64 = 10.0;
47const TAU_SCN: f64 = 0.5;
48const MAX_LC: f64 = 6.0;
49const MAX_R: f64 = 5.0;
50const MAX_VLPO: f64 = 5.0;
51const MAX_SCN: f64 = 7.0;
52
53// Neurotransmitter saturation scales.
54const GAMMA_ACH_R: f64 = 2.5;
55const GAMMA_NE: f64 = 5.0;
56const GAMMA_GABA: f64 = 2.5;
57const GAMMA_SCN: f64 = 4.0;
58
59// Sigmoid thresholds and widths of the population response functions.
60const BETA_LC: f64 = -0.4;
61const ALPHA_LC: f64 = 0.4;
62/// −0.3 gives no REM; −0.8 gives four REM bouts per sleep episode.
63const BETA_R: f64 = -0.8;
64const ALPHA_R: f64 = 0.1;
65const ALPHA_VLPO: f64 = 0.2;
66const K1: f64 = -0.1;
67const K2: f64 = -0.0045;
68const BETA_SCN: f64 = -0.1;
69const ALPHA_SCN: f64 = 0.7;

71// Homeostatic sleep drive.
72const THETA_W: f64 = 2.0;
73const TAU_HW: f64 = 946.8;
74const TAU_HS: f64 = 202.2;
75pub const H_MAX: f64 = 323.88;
76const H_MIN: f64 = 0.0;

78// Circadian oscillator (Serkh & Forger 2014 corrections).
79const TAU_X: f64 = 24.2;
80const ALPH_0: f64 = 0.05;
81const BETA: f64 = 0.0075;
82const G: f64 = 33.75;
83const P: f64 = 0.5;
84const K: f64 = 0.55;
85const I_0: f64 = 9500.0;
86const MU: f64 = 0.23;

88// Monoaminergic arousal → LC. The wake side of the switch is driven by
89// the brain's own arousal chemistry, not an exogenous drive: orexin
90// neurons project most densely onto the LC (Hagan et al. 1999, PNAS
91// 96:10911) and are the switch's stabilizer — Saper's flip-flop has
92// orexin reinforcing the monoaminergic wake populations, never VLPO
93// (VLPO neurons lack orexin receptors). Histaminergic TMN neurons
94// activate LC and are themselves silenced by VLPO during sleep — the
95// secondary wake-side input. Both are measured against their resting
96// baselines (the neurochemical defaults, `Orexin => 0.30`,
97// `Histamine => 0.30`), so resting chemistry adds nothing and the
98// source model is preserved at rest; *below*-baseline orexin weakens
99// the wake side, the loss-of-stabilizer signature of narcolepsy.
100const OXIN_REST: f64 = 0.30;
101const HTM_REST: f64 = 0.30;
102const GOXIN_LC: f64 = 1.2;
103const GHTM_LC: f64 = 0.8;

107/// Firing rate of REM-on above which the source counts REM onset.

125fn sigmoid(x: f64, max: f64, beta: f64, alpha: f64) -> f64 {
126    max * 0.5 * (1.0 + ((x - beta) / alpha).tanh())
127}

129/// Subcortical retinal afference for one integration step — the optic
130/// nerve's non-image-forming branches, measured by the body from the
131/// nerve's shared-memory frames (`daemon::optic_nerve`).
132///
133/// Two projections matter to this network:
134///
135/// - **Retinohypothalamic tract (RHT)**: melanopsin ipRGCs project
136///   (near-)exclusively to the SCN; retinal illuminance is the primary
137///   zeitgeber. `light` carries it in the oscillator's lux units.
138///   `None` means no live retinal read — the built-in photoperiod
139///   stands in for the world, as before.
140/// - **Superior colliculus → alerting**: a salient visual transient is
141///   the sensory-gaining "wake-up call" — salient stimuli are still
142///   analysed during sleep and can arouse when they matter. `alert_lc`
143///   is a transient excitation onto the LC input for this step only;
144///   the network's own flip-flop decides whether it crosses threshold.
145///
146/// Afference, not a command: it does not disable the
147/// adenosine→homeostat coupling — the body still owns the switch.
148#[derive(Clone, Copy, Debug, Default, PartialEq)]
149pub struct VisualAfference {
150    /// Retinal illuminance in lux-equivalent units (`I_0` scale).
151    pub light: Option<f64>,
152    /// Transient LC excitation from a salient visual event.
153    pub alert_lc: f64,
154}
155
156impl VisualAfference {
157    /// No retinal afference: the source model unchanged.
158    pub const NONE: VisualAfference = VisualAfference {
159        light: None,
160        alert_lc: 0.0,
161    };
162}

164/// Right-hand side of the network at time `t` (minutes), at resting
165/// arousal chemistry with no retinal or stress input — the source model.
166pub fn deriv(t: f64, x: &State) -> State {
167    deriv_with_afference(t, x, VisualAfference::NONE, 0.0, OXIN_REST, HTM_REST)
168}

170/// Right-hand side of the network at time `t` (minutes), with the optic
171/// nerve's subcortical afference, the stress-axis alerting input (`hpa`:
172/// the cortisol stress fraction, [0, 1]), and the monoaminergic arousal
173/// inputs (`oxin`, `htm`: effective orexin and histamine levels, [0, 1],
174/// measured against `OXIN_REST`/`HTM_REST`) added.
175pub fn deriv_with_afference(
176    t: f64,
177    x: &State,
178    aff: VisualAfference,
179    hpa: f64,
180    oxin: f64,
181    htm: f64,
182) -> State {
183    let [f_lc, f_vlpo, f_r, f_scn, h, c, xc, n] = *x;
184    let c_ne = (f_lc / GAMMA_NE).tanh();
185    let c_gaba = (f_vlpo / GAMMA_GABA).tanh();
186    let c_ach = (f_r / GAMMA_ACH_R).tanh();
187    let c_scn = (f_scn / GAMMA_SCN).tanh();
188
189    // Four alerting inputs sum on the same LC channel — the collicular
190    // pulse (sensory gaining), circulating cortisol (humoral stress), and
191    // the monoaminergic arousal drive. All of them are the body's own
192    // chemistry or sensory input; nothing exogenous is held on the wire.
193    let lc_input = GACH_LC * c_ach + GSCN_LC * c_scn - GGABA_LC * c_gaba
194        + aff.alert_lc
195        + GCORT_LC * hpa
196        + GOXIN_LC * (oxin - OXIN_REST)
197        + GHTM_LC * (htm - HTM_REST);
198    let vlpo_input = -GNE_VLPO * c_ne - GSCN_VLPO * c_scn;
199    let r_input = GACH_R * c_ach - GNE_R * c_ne + GSCN_R * c_scn - GGABA_R * c_gaba;
200    let beta_vlpo = K2 * h + K1;
201
202    let lc_inf = sigmoid(lc_input, MAX_LC, BETA_LC, ALPHA_LC);
203    let r_inf = sigmoid(r_input, MAX_R, BETA_R, ALPHA_R);
204    let scn_inf = sigmoid(c, MAX_SCN, BETA_SCN, ALPHA_SCN);
205    let vlpo_inf = sigmoid(vlpo_input, MAX_VLPO, beta_vlpo, ALPHA_VLPO);
206
207    // Light reaches the oscillator only through the retinohypothalamic
208    // tract — measured retinal illuminance when the eye is live, and
209    // the built-in photoperiod (the world's fallback) when it is not.
210    let i_light: f64 = aff.light.unwrap_or_else(|| scheduled_light(t));
211    let alph = ALPH_0 * (i_light.powf(P) / I_0.powf(P));
212    let b_hat = G * (1.0 - n) * alph;
213    let b = b_hat * (1.0 - 0.4 * c) * (1.0 - 0.4 * xc);
214
215    let dh = if f_lc - THETA_W >= 0.0 {
216        (H_MAX - h) / TAU_HW
217    } else {
218        (H_MIN - h) / TAU_HS
219    };
220    let w = (24.0 / (0.99669 * TAU_X)).powi(2);
221    [
222        (lc_inf - f_lc) / TAU_LC,
223        (vlpo_inf - f_vlpo) / TAU_VLPO,
224        (r_inf - f_r) / TAU_R,
225        (scn_inf - f_scn) / TAU_SCN,
226        dh,
227        (PI / 720.0) * (xc + b),
228        (PI / 720.0) * (MU * (xc - 4.0 * xc.powi(3) / 3.0) - c * (w + K * b)),
229        alph * (1.0 - n) - BETA * n,
230    ]
231}

233/// The built-in photoperiod — the world's fallback light signal when
234/// no retinal measurement exists. 500 lux during the 14 h light
235/// period, else 0. The source runs light from minute 600 of each
236/// simulated day (10:00–24:00); here t is anchored to the local wall
237/// clock, so the window is phase-shifted to 07:00–21:00 — a 14 h
238/// photoperiod, preserved — so that the entrained sleep episode
239/// (~22:11–05:50) falls inside the neurochemical circadian's
240/// biological night rather than ending ~2 h after melatonin offset as
241/// it did with the source's alignment.
242fn scheduled_light(t: f64) -> f64 {
243    if (420.0..1260.0).contains(&t.rem_euclid(1440.0)) {
244        500.0
245    } else {
246        0.0
247    }
248}

250/// One fourth-order Runge–Kutta step of `dt` minutes from time `t`, with
251/// retinal afference and chemical inputs held constant over the step.
252pub fn rk4_with_afference(
253    t: f64,
254    x: &State,
255    dt: f64,
256    aff: VisualAfference,
257    hpa: f64,
258    oxin: f64,
259    htm: f64,
260) -> State {
261    let add = |a: &State, b: &State, k: f64| -> State { std::array::from_fn(|i| a[i] + k * b[i]) };
262    let k1 = deriv_with_afference(t, x, aff, hpa, oxin, htm);
263    let k2 = deriv_with_afference(t + dt / 2.0, &add(x, &k1, dt / 2.0), aff, hpa, oxin, htm);
264    let k3 = deriv_with_afference(t + dt / 2.0, &add(x, &k2, dt / 2.0), aff, hpa, oxin, htm);
265    let k4 = deriv_with_afference(t + dt, &add(x, &k3, dt), aff, hpa, oxin, htm);
266    std::array::from_fn(|i| x[i] + dt / 6.0 * (k1[i] + 2.0 * k2[i] + 2.0 * k3[i] + k4[i]))
267}

269/// One RK4 step of the source model — resting chemistry, no afference.
270pub fn rk4(t: f64, x: &State, dt: f64) -> State {
271    rk4_with_afference(t, x, dt, VisualAfference::NONE, 0.0, OXIN_REST, HTM_REST)
272}

274/// Advance the network by `dt_min` minutes from `t_min` under the optic
275/// nerve's subcortical afference, the stress-axis alerting input (`hpa`:
276/// cortisol stress fraction, [0, 1]), and the monoaminergic arousal
277/// inputs (`oxin`, `htm`: effective levels, [0, 1]).
278///
279/// A non-finite result keeps the previous state. The network is authoritative
280/// and persistent, so one bad step must not be allowed to poison it.
281pub fn advance_with_afference(
282    t_min: f64,
283    x: &State,
284    dt_min: f64,
285    aff: VisualAfference,
286    hpa: f64,
287    oxin: f64,
288    htm: f64,
289) -> State {
290    let next = rk4_with_afference(t_min, x, dt_min, aff, hpa, oxin, htm);
291    if next.iter().all(|v| v.is_finite()) {
292        next
293    } else {
294        *x
295    }
296}

313/// saturating maximum. LC drives wake-related chemistry; REM-on drives REM.

316    crate::state::neurochemical::NetworkLevels {

318        vlpo: finite_clamp((x[1] / MAX_VLPO) as f32, 0.0, 1.0),

323/// Map the neurochemical adenosine level (resting 0.20, upper
324/// asymptote 0.95) onto the sleep-wake network's homeostatic drive
325/// `h` in `[H_MIN, H_MAX]` (`[0.0, 323.88]`).
326///
327/// Coupling `h` to `Adenosine` ensures that only her own chemical
328/// sleep pressure can flip the VLPO/LC switch into sleep, and
329/// glymphatic adenosine clearance during sleep is what releases it.
330pub fn h_from_adenosine(adenosine: f32) -> f64 {
331    let norm = ((f64::from(adenosine) - 0.20) / (0.95 - 0.20)).clamp(0.0, 1.0);
332    H_MIN + norm * (H_MAX - H_MIN)
333}

335/// Whether the network is in a sleep state at these normalized
336/// population levels — LC firing below the source's 4 Hz sleep-onset
337/// threshold. The levels are normalized by each population's maximum
338/// rate, so the threshold is `SLEEP_THRESHOLD / MAX_LC`.
339pub fn levels_asleep(l: crate::state::neurochemical::NetworkLevels) -> bool {
340    f64::from(l.lc) * MAX_LC < SLEEP_THRESHOLD
341}

343/// The sleep stage the network implies, by the source's own definitions:
344/// wake while LC fires at or above [`SLEEP_THRESHOLD`]; otherwise REM when
345/// REM-on fires above [`REM_THRESHOLD`], and NREM when it does not.
346///
347/// While awake, the stage is whatever waking phase the chemistry already
348/// names (Active, Alert, Flow, Stress, Drowsy, Overwhelmed). A sleep
349/// stage left over from the previous tick becomes Active, because the
350/// network, not the chemistry, now decides when sleep ends.
351pub fn stage(
352    x: &State,
353    awake: crate::state::zones::MentalPhase,
354) -> crate::state::zones::MentalPhase {
355    use crate::state::zones::MentalPhase;
356    if x[0] >= SLEEP_THRESHOLD {
357        match awake {
358            MentalPhase::NREM | MentalPhase::REM => MentalPhase::Active,
359            other => other,
360        }
361    } else if x[2] > REM_THRESHOLD {
362        MentalPhase::REM
363    } else {
364        MentalPhase::NREM
365    }
366}

374// ─── Bistable geometry: the fold margins ────────────────────────────
375//
376// The LC–VLPO switch is a cusp. Mutual inhibition gives the fast
377// subsystem two stable fixed points separated by a saddle, and the
378// homeostat h — through `beta_vlpo = K1 + K2·h` — moves the system
379// along the cusp's control axis: as h rises the VLPO activation
380// threshold falls until the wake attractor merges with the saddle and
381// annihilates (the upper fold = sleep onset); as h falls during sleep
382// the sleep attractor annihilates the same way (the lower fold = the
383// wake-up edge). Between the folds the system is bistable, and that
384// band is where the sleep-onset flicker lives.
385//
386// The reduction is exact for the (f_lc, f_vlpo) subsystem at frozen
387// inputs: VLPO's input depends on LC only through c_ne = tanh(f_lc/γ),
388// so the VLPO equilibrium is a closed function of f_lc and every
389// fixed point of the 2D subsystem is a root of the scalar map
390//   g(f) = LC∞(I_LC(f_vlpo*(f))) − f.
391// Stability carries over too — the 2D Jacobian is a monotone feedback
392// loop whose determinant changes sign exactly where g's slope does.
393// f_r and f_scn are held at their current values: they are inputs to
394// the switch (ACh and circadian drives), not participants in the fold
395// — the standard quasi-static reduction for a fast-slow system. The
396// reported folds are therefore the folds of the reduced switch at the
397// current operating point, computed per query rather than per tick.
398
399/// Lower bound of the homeostat sweep used to locate the folds. h is
400/// physiologically bounded to `[H_MIN, H_MAX]` at runtime, but the
401/// fold can sit outside that band — the margin is still meaningful,
402/// it just means the current regime is monostable on one side.
403const FOLD_H_LO: f64 = -400.0;
404/// Upper bound of the homeostat sweep.
405const FOLD_H_HI: f64 = 1200.0;
406/// Coarse grid for the fold scan; transitions are refined by
407/// bisection afterwards, so this only needs to bracket them.
408const FOLD_H_STEP: f64 = 1.0;
409/// Grid resolution of the fixed-point scan over f_lc ∈ [0, MAX_LC].
410const FOLD_F_STEP: f64 = 0.02;
411/// Bisection iterations refining each bracketed root and fold.
412const FOLD_BISECT_ITERS: usize = 40;
413
414/// The scalar fixed-point map of the reduced (LC, VLPO) subsystem:
415/// `g(f) = LC∞(I_LC(f_vlpo*(f))) − f` with the homeostat substituted
416/// for the state's own `x[4]` and the REM/circadian drives frozen at
417/// the state's current values. A root of `g` is a fixed point of the
418/// switch; where `g`'s slope at the root is negative the fixed point
419/// is a stable branch.
420fn switch_map(f_lc: f64, h: f64, x: &State, hpa: f64, oxin: f64, htm: f64) -> f64 {
421    let c_ne = (f_lc / GAMMA_NE).tanh();
422    let c_scn = (x[3] / GAMMA_SCN).tanh();
423    let c_ach = (x[2] / GAMMA_ACH_R).tanh();
424    let beta_vlpo = K2 * h + K1;
425    let vlpo_input = -GNE_VLPO * c_ne - GSCN_VLPO * c_scn;
426    let f_vlpo = sigmoid(vlpo_input, MAX_VLPO, beta_vlpo, ALPHA_VLPO);
427    let c_gaba = (f_vlpo / GAMMA_GABA).tanh();
428    let lc_input = GACH_LC * c_ach + GSCN_LC * c_scn - GGABA_LC * c_gaba
429        + GCORT_LC * hpa
430        + GOXIN_LC * (oxin - OXIN_REST)
431        + GHTM_LC * (htm - HTM_REST);
432    sigmoid(lc_input, MAX_LC, BETA_LC, ALPHA_LC) - f_lc
433}

435/// Every fixed point of the reduced switch at homeostat `h`, in
436/// ascending order — the roots of [`switch_map`], located by sign
437/// change on a grid and refined by bisection. One low root is the
438/// sleep branch, one high root is the wake branch, three roots are
439/// both branches plus the saddle between them.
440fn switch_fixed_points(h: f64, x: &State, hpa: f64, oxin: f64, htm: f64) -> Vec<f64> {
441    let mut roots = Vec::new();
442    let mut f_prev = 0.0;
443    let mut g_prev = switch_map(0.0, h, x, hpa, oxin, htm);
444    let n = (MAX_LC / FOLD_F_STEP).ceil() as usize;
445    for i in 1..=n {
446        let f = (i as f64 * FOLD_F_STEP).min(MAX_LC);
447        let g = switch_map(f, h, x, hpa, oxin, htm);
448        if g_prev.is_finite() && g.is_finite() && g_prev * g < 0.0 {
449            let (mut lo, mut hi, mut glo) = (f_prev, f, g_prev);
450            for _ in 0..FOLD_BISECT_ITERS {
451                let mid = 0.5 * (lo + hi);
452                let gmid = switch_map(mid, h, x, hpa, oxin, htm);
453                if !gmid.is_finite() {
454                    break;
455                }
456                if glo * gmid <= 0.0 {
457                    hi = mid;
458                } else {
459                    lo = mid;
460                    glo = gmid;
461                }
462            }
463            roots.push(0.5 * (lo + hi));
464        }
465        f_prev = f;
466        g_prev = g;
467    }
468    roots
469}

471/// The number of fixed points of the reduced switch at homeostat `h`.
472///
473/// This is the honest bistability detector: the folds are where the
474/// root count changes, not where a root crosses the behavioral
475/// [`SLEEP_THRESHOLD`]. The 4 Hz line divides *scored* wake from
476/// scored sleep; the saddle-node is where a branch annihilates —
477/// the two are different lines, and a weakened wake attractor can
478/// sit below 4 Hz while it still exists dynamically. In the cusp
479/// geometry the count runs 1 → 3 → 1 as h sweeps upward: birth of
480/// the saddle + sleep pair is the lower fold, death of the saddle +
481/// wake pair is the upper fold. Three roots means the low and high
482/// branches coexist — bistable.
483fn switch_root_count(h: f64, x: &State, hpa: f64, oxin: f64, htm: f64) -> usize {
484    switch_fixed_points(h, x, hpa, oxin, htm).len()
485}

487/// The smallest h in `lo..hi` at which `pred` first holds, assuming it
488/// holds monotonically (false below the fold, true above). Returns
489/// `lo` when the predicate already holds there (the boundary is below
490/// the bracket), `f64::NAN` when it never holds inside the window.
491fn fold_edge(lo: f64, mut hi: f64, pred: &mut dyn FnMut(f64) -> bool) -> f64 {
492    if pred(lo) {
493        return lo;
494    }
495    if !pred(hi) {
496        return f64::NAN;
497    }
498    let mut lo = lo;
499    for _ in 0..FOLD_BISECT_ITERS {
500        let mid = 0.5 * (lo + hi);
501        if pred(mid) {
502            hi = mid;
503        } else {
504            lo = mid;
505        }
506    }
507    hi
508}

510/// The bistable geometry of the LC–VLPO switch at the current
511/// operating point: the homeostat values where the sleep and wake
512/// attractors annihilate, and where `h` sits relative to them.
513///
514/// `hpa`/`oxin`/`htm` are the same alerting inputs the tick loop
515/// feeds [`deriv_with_afference`]: the cortisol stress fraction and
516/// the effective orexin/histamine levels. The collicular transient is
517/// excluded — the margin is a property of the slow state, not of a
518/// momentary sensory pulse.
519#[derive(Clone, Copy, Debug)]
520pub struct FoldMargins {
521    /// The homeostat now (`x[4]`).
522    pub h: f64,
523    /// Lower fold: below this h the sleep branch ceases to exist —
524    /// the wake-up edge. `f64::NAN` when no fold exists in the sweep
525    /// window (the sleep branch never annihilates in range).
526    pub h_sleep_fold: f64,
527    /// Upper fold: above this h the wake branch ceases to exist — the
528    /// fall-asleep edge. `f64::NAN` when no fold exists in the sweep
529    /// window.
530    pub h_wake_fold: f64,
531    /// `(h − h_sleep_fold)/(h_wake_fold − h_sleep_fold)` — 0 at the
532    /// wake-up edge, 1 at the fall-asleep edge, inside (0, 1) in the
533    /// bistable band. Values outside [0, 1] are meaningful: >1 is
534    /// forced-sleep territory, <0 forced-wake. `f64::NAN` when either
535    /// fold is unbounded.
536    pub position: f64,
537    /// Whether the switch currently has both stable branches — i.e.
538    /// `h` lies inside the bistable band where noise-driven flicker
539    /// between wake-like and sleep-like microstates can occur.
540    pub bistable: bool,
541}
542
543/// Compute the switch's fold margins at `x` with alerting inputs
544/// `hpa`/`oxin`/`htm` (see [`FoldMargins`]).
545pub fn fold_margins(x: &State, hpa: f64, oxin: f64, htm: f64) -> FoldMargins {
546    let h = x[4];
547    let bistable = switch_root_count(h, x, hpa, oxin, htm) >= 3;
548
549    // One upward sweep brackets both transitions. The cusp's count
550    // sequence is monotone in h — 1 → 3 at the lower fold (the saddle
551    // + sleep pair is born), 3 → 1 at the upper fold (the saddle +
552    // wake pair dies) — so the first ≥3 cell brackets the lower fold
553    // and the first <3 cell after it brackets the upper.
554    let mut sleep_bracket: Option<(f64, f64)> = None;
555    let mut wake_bracket: Option<(f64, f64)> = None;
556    let mut in_band = switch_root_count(FOLD_H_LO, x, hpa, oxin, htm) >= 3;
557    let mut t = FOLD_H_LO + FOLD_H_STEP;
558    while t <= FOLD_H_HI && wake_bracket.is_none() {
559        let n = switch_root_count(t, x, hpa, oxin, htm);
560        if !in_band && n >= 3 {
561            sleep_bracket = Some((t - FOLD_H_STEP, t));
562            in_band = true;
563        } else if in_band && n < 3 {
564            wake_bracket = Some((t - FOLD_H_STEP, t));
565        }
566        t += FOLD_H_STEP;
567    }
568
569    let h_sleep_fold = match sleep_bracket {
570        Some((lo, hi)) => fold_edge(lo, hi, &mut |hh| {
571            switch_root_count(hh, x, hpa, oxin, htm) >= 3
572        }),
573        // The band extends below the sweep window (or does not exist).
574        None => f64::NAN,
575    };
576    let h_wake_fold = match wake_bracket {
577        Some((lo, hi)) => fold_edge(lo, hi, &mut |hh| {
578            switch_root_count(hh, x, hpa, oxin, htm) < 3
579        }),
580        // The band extends above the sweep window.
581        None => f64::NAN,
582    };
583
584    let position = if h_sleep_fold.is_finite() && h_wake_fold.is_finite() {
585        (h - h_sleep_fold) / (h_wake_fold - h_sleep_fold)
586    } else {
587        f64::NAN
588    };
589    FoldMargins {
590        h,
591        h_sleep_fold,
592        h_wake_fold,
593        position,
594        bistable,
595    }
596}

602    /// Integrate for `minutes` at a 0.02-minute step, sampling every
603    /// `every` minutes. Returns (t, state) pairs.
604    fn simulate(minutes: f64, every: f64) -> Vec<(f64, State)> {
605        let dt = 0.02;
606        let steps = (minutes / dt).round() as usize;
607        let stride = (every / dt).round() as usize;
608        let mut x = INITIAL;
609        let mut out = Vec::new();
610        for i in 0..steps {
611            let t = i as f64 * dt;
612            if i % stride == 0 {
613                out.push((t, x));
614            }
615            x = rk4(t, &x, dt);
616        }
617        out
618    }

655    /// REM bouts in the first sleep period at or after `night_start`:

658    fn night_architecture(e: &Events, night_start: f64) -> (f64, Vec<f64>) {
659        let onset = *e
660            .sleep_on
661            .iter()
662            .find(|t| **t >= night_start)
663            .expect("sleep onset");
664        let offset = *e
665            .sleep_off
666            .iter()
667            .find(|t| **t > onset)
668            .expect("sleep offset");
669        let rems: Vec<(f64, f64)> = e
670            .rem_on
671            .iter()
672            .zip(&e.rem_off)
673            .filter(|(on, _)| **on > onset && **on < offset)
674            .map(|(on, off)| (*on, *off - *on))
675            .collect();
676        let latency = rems.first().map_or(f64::NAN, |(on, _)| on - onset);
677        (latency, rems.iter().map(|(_, d)| *d).collect())
678    }

1124    /// The position coordinate is the hysteresis-loop coordinate:
1125    /// inside the band it reads (0, 1), past the wake fold it exceeds
1126    /// 1 (forced sleep), below the sleep fold it is negative (forced
1127    /// wake) — and the bistable flag agrees with the root count.
1128    #[test]
1129    fn fold_position_tracks_the_band() {
1130        let mut x = INITIAL;
1131        let m0 = fold_margins(&x, 0.0, OXIN_REST, HTM_REST);
1132        assert!(m0.h_sleep_fold.is_finite() && m0.h_wake_fold.is_finite());
1133
1134        // Mid-band: synthetic h halfway between the folds — three
1135        // roots, position strictly inside (0, 1).
1136        x[4] = 0.5 * (m0.h_sleep_fold + m0.h_wake_fold);
1137        let mid = fold_margins(&x, 0.0, OXIN_REST, HTM_REST);
1138        assert!(mid.bistable, "mid-band must be bistable: {mid:?}");
1139        assert!(
1140            mid.position > 0.0 && mid.position < 1.0,
1141            "position in band: {}",
1142            mid.position
1143        );
1144
1145        // Forced-sleep side: h beyond the wake fold — monostable,
1146        // position past 1.
1147        x[4] = m0.h_wake_fold + 50.0;
1148        let high = fold_margins(&x, 0.0, OXIN_REST, HTM_REST);
1149        assert!(!high.bistable, "past the fold is monostable: {high:?}");
1150        assert!(
1151            high.position > 1.0,
1152            "forced-sleep position: {}",
1153            high.position
1154        );
1155
1156        // Forced-wake side: h below the sleep fold.
1157        x[4] = m0.h_sleep_fold - 50.0;
1158        let low = fold_margins(&x, 0.0, OXIN_REST, HTM_REST);
1159        assert!(!low.bistable, "below the fold is monostable: {low:?}");
1160        assert!(low.position < 0.0, "forced-wake position: {}", low.position);
1161    }
