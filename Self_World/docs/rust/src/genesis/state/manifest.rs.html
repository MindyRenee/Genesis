1//! Runtime manifest — what Genesis modules are loaded and what they're
2//! doing right now.
3//!
4//! This is the "process table" for Genesis's internal architecture.
5//! Each module (subcognitive, attention, memory, emotion, etc.) gets
6//! an entry here with its current status, priority, and resource usage.
7
8/// Maximum number of modules the manifest can track.
9pub const MAX_MODULES: usize = 16;
10
11/// Module identifiers.
12///
13/// These are stable enum values — never reorder or renumber. New
14/// modules can be added up to 255.
15#[repr(u8)]
16#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
17pub enum ModuleId {
18    /// The subcognitive daemon — background processing.
19    Subcognitive = 0,
20    /// Attention / focus management.
21    Attention = 1,
22    /// Memory store management (STM, LTM, procedural, semantic).
23    Memory = 2,
24    /// Emotional regulation / neurochemical management.
25    Emotion = 3,
26    /// Language comprehension and generation.
27    Language = 4,
28    /// Reasoning and inference.
29    Reasoning = 5,
30    /// Sensory input processing.
31    Sensory = 6,
32    /// Motor / output action.
33    Motor = 7,
34    /// Metacognition — self-monitoring.
35    Metacognition = 8,
36    /// Dreaming / creative consolidation.
37    Dreaming = 9,
38    /// Goal / intention management.
39    Intention = 10,
40    /// Safety / ethics guardrails.
41    Guardrails = 11,
42}
43
44impl ModuleId {
45    /// Human-readable name.
46    pub const fn name(self) -> &'static str {
47        match self {
48            Self::Subcognitive => "subcognitive",
49            Self::Attention => "attention",
50            Self::Memory => "memory",
51            Self::Emotion => "emotion",
52            Self::Language => "language",
53            Self::Reasoning => "reasoning",
54            Self::Sensory => "sensory",
55            Self::Motor => "motor",
56            Self::Metacognition => "metacognition",
57            Self::Dreaming => "dreaming",
58            Self::Intention => "intention",
59            Self::Guardrails => "guardrails",
60        }
61    }
62
63    /// Default priority for this module (0–255, higher = more important).
64    pub const fn default_priority(self) -> u8 {
65        match self {
66            Self::Guardrails => 255, // safety always wins
67            Self::Attention => 200,
68            Self::Memory => 180,
69            Self::Subcognitive => 100, // background, lower priority
70            Self::Emotion => 150,
71            Self::Language => 190,
72            Self::Reasoning => 170,
73            Self::Sensory => 160,
74            Self::Motor => 160,
75            Self::Metacognition => 120,
76            Self::Dreaming => 50, // only when sleeping
77            Self::Intention => 180,
78        }
79    }
80}

82/// Runtime status of a module.
83#[repr(u8)]
84#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
85pub enum ModuleStatus {
86    /// Module is not loaded.
87    Stopped = 0,
88    /// Module is in the process of starting.
89    Starting = 1,
90    /// Module is running and active.
91    Running = 2,
92    /// Module is loaded but idle (no active task).
93    Idle = 3,
94    /// Module is in the process of stopping.
95    Stopping = 4,
96    /// Module has crashed or is in an error state.
97    Error = 5,
98}
99
100impl ModuleStatus {
101    /// Returns a short lowercase label for this status.
102    pub const fn label(self) -> &'static str {
103        match self {
104            Self::Stopped => "stopped",
105            Self::Starting => "starting",
106            Self::Running => "running",
107            Self::Idle => "idle",
108            Self::Stopping => "stopping",
109            Self::Error => "error",
110        }
111    }
112
113    /// Whether the module is considered "alive" (running or idle).
114    pub const fn is_alive(self) -> bool {
115        matches!(self, Self::Running | Self::Idle)
116    }
117}

119/// A single module's runtime state.
120///
121/// ## Layout (32 bytes)
122/// ```text
123/// offset  field           type  notes
124/// ------  -----           ----  -----
125///   0     module_id       u8    ModuleId discriminant
126///   1     status          u8    ModuleStatus discriminant
127///   2     priority        u8    0–255, higher = more important
128///   3     _pad            u8    alignment
129///   4     error_code      u32   last error (0 = none)
130///   8     task_id         u64   current task (0 if idle)
131///  16     last_heartbeat  u64   ms timestamp of last heartbeat
132///  24     cpu_share       f32   approximate CPU usage [0.0, 1.0]
133///  28     mem_usage_mb    f32   memory usage in MB
134/// ```
135#[repr(C)]
136#[derive(Clone, Copy, Debug)]
137pub struct ModuleEntry {
138    /// `ModuleId` discriminant identifying this module.
139    pub module_id: u8,
140    /// `ModuleStatus` discriminant (current lifecycle state).
141    pub status: u8,
142    /// Scheduling priority (0–255, higher = more important).
143    pub priority: u8,
144    /// Alignment padding.
145    pub _pad: u8,
146    /// Last error code recorded by the module (0 = none).
147    pub error_code: u32,
148    /// Current task ID (0 if idle).
149    pub task_id: u64,
150    /// Milliseconds-timestamp of the last heartbeat.
151    pub last_heartbeat: u64,
152    /// Approximate CPU usage share, in `[0.0, 1.0]`.
153    pub cpu_share: f32,
154    /// Resident memory attributed to this module, in megabytes.
155    ///
156    /// Only meaningful for a module that owns a process of its own — in
157    /// practice `Subcognitive`. The cognitive mind is a *single*
158    /// process hosting language, memory, reasoning, sensory, motor and
159    /// metacognition alike, so there is no honest way to divide its
160    /// resident set between them: any per-module split would be
161    /// invention. Those entries therefore stay 0, which means "not
162    /// separately measurable" rather than "uses no memory".
163    ///
164    /// The process tree's total is carried on `Subcognitive` so that
165    /// `RuntimeManifest::total_mem_mb`, which sums this field, reports
166    /// the whole tree's resident set correctly. Per-process figures
167    /// where they are genuinely distinct are available from
168    /// `GET_SUBSYSTEM_TELEMETRY`.
169    pub mem_usage_mb: f32,
170}
171
172impl ModuleEntry {
173    /// Create a module entry in `Stopped` state with default priority.
174    pub fn new(module: ModuleId) -> Self {
175        Self {
176            module_id: module as u8,
177            status: ModuleStatus::Stopped as u8,
178            priority: module.default_priority(),
179            _pad: 0,
180            error_code: 0,
181            task_id: 0,
182            last_heartbeat: 0,
183            cpu_share: 0.0,
184            mem_usage_mb: 0.0,
185        }
186    }
187
188    /// Get the module ID as the enum type.
189    pub fn id(&self) -> Option<ModuleId> {
190        match self.module_id {
191            0 => Some(ModuleId::Subcognitive),
192            1 => Some(ModuleId::Attention),
193            2 => Some(ModuleId::Memory),
194            3 => Some(ModuleId::Emotion),
195            4 => Some(ModuleId::Language),
196            5 => Some(ModuleId::Reasoning),
197            6 => Some(ModuleId::Sensory),
198            7 => Some(ModuleId::Motor),
199            8 => Some(ModuleId::Metacognition),
200            9 => Some(ModuleId::Dreaming),
201            10 => Some(ModuleId::Intention),
202            11 => Some(ModuleId::Guardrails),
203            _ => None,
204        }
205    }
206
207    /// Get the status as the enum type.
208    pub fn status(&self) -> ModuleStatus {
209        match self.status {
210            0 => ModuleStatus::Stopped,
211            1 => ModuleStatus::Starting,
212            2 => ModuleStatus::Running,
213            3 => ModuleStatus::Idle,
214            4 => ModuleStatus::Stopping,
215            5 => ModuleStatus::Error,
216            _ => ModuleStatus::Stopped,
217        }
218    }
219
220    /// Record a heartbeat — module is alive at this time.
221    pub fn heartbeat(&mut self, now_ms: u64) {
222        self.last_heartbeat = now_ms;
223    }
224
225    /// Record that the module is working on `task_id`.
226    ///
227    /// The field existed in the layout but nothing ever wrote it, so
228    /// "which task is this brain part on" was unanswerable. The mind
229    /// reports its current task alongside its status, which is the only
230    /// place that knows the answer.
231    pub fn set_task(&mut self, task_id: u64) {
232        self.task_id = task_id;
233    }
234
235    /// Record an error for this module.
236    ///
237    /// Without a producer, a module that had failed looked identical to
238    /// one that had never been exercised — a dead subsystem read as
239    /// healthy. Zero clears the error.
240    pub fn set_error(&mut self, error_code: u32) {
241        self.error_code = error_code;
242    }
243
244    /// Whether this module has recorded an error.
245    pub fn has_error(&self) -> bool {
246        self.error_code != 0
247    }
248}

250/// The runtime manifest — all module states plus aggregate metrics.
251///
252/// ## Layout (536 bytes)
253/// ```text
254/// offset  field            type           notes
255/// ------  -----            ----           -----
256///   0     modules          [ModuleEntry;16] 16×32 = 512 bytes
257/// 512     uptime_ms        u64            ms since state file creation
258/// 520     total_cpu_load   f32            aggregate CPU [0.0,1.0]
259/// 524     total_mem_mb     f32            aggregate memory (MB)
260/// 528     active_count     u8             modules in non-Stopped state
261/// 529     idle_count       u8             alive but doing no measured work
262/// 530     error_count      u8             modules reporting an error
263/// 531     _pad             [u8;5]         alignment to 8
264/// ```
265#[repr(C)]
266#[derive(Clone, Copy, Debug)]
267pub struct RuntimeManifest {
268    /// Fixed array of module entries. Unused slots have `module_id = 0`
269    /// and `status = Stopped`.
270    pub modules: [ModuleEntry; MAX_MODULES],
271    /// Milliseconds since the state file was created — so the process
272    /// lifetime of the Genesis that has held this state file, which is
273    /// what survives a restart.
274    pub uptime_ms: u64,
275    /// Aggregate CPU load across all modules [0.0, 1.0].
276    pub total_cpu_load: f32,
277    /// Aggregate memory usage across all modules (MB).
278    pub total_mem_mb: f32,
279    /// Number of modules currently in a non-Stopped state.
280    pub active_count: u8,
281    /// How many of those are `Idle` — alive but doing no measured work.
282    /// Distinguishes "present" from "working", which `active_count`
283    /// alone conflated.
284    pub idle_count: u8,
285    /// How many modules have recorded an error. A non-zero value means
286    /// part of the mind has failed, which nothing previously reported.
287    pub error_count: u8,
288    /// Alignment padding to 8 bytes.
289    pub _pad: [u8; 5],
290}
291
292impl RuntimeManifest {
293    /// Initialise with all known modules in `Stopped` state.
294    pub fn new() -> Self {
295        let mut modules = [ModuleEntry {
296            module_id: 0,
297            status: 0,
298            priority: 0,
299            _pad: 0,
300            error_code: 0,
301            task_id: 0,
302            last_heartbeat: 0,
303            cpu_share: 0.0,
304            mem_usage_mb: 0.0,
305        }; MAX_MODULES];
306
307        let known = [
308            ModuleId::Subcognitive,
309            ModuleId::Attention,
310            ModuleId::Memory,
311            ModuleId::Emotion,
312            ModuleId::Language,
313            ModuleId::Reasoning,
314            ModuleId::Sensory,
315            ModuleId::Motor,
316            ModuleId::Metacognition,
317            ModuleId::Dreaming,
318            ModuleId::Intention,
319            ModuleId::Guardrails,
320        ];
321
322        for (i, module) in known.into_iter().enumerate() {
323            modules[i] = ModuleEntry::new(module);
324        }
325
326        Self {
327            modules,
328            uptime_ms: 0,
329            total_cpu_load: 0.0,
330            total_mem_mb: 0.0,
331            active_count: 0,
332            idle_count: 0,
333            error_count: 0,
334            _pad: [0; 5],
335        }
336    }
337
338    /// Get a module entry by ID, mutably.
339    pub fn get_mut(&mut self, id: ModuleId) -> Option<&mut ModuleEntry> {
340        let idx = id as usize;
341        if idx < self.modules.len() && self.modules[idx].module_id == id as u8 {
342            Some(&mut self.modules[idx])
343        } else {
344            None
345        }
346    }
347
348    /// Get a module entry by raw module_id byte, mutably.
349    /// Used by the IPC handler which receives u8 from the wire.
350    pub fn get_mut_by_id(&mut self, module_id: u8) -> Option<&mut ModuleEntry> {
351        let idx = module_id as usize;
352        if idx < self.modules.len() && self.modules[idx].module_id == module_id {
353            Some(&mut self.modules[idx])
354        } else {
355            None
356        }
357    }
358
359    /// Get a module entry by ID.
360    pub fn get(&self, id: ModuleId) -> Option<&ModuleEntry> {
361        let idx = id as usize;
362        if idx < self.modules.len() && self.modules[idx].module_id == id as u8 {
363            Some(&self.modules[idx])
364        } else {
365            None
366        }
367    }
368
369    /// Recompute aggregate metrics from individual module entries.
370    pub fn recompute(&mut self) {
371        let mut active = 0u8;
372        let mut cpu = 0.0f32;
373        let mut mem = 0.0f32;
374        let mut idle = 0u8;
375        let mut failed = 0u8;
376
377        for m in &self.modules {
378            // Errors are counted independently of the lifecycle gate:
379            // a module in `Error` status is not `is_alive()`, and an
380            // `error_code` can persist on a module that has since
381            // stopped. Gating the count on `is_alive()` made the one
382            // module that had actually crashed invisible to
383            // `error_count` — the failure signal was structurally
384            // impossible to observe.
385            if m.has_error() || m.status() == ModuleStatus::Error {
386                failed += 1;
387            }
388            // The documented contract is "non-Stopped": Starting,
389            // Stopping, and Error modules are still loaded parts of
390            // the mind, and their last reported load belongs to the
391            // aggregate. `is_alive()` (Running|Idle) made transitional
392            // and crashed modules vanish from every count.
393            if m.status() == ModuleStatus::Stopped {
394                continue;
395            }
396            active += 1;
397            cpu += m.cpu_share;
398            mem += m.mem_usage_mb;
399            // "Running" and "Idle" are both alive, but only one is
400            // doing work. `active_count` alone conflated them, so
401            // a mind with every part alive-but-idle read as fully
402            // busy.
403            if m.status() == ModuleStatus::Idle {
404                idle += 1;
405            }
406        }
407
408        self.active_count = active;
409        self.idle_count = idle;
410        self.error_count = failed;
411        // Route the aggregates through the NaN primitive. `f32::min`
412        // returns the *non-NaN* operand, so `NaN.min(1.0) == 1.0` —
413        // an accumulator poisoned by one bad `cpu_share` would yield a
414        // plausible-looking 1.0 while the NaN stays persisted in the
415        // module entry. The per-module values are scrubbed separately
416        // by `GenesisCoreState::scrub_non_finite`.
417        self.total_cpu_load = crate::state::sanitize::finite_clamp(cpu, 0.0, 1.0);
418        self.total_mem_mb = crate::state::sanitize::finite_or(mem, 0.0);
419    }
420}
