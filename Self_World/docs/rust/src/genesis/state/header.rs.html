1//! Core state header — the first bytes of the memory-mapped state block.
2//!
3//! Contains identity and versioning. Reads from the mmap'd region are
4//! volatile copies; cross-thread consistency comes from the
5//! `MmapState` writer mutex + flock, not a sequence lock.
6
7/// Magic bytes identifying a Genesis core state file. ASCII `"GNSX"`.
8pub const MAGIC: [u8; 4] = *b"GNSX";
9
10/// Current schema version. Increment this when the layout of
11/// [`GenesisCoreState`](super::core_state::GenesisCoreState) changes
12/// in any way that breaks binary compatibility.
13///
14/// ## Version history
15/// - **v1**: initial 12-chemical layout (never released)
16/// - **v2**: 12-chemical coupled dynamics with receptor adaptation
17/// - **v3**: 18-chemical layout — adds endocannabinoid, vasopressin,
18///   CRH, orexin, epinephrine, melatonin. Coupling matrix expanded
19///   from 12×12 to 18×18. Arousal/valence equations updated to
20///   include the new chemicals. Bistable flip-flop with hysteresis.
21///   Adenosine sleep pressure with configurable rates. Melatonin
22///   excluded from homeostatic force (circadian-driven). GABA
23///   disinhibition uses effective levels.
24///
25///   The struct size did not change from v2 to v3
26///   because the v2 binary was already compiled with the full
27///   18-chemical NeurochemicalVector — the v2 version label was
28///   stale. Migration from v2 to v3 initializes the 6 new chemicals
29///   at defaults (in case they were zeroed by an older binary) and
30///   fills the new coupling matrix rows/columns.
31/// - **v4**: inserts the persistent `IonState` ahead of the checksum
32///   (3416 bytes).
33/// - **v5**: inserts the sleep-wake network ahead of `ions` (3480
34///   bytes).
35pub const SCHEMA_VERSION: u32 = 5;
36
37/// Provenance marker — one letter of the creator lattice (see
38/// PROVENANCE.md). It lives beside the schema version deliberately:
39/// the body's identity block carries the author's letter too.
40pub(crate) const MARKER_G06: u64 = 0x4753_0620_43BA_C78D;

42/// Header for the memory-mapped core state.
43///
44/// ## Layout (56 bytes)
45/// ```text
46/// offset  field          type   notes
47/// ------  -----          ----   -----
48///   0     magic          [u8;4] b"GNSX"
49///   4     version        u32    schema version
50///   8     state_size     u32    sizeof(GenesisCoreState)
51///  12     _pad           [u8;4] implicit alignment padding to 8
52///  16     created_at     u64    unix epoch milliseconds
53///  24     last_updated   u64    unix epoch milliseconds
54///  32     heartbeat      u64    monotonic counter, incremented each tick
55///  40     reserved       u64    former sequence-lock slot; inert
56///  48     instance_id    u64    unique per Genesis process instance
57/// ```
58#[repr(C)]
59#[derive(Clone, Copy, Debug)]
60pub struct CoreStateHeader {
61    /// Magic bytes identifying the state file (`b"GNSX"`).
62    pub magic: [u8; 4],
63    /// Schema version of the core state.
64    pub version: u32,
65    /// Total byte size of the core state struct.
66    pub state_size: u32,
67    /// Explicit alignment padding to 8 bytes. Without this field the
68    /// compiler inserts implicit padding that is not guaranteed to be
69    /// zero-initialized, which would write stack garbage into the
70    /// mmap'd state file. The checksum covers bytes [0..40) (magic
71    /// through heartbeat) and [48..checksum_offset) (instance_id
72    /// through the fields before the checksum), so this field at
73    /// offset 12 IS covered by the CRC. Only `reserved` at offset
74    /// 40..48 is excluded.
75    pub _pad: [u8; 4],
76    /// Milliseconds-timestamp when the state was first created.
77    pub created_at: u64,
78    /// Milliseconds-timestamp of the last write.
79    pub last_updated: u64,
80    /// Monotonic heartbeat counter (incremented each tick).
81    pub heartbeat: u64,
82    /// Reserved — formerly the sequence-lock counter. Inert: nothing
83    /// reads or writes it; the slot is kept so the header layout and
84    /// checksum regions stay byte-identical to prior schema versions.
85    pub reserved: u64,
86    /// Identifier for this state instance. Currently a constant (1)
87    /// from the daemon's single production call site; it is covered by
88    /// the CRC, so it is at least stable across restarts. It is not
89    /// unique per process despite the field name — do not read it as
90    /// a way to tell two daemons apart.
91    pub instance_id: u64,
92}
93
94impl CoreStateHeader {
95    /// Create a header initialised at the current schema version.
96    pub fn new(instance_id: u64, state_size: u32, now_ms: u64) -> Self {
97        Self {
98            magic: MAGIC,
99            version: SCHEMA_VERSION,
100            state_size,
101            _pad: [0; 4],
102            created_at: now_ms,
103            last_updated: now_ms,
104            heartbeat: 0,
105            reserved: 0,
106            instance_id,
107        }
108    }
109
110    /// Verify the magic bytes match.
111    pub fn verify_magic(&self) -> bool {
112        self.magic == MAGIC
113    }
114}
