1//! Legacy schema v4 layout — the on-disk binary format of state files
2//! written by Genesis binaries before schema v5.
3//!
4//! This module exists solely for **migration**: `MmapState::open`
5//! decodes a v4 file through this `#[repr(C)]` type, then converts it
6//! into the current [`GenesisCoreState`]. Schema v5 inserted the
7//! sleep-network state before the checksum, so a v4 file cannot be
8//! reinterpreted in place.
9//!
10//! DO NOT modify these types: they are a byte-exact description of a
11//! historical format. The compile-time assertion below pins the size.
12
13use super::core_state::GenesisCoreState;
14use super::header::CoreStateHeader;
15use super::inference::InferenceSignals;
16use super::ions::IonState;
17use super::manifest::RuntimeManifest;
18use super::memory::MemoryPointers;
19use super::neurochemical::NeurochemicalVector;
20use super::sleep_network;
21use super::zones::ActiveZones;
22
23/// The v4 `GenesisCoreState` (3416 bytes).
24#[repr(C)]
25#[derive(Clone, Copy, Debug)]
26pub struct GenesisCoreStateV4 {
27    /// Identity and versioning header.
28    pub header: CoreStateHeader,
29    /// 18-chemical coupled neurochemical dynamics.
30    pub neurochemicals: NeurochemicalVector,
31    /// Active task zone, emergent phase, and subcognitive flags.
32    pub zones: ActiveZones,
33    /// Memory store pointers and emotional gating weights.
34    pub memory: MemoryPointers,
35    /// Runtime module manifest table.
36    pub manifest: RuntimeManifest,
37    /// Ion electrochemistry.
38    pub ions: IonState,
39    /// CRC32 checksum of all preceding bytes (header.reserved excluded).
40    pub checksum: u32,
41    /// Active inference signals (not covered by the checksum).
42    pub inference_signals: InferenceSignals,
43}
44
45impl GenesisCoreStateV4 {
46    /// Widen a v4 state to the current layout. The sleep network did
47    /// not exist in v4, so it starts at the source's initial conditions.
48    /// Migration recomputes the checksum afterwards.
49    pub fn into_current(self) -> GenesisCoreState {
50        GenesisCoreState {
51            header: self.header,
52            neurochemicals: self.neurochemicals,
53            zones: self.zones,
54            memory: self.memory,
55            manifest: self.manifest,
56            ions: self.ions,
57            sleep_network: sleep_network::INITIAL,
58            checksum: self.checksum,
59            inference_signals: self.inference_signals,
60        }
61    }
62}
