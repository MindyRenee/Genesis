1//! Ion electrochemistry — calcium, chloride, potassium, and sodium.
2//!
3//! This module is deliberately separate from [`NeurochemicalVector`]:
4//! neurochemicals are normalized neuromodulatory control variables, while
5//! ions are physical charge carriers with concentrations, electrochemical
6//! gradients, reversal potentials, membrane voltage, and transporter
7//! fluxes. Conflating the two layers would erase the distinction between
8//! "how strongly a modulator is signaling" and "how much charge can move
9//! across a membrane."
10//!
11//! # Model
12//!
13//! The state represents a whole-brain/population aggregate rather than a
14//! single neuron. It uses the standard equivalent-circuit approximation:
15//!
16//! ```text
17//! E_i = (RT / z_i F) ln([ion]_out / [ion]_in)       (Nernst)
18//! I_i = g_i (V_m - E_i)                            (chord current)
19//! V_m → (Σ g_i E_i - I_pump) / Σ g_i               (chord conductance)
20//! ```
21//!
22//! Conductances are dimensionless relative conductances (K⁺ leak ≈ 1).
23//! `current_density` is therefore a normalized `g·mV` quantity, not a
24//! calibrated pA/mm² measurement. Concentrations are stored in mM and
25//! membrane potential in mV. Calcium's much lower intracellular
26//! concentration is represented directly (resting ≈ 0.0001 mM = 100 nM).
27//!
28//! Activity enters through the existing neurochemical state:
29//!
30//! - Glutamate opens Na⁺ and voltage-gated Ca²⁺ conductance.
31//! - GABA opens Cl⁻ conductance.
32//! - Adenosine suppresses Ca²⁺/Na⁺ channel drive and reports ATP load.
33//! - K⁺ conductance contains a voltage-gated component plus a leak term.
34//!
35//! Homeostasis is explicit rather than folded into the channel dynamics:
36//!
37//! - Na⁺/K⁺-ATPase exports 3 Na⁺ and imports 2 K⁺ per cycle, gated by
38//!   intracellular Na⁺, extracellular K⁺, and ATP availability.
39//! - A signed K⁺/Cl⁻ cotransport term models net KCC2 extrusion (positive
40//!   flux) and reversed KCC2 import (negative flux).
41//! - A Ca²⁺ extrusion term models PMCA/NCX clearance; its Na⁺-coupled
42//!   component can weaken or reverse under intracellular Na⁺ load.
43//!
44//! This is not a Hodgkin–Huxley spike simulator. It is a bounded,
45//! physically interpretable electrochemical layer that lets the abstract
46//! neuromodulator system alter membrane excitability, chloride-dependent
47//! inhibition, calcium-dependent plasticity, and metabolic pump load.
48//!
49//! References:
50//! - Hodgkin & Huxley (1952) — ionic current formalism.
51//! - Goldman (1943); Hodgkin & Katz (1949) — multi-ion membrane potential.
52//! - Thomas (1972) — electrogenic Na⁺/K⁺-ATPase current.
53//! - Payne et al. (2003) — neuronal chloride transporters.
54//! - Berridge (1998) — intracellular calcium signaling.
55

59/// Number of explicitly modeled ion species.
60pub const ION_COUNT: usize = 4;
61
62/// Stable ordering for the four ion arrays.
63///
64/// The order follows the user-facing naming order (calcium, chloride,
65/// potassium, sodium) and is part of the persisted binary schema. Do not
66/// reorder these discriminants.
67#[repr(u8)]
68#[derive(Clone, Copy, Debug, PartialEq, Eq)]
69pub enum IonId {
70    /// Ca²⁺ — second messenger, plasticity, excitotoxic load.
71    Calcium = 0,
72    /// Cl⁻ — principal fast inhibitory charge carrier.
73    Chloride = 1,
74    /// K⁺ — dominant resting conductance and repolarizing gradient.
75    Potassium = 2,
76    /// Na⁺ — depolarizing gradient and Na⁺/K⁺ pump substrate.
77    Sodium = 3,
78}

80impl IonId {
81    /// All ion IDs in persisted array order.
82    pub const fn all() -> [Self; ION_COUNT] {
83        [Self::Calcium, Self::Chloride, Self::Potassium, Self::Sodium]
84    }
85
86    /// Convert a persisted discriminant to an ion ID.
87    pub const fn from_u8(v: u8) -> Self {
88        match v {
89            0 => Self::Calcium,
90            1 => Self::Chloride,
91            2 => Self::Potassium,
92            _ => Self::Sodium,
93        }
94    }
95
96    /// Human-readable ion name.
97    pub const fn name(self) -> &'static str {
98        match self {
99            Self::Calcium => "calcium",
100            Self::Chloride => "chloride",
101            Self::Potassium => "potassium",
102            Self::Sodium => "sodium",
103        }
104    }
105
106    /// Ion valence used by the Nernst equation.
107    const fn valence(self) -> f32 {
108        match self {
109            Self::Calcium => 2.0,
110            Self::Chloride => -1.0,
111            Self::Potassium | Self::Sodium => 1.0,
112        }
113    }
114
115    /// Return this ion's index in the fixed-size ion arrays.
116    const fn index(self) -> usize {
117        self as usize
118    }

121/// Resting intracellular concentrations in mM, in `IonId` order.
122///
123/// Ca²⁺ is 100 nM (0.0001 mM), Cl⁻ is low intracellularly, K⁺ is the
124/// dominant intracellular cation, and Na⁺ is low intracellularly (~10 mM
125/// in neurons, per hippocampal and cortical measurements).
126pub const REST_INTRACELLULAR_MM: [f32; ION_COUNT] = [0.0001, 6.0, 140.0, 10.0];
127
128/// Resting extracellular concentrations in mM, in `IonId` order.
129pub const REST_EXTRACELLULAR_MM: [f32; ION_COUNT] = [1.2, 110.0, 5.0, 145.0];
130
131/// Intracellular concentration safety bounds in mM.
132const MIN_INTRACELLULAR_MM: [f32; ION_COUNT] = [0.00002, 1.0, 50.0, 2.0];
133const MAX_INTRACELLULAR_MM: [f32; ION_COUNT] = [0.02, 30.0, 220.0, 80.0];
134
135/// Extracellular concentration safety bounds in mM.
136const MIN_EXTRACELLULAR_MM: [f32; ION_COUNT] = [0.5, 60.0, 1.0, 80.0];
137const MAX_EXTRACELLULAR_MM: [f32; ION_COUNT] = [3.0, 160.0, 25.0, 200.0];
138
139/// RT/F at 310 K expressed in mV.
140const NERNST_FACTOR_MV: f32 = 26.713;
141/// Resting membrane potential used to normalize excitability.
142const RESTING_MEMBRANE_MV: f32 = -75.0;
143/// Membrane integration time constant in seconds.
144const MEMBRANE_TAU_S: f32 = 0.025;
145/// Effective intracellular-to-reservoir volume ratio, not tissue volume fraction.
146const EXTRACELLULAR_VOLUME_RATIO: f32 = 0.20;
147/// Extracellular reservoir/glial homeostasis time constant in seconds.
148const EXTRACELLULAR_TAU_S: f32 = 20.0;
149/// Slow intracellular osmotic/metabolic restoration for Na⁺ and K⁺.
150const INTRACELLULAR_TAU_S: f32 = 600.0;
151
152/// Relative conductance activation time constants in seconds.
153const CONDUCTANCE_TAU_S: [f32; ION_COUNT] = [0.020, 0.050, 0.080, 0.015];
154/// Conversion from normalized `g·mV` current to intracellular mM/s.
155const CURRENT_TO_MM_PER_S: [f32; ION_COUNT] = [0.00175, 0.0035, 0.0035, 0.0035];
156const CALCIUM_FREE_FRACTION: f32 = 0.00015 / CURRENT_TO_MM_PER_S[IonId::Calcium as usize];
157
158/// Maximum Na⁺/K⁺-ATPase cycle flux in mM/s. Each cycle exports 3 Na⁺
159/// and imports 2 K⁺, so the Na⁺ rate is `3 × nak_pump_rate`.
160const NAK_MAX_RATE_MM_PER_S: f32 = 0.042;
161/// Normalized electrogenic pump contribution in the chord equation.
162const PUMP_CURRENT_EQUIV_MV: f32 =
163    NAK_MAX_RATE_MM_PER_S / CURRENT_TO_MM_PER_S[IonId::Sodium as usize];
164/// Half-saturation constants for Na⁺i and K⁺o pump activation.
165const NAK_KM_NA_I_MM: f32 = 10.0;
166const NAK_KM_K_O_MM: f32 = 1.5;
167/// Hill exponent for cooperative Na⁺i activation. The pump binds three
168/// cytosolic Na⁺ per cycle; measured activation is sigmoidal with an
169/// effective exponent of ~2–3 (Gadsby & Nakao 1989). The canonical
170/// reduced form used by Luo & Rudy (1994) and most neural models is
171/// `1/(1 + (Km/Na_i)^1.5)`.
172const NAK_NA_HILL: f32 = 1.5;
173/// Extracellular Na⁺ scale (mM) inside the Luo–Rudy voltage factor —
174/// sets the σ(Na_o) term of `f_NaK` (Luo & Rudy 1994).
175const NAK_NA_O_SCALE_MM: f32 = 67.3;
176/// Half-saturation of the pump's ATP dependence as a fraction of the
177/// normalized ATP pool. Physiological ATP saturates the pump
178/// (Km ≈ 0.1–0.5 mM against ~5 mM cytosolic ATP), so availability
179/// gates hyperbolically, not linearly.
180const NAK_KM_ATP: f32 = 0.15;
181
182/// Baseline outward Cl⁻ transport at the resting intracellular level.
183const CL_TRANSPORT_BASE_MM_PER_S: f32 = 0.010;
184/// Phenomenological increase of KCC2 transport capacity with Cl⁻ load.
185const CL_TRANSPORT_GAIN_PER_S: f32 = 0.020;
186/// K⁺ carried per Cl⁻ by electroneutral KCC2 in either direction;
187/// NKCC1's distinct Na⁺:K⁺:Cl⁻ = 1:1:2 transport is not modeled.
188const CL_TRANSPORT_K_STOICH: f32 = 1.0;
189
190/// Baseline Ca²⁺ extrusion balancing resting Ca²⁺ leak/current.
191const CA_EXTRUSION_BASE_MM_PER_S: f32 = 0.00016;
192/// First-order PMCA/NCX clearance gain for free intracellular Ca²⁺.
193const CA_EXTRUSION_GAIN_PER_S: f32 = 2.0;
194/// High intracellular Na⁺ weakens/reverses Na⁺/Ca²⁺ exchange.
195const NCX_REVERSE_GAIN_MM_PER_S: f32 = 0.0004;
196/// Ca²⁺ level treated as a full-strength normalized signal.
197const CA_SIGNAL_FULL_MM: f32 = 0.005;

199/// Persistent electrochemical ion state.
200///
201/// # Layout (124 bytes)
202///
203/// ```text
204/// offset  field                     type          size
205/// ------  -----                     ----          ----
206///   0     intracellular_mm          [f32;4]       16
207///  16     extracellular_mm          [f32;4]       16
208///  32     reversal_potential_mv     [f32;4]       16
209///  48     conductance               [f32;4]       16
210///  64     current_density           [f32;4]       16
211///  80     membrane_potential_mv     f32            4
212///  84     nak_pump_rate             f32            4
213///  88     kcl_cotransporter_flux    f32            4
214///  92     ncx_flux                  f32            4
215///  96     atp_availability          f32            4
216/// 100     calcium_signal            f32            4
217/// 104     chloride_efficacy         f32            4
218/// 108     excitability              f32            4
219/// 112     gradient_integrity        f32            4
220/// 116     energy_load               f32            4
221/// 120     net_membrane_current      f32            4
222/// 124     TOTAL                                  124 bytes
223/// ```
224#[repr(C)]
225#[derive(Clone, Copy, Debug, PartialEq)]
226pub struct IonState {
227    /// Intracellular concentration in mM (`IonId` order).
228    pub intracellular_mm: [f32; ION_COUNT],
229    /// Extracellular concentration in mM (`IonId` order).
230    pub extracellular_mm: [f32; ION_COUNT],
231    /// Nernst reversal potential in mV (`IonId` order).
232    pub reversal_potential_mv: [f32; ION_COUNT],
233    /// Relative membrane conductance (`IonId` order; K⁺ leak ≈ 1).
234    pub conductance: [f32; ION_COUNT],
235    /// Normalized ionic current `g·(V-E)` (`IonId` order). Positive is
236    /// outward conventional current; for Cl⁻ that corresponds to Cl⁻ influx.
237    pub current_density: [f32; ION_COUNT],
238    /// Aggregate membrane potential in mV.
239    pub membrane_potential_mv: f32,
240    /// Na⁺/K⁺-ATPase cycle rate in mM/s.
241    pub nak_pump_rate: f32,
242    /// Net K⁺/Cl⁻ cotransport in mM/s. Positive = outward Cl⁻/K⁺ flux
243    /// (KCC2-like); negative = reversed KCC2 inward flux.
244    pub kcl_cotransporter_flux: f32,
245    /// Na⁺/Ca²⁺ exchange rate in mM/s of Ca²⁺. Positive = Ca²⁺ extrusion
246    /// coupled to Na⁺ entry; negative = exchange reversal.
247    pub ncx_flux: f32,
248    /// ATP availability for ion pumps and transporters [0, 1].
249    pub atp_availability: f32,
250    /// Log-normalized intracellular Ca²⁺ signal [0, 1].
251    pub calcium_signal: f32,
252    /// Effective inhibitory Cl⁻ driving force [0, 1].
253    pub chloride_efficacy: f32,
254    /// Depolarization above the resting potential, normalized [0, 1].
255    pub excitability: f32,
256    /// Mean preservation of all four resting electrochemical gradients [0, 1].
257    pub gradient_integrity: f32,
258    /// Normalized channel/transporter workload [0, 1].
259    pub energy_load: f32,
260    /// Sum of channel currents plus electrogenic pump current.
261    pub net_membrane_current: f32,
262}

267        let mut state = Self {
268            intracellular_mm: REST_INTRACELLULAR_MM,
269            extracellular_mm: REST_EXTRACELLULAR_MM,
270            reversal_potential_mv: [0.0; ION_COUNT],
271            conductance: [0.0055, 0.475, 0.945, 0.11],
272            current_density: [0.0; ION_COUNT],
273            membrane_potential_mv: RESTING_MEMBRANE_MV,
274            nak_pump_rate: 0.0,
275            kcl_cotransporter_flux: 0.0,
276            ncx_flux: 0.0,
277            atp_availability: 0.9,
278            calcium_signal: 0.0,
279            chloride_efficacy: 0.5,

281            gradient_integrity: 1.0,

304    /// Calcium-dependent plasticity modulation for the memory gating layer.
305    ///
306    /// Resting Ca²⁺ leaves the neurochemical plasticity gate at ~70% of its
307    /// neurotransmitter-derived value; strong but bounded Ca²⁺ transients
308    /// restore full plasticity. Extreme Ca²⁺ is clamped rather than allowed
309    /// to amplify plasticity without bound.
310    pub fn plasticity_modulator(&self) -> f32 {
311        finite_clamp(0.70 + 0.60 * self.calcium_signal, 0.0, 1.0)
312    }

314    /// Advance ion electrochemistry by `dt` seconds, driven by the current
315    /// effective neurochemical state.
316    ///
317    /// The update is stable at the daemon's 100 ms tick: conductance and
318    /// voltage approach their targets by exact exponentials rather than
319    /// explicit Euler steps, while concentration changes are bounded and
320    /// followed by physiological range clamps.
321    pub fn tick(&mut self, neuro: &NeurochemicalVector, dt: f32) {
322        let dt = finite_clamp(finite_or(dt, super::neurochemical::DT), 0.001, 1.0);
323        let steps = (dt / 0.01).ceil() as usize;
324        for _ in 0..steps {
325            self.tick_step(neuro, dt / steps as f32);
326        }
327    }
328
329    fn tick_step(&mut self, neuro: &NeurochemicalVector, dt: f32) {
330        self.sanitize();
331        self.recompute_derived();
332        let dt = finite_clamp(finite_or(dt, super::neurochemical::DT), 0.001, 1.0);
333
334        let glutamate = finite_clamp(neuro.effective(NeurochemicalId::Glutamate), 0.0, 2.0);
335        let gaba = finite_clamp(neuro.effective(NeurochemicalId::GABA), 0.0, 2.0);
336        let acetylcholine = finite_clamp(neuro.effective(NeurochemicalId::Acetylcholine), 0.0, 2.0);
337        let adenosine = finite_clamp(neuro.effective(NeurochemicalId::Adenosine), 0.0, 1.0);
338        let cortisol = finite_clamp(neuro.effective(NeurochemicalId::Cortisol), 0.0, 1.0);
339
340        let voltage = self.membrane_potential_mv;
341        let sodium_gate = sigmoid((voltage + 35.0) / 6.0);
342        let calcium_gate = sigmoid((voltage + 45.0) / 6.0);
343        let na_i = IonId::Sodium as usize;
344        let k_i = IonId::Potassium as usize;
345        let cation_reversal = NERNST_FACTOR_MV
346            * ((self.extracellular_mm[na_i] + self.extracellular_mm[k_i])
347                / (self.intracellular_mm[na_i] + self.intracellular_mm[k_i]))
348                .ln();
349        let sodium_fraction = finite_clamp(
350            (cation_reversal - self.reversal_potential_mv[k_i])
351                / (self.reversal_potential_mv[na_i] - self.reversal_potential_mv[k_i]),
352            0.0,
353            1.0,
354        );
355        let synaptic_cation_conductance =
356            (0.25 * glutamate + 0.05 * acetylcholine) * (1.0 - 0.25 * adenosine);
357
358        let target = [
359            // Voltage-gated Ca²⁺: glutamate opens NMDA/L-type-like influx,
360            // while adenosine presynaptically suppresses Ca²⁺ entry.
361            0.005 + 0.15 * glutamate * calcium_gate * (1.0 - 0.5 * adenosine),
362            // GABA-A-like Cl⁻ conductance plus a resting leak.
363            0.20 + 0.55 * gaba,
364            // K⁺ leak plus voltage-gated delayed-rectifier-like activation;
365            // adenosine/GABA modestly increase outward stabilizing current.
366            0.90 + 0.40 * sigmoid((voltage + 35.0) / 10.0)
367                + 0.08 * adenosine
368                + 0.04 * gaba
369                + synaptic_cation_conductance * (1.0 - sodium_fraction),
370            // AMPA-like glutamate drive plus voltage-gated Na⁺ recruitment.
371            (0.035 + 0.20 * sodium_gate) * (1.0 - 0.25 * adenosine)
372                + synaptic_cation_conductance * sodium_fraction,
373        ];
374        for i in 0..ION_COUNT {
375            let alpha = 1.0 - (-dt / CONDUCTANCE_TAU_S[i]).exp();
376            self.conductance[i] += (target[i] - self.conductance[i]) * alpha;
377            self.conductance[i] = finite_clamp(self.conductance[i], 0.0, 4.0);
378        }
379
380        // Na⁺/K⁺-ATPase — Luo & Rudy (1994) forward rate:
381        //   i_p = Ī · f_NaK(V, Na_o) · 1/(1 + (Km_Na/Na_i)^1.5)
382        //         · K_o/(K_o + Km_K) · ATP/(ATP + Km_ATP)
383        // Three cytosolic Na⁺ sites make the Na⁺i drive cooperative
384        // (Hill 1.5) rather than single-site Michaelis; the electrogenic
385        // forward cycle accelerates with depolarization and with
386        // extracellular Na⁺ load — the pump speeds up under exactly the
387        // ionic influx it exists to reverse. ATP gates through a
388        // saturating Michaelis term; the previous linear factor made
389        // the pump stall far earlier than its real Km allows.
390        let na_i_conc = self.intracellular_mm[na_i].max(0.01);
391        let k_o_conc = self.extracellular_mm[k_i];
392        let na_o_conc = self.extracellular_mm[na_i];
393        let na_drive = 1.0 / (1.0 + (NAK_KM_NA_I_MM / na_i_conc).powf(NAK_NA_HILL));
394        let ko_drive = k_o_conc / (k_o_conc + NAK_KM_K_O_MM);
395        let v_over_rt_f = self.membrane_potential_mv / NERNST_FACTOR_MV;
396        let sigma = ((na_o_conc / NAK_NA_O_SCALE_MM).exp() - 1.0) / 7.0;
397        let f_nak = 1.0
398            / (1.0 + 0.1245 * (-0.1 * v_over_rt_f).exp() + 0.0365 * sigma * (-v_over_rt_f).exp());
399        let atp_drive = self.atp_availability / (self.atp_availability + NAK_KM_ATP);
400        self.nak_pump_rate = finite_clamp(
401            NAK_MAX_RATE_MM_PER_S * na_drive * ko_drive * f_nak * atp_drive,
402            0.0,
403            NAK_MAX_RATE_MM_PER_S,
404        );
405
406        // Chord-conductance voltage target. The Na⁺/K⁺ pump contributes a
407        // small net outward (hyperpolarizing) current due to its 3:2 ratio.
408        let ca_i = IonId::Calcium as usize;
409        let ncx_reversal =
410            3.0 * self.reversal_potential_mv[na_i] - 2.0 * self.reversal_potential_mv[ca_i];
411        let ncx = (NCX_REVERSE_GAIN_MM_PER_S / CALCIUM_FREE_FRACTION)
412            * ((ncx_reversal - voltage) / (2.0 * NERNST_FACTOR_MV)).tanh();
413        self.ncx_flux = ncx;
414        let pmca_max = 2.0 * CA_EXTRUSION_BASE_MM_PER_S;
415        let pmca_affinity = pmca_max / CA_EXTRUSION_GAIN_PER_S;
416        let pmca = (pmca_max / CALCIUM_FREE_FRACTION)
417            * self.atp_availability
418            * self.intracellular_mm[ca_i]
419            / (self.intracellular_mm[ca_i] + pmca_affinity);
420        let pump_current = PUMP_CURRENT_EQUIV_MV * (self.nak_pump_rate / NAK_MAX_RATE_MM_PER_S)
421            - ncx / CURRENT_TO_MM_PER_S[na_i];
422        let mut conductance_sum = 0.0;
423        let mut voltage_sum = 0.0;
424        for i in 0..ION_COUNT {
425            conductance_sum += self.conductance[i];
426            voltage_sum += self.conductance[i] * self.reversal_potential_mv[i];
427        }
428        let voltage_target = if conductance_sum > 0.001 {
429            (voltage_sum - pump_current) / conductance_sum
430        } else {
431            RESTING_MEMBRANE_MV
432        };
433        let voltage_decay = dt * conductance_sum / MEMBRANE_TAU_S;
434        let voltage_alpha = -(-voltage_decay).exp_m1();
435        let mean_voltage = voltage_target
436            + (voltage - voltage_target) * voltage_alpha / voltage_decay.max(f32::MIN_POSITIVE);
437        self.membrane_potential_mv += (voltage_target - self.membrane_potential_mv) * voltage_alpha;
438        self.membrane_potential_mv = finite_clamp(self.membrane_potential_mv, -120.0, 80.0);
439
440        // Channel currents and concentration changes. Positive current is
441        // outward conventional current. For cations, outward current removes
442        // intracellular ion; for Cl⁻, outward conventional current is an
443        // inward anion flux.
444        let mut channel_flux = [0.0f32; ION_COUNT];
445        for (i, ion) in IonId::all().into_iter().enumerate() {
446            self.current_density[i] =
447                self.conductance[i] * (mean_voltage - self.reversal_potential_mv[i]);
448            let inward_flux = if ion == IonId::Chloride {
449                self.current_density[i]
450            } else {
451                -self.current_density[i]
452            } * CURRENT_TO_MM_PER_S[i];
453            channel_flux[i] = inward_flux;
454        }
455
456        // Na⁺/K⁺ pump stoichiometry.
457        let nak = self.nak_pump_rate;
458
459        // Net KCC2 transport follows ln(Ki*Cli / Ko*Clo); positive is outward.
460        // Reversed KCC2 imports K⁺ and Cl⁻, still with electroneutral 1:1 flux.
461        let cl_i = IonId::Chloride as usize;
462        let cl_drive = ((self.intracellular_mm[k_i] * self.intracellular_mm[cl_i])
463            / (self.extracellular_mm[k_i] * self.extracellular_mm[cl_i]))
464            .ln();
465        let rest_cl_drive = ((REST_INTRACELLULAR_MM[k_i] * REST_INTRACELLULAR_MM[cl_i])
466            / (REST_EXTRACELLULAR_MM[k_i] * REST_EXTRACELLULAR_MM[cl_i]))
467            .ln();
468        let cl_capacity = CL_TRANSPORT_BASE_MM_PER_S
469            + CL_TRANSPORT_GAIN_PER_S
470                * (self.intracellular_mm[cl_i] - REST_INTRACELLULAR_MM[cl_i]).max(0.0);
471        let cl_transport = finite_clamp(cl_capacity * cl_drive / rest_cl_drive, -0.08, 0.12);
472        self.kcl_cotransporter_flux = cl_transport;
473
474        // PMCA/NCX Ca²⁺ extrusion. Positive ncx_flux extrudes one Ca²⁺ and
475        // imports approximately three Na⁺. High intracellular Na⁺ weakens
476        // the forward exchange and can reverse it under sustained load.
477        let calcium_outflow = ncx + pmca;
478        let transporter_flux = [
479            -calcium_outflow,
480            -cl_transport,
481            2.0 * nak - CL_TRANSPORT_K_STOICH * cl_transport,
482            3.0 * ncx - 3.0 * nak,
483        ];
484
485        // Extracellular reservoirs and glial buffering pull concentrations
486        // back toward their physiological bath values. Slow intracellular
487        // restoration for Na⁺/K⁺ is gated by ATP availability: without energy,
488        // passive restoration is disabled and pump/transporter fluxes alone
489        // determine concentration drift. This prevents the restoration term
490        // from masking pump failure or ATP depletion.
491        let dt64 = f64::from(dt);
492        let extracellular_alpha = -(-dt64 / f64::from(EXTRACELLULAR_TAU_S)).exp_m1();
493        let intracellular_alpha = -(-dt64 / f64::from(INTRACELLULAR_TAU_S)).exp_m1();
494        let atp_factor = f64::from(self.atp_availability);
495        for (i, ion) in IonId::all().into_iter().enumerate() {
496            let flux = f64::from(channel_flux[i]) + f64::from(transporter_flux[i]);
497            let inside = f64::from(self.intracellular_mm[i]);
498            let outside = f64::from(self.extracellular_mm[i]);
499            let inside_delta = if ion == IonId::Sodium || ion == IonId::Potassium {
500                // Restoration is scaled by ATP: when ATP is low, only the
501                // explicit pump/transporter fluxes drive concentration change.
502                let restoration = (f64::from(REST_INTRACELLULAR_MM[i]) - inside
503                    + flux * f64::from(INTRACELLULAR_TAU_S))
504                    * intracellular_alpha;
505                flux * dt64 + restoration * atp_factor
506            } else {
507                flux * dt64
508                    * if ion == IonId::Calcium {
509                        f64::from(CALCIUM_FREE_FRACTION)
510                    } else {
511                        1.0
512                    }
513            };
514            let outside_delta = (f64::from(REST_EXTRACELLULAR_MM[i])
515                - outside
516                - flux * f64::from(EXTRACELLULAR_VOLUME_RATIO) * f64::from(EXTRACELLULAR_TAU_S))
517                * extracellular_alpha;
518            self.intracellular_mm[i] = (inside + inside_delta) as f32;
519            self.extracellular_mm[i] = (outside + outside_delta) as f32;
520        }
521
522        // ATP consumption counts Na⁺/K⁺-ATPase and PMCA cycles, not passive
523        // channels, NCX, or KCC2. ATP availability gates these pumps: adenosine is
524        // treated as an ATP-depletion signal, cortisol adds a smaller stress
525        // cost, and sustained ionic workload consumes capacity. Cortisol is
526        // expressed as a fraction of its reachable envelope (~0.02 rest,
527        // ~0.08 peak) — the raw level could never move this term.
528        let atp_consumption = nak + pmca;
529        let pump_capacity = NAK_MAX_RATE_MM_PER_S + pmca_max / CALCIUM_FREE_FRACTION;
530        let load_target = finite_clamp(atp_consumption / pump_capacity, 0.0, 1.0);
531        let load_alpha = 1.0 - (-dt / 5.0).exp();
532        self.energy_load += (load_target - self.energy_load) * load_alpha;
533        let atp_target = finite_clamp(
534            1.0 - 0.35 * adenosine
535                - 0.35 * self.energy_load
536                - 0.10 * crate::state::neurochemical::cortisol_stress_fraction(cortisol),
537            0.0,
538            1.0,
539        );
540        let atp_alpha = 1.0 - (-dt / 2.0).exp();
541        self.atp_availability += (atp_target - self.atp_availability) * atp_alpha;
542
543        self.recompute_derived();
544        self.sanitize();
545    }
546
547    /// Recompute reversal potentials, currents, and normalized summaries.
548    fn recompute_derived(&mut self) {
549        let mut conductance_sum = 0.0;
550        let mut gradient_sum = 0.0;
551        let mut net_current = PUMP_CURRENT_EQUIV_MV * (self.nak_pump_rate / NAK_MAX_RATE_MM_PER_S)
552            - self.ncx_flux / CURRENT_TO_MM_PER_S[IonId::Sodium as usize];
553        for (i, ion) in IonId::all().into_iter().enumerate() {
554            let inside = finite_clamp(
555                finite_or(self.intracellular_mm[i], REST_INTRACELLULAR_MM[i]),
556                MIN_INTRACELLULAR_MM[i],
557                MAX_INTRACELLULAR_MM[i],
558            );
559            let outside = finite_clamp(
560                finite_or(self.extracellular_mm[i], REST_EXTRACELLULAR_MM[i]),
561                MIN_EXTRACELLULAR_MM[i],
562                MAX_EXTRACELLULAR_MM[i],
563            );
564            self.reversal_potential_mv[i] =
565                (NERNST_FACTOR_MV / ion.valence()) * (outside / inside).ln();
566            self.current_density[i] =
567                self.conductance[i] * (self.membrane_potential_mv - self.reversal_potential_mv[i]);
568            conductance_sum += self.conductance[i];
569            net_current += self.current_density[i];
570
571            let gradient_ratio = (inside / outside).ln();
572            let rest_ratio = (REST_INTRACELLULAR_MM[i] / REST_EXTRACELLULAR_MM[i]).ln();
573            gradient_sum += finite_clamp(gradient_ratio / rest_ratio, 0.0, 1.0);
574        }
575        self.net_membrane_current = finite_clamp(net_current, -1000.0, 1000.0);
576        self.gradient_integrity = finite_clamp(gradient_sum / ION_COUNT as f32, 0.0, 1.0);
577
578        let ca_i = IonId::Calcium as usize;
579        let ca_ratio = finite_clamp(
580            self.intracellular_mm[ca_i] / REST_INTRACELLULAR_MM[ca_i],
581            1.0,
582            CA_SIGNAL_FULL_MM / REST_INTRACELLULAR_MM[ca_i],
583        );
584        self.calcium_signal = finite_clamp(
585            ca_ratio.ln() / (CA_SIGNAL_FULL_MM / REST_INTRACELLULAR_MM[ca_i]).ln(),
586            0.0,
587            1.0,
588        );
589
590        let cl_i = IonId::Chloride as usize;
591        let cl_drive = self.membrane_potential_mv - self.reversal_potential_mv[cl_i];
592        self.chloride_efficacy = finite_clamp(0.5 + cl_drive / 20.0, 0.0, 1.0);
593        self.excitability = finite_clamp(
594            (self.membrane_potential_mv - RESTING_MEMBRANE_MV) / 60.0,
595            0.0,
596            1.0,
597        );
598
599        if !conductance_sum.is_finite() {
600            self.conductance = [0.0055, 0.475, 0.945, 0.11];
601        }
602    }
603
604    /// Clamp/replace all persisted fields into physiologically bounded,
605    /// finite ranges. This is the NaN/∞ circuit breaker for state loaded
606    /// from mmap or perturbed by malformed inputs.
607    pub fn sanitize(&mut self) {
608        for i in 0..ION_COUNT {
609            self.intracellular_mm[i] = finite_clamp(
610                finite_or(self.intracellular_mm[i], REST_INTRACELLULAR_MM[i]),
611                MIN_INTRACELLULAR_MM[i],
612                MAX_INTRACELLULAR_MM[i],
613            );
614            self.extracellular_mm[i] = finite_clamp(
615                finite_or(self.extracellular_mm[i], REST_EXTRACELLULAR_MM[i]),
616                MIN_EXTRACELLULAR_MM[i],
617                MAX_EXTRACELLULAR_MM[i],
618            );
619            self.reversal_potential_mv[i] =
620                finite_clamp(self.reversal_potential_mv[i], -200.0, 250.0);
621            self.conductance[i] = finite_clamp(self.conductance[i], 0.0, 4.0);
622            self.current_density[i] = finite_clamp(self.current_density[i], -1000.0, 1000.0);
623        }
624        self.membrane_potential_mv = finite_clamp(
625            finite_or(self.membrane_potential_mv, RESTING_MEMBRANE_MV),
626            -120.0,
627            80.0,
628        );
629        self.nak_pump_rate = finite_clamp(self.nak_pump_rate, 0.0, NAK_MAX_RATE_MM_PER_S);
630        self.kcl_cotransporter_flux = finite_clamp(self.kcl_cotransporter_flux, -0.08, 0.12);
631        self.ncx_flux = finite_clamp(
632            self.ncx_flux,
633            -NCX_REVERSE_GAIN_MM_PER_S / CALCIUM_FREE_FRACTION,
634            NCX_REVERSE_GAIN_MM_PER_S / CALCIUM_FREE_FRACTION,
635        );
636        self.atp_availability = finite_clamp(self.atp_availability, 0.0, 1.0);
637        self.calcium_signal = finite_clamp(self.calcium_signal, 0.0, 1.0);
638        self.chloride_efficacy = finite_clamp(self.chloride_efficacy, 0.0, 1.0);
639        self.excitability = finite_clamp(self.excitability, 0.0, 1.0);
640        self.gradient_integrity = finite_clamp(self.gradient_integrity, 0.0, 1.0);
641        self.energy_load = finite_clamp(self.energy_load, 0.0, 1.0);
642        self.net_membrane_current = finite_clamp(self.net_membrane_current, -1000.0, 1000.0);
643    }
644}

653/// Numerically stable logistic function used for bounded pump modulation.
654#[inline]
655fn sigmoid(x: f32) -> f32 {
656    if x > 20.0 {
657        1.0
658    } else if x < -20.0 {
659        0.0
660    } else {
661        1.0 / (1.0 + (-x).exp())
662    }
663}

698    /// imposed voltage from relaxing before the rate is read.
699    fn pump_rate_with_dt(dt: f32, f: impl Fn(&mut IonState)) -> f32 {
700        let neuro = NeurochemicalVector::new(0);
701        let mut state = IonState::new();
702        f(&mut state);
703        state.tick(&neuro, dt);
704        state.nak_pump_rate
705    }

707    /// Intracellular Na⁺ load accelerates the pump — the cooperative
708    /// Hill drive exists precisely to meet the influx it reverses.
709    #[test]
710    fn nak_pump_accelerates_with_sodium_load() {
711        let rest = pump_rate_with(|_| {});
712        let loaded = pump_rate_with(|s| {
713            s.intracellular_mm[IonId::Sodium as usize] = 40.0;
714        });
715        assert!(
716            loaded > rest * 1.2,
717            "Na_i 40 mM should speed the pump: rest {rest} vs loaded {loaded}"
718        );
719    }
720
721    /// Depolarization raises the electrogenic forward rate (Luo–Rudy
722    /// `f_NaK`), so the pump runs faster at −20 mV than at rest.
723    /// Measured at 1 ms: the 25 ms membrane time constant would relax
724    /// an imposed voltage back toward rest inside a full tick.
725    #[test]
726    fn nak_pump_accelerates_on_depolarization() {
727        let rest = pump_rate_with_dt(0.001, |_| {});
728        let depol = pump_rate_with_dt(0.001, |s| s.membrane_potential_mv = -20.0);
729        assert!(
730            depol > rest * 1.2,
731            "depolarized pump should run faster: rest {rest} vs depol {depol}"
732        );
733    }

735    /// The dynamic chain end-to-end: elevated glutamate opens Na⁺
736    /// conductance, the chord equation depolarizes the membrane, and
737    /// the voltage factor accelerates the pump — driven by the model's
738    /// own trajectory, not an imposed field.
739    #[test]
740    fn nak_pump_follows_glutamate_driven_depolarization() {
741        use crate::state::neurochemical::NeuroTickParams;
742
743        let neuro_rest = NeurochemicalVector::new(0);
744        let mut rest_state = IonState::new();
745        rest_state.tick(&neuro_rest, 0.1);
746        let rest = rest_state.nak_pump_rate;
747
748        let mut neuro_glu = NeurochemicalVector::new(0);
749        neuro_glu.apply_impulse(NeurochemicalId::Glutamate, 2.0, 0);
750        // Impulses touch level/velocity directly; one neuro tick
751        // refreshes effective levels so the ion layer sees the drive.
752        neuro_glu.tick_with_params(&NeuroTickParams::DEFAULT);
753        let mut glu_state = IonState::new();
754        glu_state.tick(&neuro_glu, 0.3);
755
756        // The K⁺-bucket split of the synaptic cation conductance
757        // opposes part of the depolarization, so the achieved ΔV is
758        // modest — assert the chain, not an arbitrary magnitude:
759        // the membrane depolarized and the pump tracked its live
760        // voltage rather than a static resting factor.
761        assert!(
762            glu_state.membrane_potential_mv > rest_state.membrane_potential_mv + 1.0,
763            "glutamate drive should depolarize: rest {:.1} mV vs driven {:.1} mV",
764            rest_state.membrane_potential_mv,
765            glu_state.membrane_potential_mv
766        );
767        assert!(
768            glu_state.nak_pump_rate > rest,
769            "depolarized pump should outrun rest: {rest} vs {}",
770            glu_state.nak_pump_rate
771        );
772    }

774    /// ATP gates the pump hyperbolically: severe depletion slows but
