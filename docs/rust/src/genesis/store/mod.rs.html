1//! Storage layer — persistence for Genesis's state and memories.
2//!
3//! This module provides the memory-mapped file layer that backs the
4//! core state, and (in future) the short-term and long-term memory
5//! stores.
6//!
7//! ## Design principles
8//!
9//! - **Memory-mapped, not heap-allocated**: on a 4.7 GB RAM machine,
10//!   we let the OS manage paging. State lives on disk, mapped into
11//!   the process's address space.
12//! - **Zero-copy reads**: readers access the mmap'd memory directly
13//!   through volatile copies.
14//! - **Crash-safe**: the state file has magic bytes, a schema version,
15//!   and a CRC32 checksum. Corruption is detected on open.
16//! - **No external services**: no database daemon, no server process.
17//!   Just files on disk, mapped into memory.

24// ─────────────────────────────────────────────────────────────────
25//  Page-alignment utility
26// ─────────────────────────────────────────────────────────────────

28/// Return the OS page size in bytes.
29///
30/// Uses `sysconf(_SC_PAGESIZE)` at runtime to support architectures
31/// with non-4K pages (e.g., aarch64 with 64K pages). Falls back to
32/// 4096 if `sysconf` fails, which is the minimum page size on any
33/// Linux architecture and therefore always safe.
34pub fn page_size() -> usize {
35    // SAFETY: `sysconf` is async-signal-safe and has no side effects.
36    // `_SC_PAGESIZE` is a valid configuration parameter.
37    let ps = unsafe { libc::sysconf(libc::_SC_PAGESIZE) };
38    if ps > 0 { ps as usize } else { 4096 }
39}
40
41/// Round `len` up to the next multiple of the OS page size.
42///
43/// When `mmap` is called with a length that is not a multiple of the
44/// page size, the kernel maps enough pages to cover the requested
45/// length, but the tail of the last page (beyond `len`) is not backed
46/// by the file. Accessing those tail bytes causes `SIGBUS`. By
47/// `ftruncate`-ing the file to the page-aligned length and `mmap`-ing
48/// the same, the entire mapping is backed by the file, eliminating
49/// this hazard.
50pub fn page_align_up(len: usize) -> usize {
51    let ps = page_size();
52    // If len is already aligned, return as-is. Otherwise round up.
53    // This is `len.div_ceil(ps) * ps` but without the nightly
54    // `div_ceil` method.
55    let remainder = len % ps;
56    if remainder == 0 {
57        len
58    } else {
59        // Guard against overflow: `len + (ps - remainder)` can
60        // overflow if `len` is near `usize::MAX`. In that case,
61        // the rounded-up value would wrap to a small number,
62        // causing `ftruncate` to create a tiny file and `mmap` to
63        // produce a mapping that `SIGBUS`s on access. Return `len`
64        // unchanged — it's already absurdly large and the caller
65        // will likely fail on the subsequent `ftruncate`/`mmap`.
66        match len.checked_add(ps - remainder) {
67            Some(aligned) => aligned,
68            None => len,
69        }
70    }
71}
