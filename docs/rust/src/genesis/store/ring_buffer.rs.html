1//! Short-term memory (STM) ring buffer.
2//!
3//! This is Genesis's "working memory" — a fixed-capacity ring buffer
4//! of recent events that lives in its own mmap'd file. Events enter
5//! the buffer as they happen, and the subcognitive daemon consolidates
6//! important ones into long-term memory during idle periods.
7//!
8//! # What makes this different from a normal ring buffer
9//!
10//! Each entry carries an **emotional tag** — a snapshot of the
11//! neurochemical state at the time the event was recorded. This
12//! means the consolidation process can prioritise emotionally
13//! significant events (high cortisol = threatening, high dopamine =
14//! rewarding) without re-reading the full neurochemical history.
15//!
16//! The emotional tag also includes a **salience score** — a single
17//! float that summarises how important this event is. It's computed
18//! at insertion time from the neurochemical state, so the
19//! consolidation process can simply sort by salience rather than
20//! re-evaluating each event's importance.
21//!
22//! # Layout
23//!
24//! The file is a header followed by a fixed array of slots:
25//!
26//! ```text
27//! ┌─────────────────────────────────┐
28//! │       RingBufferHeader          │  64 bytes
29//! ├─────────────────────────────────┤
30//! │  Slot 0: RingBufferEntry        │  256 bytes
31//! │  Slot 1: RingBufferEntry        │  256 bytes
32//! │  ...                            │
33//! │  Slot N-1: RingBufferEntry      │  256 bytes
34//! └─────────────────────────────────┘
35//! ```
36

70/// Header for the STM ring buffer file.
71///
72/// ## Layout (64 bytes)
73#[repr(C)]
74#[derive(Clone, Copy, Debug)]
75pub struct RingBufferHeader {
76    /// Magic bytes identifying the STM file (`STM_MAGIC`).
77    pub magic: [u8; 4],
78    /// Schema version of the ring buffer file.
79    pub version: u32,
80    /// Number of slots in the ring buffer.
81    pub capacity: u32,
82    /// Alignment padding.
83    pub _pad0: u32,
84    /// Next write index (wraps around at `capacity`).
85    pub head: u64,
86    /// Oldest unread index.
87    pub tail: u64,
88    /// Number of entries currently in the buffer.
89    pub count: u64,
90    /// Total entries ever written (monotonic, never wraps).
91    pub total_written: u64,
92    /// Reserved bytes for future expansion.
93    pub _reserved: [u8; 16],
94}
95
96impl RingBufferHeader {
97    /// Create a fresh, empty header for a buffer of the given capacity.
98    pub fn new(capacity: u32) -> Self {
99        Self {
100            magic: STM_MAGIC,
101            version: STM_SCHEMA_VERSION,
102            capacity,
103            _pad0: 0,
104            head: 0,
105            tail: 0,
106            count: 0,
107            total_written: 0,
108            _reserved: [0; 16],
109        }
110    }
111
112    /// Returns `true` if the magic bytes match `STM_MAGIC`.
113    pub fn verify_magic(&self) -> bool {
114        self.magic == STM_MAGIC
115    }
116}

126/// The type of event recorded in the ring buffer.
127#[repr(u8)]
128#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
129pub enum EventType {
130    /// User interaction (message, command, etc.)
131    UserInput = 0,
132    /// Genesis's own output (response, action, etc.)
133    Output = 1,
134    /// Internal event (module state change, error, etc.)
135    Internal = 2,
136    /// Observation from the environment (file change, sensor, etc.)
137    Observation = 3,
138    /// A memory was recalled from long-term storage.
139    MemoryRecall = 4,
140    /// A new memory was consolidated to long-term storage.
141    MemoryConsolidated = 5,
142    /// An error occurred.
143    Error = 6,
144    /// A neurochemical shift event.
145    NeurochemicalShift = 7,
146}
147
148impl EventType {
149    /// Returns a short kebab-case label for this event type.
150    pub const fn label(self) -> &'static str {
151        match self {
152            Self::UserInput => "user-input",
153            Self::Output => "output",
154            Self::Internal => "internal",
155            Self::Observation => "observation",
156            Self::MemoryRecall => "memory-recall",
157            Self::MemoryConsolidated => "memory-consolidated",
158            Self::Error => "error",
159            Self::NeurochemicalShift => "neurochemical-shift",
160        }
161    }
162
163    /// Convert a raw `u8` into an `EventType`, defaulting to
164    /// [`Internal`](Self::Internal) for unknown values.
165    pub fn from_u8(v: u8) -> Self {
166        match v {
167            0 => Self::UserInput,
168            1 => Self::Output,
169            2 => Self::Internal,
170            3 => Self::Observation,
171            4 => Self::MemoryRecall,
172            5 => Self::MemoryConsolidated,
173            6 => Self::Error,
174            7 => Self::NeurochemicalShift,
175            _ => Self::Internal,
176        }
177    }
178}

180/// A single entry in the STM ring buffer.
181///
182/// ## Layout (256 bytes)
183/// ```text
184/// offset  field              type       notes
185/// ------  -----              ----       -----
186///   0     timestamp          u64        ms since epoch
187///   8     event_type         u8         EventType discriminant
188///   9     source_module      u8         ModuleId that produced this
189///  10     flags              u8         bit 0: consolidated to LTM
190///  11     _pad               u8         alignment
191///  12     salience           f32        finite evidence value
192///  16     emotional_tag      [f32;12]   effective neurochemical levels
193///  64     text_len           u16        length of text payload
194///  66     _pad2              [u8;2]     alignment
195///  68     text               [u8;188]   event description (UTF-8)
196/// ```
197#[repr(C)]
198#[derive(Clone, Copy, Debug)]
199pub struct RingBufferEntry {
200    /// Milliseconds since the Unix epoch when the entry was recorded.
201    pub timestamp: u64,
202    /// `EventType` discriminant (stored as `u8`).
203    pub event_type: u8,
204    /// `ModuleId` of the module that produced this entry.
205    pub source_module: u8,
206    /// Bit 0: entry has been consolidated to LTM (don't re-consolidate).
207    /// Other bits reserved for future use.
208    pub flags: u8,
209    /// Alignment padding.
210    pub _pad: u8,
211    /// Finite evidence recorded by the producing module.
212    /// Higher = more important. Used by consolidation to prioritise.
213    pub salience: f32,
214    /// Snapshot of effective neurochemical levels at insertion time.
215    ///
216    /// Order matches the 12-element emotional tag on the IPC wire (see
217    /// `ipc.rs` PUSH_EVENT), **not** [`NeurochemicalId::all()`](crate::state::neurochemical::NeurochemicalId::all) — the
218    /// six v3 chemicals are not carried here.
219    /// `consolidation::expand_emotional_tag` maps this into the full
220    /// 18-chemical LTM tag.
221    pub emotional_tag: [f32; EMOTIONAL_TAG_SIZE],
222    /// Length of the UTF-8 text payload (bytes).
223    pub text_len: u16,
224    /// Alignment padding.
225    pub _pad2: [u8; 2],
226    /// Fixed-size UTF-8 text payload (truncated to `ENTRY_TEXT_SIZE`).
227    pub text: [u8; ENTRY_TEXT_SIZE],
228}
229
230impl RingBufferEntry {
231    /// Flag bit: entry has been consolidated to LTM.
232    pub const CONSOLIDATED: u8 = 1 << 0;
233
234    /// Create a new entry with the given text and emotional tag.
235    pub fn new(
236        timestamp: u64,
237        event_type: EventType,
238        source_module: u8,
239        salience: f32,
240        emotional_tag: [f32; EMOTIONAL_TAG_SIZE],
241        text: &str,
242    ) -> Self {
243        let mut text_buf = [0u8; ENTRY_TEXT_SIZE];
244        let bytes = text.as_bytes();
245        // Truncate on a UTF-8 char boundary. Cutting mid-character
246        // would make `text()`'s `from_utf8` fail and drop the entire
247        // payload, not just the tail.
248        let mut len = bytes.len().min(ENTRY_TEXT_SIZE);
249        while len > 0 && !text.is_char_boundary(len) {
250            len -= 1;
251        }
252        text_buf[..len].copy_from_slice(&bytes[..len]);
253
254        Self {
255            timestamp,
256            event_type: event_type as u8,
257            source_module,
258            flags: 0,
259            _pad: 0,
260            salience: crate::state::sanitize::finite_or(salience, 0.0),
261            // Sanitise the tag here rather than relying on every call
262            // site: the value is persisted, and a NaN in it silently
263            // poisons the SDR/SimHash encoding used by LTM associative
264            // retrieval, with no error anywhere downstream.
265            emotional_tag: emotional_tag.map(|v| crate::state::sanitize::finite_or(v, 0.0)),
266            text_len: len as u16,
267            _pad2: [0; 2],
268            text: text_buf,
269        }
270    }
271
272    /// Whether this entry has been consolidated to LTM.
273    pub fn is_consolidated(&self) -> bool {
274        self.flags & Self::CONSOLIDATED != 0
275    }
276
277    /// Mark this entry as consolidated to LTM.
278    pub fn mark_consolidated(&mut self) {
279        self.flags |= Self::CONSOLIDATED;
280    }
281
282    /// Get the event type as the enum.
283    pub fn event_type(&self) -> EventType {
284        EventType::from_u8(self.event_type)
285    }
286
287    /// Get the text payload as a string (lossy UTF-8).
288    pub fn text(&self) -> &str {
289        let len = self.text_len as usize;
290        if len > ENTRY_TEXT_SIZE {
291            return "";
292        }
293        std::str::from_utf8(&self.text[..len]).unwrap_or("")
294    }
295}

298    assert!(core::mem::size_of::<RingBufferEntry>() == ENTRY_SIZE);

305/// Errors for the STM ring buffer.
306#[derive(Debug)]
307pub enum RingBufferError {
308    /// An underlying I/O error from the filesystem.
309    Io(std::io::Error),
310    /// The STM file was not found.
311    NotFound,
312    /// The file's magic bytes did not match.
313    BadMagic,
314    /// The file's schema version did not match the expected one.
315    VersionMismatch {
316        /// The version the code expected.
317        expected: u32,
318        /// The version found in the file.
319        found: u32,
320    },
321    /// Memory-mapping the file failed.
322    MmapFailed,
323    /// `msync` failed while flushing mapped pages to disk.
324    MsyncFailed,
325    /// `fdatasync` failed while flushing the file descriptor.
326    FsyncFailed,
327    /// `ftruncate` failed while sizing the file.
328    FtruncateFailed,
329    /// The backing file is smaller than the required size.
330    FileTooSmall {
331        /// The expected file size.
332        expected: u64,
333        /// The actual file size found.
334        found: u64,
335    },
336    /// The requested capacity was zero or otherwise invalid.
337    InvalidCapacity {
338        /// The invalid capacity value found.
339        found: u32,
340    },
341    /// The requested capacity exceeds the maximum allowed.
342    CapacityTooLarge {
343        /// The capacity value found.
344        found: u32,
345        /// The maximum allowed capacity.
346        max: u32,
347    },
348    /// The header's cursor fields are outside the buffer's own bounds.
349    ///
350    /// `head` and `tail` are full `u64`s used directly to compute slot
351    /// offsets, so a value ≥ `capacity` would address outside the
352    /// mapping. The header carries no integrity check of its own, so
353    /// these fields are untrusted on open.
354    CorruptHeader {
355        /// The out-of-range `head` found.
356        head: u64,
357        /// The out-of-range `tail` found.
358        tail: u64,
359        /// The out-of-range `count` found.
360        count: u64,
361        /// The capacity the other fields were checked against.
362        capacity: u32,
363    },
364    /// The sole-mapper flock could not be acquired: another process
365    /// holds a live mapping of this file. Returned instead of blocking
366    /// so the caller can retry or degrade rather than silently racing
367    /// a second writer into the same `MAP_SHARED` pages.
368    LockBusy,
369}
370
371impl std::fmt::Display for RingBufferError {
372    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
373        match self {
374            Self::Io(e) => write!(f, "I/O error: {e}"),
375            Self::NotFound => write!(f, "STM file not found"),
376            Self::BadMagic => write!(f, "bad magic bytes"),
377            Self::VersionMismatch { expected, found } => {
378                write!(f, "version mismatch: expected {expected}, found {found}")
379            }
380            Self::MmapFailed => write!(f, "mmap failed"),
381            Self::MsyncFailed => write!(f, "msync failed"),
382            Self::FsyncFailed => write!(f, "fdatasync failed"),
383            Self::FtruncateFailed => write!(f, "ftruncate failed"),
384            Self::FileTooSmall { expected, found } => {
385                write!(f, "file too small: expected {expected}, found {found}")
386            }
387            Self::InvalidCapacity { found } => {
388                write!(f, "invalid capacity: {found} (must be > 0)")
389            }
390            Self::CapacityTooLarge { found, max } => {
391                write!(f, "capacity {found} exceeds maximum {max}")
392            }
393            Self::CorruptHeader {
394                head,
395                tail,
396                count,
397                capacity,
398            } => {
399                write!(
400                    f,
401                    "corrupt STM header: head={head} tail={tail} count={count} \
402                     outside capacity {capacity}"
403                )
404            }
405            Self::LockBusy => write!(
406                f,
407                "STM file is locked by another process (sole-mapper invariant)"
408            ),
409        }
410    }
411}

415/// Compute the file size for a ring buffer with the given capacity.
416///
417/// This is the **logical** size — the actual data occupies exactly
418/// `header_size + capacity * ENTRY_SIZE` bytes. The on-disk file is
419/// page-aligned (see [`mmap_size`]) so the entire `mmap` mapping is
420/// backed by the file.
421pub fn file_size(capacity: u32) -> usize {
422    core::mem::size_of::<RingBufferHeader>() + capacity as usize * ENTRY_SIZE
423}
424
425/// Compute the page-aligned mmap size for a ring buffer with the given
426/// capacity.
427///
428/// `ftruncate` and `mmap` use this size so the entire mapping is
429/// backed by the file, eliminating the risk of `SIGBUS` when accessing
430/// the tail of the last page. Data access is still bounded by
431/// [`file_size`] (the logical size).
432pub fn mmap_size(capacity: u32) -> usize {
433    crate::store::page_align_up(file_size(capacity))
434}

436/// A memory-mapped STM ring buffer.
437///
438/// Owns the mmap'd region and file descriptor. When dropped, unmaps
439/// and closes.
440pub struct RingBuffer {
441    ptr: *mut u8,
442    fd: i32,
443    capacity: u32,
444    file_len: usize,
445    /// Coarse lock protecting all access to the mmap'd region.
446    ///
447    /// The ring buffer is shared between the IPC thread (which writes
448    /// new events) and the daemon's reactive handler (which iterates
449    /// and marks entries as consolidated). This mutex makes every
450    /// public operation
451    /// atomic with respect to other threads, removing the previous
452    /// data race on the header and slot array.
453    ///
454    /// A poisoned lock means an earlier writer panicked and may have
455    /// left the mmap in an inconsistent state. All public methods
456    /// `.expect("access_lock poisoned: mmap may be inconsistent")` the guard and intentionally panic the daemon in
457    /// that case — crashing is safer than continuing with a torn
458    /// in-memory event log.
459    access_lock: crate::inhibition::Inhibition<()>,
460}
461
462// SAFETY (Send/Sync): all access to the mapped region goes through
463// `access_lock`, and every public method returns owned copies rather
464// than references into `MAP_SHARED` memory. Cross-process exclusion
465// is enforced at open/create time by the sole-mapper `flock` — the
466// lock is held for the life of the fd and released when Drop closes
467// it, so no second process can map the file while a RingBuffer lives.
468unsafe impl Send for RingBuffer {}
469unsafe impl Sync for RingBuffer {}
470
471/// Maximum capacity of the ring buffer.
472///
473/// Two independent limits apply, and this is the smaller of them:
474///
475/// 1. `capacity * ENTRY_SIZE` must fit in a `usize`, so the slot array
476///    is addressable at all. On a 32-bit target this binds (the 2^24
477///    cap below is reached at exactly 2^32 bytes).
478/// 2. A 4 GB ceiling, so a corrupt or absurd caller-supplied value
479///    cannot request an enormous mapping.
480pub const MAX_CAPACITY: u32 = {
481    // `usize::MAX / ENTRY_SIZE` saturates to `u64::MAX` on 64-bit, so
482    // compare in u64 before narrowing back to u32.
483    let by_address_space =
484        ((usize::MAX - core::mem::size_of::<RingBufferHeader>()) / ENTRY_SIZE) as u64;
485    let ceiling = 1u64 << 24;
486    if by_address_space > ceiling {
487        ceiling as u32
488    } else {
489        by_address_space as u32
490    }
491};

493impl RingBuffer {
494    /// Create a new ring buffer file with the given capacity.
495    pub fn create(path: impl AsRef<Path>, capacity: u32) -> Result<Self, RingBufferError> {
496        Self::create_inner(path, capacity, false)
497    }
498
499    /// Internal create with an `exclusive` flag. When `exclusive` is
500    /// true, uses `create_new` (fails with `EEXIST` if the file already
501    /// exists) instead of `create` + `truncate`. This is used by
502    /// `open_or_create` to avoid the TOCTOU race of `exists()` + `create`.
503    fn create_inner(
504        path: impl AsRef<Path>,
505        capacity: u32,
506        exclusive: bool,
507    ) -> Result<Self, RingBufferError> {
508        if capacity == 0 {
509            return Err(RingBufferError::InvalidCapacity { found: 0 });
510        }
511        if capacity > MAX_CAPACITY {
512            return Err(RingBufferError::CapacityTooLarge {
513                found: capacity,
514                max: MAX_CAPACITY,
515            });
516        }
517        let path = path.as_ref();
518        let mmap_len = mmap_size(capacity);
519
520        // No `truncate` on open: the sole-mapper flock must be held
521        // *before* the file is resized, or a racing create would
522        // truncate the file under another process's live mapping.
523        // `ftruncate(mmap_len)` below performs the sizing once the
524        // lock is held.
525        let file = std::fs::OpenOptions::new()
526            .read(true)
527            .write(true)
528            .create(!exclusive)
529            .create_new(exclusive)
530            .open(path)
531            .map_err(RingBufferError::Io)?;
532
533        let fd = file.as_raw_fd();
534        Self::acquire_sole_mapper_lock(fd)?;
535
536        // Set the file size to the page-aligned mmap length so the
537        // entire mapping is backed by the file. Without this, the tail
538        // of the last page would be unbacked and touching it would
539        // cause SIGBUS.
540        // SAFETY: ftruncate is a POSIX call on a valid, owned fd. The fd is
541        // obtained from `file.as_raw_fd()` and `file` is kept alive (forgotten
542        // later) so the fd remains valid. `mmap_len` is the page-aligned
543        // file size, bounded by the requested capacity.
544        unsafe {
545            if ftruncate(fd, mmap_len as i64) != 0 {
546                return Err(RingBufferError::FtruncateFailed);
547            }
548        }
549        // Persist the file size metadata so a crash doesn't leave a
550        // zero-length or truncated file. `msync` only flushes data
551        // pages, not the inode's i_size.
552        // SAFETY: `fd` is a valid open file descriptor.
553        unsafe {
554            if fdatasync(fd) != 0 {
555                return Err(RingBufferError::FsyncFailed);
556            }
557        }
558
559        let ptr = Self::do_mmap(fd, mmap_len)?;
560        std::mem::forget(file);
561
562        // Initialise the header
563        let header = RingBufferHeader::new(capacity);
564        // SAFETY: `ptr` was just returned by mmap with PROT_WRITE.
565        // The region is exclusively owned (no other references exist).
566        unsafe {
567            std::ptr::write_volatile(ptr as *mut RingBufferHeader, header);
568        }
569
570        // Zero out the entry slots
571        // SAFETY: `entries_start` is within the mmap'd region
572        // (header size + capacity * ENTRY_SIZE ≤ logical_len ≤ mmap_len).
573        unsafe {
574            let entries_start = ptr.add(core::mem::size_of::<RingBufferHeader>());
575            std::ptr::write_bytes(entries_start, 0, capacity as usize * ENTRY_SIZE);
576        }
577
578        // `file` was forgotten above, so `fd` and `ptr` are ours alone
579        // from here: a failed flush must release both rather than leak
580        // them.
581        if let Err(e) = Self::do_msync(ptr, mmap_len) {
582            // SAFETY: `ptr` is a live mapping of `mmap_len` bytes and
583            // `fd` is live, both exclusively owned.
584            unsafe { Self::discard_mapping(ptr, mmap_len, fd) };
585            return Err(e);
586        }
587
588        Ok(Self {
589            ptr,
590            fd,
591            capacity,
592            file_len: mmap_len,
593            access_lock: crate::inhibition::Inhibition::new(()),
594        })
595    }
596
597    /// Open an existing ring buffer file.
598    pub fn open(path: impl AsRef<Path>) -> Result<Self, RingBufferError> {
599        let path = path.as_ref();
600
601        if !path.exists() {
602            return Err(RingBufferError::NotFound);
603        }
604
605        let file = std::fs::OpenOptions::new()
606            .read(true)
607            .write(true)
608            .open(path)
609            .map_err(RingBufferError::Io)?;
610
611        let fd = file.as_raw_fd();
612        Self::acquire_sole_mapper_lock(fd)?;
613        let metadata = file.metadata().map_err(RingBufferError::Io)?;
614
615        // Read the header to get capacity
616        if metadata.len() < core::mem::size_of::<RingBufferHeader>() as u64 {
617            return Err(RingBufferError::FileTooSmall {
618                expected: core::mem::size_of::<RingBufferHeader>() as u64,
619                found: metadata.len(),
620            });
621        }
622
623        let len = metadata.len() as usize;
624        let ptr = Self::do_mmap(fd, len)?;
625        std::mem::forget(file);
626
627        // Verify header.
628        //
629        // `RingBufferHeader` is `Copy` and has no implicit padding, so
630        // reading it *by value* yields an owned snapshot. That matters:
631        // the remapping path below `munmap`s this mapping, and the
632        // capacity handed back to the caller must be read from our own
633        // copy, not from unmapped memory.
634        // SAFETY: `ptr` is a valid mmap'd region of `len` bytes,
635        // len ≥ size_of::<RingBufferHeader>() (checked above).
636        let header: RingBufferHeader = unsafe { *(ptr as *const RingBufferHeader) };
637        if !header.verify_magic() {
638            // SAFETY: `ptr` and `fd` are valid and exclusively owned.
639            unsafe { Self::discard_mapping(ptr, len, fd) };
640            return Err(RingBufferError::BadMagic);
641        }
642        if header.version != STM_SCHEMA_VERSION {
643            // SAFETY: `ptr` and `fd` are valid and exclusively owned.
644            unsafe { Self::discard_mapping(ptr, len, fd) };
645            return Err(RingBufferError::VersionMismatch {
646                expected: STM_SCHEMA_VERSION,
647                found: header.version,
648            });
649        }
650        if header.capacity == 0 {
651            // SAFETY: `ptr` and `fd` are valid and exclusively owned.
652            unsafe { Self::discard_mapping(ptr, len, fd) };
653            return Err(RingBufferError::InvalidCapacity { found: 0 });
654        }
655        if header.capacity > MAX_CAPACITY {
656            // SAFETY: `ptr` and `fd` are valid and exclusively owned.
657            unsafe { Self::discard_mapping(ptr, len, fd) };
658            return Err(RingBufferError::CapacityTooLarge {
659                found: header.capacity,
660                max: MAX_CAPACITY,
661            });
662        }
663        let capacity = header.capacity;
664
665        // Validate the cursor fields against the capacity.
666        //
667        // The header has no integrity check of its own (unlike the core
668        // state, which is CRC-protected), so `head`, `tail` and `count`
669        // are untrusted: bit-rot in the first page, a foreign/older
670        // writer, or a page-tearing writeback can set them to
671        // anything. `head` and `tail` are full `u64`s that are used
672        // directly to compute slot offsets, so an out-of-range value
673        // would read or write outside the mapping. `push` in
674        // particular trusts `head` — this check is what makes its
675        // SAFETY argument true.
676        if header.head >= capacity as u64
677            || header.tail >= capacity as u64
678            || header.count > capacity as u64
679        {
680            // SAFETY: `ptr` and `fd` are valid and exclusively owned.
681            unsafe { Self::discard_mapping(ptr, len, fd) };
682            return Err(RingBufferError::CorruptHeader {
683                head: header.head,
684                tail: header.tail,
685                count: header.count,
686                capacity,
687            });
688        }
689
690        // Verify the file is large enough to hold all `capacity` entries.
691        // Without this, a truncated file could pass the header check but
692        // cause out-of-bounds access when `push` or `iter` compute
693        // `entry_ptr(index)` beyond the actual file length.
694        let expected = file_size(capacity);
695        if len < expected {
696            // SAFETY: `ptr` and `fd` are valid and exclusively owned.
697            unsafe { Self::discard_mapping(ptr, len, fd) };
698            return Err(RingBufferError::FileTooSmall {
699                expected: expected as u64,
700                found: len as u64,
701            });
702        }
703
704        // Ensure the file is page-aligned so the entire mmap mapping is
705        // backed by the file. Files created by older versions of this
706        // code may have a non-page-aligned size. We extend the file to
707        // the page-aligned mmap size and re-mmap if necessary.
708        let needed_mmap_len = mmap_size(capacity);
709        let (final_ptr, final_mmap_len) = if len < needed_mmap_len {
710            // The file is smaller than the page-aligned size. We need
711            // to unmap, ftruncate, fsync, and re-mmap.
712            // SAFETY: `ptr` and `fd` are valid and exclusively owned.
713            // `header` is an owned copy, so nothing refers to the
714            // mapping after this point.
715            unsafe {
716                munmap(ptr as *mut c_void, len);
717            }
718            // SAFETY: `fd` is a valid open file descriptor with write
719            // permissions. ftruncate extends the file to
720            // `needed_mmap_len`, zero-filling the new bytes.
721            unsafe {
722                if ftruncate(fd, needed_mmap_len as i64) != 0 {
723                    close(fd);
724                    return Err(RingBufferError::FtruncateFailed);
725                }
726            }
727            // SAFETY: `fd` is a valid open file descriptor.
728            unsafe {
729                if fdatasync(fd) != 0 {
730                    close(fd);
731                    return Err(RingBufferError::FsyncFailed);
732                }
733            }
734            // Re-mmap with the page-aligned size. On failure the mapping
735            // is already gone, so only `fd` is left to release.
736            let new_ptr = match Self::do_mmap(fd, needed_mmap_len) {
737                Ok(p) => p,
738                Err(e) => {
739                    // SAFETY: `fd` is valid and no mapping is live.
740                    unsafe { close(fd) };
741                    return Err(e);
742                }
743            };
744            (new_ptr, needed_mmap_len)
745        } else {
746            // File is already large enough (page-aligned or larger).
747            // Use the existing mapping.
748            (ptr, len)
749        };
750
751        Ok(Self {
752            ptr: final_ptr,
753            fd,
754            capacity,
755            file_len: final_mmap_len,
756            access_lock: crate::inhibition::Inhibition::new(()),
757        })
758    }
759
760    /// Open or create.
761    ///
762    /// Tries `create_new` first (which fails with `EEXIST` if the file
763    /// already exists), then falls back to `open`. This avoids the
764    /// TOCTOU race of checking `exists()` then calling `create()` —
765    /// between the check and the create, another process could create
766    /// the file, and `create` (with truncate) would destroy it.
767    pub fn open_or_create(path: impl AsRef<Path>, capacity: u32) -> Result<Self, RingBufferError> {
768        match Self::create_inner(path.as_ref(), capacity, true) {
769            Ok(rb) => Ok(rb),
770            Err(RingBufferError::Io(ref e)) if e.raw_os_error() == Some(EEXIST) => Self::open(path),
771            Err(e) => Err(e),
772        }
773    }

801    /// Read the entry at the given slot index (without removing).
802    pub fn peek(&self, index: u64) -> Option<RingBufferEntry> {
803        let _guard = self
804            .access_lock
805            .lock()
806            .expect("access_lock poisoned: mmap may be inconsistent");
807        // SAFETY: `self.ptr` is a valid mmap'd region of at least
808        // `size_of::<RingBufferHeader>()` bytes, validated on open/create.
809        let header = unsafe { &*self.header_ptr() };
810        if index >= header.capacity as u64 {
811            return None;
812        }
813        // SAFETY: `index` < `header.capacity` (checked above), so
814        // `entry_ptr(index)` points within the mmap'd entry slot array.
815        Some(unsafe { *self.entry_ptr(index) })
816    }

818    /// Mark the entry at `index` as consolidated, but only if the entry
819    /// currently occupying that slot is the same one we consolidated.
820    ///
821    /// This prevents the race where a concurrent `push()` wraps the ring
822    /// buffer and overwrites the slot between the snapshot (from
823    /// `iter_with_slots()`, which releases the lock) and this mark. The
824    /// `expected_timestamp` identifies the entry we actually promoted to
825    /// LTM; if the slot now holds a different entry, we leave it
826    /// unconsolidated so it can be promoted on a future tick.
827    ///
828    /// Returns `true` if the entry was matched and marked, `false` if the
829    /// slot was overwritten by a newer entry (race lost) or the index is
830    /// out of bounds.
831    pub fn mark_consolidated_if_match(&self, index: u64, expected_timestamp: u64) -> bool {
832        let _guard = self
833            .access_lock
834            .lock()
835            .expect("access_lock poisoned: mmap may be inconsistent");
836        // SAFETY: `self.ptr` is a valid mmap'd region of at least
837        // `size_of::<RingBufferHeader>()` bytes, validated on open/create.
838        let header = unsafe { &*self.header_ptr() };
839        if index >= header.capacity as u64 {
840            return false;
841        }
842        // SAFETY: `index` < `header.capacity` (checked above), so
843        // `entry_ptr(index)` points within the mmap'd entry slot array.
844        unsafe {
845            let entry = &mut *self.entry_ptr(index);
846            // If the slot was overwritten by a push() while we were
847            // storing to LTM, the timestamp will differ. Don't mark
848            // the new entry — let it be consolidated on a future tick.
849            if entry.timestamp != expected_timestamp {
850                return false;
851            }
852            entry.flags |= RingBufferEntry::CONSOLIDATED;
853        }
854        true
855    }

859    /// Push a new entry into the ring buffer.
860    ///
861    /// If the buffer is full, the oldest entry is overwritten.
862    /// Returns the slot index where the entry was written.
863    pub fn push(&self, entry: RingBufferEntry) -> u64 {
864        let _guard = self
865            .access_lock
866            .lock()
867            .expect("access_lock poisoned: mmap may be inconsistent");
868        // SAFETY: `self.ptr` is a valid mmap'd region of at least
869        // `size_of::<RingBufferHeader>()` bytes, validated on open/create.
870        let header = unsafe { &mut *self.header_ptr() };
871        let slot = header.head;
872        let cap = header.capacity as u64;
873
874        // Guard against a corrupt header. `open` rejects out-of-range
875        // cursors, and every push below maintains `head < cap` by
876        // modular arithmetic — but this is the one place a bad `head`
877        // turns directly into an out-of-bounds 256-byte write via
878        // `entry_ptr`, so re-check rather than trust the invariant.
879        // Return a sentinel (u64::MAX) instead of slot 0 so callers
880        // cannot mistake the no-op for a successful write to slot 0.
881        if cap == 0 || slot >= cap {
882            debug_assert!(false, "ring buffer head out of range (corrupt header)");
883            return u64::MAX;
884        }
885
886        // Write the entry
887        // SAFETY: `slot` is `header.head`, and the guard above plus the
888        // validated capacity guarantee `slot < cap`, so `entry_ptr(slot)`
889        // points within the mmap'd entry slot array.
890        // We use `write_volatile` to ensure the compiler does not
891        // reorder or elide this store, which is critical for mmap
892        // persistence — the entry must reach the page cache before
893        // we update the header below.
894        unsafe {
895            std::ptr::write_volatile(self.entry_ptr(slot), entry);
896        }
897
898        // Release fence: ensures the entry write above is visible
899        // before the header update below. On x86 (TSO) this is a
900        // no-op, but on ARM/RISC-V the CPU could reorder the header
901        // store before the entry store, leaving a crash-recovery
902        // scenario where the header points to a slot that hasn't
903        // been written yet.
904        std::sync::atomic::fence(std::sync::atomic::Ordering::Release);
905
906        // Advance head
907        header.head = (slot + 1) % cap;
908        // This counter is defined as monotonic; wrapping at u64::MAX would violate
909        // that invariant and could make persisted telemetry appear to go backwards.
910        header.total_written = header.total_written.saturating_add(1);
911
912        // Update count and tail
913        if header.count < cap {
914            header.count += 1;
915        } else {
916            // Buffer is full — tail advances too (overwrite oldest)
917            header.tail = (header.tail + 1) % cap;
918        }
919
920        slot
921    }

923    /// Iterate over entries in order from oldest to newest.
924    ///
925    /// Returns an owned snapshot so the lock is not held across the
926    /// iteration. Each entry is a 256-byte `Copy` value.
927    pub fn iter(&self) -> impl Iterator<Item = RingBufferEntry> {
928        let _guard = self
929            .access_lock
930            .lock()
931            .expect("access_lock poisoned: mmap may be inconsistent");
932        // SAFETY: `self.ptr` is a valid mmap'd region of at least
933        // `size_of::<RingBufferHeader>()` bytes, validated on open/create.
934        let header = unsafe { &*self.header_ptr() };
935        let cap = header.capacity as u64;
936        // Clamp count to capacity — a corrupted or truncated mmap header
937        // could report a count exceeding capacity, causing a huge
938        // allocation and iterating over garbage entries.
939        let count = header.count.min(cap);
940        let start = header.tail;
941        let mut entries = Vec::with_capacity(count as usize);
942        // Guard against corrupted header with capacity == 0.
943        if cap > 0 {
944            for i in 0..count {
945                let index = (start + i) % cap;
946                // SAFETY: `index` < `cap` (modular arithmetic), and `cap`
947                // is `header.capacity`, so `entry_ptr(index)` is within the
948                // mmap'd entry slot array.
949                entries.push(unsafe { *self.entry_ptr(index) });
950            }
951        }
952        entries.into_iter()
953    }

955    /// Snapshot all entries with their slot indices, holding the lock
956    /// for the entire scan.
957    ///
958    /// This is the atomic variant of `iter()` for callers that need
959    /// slot indices (e.g. consolidation, which marks entries
960    /// consolidated by slot). Without this, calling `count()`,
961    /// `header()`, `peek()`, and `mark_consolidated_if_match()` separately
962    /// would acquire and release `access_lock` for each operation,
963    /// allowing a concurrent `push()` to change `head`/`tail` and
964    /// overwrite slots mid-scan.
965    ///
966    /// Returns `(slot_index, entry)` pairs in oldest-to-newest order.
967    pub fn iter_with_slots(&self) -> Vec<(u64, RingBufferEntry)> {
968        let _guard = self
969            .access_lock
970            .lock()
971            .expect("access_lock poisoned: mmap may be inconsistent");
972        // SAFETY: `self.ptr` is a valid mmap'd region of at least
973        // `size_of::<RingBufferHeader>()` bytes, validated on open/create.
974        let header = unsafe { &*self.header_ptr() };
975        let cap = header.capacity as u64;
976        let count = header.count.min(cap);
977        let start = header.tail;
978        let mut entries = Vec::with_capacity(count as usize);
979        if cap > 0 {
980            for i in 0..count {
981                let slot = (start + i) % cap;
982                // SAFETY: `slot` < `cap`, so `entry_ptr(slot)` is within
983                // the mmap'd entry slot array.
984                entries.push((slot, unsafe { *self.entry_ptr(slot) }));
985            }
986        }
987        entries
988    }

1051

1053        // SAFETY: `entries_start` is within the mmap'd region
1054        // (header size + capacity * ENTRY_SIZE ≤ file_len).
1055        unsafe {
1056            let entries_start = self.ptr.add(core::mem::size_of::<RingBufferHeader>());
1057            std::ptr::write_bytes(entries_start, 0, self.capacity as usize * ENTRY_SIZE);
1058        }
1059    }

1063    /// Take the sole-mapper advisory lock on the backing file.
1064    ///
1065    /// `RingBuffer` forms `&`/`&mut` references into `MAP_SHARED`
1066    /// memory, which is only sound when no second process can map and
1067    /// write the same file concurrently. `flock(LOCK_EX | LOCK_NB)` is
1068    /// held for the life of the file descriptor — released implicitly
1069    /// when `Drop` closes `fd` — so any other `open`/`create` (in this
1070    /// process or another) fails fast with [`RingBufferError::LockBusy`]
1071    /// instead of silently violating the aliasing contract.
1072    ///
1073    /// Callers must hold the lock *before* resizing or mapping the
1074    /// file; that is why `create_inner` does not pass `.truncate()` to
1075    /// `OpenOptions`.
1076    fn acquire_sole_mapper_lock(fd: i32) -> Result<(), RingBufferError> {
1077        // SAFETY: `fd` is a valid open file descriptor. flock is
1078        // advisory and a failed call changes no state.
1079        if unsafe { flock(fd, LOCK_EX | LOCK_NB) } != 0 {
1080            return Err(RingBufferError::LockBusy);
1081        }
1082        Ok(())
1083    }
1084
1085    fn do_mmap(fd: i32, len: usize) -> Result<*mut u8, RingBufferError> {
1086        // SAFETY: `fd` is a valid open file descriptor with read+write
1087        // permissions. `len` is the file size (validated above).
1088        let ptr = unsafe {
1089            mmap(
1090                std::ptr::null_mut(),
1091                len,
1092                PROT_READ | PROT_WRITE,
1093                MAP_SHARED,
1094                fd,
1095                0,
1096            )
1097        };
1098        if ptr == MAP_FAILED {
1099            return Err(RingBufferError::MmapFailed);
1100        }
1101        Ok(ptr as *mut u8)
1102    }
1103
1104    fn do_msync(ptr: *mut u8, len: usize) -> Result<(), RingBufferError> {
1105        // SAFETY: `ptr` is a valid mmap'd region of `len` bytes.
1106        let rc = unsafe { msync(ptr as *mut c_void, len, MS_SYNC) };
1107        if rc != 0 {
1108            return Err(RingBufferError::MsyncFailed);
1109        }
1110        Ok(())
1111    }

1113    /// Release a mapping and descriptor on a failed `open`/`create`
1114    /// path, so an error return does not leak either.
1115    ///
1116    /// # Safety
1117    /// `ptr` must be a live mapping of exactly `len` bytes and `fd` a
1118    /// live descriptor, both exclusively owned by the caller and not
1119    /// referenced anywhere else.
1120    unsafe fn discard_mapping(ptr: *mut u8, len: usize, fd: i32) {
1121        // SAFETY (unsafe fn): the caller guarantees both are live and
1122        // exclusively owned, per the contract above.
1123        unsafe {
1124            munmap(ptr as *mut c_void, len);
1125            close(fd);
1126        }
1127    }
1128}

1130impl Drop for RingBuffer {
1131    fn drop(&mut self) {
1132        // Best-effort msync before munmap. Without this, any STM
1133        // events written since the last explicit sync() are lost
1134        // when the mapping is removed. We ignore errors here because
1135        // Drop cannot propagate them, and a failed msync is better
1136        // than skipping it entirely.
1137        let _ = Self::do_msync(self.ptr, self.file_len);
1138        // SAFETY: `self.ptr` and `self.fd` are valid and exclusively
1139        // owned by this instance. No other references exist at drop time.
1140        unsafe {
1141            munmap(self.ptr as *mut c_void, self.file_len);
1142            close(self.fd);
1143        }
1144    }
