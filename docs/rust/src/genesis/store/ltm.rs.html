1//! Long-term episodic memory (LTM) store — SDR/LogHD architecture.
2//!
3//! This is Genesis's durable memory. Episodes consolidated from
4//! short-term memory persist here across restarts.
5//!
6//! # Architecture: three-file design
7//!
8//! ```text
9//! ┌─────────────────────────────────────────────────┐
10//! │  Bundles file (.bundles) — mmap'd, fixed 800 KB │
11//! │  20 dense f32 hypervectors (10,000-dim each)    │
12//! │  LogHD class-axis compressed SDR index          │
13//! │  Fixed size regardless of episode count         │
14//! └─────────────────────────────────────────────────┘
15//!
16//! ┌─────────────────────────────────────────────────┐
17//! │  Metadata file (.meta) — append-only            │
18//! │  64-byte IndexEntry per episode                 │
19//! │  Grows with episodes (no fixed capacity)        │
20//! └─────────────────────────────────────────────────┘
21//!
22//! ┌─────────────────────────────────────────────────┐
23//! │  Data file (.dat) — append-only                 │
24//! │  Compressed episode payloads                    │
25//! │  Same format as v1                              │
26//! └─────────────────────────────────────────────────┘
27//! ```
28//!
29//! # What changed from v1
30//!
31//! v1 used a fixed-capacity mmap'd index (65,536 slots × 64 bytes).
32//! When it filled up, no new memories could be stored — Genesis
33//! stopped learning.
34//!
35//! v2 replaces the fixed index with:
36//! - A **LogHD bundle store** (fixed 800 KB) for SDR-based
37//!   similarity matching with effectively unlimited capacity
38//!   (4^20 ≈ 1 trillion episodes)
39//! - An **append-only metadata file** that grows with episodes
40//!   (no capacity limit)
41//!
42//! The SimHash (64-bit) is retained per episode for backward
43//! compatibility with the association engine's Hamming distance
44//! matching. The SDR provides higher-dimensional similarity for
45//! `find_similar` queries.
46

113/// A single metadata entry — same layout as v1 for IPC compatibility.
114///
115/// ## Layout (64 bytes)
116/// ```text
117/// offset  field              type    notes
118/// ------  -----              ----    -----
119///   0     episode_id         u64     unique monotonic ID
120///   8     data_offset        u64     offset into the data file
121///  16     timestamp          u64     when the episode occurred
122///  24     association_hash   u64     SimHash fingerprint (backward compat)
123///  32     compressed_len     u32     bytes in the data file
124///  36     uncompressed_len   u32     original payload size
125///  40     salience           f32     finite evidence value
126///  44     emotional_tag      [f32;4] compact: arousal, valence, cortisol, dopamine
127///  60     flags              u8      bit 0: deleted, bits 1-7: reserved
128///  61     source_module      u8      module that created the episode
129///  62     _pad               [u8;2]  alignment
130/// ```
131#[repr(C)]
132#[derive(Clone, Copy, Debug)]
133pub struct IndexEntry {
134    /// Unique monotonic episode identifier.
135    pub episode_id: u64,
136    /// Byte offset of the compressed payload in the data file.
137    pub data_offset: u64,
138    /// Unix timestamp (ms) when the episode was recorded.
139    pub timestamp: u64,
140    /// 64-bit SimHash fingerprint used for association matching.
141    pub association_hash: u64,
142    /// Length of the compressed payload in the data file (bytes).
143    pub compressed_len: u32,
144    /// Original uncompressed payload size (bytes).
145    pub uncompressed_len: u32,
146    /// Emotional importance of the episode, in `[0,1]`.
147    pub salience: f32,
148    /// Compact emotional snapshot: arousal, valence, cortisol, dopamine.
149    pub emotional_tag: [f32; 4],
150    /// Bit flags (bit 0: deleted, bit 1: meta-memory, bit 2: archived).
151    pub flags: u8,
152    /// Identifier of the module that created the episode.
153    pub source_module: u8,
154    /// Padding to keep the entry at exactly 64 bytes.
155    pub _pad: [u8; 2],
156}
157
158impl IndexEntry {
159    /// Flag bit marking an episode as logically deleted.
160    pub const DELETED_FLAG: u8 = 1 << 0;
161    /// Bit 1: this episode is a meta-memory (association trace or
162    /// dream insight), not a real source episode. Meta-memories are
163    /// excluded from `find_similar` results so that retrieval returns
164    /// real conversation/learning episodes, not association bridges.
165    pub const META_MEMORY_FLAG: u8 = 1 << 1;
166    /// Bit 2: this episode has been archived by sleep compression.
167    /// Archived episodes are excluded from `find_similar`,
168    /// `recent_episodes`, and the bundle store (same as deleted),
169    /// but unlike deleted episodes they remain retrievable by ID.
170    /// This preserves dream insight references and conversation
171    /// history — nothing is ever lost.
172    pub const ARCHIVED_FLAG: u8 = 1 << 2;
173
174    #[allow(clippy::too_many_arguments)]
175    /// Construct a new index entry with no flags set.
176    pub fn new(
177        episode_id: u64,
178        data_offset: u64,
179        compressed_len: u32,
180        uncompressed_len: u32,
181        timestamp: u64,
182        salience: f32,
183        association_hash: u64,
184        emotional_tag: [f32; 4],
185        source_module: u8,
186    ) -> Self {
187        Self {
188            episode_id,
189            data_offset,
190            compressed_len,
191            uncompressed_len,
192            timestamp,
193            salience,
194            association_hash,
195            emotional_tag,
196            flags: 0,
197            source_module,
198            _pad: [0; 2],
199        }
200    }
201
202    /// Deserialize an `IndexEntry` from a byte array.
203    ///
204    /// # Safety
205    ///
206    /// `bytes` must be exactly `INDEX_ENTRY_SIZE` bytes. The function
207    /// uses `ptr::read_unaligned` so the buffer does not need to be
208    /// aligned to `IndexEntry`'s normal alignment. This is sound because
209    /// `IndexEntry` is `#[repr(C)]`, is `Copy`, and contains only
210    /// primitive types with no invalid bit patterns.
211    pub fn from_bytes(bytes: &[u8; INDEX_ENTRY_SIZE]) -> Self {
212        // SAFETY: `bytes` has the same size as `IndexEntry` and the type
213        // is plain data. `read_unaligned` avoids alignment requirements
214        // so this is safe regardless of the buffer's address.
215        unsafe { std::ptr::read_unaligned(bytes.as_ptr() as *const IndexEntry) }
216    }
217
218    /// Returns `true` if this episode has been logically deleted.
219    pub fn is_deleted(&self) -> bool {
220        self.flags & Self::DELETED_FLAG != 0
221    }
222
223    /// Mark this episode as logically deleted.
224    pub fn mark_deleted(&mut self) {
225        self.flags |= Self::DELETED_FLAG;
226    }
227
228    /// Whether this episode is a meta-memory (association trace or
229    /// dream insight) that should be excluded from retrieval results.
230    pub fn is_meta_memory(&self) -> bool {
231        self.flags & Self::META_MEMORY_FLAG != 0
232    }
233
234    /// Mark this episode as a meta-memory.
235    pub fn mark_meta_memory(&mut self) {
236        self.flags |= Self::META_MEMORY_FLAG;
237    }
238
239    /// Whether this episode has been archived by sleep compression.
240    /// Archived episodes are excluded from similarity search and
241    /// recent-episode scans but remain retrievable by ID.
242    pub fn is_archived(&self) -> bool {
243        self.flags & Self::ARCHIVED_FLAG != 0
244    }
245
246    /// Mark this episode as archived.
247    pub fn mark_archived(&mut self) {
248        self.flags |= Self::ARCHIVED_FLAG;
249    }
250
251    /// Whether this episode should be excluded from similarity search,
252    /// recent-episode scans, and the bundle store. Deleted, archived,
253    /// and meta-memory entries are all excluded.
254    pub fn is_excluded_from_search(&self) -> bool {
255        self.is_deleted() || self.is_archived() || self.is_meta_memory()
256    }
257
258    /// Hamming distance between this entry's association hash and
259    /// `other_hash` (number of differing bits).
260    pub fn hash_distance(&self, other_hash: u64) -> u32 {
261        (self.association_hash ^ other_hash).count_ones()
262    }
263}

360/// Compute a 64-bit SimHash fingerprint of the given text.
361pub fn simhash(text: &str) -> u64 {
362    let mut counts = [0i32; 64];
363    let words: Vec<&str> = text
364        .split(|c: char| !c.is_alphanumeric())
365        .filter(|s| !s.is_empty())
366        .collect();
367    for word in &words {
368        let h = fnv1a_lower(word.as_bytes());
369        update_counts(&mut counts, h);
370    }
371    for i in 0..words.len().saturating_sub(1) {
372        let h = fnv1a_lower_bigram(words[i].as_bytes(), words[i + 1].as_bytes());
373        update_counts(&mut counts, h);
374    }
375    let mut result: u64 = 0;
376    for (i, &count) in counts.iter().enumerate() {
377        if count > 0 {
378            result |= 1 << i;
379        }
380    }
381    result
382}
383
384fn fnv1a_lower(data: &[u8]) -> u64 {
385    let mut hash: u64 = 0xcbf29ce484222325;
386    for &byte in data {
387        hash ^= byte.to_ascii_lowercase() as u64;
388        hash = hash.wrapping_mul(0x100000001b3);
389    }
390    hash
391}
392
393fn fnv1a_lower_bigram(a: &[u8], b: &[u8]) -> u64 {
394    let mut hash: u64 = 0xcbf29ce484222325;
395    for &byte in a {
396        hash ^= byte.to_ascii_lowercase() as u64;
397        hash = hash.wrapping_mul(0x100000001b3);
398    }
399    hash ^= b' ' as u64;
400    hash = hash.wrapping_mul(0x100000001b3);
401    for &byte in b {
402        hash ^= byte.to_ascii_lowercase() as u64;
403        hash = hash.wrapping_mul(0x100000001b3);
404    }
405    hash
406}
407
408fn update_counts(counts: &mut [i32; 64], hash: u64) {
409    for (i, count) in counts.iter_mut().enumerate() {
410        if hash & (1 << i) != 0 {
411            *count += 1;
412        } else {
413            *count -= 1;
414        }
415    }
416}

418/// Hamming distance between two 64-bit hashes (number of differing bits).
419pub fn hamming_distance(a: u64, b: u64) -> u32 {
420    (a ^ b).count_ones()

454/// Errors that can occur when working with the LTM store.
455#[derive(Debug)]
456pub enum LtmError {
457    /// An underlying I/O error from the filesystem.
458    Io(std::io::Error),
459    /// The metadata file's magic bytes did not match.
460    IndexBadMagic,
461    /// The data file's magic bytes did not match.
462    DataBadMagic,
463    /// A file's schema version did not match the expected one.
464    VersionMismatch {
465        /// The version the code expected.
466        expected: u32,
467        /// The version found in the file.
468        found: u32,
469    },
470    /// Memory-mapping a file failed.
471    MmapFailed,
472    /// `msync` failed while flushing mapped pages to disk.
473    MsyncFailed,
474    /// `ftruncate` failed while sizing a new file.
475    FtruncateFailed,
476    /// The (legacy) fixed-capacity index is full.
477    IndexFull,
478    /// No episode matched the requested ID.
479    EpisodeNotFound,
480    /// An episode payload exceeded the maximum allowed size.
481    PayloadTooLarge {
482        /// The actual payload size.
483        size: usize,
484        /// The maximum allowed size.
485        max: usize,
486    },
487    /// Decompressing an episode payload failed (with a reason string).
488    DecompressFailed(String),
489    /// Compressing an episode payload failed (with a reason string).
490    CompressFailed(String),
491    /// The episode text was empty or all whitespace — an episode with
492    /// no content is noise, not a memory (it also degenerates the
493    /// SimHash to 0, which makes it "associated" with every other
494    /// empty episode at distance 0).
495    EmptyPayload,
496}
497
498impl std::fmt::Display for LtmError {
499    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
500        match self {
501            Self::Io(e) => write!(f, "I/O error: {e}"),
502            Self::IndexBadMagic => write!(f, "metadata file bad magic"),
503            Self::DataBadMagic => write!(f, "data file bad magic"),
504            Self::VersionMismatch { expected, found } => {
505                write!(f, "version mismatch: expected {expected}, found {found}")
506            }
507            Self::MmapFailed => write!(f, "mmap failed"),
508            Self::MsyncFailed => write!(f, "msync failed"),
509            Self::FtruncateFailed => write!(f, "ftruncate failed"),
510            Self::IndexFull => write!(f, "index is full"),
511            Self::EpisodeNotFound => write!(f, "episode not found"),
512            Self::PayloadTooLarge { size, max } => {
513                write!(f, "payload too large: {size} bytes (max {max})")
514            }
515            Self::DecompressFailed(e) => write!(f, "decompression failed: {e}"),
516            Self::CompressFailed(e) => write!(f, "compression failed: {e}"),
517            Self::EmptyPayload => write!(f, "episode text is empty"),
518        }
519    }
520}

528/// The LTM episodic store.
529///
530/// Uses a LogHD bundle store for SDR-based similarity matching and
531/// an append-only metadata file for per-episode data. The data file
532/// holds compressed payloads (same format as v1).
533pub struct LtmStore {
534    /// LogHD bundle store (mmap'd, fixed 800 KB)
535    bundles: BundleStore,
536    /// Metadata file (append-only)
537    meta_file: std::fs::File,
538    /// Data file (append-only, compressed payloads)
539    data_file: std::fs::File,
540    /// In-memory metadata entries (all active + deleted)
541    entries: Vec<IndexEntry>,
542    /// episode_id → index in entries vec
543    id_to_idx: HashMap<u64, usize>,
544    /// Active (non-deleted) episode IDs
545    active_ids: Vec<u64>,
546    /// Metadata header (kept in sync with file)
547    meta_header: IndexHeader,
548    /// Base path (without extension) for rebuilding the bundle store
549    base_path: std::path::PathBuf,
550    /// SimHash fingerprints of active meta-memory texts — the
551    /// association engine's global dedup index. A bridge's fingerprint
552    /// is `simhash` of its full meta text (episode IDs embedded), so
553    /// a pair is bridged at most once across the whole store, no
554    /// matter which later pass re-encounters it. Rebuilt from active
555    /// meta entries at `open` (archived connections may legitimately
556    /// re-form) and maintained on every meta store.
557    meta_hashes: HashSet<u64>,
558    /// Rehearsal/access traces — the forgetting module's sidecar.
559    /// Every encoding and every cognitive retrieval leaves a trace;
560    /// the power-law decay of those traces is what `forgetting`
561    /// measures. Kept out of `.meta` so the frozen 64-byte index row
562    /// never changes.
563    traces: TraceStore,
564}

569impl LtmStore {
570    /// Create a new LTM store at the given base path.
571    /// Creates `{base}.bundles`, `{base}.meta`, and `{base}.dat`.
572    pub fn create(base_path: impl AsRef<Path>, _capacity: u32) -> Result<Self, LtmError> {
573        let base = base_path.as_ref().to_path_buf();
574        let bundles_path = base.with_extension("bundles");
575        let meta_path = base.with_extension("meta");
576        let dat_path = base.with_extension("dat");
577
578        // Create bundle store
579        let bundles = BundleStore::create(&bundles_path).map_err(LtmError::Io)?;
580
581        // Create metadata file
582        let mut meta_file = OpenOptions::new()
583            .read(true)
584            .write(true)
585            .create(true)
586            .truncate(true)
587            .open(&meta_path)
588            .map_err(LtmError::Io)?;
589
590        let meta_header = IndexHeader::new(DEFAULT_INDEX_CAPACITY);
591        meta_file.seek(SeekFrom::Start(0)).map_err(LtmError::Io)?;
592        // SAFETY: `IndexHeader` is `#[repr(C)]`, `Copy`, contains only
593        // primitive types, and `size_of::<IndexHeader>() == 64`
594        // (statically asserted). The slice covers exactly the struct's
595        // in-memory representation.
596        let meta_header_bytes = unsafe {
597            std::slice::from_raw_parts(
598                &meta_header as *const IndexHeader as *const u8,
599                core::mem::size_of::<IndexHeader>(),
600            )
601        };
602        meta_file
603            .write_all(meta_header_bytes)
604            .map_err(LtmError::Io)?;
605        meta_file.sync_all().map_err(LtmError::Io)?;
606
607        // Create data file
608        let mut data_file = OpenOptions::new()
609            .read(true)
610            .write(true)
611            .create(true)
612            .truncate(true)
613            .open(&dat_path)
614            .map_err(LtmError::Io)?;
615
616        let data_header = DataHeader::new();
617        data_file.seek(SeekFrom::Start(0)).map_err(LtmError::Io)?;
618        // SAFETY: `DataHeader` is `#[repr(C)]`, `Copy`, contains only
619        // primitive types, and `size_of::<DataHeader>() == 16`
620        // (statically asserted). The slice covers exactly the struct's
621        // in-memory representation.
622        let data_header_bytes = unsafe {
623            std::slice::from_raw_parts(
624                &data_header as *const DataHeader as *const u8,
625                core::mem::size_of::<DataHeader>(),
626            )
627        };
628        data_file
629            .write_all(data_header_bytes)
630            .map_err(LtmError::Io)?;
631        data_file.sync_all().map_err(LtmError::Io)?;
632
633        let traces = TraceStore::open_or_create(&base).map_err(LtmError::Io)?;
634
635        Ok(Self {
636            bundles,
637            meta_file,
638            data_file,
639            entries: Vec::new(),
640            id_to_idx: HashMap::new(),
641            active_ids: Vec::new(),
642            meta_header,
643            base_path: base,
644            meta_hashes: HashSet::new(),
645            traces,
646        })
647    }

649    /// Open an existing LTM store.
650    pub fn open(base_path: impl AsRef<Path>) -> Result<Self, LtmError> {
651        let base = base_path.as_ref().to_path_buf();
652
653        // Finish or clean up a compaction that was interrupted before
654        // it completed. The journal marks the point after which the
655        // staged files were durable, so redoing the renames is safe.
656        Self::recover_compaction(&base);
657
658        let bundles_path = base.with_extension("bundles");
659        let meta_path = base.with_extension("meta");
660        let dat_path = base.with_extension("dat");
661
662        // Open bundle store
663        let mut bundles = BundleStore::open(&bundles_path).map_err(LtmError::Io)?;
664
665        // Open metadata file
666        let mut meta_file = OpenOptions::new()
667            .read(true)
668            .write(true)
669            .open(&meta_path)
670            .map_err(LtmError::Io)?;
671
672        let meta_len = meta_file.metadata().map_err(LtmError::Io)?.len();
673        if meta_len < 64 || (meta_len - 64) % INDEX_ENTRY_SIZE as u64 != 0 {
674            return Err(LtmError::Io(std::io::Error::new(
675                std::io::ErrorKind::InvalidData,
676                "LTM metadata contains an incomplete header or entry",
677            )));
678        }
679
680        // Read metadata header
681        let mut magic = [0u8; 4];
682        let mut version_bytes = [0u8; 4];
683        let mut capacity_bytes = [0u8; 4];
684        let mut count_bytes = [0u8; 4];
685        let mut next_id_bytes = [0u8; 8];
686        let mut reserved = [0u8; 40];
687        meta_file.seek(SeekFrom::Start(0)).map_err(LtmError::Io)?;
688        meta_file.read_exact(&mut magic).map_err(LtmError::Io)?;
689        meta_file
690            .read_exact(&mut version_bytes)
691            .map_err(LtmError::Io)?;
692        meta_file
693            .read_exact(&mut capacity_bytes)
694            .map_err(LtmError::Io)?;
695        meta_file
696            .read_exact(&mut count_bytes)
697            .map_err(LtmError::Io)?;
698        meta_file
699            .read_exact(&mut next_id_bytes)
700            .map_err(LtmError::Io)?;
701        meta_file.read_exact(&mut reserved).map_err(LtmError::Io)?;
702
703        if magic != LTM_META_MAGIC {
704            return Err(LtmError::IndexBadMagic);
705        }
706        let version = u32::from_le_bytes(version_bytes);
707        if version != LTM_SCHEMA_VERSION {
708            return Err(LtmError::VersionMismatch {
709                expected: LTM_SCHEMA_VERSION,
710                found: version,
711            });
712        }
713
714        let meta_header = IndexHeader {
715            magic,
716            version,
717            capacity: u32::from_le_bytes(capacity_bytes),
718            episode_count: u32::from_le_bytes(count_bytes),
719            next_episode_id: u64::from_le_bytes(next_id_bytes),
720            _reserved: reserved,
721        };
722
723        // Read all metadata entries in one bulk read. The per-entry
724        // `read_exact` loop was one syscall per entry (320K syscalls
725        // at 320K entries); a single bulk read is 1 syscall plus
726        // in-memory parsing. Pre-allocate all collections with the
727        // exact entry count to avoid reallocations.
728        let entry_bytes_len = meta_len as usize - 64;
729        let entry_count = entry_bytes_len / INDEX_ENTRY_SIZE;
730        let mut all_entry_bytes = vec![0u8; entry_bytes_len];
731        meta_file
732            .read_exact(&mut all_entry_bytes)
733            .map_err(LtmError::Io)?;
734
735        let mut entries: Vec<IndexEntry> = Vec::with_capacity(entry_count);
736        let mut id_to_idx: HashMap<u64, usize> = HashMap::with_capacity(entry_count);
737        let mut active_ids: Vec<u64> = Vec::with_capacity(entry_count);
738
739        for i in 0..entry_count {
740            let offset = i * INDEX_ENTRY_SIZE;
741            // SAFETY: `offset` is `i * INDEX_ENTRY_SIZE` where
742            // `i < entry_count` and `entry_bytes_len =
743            // entry_count * INDEX_ENTRY_SIZE`, so
744            // `offset + INDEX_ENTRY_SIZE <= entry_bytes_len`.
745            // The slice is exactly `INDEX_ENTRY_SIZE` bytes.
746            let entry_bytes: &[u8; INDEX_ENTRY_SIZE] = all_entry_bytes
747                [offset..offset + INDEX_ENTRY_SIZE]
748                .try_into()
749                .expect("entry slice length equals INDEX_ENTRY_SIZE");
750            let entry = IndexEntry::from_bytes(entry_bytes);
751
752            // Episode IDs are the primary key used by both the metadata
753            // index and the bundle code. A zero or duplicate ID makes
754            // the in-memory HashMap silently overwrite an earlier entry,
755            // causing retrieval to return the wrong episode. Treat either
756            // condition as on-disk corruption instead.
757            if entry.episode_id == 0 || id_to_idx.contains_key(&entry.episode_id) {
758                return Err(LtmError::Io(std::io::Error::new(
759                    std::io::ErrorKind::InvalidData,
760                    format!(
761                        "LTM metadata contains invalid or duplicate episode ID {}",
762                        entry.episode_id
763                    ),
764                )));
765            }
766
767            // These values are persisted and later used in arithmetic and
768            // ordering. Reject non-finite floats at the trust boundary
769            // rather than allowing a valid-looking metadata file to inject
770            // NaN into retrieval or salience sorting.
771            if !entry.salience.is_finite() || entry.emotional_tag.iter().any(|v| !v.is_finite()) {
772                return Err(LtmError::Io(std::io::Error::new(
773                    std::io::ErrorKind::InvalidData,
774                    format!(
775                        "LTM metadata entry {} contains non-finite values",
776                        entry.episode_id
777                    ),
778                )));
779            }
780
781            let idx = entries.len();
782            if !entry.is_deleted() {
783                // Archived episodes stay in id_to_idx (retrievable
784                // by ID) but are excluded from active_ids (similarity
785                // search, recent_episodes, bundle store).
786                id_to_idx.insert(entry.episode_id, idx);
787                if !entry.is_archived() {
788                    active_ids.push(entry.episode_id);
789                }
790            }
791            entries.push(entry);
792        }
793
794        // Open data file
795        let mut data_file = OpenOptions::new()
796            .read(true)
797            .write(true)
798            .open(&dat_path)
799            .map_err(LtmError::Io)?;
800
801        // Verify data header
802        let mut dmagic = [0u8; 4];
803        let mut dversion = [0u8; 4];
804        data_file.seek(SeekFrom::Start(0)).map_err(LtmError::Io)?;
805        data_file.read_exact(&mut dmagic).map_err(LtmError::Io)?;
806        data_file.read_exact(&mut dversion).map_err(LtmError::Io)?;
807        if dmagic != LTM_DATA_MAGIC {
808            return Err(LtmError::DataBadMagic);
809        }
810        let dver = u32::from_le_bytes(dversion);
811        if dver != LTM_SCHEMA_VERSION {
812            return Err(LtmError::VersionMismatch {
813                expected: LTM_SCHEMA_VERSION,
814                found: dver,
815            });
816        }
817
818        // Validate next_write_offset: must be >= 16 (past the header)
819        // and <= file size. A corrupted or zeroed offset could cause
820        // store() to seek to offset 0 and overwrite the header, or
821        // to seek beyond EOF into a sparse file.
822        let mut offset_bytes = [0u8; 8];
823        data_file
824            .read_exact(&mut offset_bytes)
825            .map_err(LtmError::Io)?;
826        let next_write_offset = u64::from_le_bytes(offset_bytes);
827        let data_file_len = data_file.metadata().map_err(LtmError::Io)?.len();
828
829        // Every non-deleted episode must point to a complete payload record.
830        // Validate this once at startup so a corrupt metadata entry cannot
831        // later seek outside the data file or allocate based on an invalid
832        // compressed length during retrieval.
833        for entry in &entries {
834            if entry.is_deleted() {
835                continue;
836            }
837            if entry.data_offset < 16
838                || entry
839                    .data_offset
840                    .checked_add(4 + entry.compressed_len as u64)
841                    .is_none_or(|end| end > data_file_len)
842                || entry.compressed_len as usize > MAX_COMPRESSED_SIZE
843                || entry.uncompressed_len as usize > MAX_PAYLOAD_SIZE
844            {
845                return Err(LtmError::Io(std::io::Error::new(
846                    std::io::ErrorKind::InvalidData,
847                    format!(
848                        "LTM episode {} points to an invalid payload",
849                        entry.episode_id
850                    ),
851                )));
852            }
853        }
854
855        if next_write_offset < 16 || next_write_offset > data_file_len {
856            return Err(LtmError::Io(std::io::Error::new(
857                std::io::ErrorKind::InvalidData,
858                format!(
859                    "LTM data file has invalid next_write_offset: {} (file size: {})",
860                    next_write_offset, data_file_len
861                ),
862            )));
863        }
864
865        // Reconcile in-memory state with disk. The metadata header's
866        // episode_count and next_episode_id may be stale if a prior
867        // process crashed between writing entries and updating the
868        // header. Recompute from the actual loaded entries and the
869        // bundle store's next_episode_id to ensure consistency.
870        let active_count = active_ids.len() as u32;
871        let max_loaded_id = entries.iter().map(|e| e.episode_id).max().unwrap_or(0);
872        let bundle_next = bundles.next_episode_id();
873        let reconciled_next = max_loaded_id
874            .saturating_add(1)
875            .max(bundle_next)
876            .max(meta_header.next_episode_id);
877        bundles.ensure_next_episode_id(reconciled_next);
878        bundles.sync().map_err(LtmError::Io)?;
879
880        // No meta-memory migration: every episode is a real memory and
881        // stays recallable regardless of which module stored it.
882
883        // Rebuild the association dedup index from every active
884        // entry — O(active) over the in-memory index, no payload
885        // reads. Archived bridges stay out of the set: a connection
886        // that was archived may legitimately re-form.
887        let meta_hashes: HashSet<u64> = active_ids
888            .iter()
889            .filter_map(|id| id_to_idx.get(id))
890            .map(|&idx| &entries[idx])
891            .map(|e| e.association_hash)
892            .collect();
893
894        // Check the trace ledger's existence *before* open_or_create
895        // makes the file — an absent sidecar means this store's access
896        // history is unknowable (first run after the forgetting
897        // subsystem arrived, or a deleted/corrupt ledger).
898        let traces_existed = base.with_extension("traces").exists();
899        let traces = TraceStore::open_or_create(&base).map_err(LtmError::Io)?;
900
901        let mut reconciled_store = Self {
902            bundles,
903            meta_file,
904            data_file,
905            entries,
906            id_to_idx,
907            active_ids,
908            meta_hashes,
909            traces,
910            meta_header: IndexHeader {
911                episode_count: active_count,
912                next_episode_id: reconciled_next,
913                ..meta_header
914            },
915            base_path: base,
916        };
917        // Write the reconciled header back to disk so the next open
918        // sees consistent values.
919        reconciled_store.write_meta_header()?;
920
921        // Missing ledger: seed every surviving memory with a single
922        // fresh-ink trace stamped now — a mass reconsolidation event.
923        // Without this, a memory she retrieved yesterday but encoded
924        // long ago would wear its encoding age alone and could be
925        // pruned at the very next sweep. Data loss must fail toward
926        // retention, never deletion; from here real activity rebuilds
927        // the ledger and honest fading resumes.
928        if !traces_existed {
929            let now = wall_now_ms();
930            for entry in &reconciled_store.entries {
931                if entry.is_deleted() {
932                    continue;
933                }
934                if let Err(e) = reconciled_store
935                    .traces
936                    .record_encoding(entry.episode_id, now)
937                {
938                    eprintln!("[ltm] trace seeding failed for {}: {e}", entry.episode_id);
939                }
940            }
941        }
942
943        // Rebuild the bundle store when its episode_count is much
944        // higher than the number of real (non-meta) episodes, or when
945        // real episodes are missing from it entirely — e.g. after a
946        // flags repair that restored previously hidden episodes.
947        let real_count = reconciled_store
948            .active_ids
949            .iter()
950            .filter(|&&id| {
951                let Some(&idx) = reconciled_store.id_to_idx.get(&id) else {
952                    return false;
953                };
954                !reconciled_store.entries[idx].is_meta_memory()
955            })
956            .count();
957        let bundle_count = reconciled_store.bundles.episode_count() as usize;
958        // archive/delete/store_meta subtract SDRs as episodes leave
959        // the bundled set, but crash-torn writes and pre-subtraction
960        // stores leave residual dead weight. A small excess of dead
961        // contributions is cheap noise in query_activation — rebuild
962        // only when the pollution is material (over ~25% of the real
963        // set, floor 64) or when real episodes are missing from the
964        // bundles entirely.
965        let drift = bundle_count.saturating_sub(real_count);
966        let drift_threshold = (real_count / 4).max(64);
967        let needs_rebuild =
968            bundle_count != real_count && (bundle_count < real_count || drift > drift_threshold);
969        if needs_rebuild {
970            eprintln!(
971                "[ltm] Bundle store has {bundle_count} episodes but \
972                 {real_count} are real — rebuilding to match"
973            );
974            reconciled_store.rebuild_bundles()?;
975        }
976        Ok(reconciled_store)
977    }

979    /// Open or create.
980    pub fn open_or_create(base_path: impl AsRef<Path>, capacity: u32) -> Result<Self, LtmError> {
981        let base = base_path.as_ref();
982        let has_bundles = base.with_extension("bundles").exists();
983        let has_meta = base.with_extension("meta").exists();
984        let has_dat = base.with_extension("dat").exists();
985        let has_old_idx = base.with_extension("idx").exists();
986
987        // Migrate from v1 if old .idx exists but new files don't
988        if has_old_idx && !has_meta && !has_bundles {
989            return Self::migrate_from_v1(base, capacity);
990        }
991
992        if has_bundles && has_meta && has_dat {
993            Self::open(base)
994        } else if has_bundles || has_meta || has_dat || has_old_idx {
995            Err(LtmError::Io(std::io::Error::new(
996                std::io::ErrorKind::InvalidData,
997                "LTM store is incomplete; existing files were preserved",
998            )))
999        } else {
1000            Self::create(base, capacity)
1001        }
1002    }

1004    /// Migrate from v1 (.idx/.dat) to v2 (.bundles/.meta/.dat).
1005    fn migrate_from_v1(base: &Path, _capacity: u32) -> Result<Self, LtmError> {
1006        eprintln!("[ltm] Migrating from v1 to v2 (SDR/LogHD)...");
1007        let idx_path = base.with_extension("idx");
1008        let old_dat_path = base.with_extension("dat");
1009
1010        // Open old store using v1 format — we need to read entries.
1011        // We'll read the old index directly since we can't import the
1012        // old code. The v1 format is: 64-byte header + 64-byte entries.
1013        let old_idx = std::fs::read(&idx_path).map_err(LtmError::Io)?;
1014        if old_idx.len() < 64 {
1015            return Err(LtmError::IndexBadMagic);
1016        }
1017        if &old_idx[0..4] != b"LTMI" {
1018            return Err(LtmError::IndexBadMagic);
1019        }
1020
1021        let old_version = u32::from_le_bytes(old_idx[4..8].try_into().unwrap_or([0; 4]));
1022        if old_version != 1 {
1023            return Err(LtmError::VersionMismatch {
1024                expected: 1,
1025                found: old_version,
1026            });
1027        }
1028        let old_capacity = u32::from_le_bytes(old_idx[8..12].try_into().unwrap_or([0; 4])) as usize;
1029        if old_idx.len() != 64 + old_capacity * INDEX_ENTRY_SIZE {
1030            return Err(LtmError::Io(std::io::Error::new(
1031                std::io::ErrorKind::InvalidData,
1032                "v1 metadata length does not match its capacity",
1033            )));
1034        }
1035        let mut next_id = u64::from_le_bytes(old_idx[16..24].try_into().unwrap_or([0; 8])).max(1);
1036        let idx_backup = base.with_extension("idx.v1bak");
1037        let dat_backup = base.with_extension("dat.v1bak");
1038        if idx_backup.exists() || dat_backup.exists() {
1039            return Err(LtmError::Io(std::io::Error::new(
1040                std::io::ErrorKind::AlreadyExists,
1041                "v1 migration backups already exist; recovery is required before retrying",
1042            )));
1043        }
1044
1045        // Read the old data file BEFORE creating the new v2 store.
1046        // `Self::create()` truncates `{base}.dat`, so if we create
1047        // the new store first, the old data file is destroyed before
1048        // we can read it.
1049        let mut old_data = OpenOptions::new()
1050            .read(true)
1051            .open(&old_dat_path)
1052            .map_err(LtmError::Io)?;
1053
1054        let mut data_header = [0u8; 16];
1055        old_data
1056            .read_exact(&mut data_header)
1057            .map_err(LtmError::Io)?;
1058        if data_header[..4] != LTM_DATA_MAGIC {
1059            return Err(LtmError::DataBadMagic);
1060        }
1061        let data_version = u32::from_le_bytes(data_header[4..8].try_into().unwrap_or([0; 4]));
1062        if data_version != 1 {
1063            return Err(LtmError::VersionMismatch {
1064                expected: 1,
1065                found: data_version,
1066            });
1067        }
1068
1069        // Read all old entries and their payloads into memory before
1070        // the new store is created in a separate staging directory.
1071        struct OldEntry {
1072            episode_id: u64,
1073            timestamp: u64,
1074            salience: f32,
1075            emotional_tag: [f32; 4],
1076        }
1077        let mut old_entries: Vec<(OldEntry, Vec<u8>)> = Vec::new();
1078        for i in 0..old_capacity {
1079            let offset = 64 + i * INDEX_ENTRY_SIZE;
1080            let entry_bytes: &[u8; INDEX_ENTRY_SIZE] = old_idx[offset..offset + INDEX_ENTRY_SIZE]
1081                .try_into()
1082                .map_err(|_| {
1083                    LtmError::Io(std::io::Error::new(
1084                        std::io::ErrorKind::InvalidData,
1085                        "v1 index entry slice length mismatch",
1086                    ))
1087                })?;
1088            let entry = IndexEntry::from_bytes(entry_bytes);
1089            next_id = next_id.max(entry.episode_id.saturating_add(1));
1090            if entry.episode_id == 0 || entry.is_deleted() {
1091                continue;
1092            }
1093            if entry.data_offset < 16 {
1094                return Err(LtmError::Io(std::io::Error::new(
1095                    std::io::ErrorKind::InvalidData,
1096                    "v1 episode points into the data header",
1097                )));
1098            }
1099
1100            // Read the compressed payload from the old data file
1101            old_data
1102                .seek(SeekFrom::Start(entry.data_offset))
1103                .map_err(LtmError::Io)?;
1104            let mut len_bytes = [0u8; 4];
1105            old_data.read_exact(&mut len_bytes).map_err(LtmError::Io)?;
1106            let compressed_len = u32::from_le_bytes(len_bytes) as usize;
1107            if compressed_len > MAX_COMPRESSED_SIZE
1108                || compressed_len != entry.compressed_len as usize
1109            {
1110                return Err(LtmError::DecompressFailed(
1111                    "invalid v1 compressed length".into(),
1112                ));
1113            }
1114            let mut compressed = vec![0u8; compressed_len];
1115            old_data.read_exact(&mut compressed).map_err(LtmError::Io)?;
1116
1117            // Decompress to get the payload
1118            let payload = decompress(&compressed, entry.uncompressed_len as usize)?;
1119            if payload.len() < 52 {
1120                return Err(LtmError::DecompressFailed("v1 payload too short".into()));
1121            }
1122
1123            old_entries.push((
1124                OldEntry {
1125                    episode_id: entry.episode_id,
1126                    timestamp: entry.timestamp,
1127                    salience: entry.salience,
1128                    emotional_tag: entry.emotional_tag,
1129                },
1130                payload,
1131            ));
1132        }
1133
1134        old_entries.sort_unstable_by_key(|(entry, _)| entry.episode_id);
1135        if old_entries
1136            .windows(2)
1137            .any(|pair| pair[0].0.episode_id == pair[1].0.episode_id)
1138        {
1139            return Err(LtmError::Io(std::io::Error::new(
1140                std::io::ErrorKind::InvalidData,
1141                "v1 metadata contains duplicate episode IDs",
1142            )));
1143        }
1144
1145        // Create the new v2 store separately without truncating .dat,
1146        // and keep the original files intact until it is fully durable.
1147        let stage_dir = base.with_extension(format!(
1148            "migration-{}-{}",
1149            std::process::id(),
1150            std::time::SystemTime::now()
1151                .duration_since(std::time::UNIX_EPOCH)
1152                .unwrap_or_default()
1153                .as_nanos(),
1154        ));
1155        std::fs::create_dir(&stage_dir).map_err(LtmError::Io)?;
1156        let result = (|| -> Result<Self, LtmError> {
1157            let stage_base = stage_dir.join("ltm");
1158            let mut new_store = Self::create(&stage_base, DEFAULT_INDEX_CAPACITY)?;
1159            let migrated = old_entries.len();
1160            for (old, payload) in old_entries {
1161                let event_type = payload[0];
1162                let source_module = payload[1];
1163                // Parse the 12 f32 emotional tag from bytes 4..52.
1164                // `try_into` to `[u8; 48]` is exact (48 bytes) and
1165                // avoids the partial-chunk risk of `chunks(4)`.
1166                let tag_bytes: [u8; 48] = payload[4..52]
1167                    .try_into()
1168                    .expect("payload[4..52] is exactly 48 bytes");
1169                let mut full_tag = [0.0f32; EMOTIONAL_TAG_SIZE];
1170                for (j, chunk) in tag_bytes.chunks_exact(4).enumerate() {
1171                    full_tag[j] = f32::from_le_bytes(
1172                        chunk
1173                            .try_into()
1174                            .expect("chunks_exact(4) yields 4-byte chunks"),
1175                    );
1176                }
1177                let text = String::from_utf8_lossy(&payload[52..]).into_owned();
1178
1179                // Store in the new v2 format
1180                new_store.bundles.ensure_next_episode_id(old.episode_id);
1181                new_store.store(
1182                    old.timestamp,
1183                    old.salience,
1184                    full_tag,
1185                    old.emotional_tag,
1186                    event_type,
1187                    source_module,
1188                    &text,
1189                )?;
1190            }
1191            new_store.meta_header.next_episode_id = next_id;
1192            new_store.bundles.ensure_next_episode_id(next_id);
1193            new_store.write_meta_header()?;
1194            new_store.sync()?;
1195            let parent = base.parent().filter(|p| !p.as_os_str().is_empty());
1196            let directory =
1197                std::fs::File::open(parent.unwrap_or(Path::new("."))).map_err(LtmError::Io)?;
1198            std::fs::hard_link(&old_dat_path, &dat_backup).map_err(LtmError::Io)?;
1199            directory.sync_all().map_err(LtmError::Io)?;
1200            for ext in ["bundles", "dat", "meta"] {
1201                std::fs::rename(stage_base.with_extension(ext), base.with_extension(ext))
1202                    .map_err(LtmError::Io)?;
1203            }
1204            new_store.base_path = base.to_path_buf();
1205            eprintln!("[ltm] Migrated {migrated} episodes from v1 to v2");
1206
1207            // Rename the old index as backup; the original data is already backed up.
1208            std::fs::rename(&idx_path, &idx_backup).map_err(LtmError::Io)?;
1209            directory.sync_all().map_err(LtmError::Io)?;
1210            Ok(new_store)
1211        })();
1212        let _ = std::fs::remove_dir_all(&stage_dir);
1213        result
1214    }

1218    /// Store a new episode.
1219    #[allow(clippy::too_many_arguments)]
1220    pub fn store(
1221        &mut self,
1222        timestamp: u64,
1223        salience: f32,
1224        emotional_tag: [f32; EMOTIONAL_TAG_SIZE],
1225        compact_tag: [f32; 4],
1226        event_type: u8,
1227        source_module: u8,
1228        text: &str,
1229    ) -> Result<u64, LtmError> {
1230        let meta_len = self.meta_file.metadata().map_err(LtmError::Io)?.len();
1231        if meta_len != 64 + self.entries.len() as u64 * INDEX_ENTRY_SIZE as u64 {
1232            return Err(LtmError::Io(std::io::Error::new(
1233                std::io::ErrorKind::InvalidData,
1234                "LTM metadata changed or has an incomplete entry; reopen before storing",
1235            )));
1236        }
1237        let next_episode_id = self
1238            .meta_header
1239            .next_episode_id
1240            .max(self.bundles.next_episode_id())
1241            .checked_add(1)
1242            .ok_or_else(|| {
1243                LtmError::Io(std::io::Error::new(
1244                    std::io::ErrorKind::InvalidData,
1245                    "LTM episode ID space exhausted",
1246                ))
1247            })?;
1248        let text_bytes = text.as_bytes();
1249        // An episode with no text is noise, not a memory — and its
1250        // SimHash degenerates to 0, "associating" it with every other
1251        // empty episode at distance 0. Reject at the boundary so all
1252        // producers (STM consolidation, IPC, meta stores) share the
1253        // guard.
1254        if text.trim().is_empty() {
1255            return Err(LtmError::EmptyPayload);
1256        }
1257        // The full payload is 52 bytes (header + emotional tag) + text.
1258        // Check the total payload size, not just the text size, so that
1259        // a text of exactly MAX_PAYLOAD_SIZE doesn't produce a payload
1260        // that exceeds the limit by 52 bytes (which would then fail on
1261        // retrieval).
1262        if 52 + text_bytes.len() > MAX_PAYLOAD_SIZE {
1263            return Err(LtmError::PayloadTooLarge {
1264                size: 52 + text_bytes.len(),
1265                max: MAX_PAYLOAD_SIZE,
1266            });
1267        }
1268
1269        // Sanitize emotional tag values to prevent NaN/inf from
1270        // corrupting the SDR encoding and downstream retrieval. Finite
1271        // evidence is preserved verbatim, including values outside the
1272        // historical normalized range.
1273        let emotional_tag: [f32; EMOTIONAL_TAG_SIZE] = {
1274            let mut sanitized = [0.0f32; EMOTIONAL_TAG_SIZE];
1275            for (i, &v) in emotional_tag.iter().enumerate() {
1276                sanitized[i] = crate::state::sanitize::finite_or(v, 0.0);
1277            }
1278            sanitized
1279        };
1280
1281        // Build payload: [event_type][source_module][pad:2][tag:48][text...]
1282        let mut payload = Vec::with_capacity(52 + text_bytes.len());
1283        payload.push(event_type);
1284        payload.push(source_module);
1285        payload.push(0);
1286        payload.push(0);
1287        for &v in &emotional_tag {
1288            payload.extend_from_slice(&v.to_le_bytes());
1289        }
1290        payload.extend_from_slice(text_bytes);
1291
1292        let compressed = compress(&payload)?;
1293
1294        // Read current data write offset
1295        self.data_file
1296            .seek(SeekFrom::Start(8))
1297            .map_err(LtmError::Io)?;
1298        let mut offset_bytes = [0u8; 8];
1299        self.data_file
1300            .read_exact(&mut offset_bytes)
1301            .map_err(LtmError::Io)?;
1302        let recorded_offset = u64::from_le_bytes(offset_bytes);
1303        let data_offset = self.data_file.metadata().map_err(LtmError::Io)?.len();
1304
1305        // Validate the recorded offset against the header and EOF.
1306        // Always append at EOF, not at a potentially stale offset.
1307        // A crash can leave a payload durable before its offset update;
1308        // retaining those trailing bytes avoids overwriting episodes.
1309        // Offsets outside the existing file indicate corruption.
1310        if recorded_offset < 16 || recorded_offset > data_offset {
1311            return Err(LtmError::Io(std::io::Error::new(
1312                std::io::ErrorKind::InvalidData,
1313                format!("LTM data_offset {recorded_offset} is outside the data file"),
1314            )));
1315        }
1316
1317        // Append compressed payload
1318        let compressed_len = compressed.len() as u32;
1319        self.data_file
1320            .seek(SeekFrom::Start(data_offset))
1321            .map_err(LtmError::Io)?;
1322        self.data_file
1323            .write_all(&compressed_len.to_le_bytes())
1324            .map_err(LtmError::Io)?;
1325        self.data_file
1326            .write_all(&compressed)
1327            .map_err(LtmError::Io)?;
1328        // fsync the payload BEFORE writing the offset that points to
1329        // it. A single fsync after both does not guarantee ordering:
1330        // the kernel may flush the offset (a small write near the
1331        // start of the file) before the much larger payload at the
1332        // end. If a crash occurs between the fsync and the next
1333        // operation, the offset could point to data that hasn't
1334        // reached disk yet, producing a corrupt or zero-filled read
1335        // on reopen.
1336        self.data_file.sync_all().map_err(LtmError::Io)?;
1337
1338        // Update data file's next_write_offset, then fsync again to
1339        // make the offset durable. The payload is already on disk
1340        // from the sync above, so this fsync only needs to flush the
1341        // 8-byte offset update.
1342        let new_offset = data_offset + 4 + compressed.len() as u64;
1343        self.data_file
1344            .seek(SeekFrom::Start(8))
1345            .map_err(LtmError::Io)?;
1346        self.data_file
1347            .write_all(&new_offset.to_le_bytes())
1348            .map_err(LtmError::Io)?;
1349        self.data_file.sync_all().map_err(LtmError::Io)?;
1350
1351        // Compute SimHash (backward compat for association engine)
1352        let assoc_hash = simhash(text);
1353
1354        // Assign episode ID
1355        let episode_id = next_episode_id - 1;
1356        // Every episode's SDR enters the superposition — all stored
1357        // content is recallable regardless of source module.
1358        self.bundles.ensure_next_episode_id(episode_id);
1359        let code = episode_id_to_code(episode_id);
1360
1361        // Encode SDR and add to bundle store
1362        let sdr = encode_episode(text, &emotional_tag, timestamp);
1363        self.bundles.add_episode(&code, &sdr);
1364
1365        // Make the bundle store durable BEFORE writing the metadata
1366        // entry that references this episode ID. Without this, a crash
1367        // after `write_meta_header()` but before the next `sync()` would
1368        // leave the metadata file with episode N but the bundle file's
1369        // `next_episode_id` still at N-1. On reopen, `store()` would
1370        // reuse N-1, producing a duplicate episode ID that corrupts
1371        // `id_to_idx` and `active_ids`.
1372        self.bundles.sync().map_err(LtmError::Io)?;
1373
1374        // Sanitize salience and compact_tag to prevent NaN/inf from
1375        // corrupting the index entry, SimHash scoring, and SDR encoding.
1376        // Salience is evidence and preserves any finite value; compact_tag
1377        // remains a bounded derived affective summary.
1378        let salience = crate::state::sanitize::finite_or(salience, 0.0);
1379        let compact_tag = [
1380            crate::state::sanitize::finite_clamp(compact_tag[0], 0.0, 1.0),
1381            crate::state::sanitize::finite_clamp(compact_tag[1], -1.0, 1.0),
1382            crate::state::sanitize::finite_clamp(compact_tag[2], 0.0, 1.0),
1383            crate::state::sanitize::finite_clamp(compact_tag[3], 0.0, 1.0),
1384        ];
1385
1386        // Build index entry
1387        let entry = IndexEntry::new(
1388            episode_id,
1389            data_offset,
1390            compressed_len,
1391            payload.len() as u32,
1392            timestamp,
1393            salience,
1394            assoc_hash,
1395            compact_tag,
1396            source_module,
1397        );
1398
1399        // No automatic meta-memory marking: episodes stay recallable
1400        // no matter which module stored them. Only an explicit
1401        // `store_meta` call sets the flag now.
1402
1403        // Append to metadata file
1404        // SAFETY: `entry` is a valid local `IndexEntry` reference that lives
1405        // for the entire scope. `INDEX_ENTRY_SIZE` equals
1406        // `size_of::<IndexEntry>()`, so the slice covers exactly the struct's
1407        // in-memory representation with no out-of-bounds access.
1408        let entry_bytes = unsafe {
1409            std::slice::from_raw_parts(&entry as *const IndexEntry as *const u8, INDEX_ENTRY_SIZE)
1410        };
1411        self.meta_file
1412            .seek(SeekFrom::End(0))
1413            .map_err(LtmError::Io)?;
1414        self.meta_file
1415            .write_all(entry_bytes)
1416            .map_err(LtmError::Io)?;
1417
1418        // Update metadata header in file. `write_meta_header` fsyncs
1419        // the metadata file, which also makes the entry appended
1420        // above durable — one fsync for both.
1421        self.meta_header.episode_count += 1;
1422        self.meta_header.next_episode_id = next_episode_id;
1423        self.write_meta_header()?;
1424
1425        // Update in-memory indices
1426        let idx = self.entries.len();
1427        self.entries.push(entry);
1428        self.id_to_idx.insert(episode_id, idx);
1429        self.active_ids.push(episode_id);
1430        // Record the fingerprint so the association engine's dedup
1431        // sees this episode — the set covers every stored episode,
1432        // not just bookkeeping ones.
1433        self.meta_hashes.insert(assoc_hash);
1434
1435        // The encoding trace — forgetting's first presentation.
1436        // Best-effort: a lost record just reads as the same default
1437        // the absent-record path already computes.
1438        if let Err(e) = self.traces.record_encoding(episode_id, timestamp) {
1439            eprintln!("[ltm] could not record encoding trace for {episode_id}: {e}");
1440        }
1441
1442        Ok(episode_id)
1443    }

1497    ///
1498    /// Archived episodes are retrievable (unlike deleted ones) — this
1499    /// is the key difference between archiving and deletion. Dream
1500    /// insights and association chains that reference archived episodes
1501    /// still resolve, preserving memory integrity across sleep
1502    /// compression cycles.
1503    ///
1504    /// Retrieval is rehearsal — the opposite of forgetting. Every
1505    /// episode the mind actually reaches for adds a trace, pushing its
1506    /// accessibility boundary outward. Maintenance reads (unbundling,
1507    /// bundle rebuilds) go through [`Self::retrieve_raw`] so the
1508    /// janitor's broom never counts as a rehearsal.
1509    pub fn retrieve(&mut self, episode_id: u64) -> Result<Episode, LtmError> {
1510        let episode = self.retrieve_raw(episode_id)?;
1511        if let Err(e) = self.traces.touch(episode_id, wall_now_ms()) {
1512            eprintln!("[ltm] could not record rehearsal for {episode_id}: {e}");
1513        }
1514        Ok(episode)
1515    }

1517    /// Read an episode's payload without rehearsing it.
1518    ///
1519    /// For the store's own bookkeeping — subtracting an SDR during
1520    /// archive/delete, re-encoding during bundle rebuilds. Reading a
1521    /// memory to maintain the shelves it sits on is not the mind
1522    /// remembering it, so no trace is added.
1523    fn retrieve_raw(&mut self, episode_id: u64) -> Result<Episode, LtmError> {
1524        let idx = *self
1525            .id_to_idx
1526            .get(&episode_id)
1527            .ok_or(LtmError::EpisodeNotFound)?;
1528        let entry = self.entries[idx];
1529
1530        // Guard against deleted entries. `id_to_idx` should not
1531        // contain deleted entries (delete removes them), but
1532        // defense-in-depth: if a bug or corruption leaves a stale
1533        // mapping, don't return a deleted episode.
1534        // Archived episodes ARE retrievable — that's the point.
1535        if entry.is_deleted() {
1536            return Err(LtmError::EpisodeNotFound);
1537        }
1538
1539        // Validate entry.data_offset: must be >= 16 (past the data
1540        // file header). A corrupted index entry with data_offset < 16
1541        // would read the file header as if it were a compressed payload.
1542        if entry.data_offset < 16 {
1543            return Err(LtmError::Io(std::io::Error::new(
1544                std::io::ErrorKind::InvalidData,
1545                format!(
1546                    "LTM entry {} has invalid data_offset {} (< 16)",
1547                    episode_id, entry.data_offset
1548                ),
1549            )));
1550        }
1551
1552        self.data_file
1553            .seek(SeekFrom::Start(entry.data_offset))
1554            .map_err(LtmError::Io)?;
1555        let mut len_bytes = [0u8; 4];
1556        self.data_file
1557            .read_exact(&mut len_bytes)
1558            .map_err(LtmError::Io)?;
1559        let compressed_len = u32::from_le_bytes(len_bytes) as usize;
1560
1561        if compressed_len > MAX_COMPRESSED_SIZE {
1562            return Err(LtmError::DecompressFailed(format!(
1563                "compressed length {compressed_len} exceeds maximum {MAX_COMPRESSED_SIZE}"
1564            )));
1565        }
1566
1567        let mut compressed = vec![0u8; compressed_len];
1568        self.data_file
1569            .read_exact(&mut compressed)
1570            .map_err(LtmError::Io)?;
1571
1572        // Cap uncompressed_len at MAX_PAYLOAD_SIZE to prevent OOM
1573        // from a corrupted index entry with a huge uncompressed_len.
1574        let unc_len = entry.uncompressed_len as usize;
1575        let payload = decompress(&compressed, unc_len)?;
1576        if payload.len() > MAX_PAYLOAD_SIZE {
1577            return Err(LtmError::DecompressFailed(format!(
1578                "decompressed length {} exceeds maximum {}",
1579                payload.len(),
1580                MAX_PAYLOAD_SIZE
1581            )));
1582        }
1583        // Verify the decompressed length matches the index entry's
1584        // uncompressed_len (capped). A mismatch indicates corruption.
1585        if payload.len() != unc_len {
1586            return Err(LtmError::DecompressFailed(format!(
1587                "decompressed length {} does not match expected {}",
1588                payload.len(),
1589                unc_len
1590            )));
1591        }
1592
1593        if payload.len() < 52 {
1594            return Err(LtmError::DecompressFailed("payload too short".to_string()));
1595        }
1596        let event_type = payload[0];
1597        let source_module = payload[1];
1598
1599        // Parse the 12 f32 emotional tag from bytes 4..52. The slice
1600        // is exactly 48 bytes (12 × 4), so `try_into` to `[u8; 48]`
1601        // is exact — no partial-chunk risk that `chunks(4)` would hide.
1602        let tag_bytes: [u8; 48] = payload[4..52]
1603            .try_into()
1604            .expect("payload[4..52] is exactly 48 bytes");
1605        let mut full_tag = [0.0f32; EMOTIONAL_TAG_SIZE];
1606        for (i, chunk) in tag_bytes.chunks_exact(4).enumerate() {
1607            full_tag[i] = f32::from_le_bytes(
1608                chunk
1609                    .try_into()
1610                    .expect("chunks_exact(4) yields 4-byte chunks"),
1611            );
1612        }
1613
1614        let text = String::from_utf8_lossy(&payload[52..]).into_owned();
1615
1616        Ok(Episode {
1617            episode_id: entry.episode_id,
1618            timestamp: entry.timestamp,
1619            salience: entry.salience,
1620            association_hash: entry.association_hash,
1621            emotional_tag: entry.emotional_tag,
1622            full_emotional_tag: full_tag,
1623            event_type,
1624            source_module,
1625            text,
1626        })
1627    }

1629    /// Find episodes similar to the given query text.
1630    ///
1631    /// Uses the LogHD bundle store for SDR-based similarity matching.
1632    /// Returns up to `limit` results sorted by distance (ascending).
1633    ///
1634    /// Forgetting applies: candidates whose trace activation a cue of
1635    /// this strength can no longer reach are filtered before ranking.
1636    /// This convenience form measures decay from the store's own
1637    /// newest timestamp with neutral dopamine — the explicit form is
1638    /// [`Self::find_similar_at`].
1639    pub fn find_similar(&self, query: &str, limit: usize) -> Vec<(u64, u32, &IndexEntry)> {
1640        self.find_similar_at(
1641            query,
1642            limit,
1643            self.reference_now_ms(),
1644            forgetting::DA_BASELINE,
1645        )
1646    }

1648    /// `find_similar` with an explicit clock and dopamine level — the
1649    /// form the daemon calls, so decay is measured against the body's
1650    /// time and the forgetting gate responds to live chemistry.
1651    pub fn find_similar_at(
1652        &self,
1653        query: &str,
1654        limit: usize,
1655        now_ms: u64,
1656        dopamine: f64,
1657    ) -> Vec<(u64, u32, &IndexEntry)> {
1658        // Encode query as SDR (content only — no emotional/temporal
1659        // context for a text query)
1660        let query_sdr = crate::store::sdr::encode_content(query);
1661        let activation = self.bundles.query_activation(&query_sdr);
1662
1663        // Score each active episode by the raw L2 distance
1664        // ||A − E||² between the query activation A and the episode's
1665        // expected activation E. Lower = more similar.
1666        //
1667        // We deliberately do NOT normalize the activation vector. The
1668        // LogHD bundle store is additive — each `add_episode` call
1669        // superimposes `weight * SDR` onto every bundle — so the
1670        // activation magnitude encodes how strongly the query overlaps
1671        // with each episode's content. Normalizing A to unit L2 norm
1672        // discards that magnitude information. When episode codes are
1673        // concentrated in a few of the 20 code dimensions (the common
1674        // case for small episode counts, since IDs 1–4 only use
1675        // positions 18–19), the expected activations are nearly
1676        // collinear and normalization makes them indistinguishable
1677        // regardless of their code digit values. The sort then
1678        // degenerates, returning wrong matches.
1679        //
1680        // Instead, we compute distances as f32 (which cannot overflow
1681        // regardless of episode count) and then scale to u32
1682        // adaptively — dividing by the maximum distance so the full
1683        // u32 range is used without saturation. This preserves
1684        // ordering for both small and large episode counts.
1685        let mut scored: Vec<(u64, f32, &IndexEntry)> = Vec::new();
1686
1687        for &id in &self.active_ids {
1688            let Some(&idx) = self.id_to_idx.get(&id) else {
1689                continue;
1690            };
1691            let entry = &self.entries[idx];
1692
1693            // Skip meta-memories (association traces, dream insights),
1694            // deleted, and archived entries. These are internal
1695            // bookkeeping episodes or excluded entries, not real
1696            // experiences. Including them in retrieval results floods
1697            // find_similar with `[association] episode X ↔ episode Y`
1698            // entries that crowd out actual conversation/learning
1699            // memories. `is_excluded_from_search()` is defense-in-depth:
1700            // active_ids should already exclude deleted/archived entries,
1701            // but a bug or corruption could leave a stale mapping.
1702            if entry.is_excluded_from_search() {
1703                continue;
1704            }
1705
1706            let code = episode_id_to_code(id);
1707            let expected = code_to_expected_activation(&code);
1708
1709            let mut dist_sq = 0.0f32;
1710            for j in 0..LOGHD_N {
1711                let diff = activation[j] - expected[j];
1712                dist_sq += diff * diff;
1713            }
1714
1715            scored.push((id, dist_sq, entry));
1716        }
1717
1718        // The forgetting gate. The bundle distance has no natural
1719        // similarity scale, so the cue strength is normalized against
1720        // this query's own spread: the best candidate simulates
1721        // sim = 1 (a perfect cue), the worst sim = 0. A degenerate
1722        // spread (max 0 or non-finite) means every candidate is
1723        // equidistant — a uniform perfect cue, and the gate still
1724        // applies.
1725        let query_hash = simhash(query);
1726        let max_raw = scored.iter().map(|(_, d, _)| *d).fold(0.0f32, f32::max);
1727        let spread_ok = max_raw > 0.0 && max_raw.is_finite();
1728        scored.retain(|(id, dist, entry)| {
1729            let cue = if spread_ok {
1730                1.0 - (dist / max_raw) as f64
1731            } else {
1732                1.0
1733            };
1734            let base = forgetting::base_level(
1735                self.traces.get(*id, entry.timestamp),
1736                entry.timestamp,
1737                entry.salience,
1738                now_ms,
1739            );
1740            forgetting::is_accessible(base, cue, *id, query_hash, dopamine)
1741        });
1742
1743        // Sort by f32 distance (ascending) before converting to u32,
1744        // so the ordering is determined by full-precision distances.
1745        scored.sort_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal));
1746        scored.truncate(limit);
1747
1748        // Adaptive scaling: map f32 distances to u32 preserving order.
1749        // Dividing by the max distance uses the full u32 range without
1750        // overflow, regardless of whether there are 4 episodes or 4

1798    ///
1799    /// Each candidate's cue strength is `1 − dist/64` — a SimHash
1800    /// match is a continuum from a perfect reminder (dist 0) to
1801    /// anti-correlation (dist 64). Candidates whose base-level
1802    /// activation plus this cue no longer reaches the retrieval
1803    /// threshold are forgotten *for this probe*: retrieval fades with
1804    /// time while the trace itself persists for pruning to release.

1806        &self,
1807        query: &str,
1808        limit: usize,
1809        now_ms: u64,
1810        dopamine: f64,
1811    ) -> Vec<(u64, u32, &IndexEntry)> {
1812        let query_hash = simhash(query);
1813
1814        let mut scored: Vec<(u64, u32, &IndexEntry)> = Vec::new();
1815        for &id in &self.active_ids {
1816            let Some(&idx) = self.id_to_idx.get(&id) else {
1817                continue;
1818            };
1819            let entry = &self.entries[idx];
1820
1821            // Skip meta-memories, deleted, and archived entries.
1822            // (active_ids should already exclude deleted/archived,
1823            // but defense-in-depth.)
1824            if entry.is_excluded_from_search() {
1825                continue;
1826            }
1827
1828            let dist = entry.hash_distance(query_hash);
1829            let cue = 1.0 - f64::from(dist) / 64.0;
1830            let base = forgetting::base_level(
1831                self.traces.get(id, entry.timestamp),
1832                entry.timestamp,
1833                entry.salience,
1834                now_ms,
1835            );
1836            if !forgetting::is_accessible(base, cue, id, query_hash, dopamine) {
1837                continue;
1838            }

1840        }
1841

1843        scored.sort_by_key(|a| a.1);
1844        scored.truncate(limit);

1846    }

2007    /// any retrieval cue could reach them — the functionally
2008    /// dismantled, not merely the faded.

2010    /// Returns `(episode_id, base_level)` pairs. The predicate is
2011    /// conservative by construction: a memory must fail even a
2012    /// maximal-similarity cue *and* be older than the age floor

2019            // The body's autonomous sweep governs only the plain
2020            // active episodic tier. The excluded classes all carry
2021            // their own retention authority:
2022            //   deleted  — already gone.
2023            //   archived — she chose cold storage; the mind's release
2024            //              verdict (release_fn → delete_episode) owns
2025            //              this tier, and it shrinks via compact only
2026            //              under her say-so.
2027            //   meta     — association bridges and dream insights:
2028            //              the cortical residue that *justifies* the
2029            //              forgetting. Deleting the residue would
2030            //              erase the record of why things were lost.
2031            //   dream    — the mind's policy preserves dreaming-module
2032            //              episodes even if one arrives unflagged.
2033            if entry.is_deleted()
2034                || entry.is_archived()
2035                || entry.is_meta_memory()
2036                || entry.source_module == SOURCE_MODULE_DREAM
2037            {
2038                continue;
2039            }
2040            let base = forgetting::base_level(
2041                self.traces.get(entry.episode_id, entry.timestamp),
2042                entry.timestamp,
2043                entry.salience,
2044                now_ms,
2045            );
2046            let age_days = (now_ms.saturating_sub(entry.timestamp) as f64) / forgetting::DAY_MS;
2047            if forgetting::should_forget(base, age_days) {
2048                faded.push((entry.episode_id, base));
2049            }
2050        }
2051        faded
2052    }

2311            .collect();

2314        // rebuild it from the survivors, then drop their traces.

2317        self.traces.retain(&live).map_err(LtmError::Io)?;

2320        let new_dat_len = std::fs::metadata(&dat_path).map_err(LtmError::Io)?.len();

2329        let journal_path = base.with_extension("compacting");
2330        if journal_path.exists() {

2362    /// Soft-delete an episode.
2363    pub fn delete(&mut self, episode_id: u64) -> Result<(), LtmError> {
2364        let idx = *self
2365            .id_to_idx
2366            .get(&episode_id)
2367            .ok_or(LtmError::EpisodeNotFound)?;
2368
2369        // Read the payload while the entry is still readable —
2370        // `retrieve` rejects deleted entries — so its SDR
2371        // contribution can be subtracted once the durable writes
2372        // land. Skipped for meta-memories (never bundled) and
2373        // already-archived entries (unbundled when they left the
2374        // active set).
2375        let unbundle_episode =
2376            if !self.entries[idx].is_meta_memory() && !self.entries[idx].is_archived() {
2377                self.retrieve_raw(episode_id).ok()
2378            } else {
2379                None
2380            };
2381
2382        self.entries[idx].mark_deleted();
2383
2384        // Only the flags byte changed — write just that byte instead
2385        // of rewriting the full 64-byte entry. The flags field is at
2386        // offset 60 within each 64-byte IndexEntry.
2387        let flags_offset = 64 + (idx as u64 * INDEX_ENTRY_SIZE as u64) + 60;
2388        self.meta_file
2389            .seek(SeekFrom::Start(flags_offset))
2390            .map_err(LtmError::Io)?;
2391        self.meta_file
2392            .write_all(&[self.entries[idx].flags])
2393            .map_err(LtmError::Io)?;
2394        // Decrement episode_count and write the header (which fsyncs
2395        // the metadata file, covering the flags write above).
2396        let previous_count = self.meta_header.episode_count;
2397        if !self.entries[idx].is_archived() {
2398            self.meta_header.episode_count = previous_count.saturating_sub(1);
2399        }
2400        if let Err(error) = self.write_meta_header() {
2401            self.meta_header.episode_count = previous_count;
2402            return Err(error);
2403        }
2404
2405        // Only update in-memory indices after the durable write
2406        // succeeds. If `write_meta_header` failed above, the entry is
2407        // marked deleted in `entries` but still present in
2408        // `id_to_idx`/`active_ids` — the next `open()` will see the
2409        // on-disk deleted flag and exclude it from `active_ids`.
2410        if let Some(episode) = unbundle_episode {
2411            self.unbundle(episode_id, &episode);
2412        }
2413        self.id_to_idx.remove(&episode_id);
2414        self.active_ids.retain(|id| *id != episode_id);
2415        self.traces.remove(episode_id);
2416
2417        Ok(())
2418    }

2420    /// Delete many episodes in one pass — the bulk form of
2421    /// [`Self::delete`], mirroring [`Self::archive_many`].
2422    ///
2423    /// Per-episode work is identical (flags byte, unbundle, index
2424    /// removal) but the metadata header is written and fsynced once
2425    /// for the whole batch — the difference between seconds and
2426    /// minutes when a forgetting sweep releases a large cohort at
2427    /// once. Unknown, already-deleted, and already-archived-but-deleted
2428    /// IDs are skipped. Returns the number deleted.
2429    pub fn delete_many(&mut self, ids: &[u64]) -> Result<u32, LtmError> {
2430        let mut deleted_ids: HashSet<u64> = HashSet::with_capacity(ids.len());
2431        for &episode_id in ids {
2432            let Some(&idx) = self.id_to_idx.get(&episode_id) else {
2433                continue;
2434            };
2435            if self.entries[idx].is_deleted() {
2436                continue;
2437            }
2438
2439            // Read the payload while still readable so its SDR can be
2440            // subtracted — same ordering as `delete`. Maintenance read:
2441            // no rehearsal.
2442            let unbundle_episode =
2443                if !self.entries[idx].is_meta_memory() && !self.entries[idx].is_archived() {
2444                    self.retrieve_raw(episode_id).ok()
2445                } else {
2446                    None
2447                };
2448
2449            // Flags byte at offset 60 within each 64-byte entry.
2450            let flags_offset = 64 + (idx as u64 * INDEX_ENTRY_SIZE as u64) + 60;
2451            self.meta_file
2452                .seek(SeekFrom::Start(flags_offset))
2453                .map_err(LtmError::Io)?;
2454            self.meta_file
2455                .write_all(&[self.entries[idx].flags | IndexEntry::DELETED_FLAG])
2456                .map_err(LtmError::Io)?;
2457            self.entries[idx].mark_deleted();
2458
2459            if let Some(episode) = unbundle_episode {
2460                self.unbundle(episode_id, &episode);
2461            }
2462            deleted_ids.insert(episode_id);
2463        }
2464
2465        if deleted_ids.is_empty() {
2466            return Ok(0);
2467        }
2468
2469        let previous_count = self.meta_header.episode_count;
2470        let active_lost = deleted_ids
2471            .iter()
2472            .filter(|id| {
2473                self.id_to_idx
2474                    .get(id)
2475                    .is_some_and(|&idx| !self.entries[idx].is_archived())
2476            })
2477            .count() as u32;
2478        self.meta_header.episode_count = previous_count.saturating_sub(active_lost);
2479        if let Err(error) = self.write_meta_header() {
2480            self.meta_header.episode_count = previous_count;
2481            return Err(error);
2482        }
2483
2484        for &id in &deleted_ids {
2485            self.id_to_idx.remove(&id);
2486            self.traces.remove(id);
2487        }
2488        self.active_ids.retain(|id| !deleted_ids.contains(id));

2491    }

2493    /// Archive an episode.
2494    ///
2495    /// Unlike `delete`, archiving preserves the episode in `id_to_idx`
2496    /// so it remains retrievable by ID. The episode is only removed
2497    /// from `active_ids`, which excludes it from similarity search,
2498    /// recent-episode scans, and the bundle store — but not from
2499    /// direct retrieval by ID.
2500    ///
2501    /// This is the correct operation for sleep compression: the
2502    /// episode's semantic content has been extracted into the concept
2503    /// network, so it no longer needs to be in the active similarity
2504    /// index. But the episode itself must survive — dream insights
2505    /// and association chains reference it by ID, and the conversation
2506    /// history is irreplaceable ground truth.
2507    pub fn archive(&mut self, episode_id: u64) -> Result<(), LtmError> {
2508        let idx = *self
2509            .id_to_idx
2510            .get(&episode_id)
2511            .ok_or(LtmError::EpisodeNotFound)?;
2512
2513        // Already archived — nothing to do.
2514        if self.entries[idx].is_archived() {
2515            return Ok(());
2516        }
2517
2518        let flags = self.entries[idx].flags | IndexEntry::ARCHIVED_FLAG;
2519
2520        // Only the flags byte changed — write just that byte instead
2521        // of rewriting the full 64-byte entry. The flags field is at
2522        // offset 60 within each 64-byte IndexEntry.
2523        let flags_offset = 64 + (idx as u64 * INDEX_ENTRY_SIZE as u64) + 60;
2524        self.meta_file
2525            .seek(SeekFrom::Start(flags_offset))
2526            .map_err(LtmError::Io)?;
2527        self.meta_file.write_all(&[flags]).map_err(LtmError::Io)?;
2528        // Decrement episode_count (active count) and fsync via header.
2529        let previous_count = self.meta_header.episode_count;
2530        self.meta_header.episode_count = previous_count.saturating_sub(1);
2531        if let Err(error) = self.write_meta_header() {
2532            self.meta_header.episode_count = previous_count;
2533            return Err(error);
2534        }
2535        self.entries[idx].mark_archived();
2536
2537        // The SDR contribution stays in the bundle superposition
2538        // after archival — subtract it so routine sleep-cycle
2539        // churn does not accumulate as dead weight that the
2540        // startup reconciliation has to rebuild away every boot.
2541        // Meta-memories were never bundled. A failed read leaves
2542        // bounded drift, which the open() tolerance absorbs.
2543        if !self.entries[idx].is_meta_memory() {
2544            match self.retrieve_raw(episode_id) {
2545                Ok(episode) => self.unbundle(episode_id, &episode),
2546                Err(e) => eprintln!("[ltm] could not unbundle archived episode {episode_id}: {e}"),
2547            }
2548        }
2549
2550        // Remove from active_ids but keep in id_to_idx.
2551        self.active_ids.retain(|id| *id != episode_id);
2552
2553        Ok(())
2554    }

2556    /// Un-archive an episode — reconsolidation.
2557    ///
2558    /// The mirror of `archive`: retrieval is what earns a memory its
2559    /// place in the active set, so when a deep-recall cue reaches into
2560    /// the cold tier and surfaces an episode, the episode comes back.
2561    /// This clears the archived flag on disk, returns the ID to
2562    /// `active_ids`, re-adds its SDR to the bundle superposition (the
2563    /// inverse of the subtraction `archive` performs), and floors its
2564    /// salience at `RECONSOLIDATION_SALIENCE_FLOOR` so the next
2565    /// compaction pass retains what recall actually used. A revived
2566    /// episode that is never touched again can still be re-archived by
2567    /// a later sweep — forgetting stays earned in both directions.
2568    pub fn unarchive(&mut self, episode_id: u64) -> Result<(), LtmError> {
2569        let idx = *self
2570            .id_to_idx
2571            .get(&episode_id)
2572            .ok_or(LtmError::EpisodeNotFound)?;
2573
2574        // Already active — nothing to do.
2575        if !self.entries[idx].is_archived() {
2576            return Ok(());
2577        }
2578
2579        let flags = self.entries[idx].flags & !IndexEntry::ARCHIVED_FLAG;
2580        let new_salience = self.entries[idx]
2581            .salience
2582            .max(RECONSOLIDATION_SALIENCE_FLOOR);
2583        let salience_changed = (new_salience - self.entries[idx].salience).abs() > f32::EPSILON;
2584
2585        // flags sits at offset 60 and salience at offset 40 inside the
2586        // 64-byte entry — same surgical writes `archive` uses.
2587        let entry_base = 64 + idx as u64 * INDEX_ENTRY_SIZE as u64;
2588        self.meta_file
2589            .seek(SeekFrom::Start(entry_base + 60))
2590            .map_err(LtmError::Io)?;
2591        self.meta_file.write_all(&[flags]).map_err(LtmError::Io)?;
2592        if salience_changed {
2593            self.meta_file
2594                .seek(SeekFrom::Start(entry_base + 40))
2595                .map_err(LtmError::Io)?;
2596            self.meta_file
2597                .write_all(&new_salience.to_le_bytes())
2598                .map_err(LtmError::Io)?;
2599        }
2600        let previous_count = self.meta_header.episode_count;
2601        self.meta_header.episode_count = previous_count.saturating_add(1);
2602        if let Err(error) = self.write_meta_header() {
2603            self.meta_header.episode_count = previous_count;
2604            return Err(error);
2605        }
2606        self.entries[idx].flags = flags;
2607        self.entries[idx].salience = new_salience;
2608
2609        // Re-bundle the SDR so the bundle-store path sees the revived
2610        // episode again — the inverse of `archive`'s subtraction.
2611        // Meta-memories were never bundled; a failed read leaves
2612        // bounded drift the startup reconciliation tolerates.
2613        if !self.entries[idx].is_meta_memory() {
2614            match self.retrieve(episode_id) {
2615                Ok(episode) => {
2616                    let code = episode_id_to_code(episode_id);
2617                    let sdr = encode_episode(
2618                        &episode.text,
2619                        &episode.full_emotional_tag,
2620                        episode.timestamp,
2621                    );
2622                    self.bundles.add_episode(&code, &sdr);
2623                }
2624                Err(e) => {
2625                    eprintln!("[ltm] could not re-bundle revived episode {episode_id}: {e}")
2626                }
2627            }
2628        }
2629
2630        // Back on the active list — appended like a fresh store, which
2631        // is what reconsolidation semantically is.
2632        self.active_ids.push(episode_id);
2633
2634        Ok(())
2635    }

2737    /// Read source_module from a compressed payload header.
2738    ///
2739    /// After successfully reading the source_module, **backfills** it
2740    /// into the index entry (both in memory and on disk) so that future
2741    /// calls never need to decompress the same entry again. This is
2742    /// critical for performance: without backfilling, every call to
2743    /// `recent_episodes` with a source_module filter would decompress
2744    /// all old v2 entries (which have `source_module == 0` in the
2745    /// index) on every single call — potentially hundreds of thousands
2746    /// of decompressions per invocation.
2747    fn source_module_of(&mut self, idx: usize) -> Option<u8> {
2748        let entry = &self.entries[idx];
2749        let data_offset = entry.data_offset;
2750        // Validate data_offset before seeking — a corrupted entry with
2751        // data_offset < 16 would read the data file header as if it
2752        // were a compressed payload. `retrieve()` validates this too,
2753        // but `source_module_of` is called independently during
2754        // `recent_episodes` filtering.
2755        if data_offset < 16 {
2756            return None;
2757        }
2758        self.data_file.seek(SeekFrom::Start(data_offset)).ok()?;
2759        let mut len_bytes = [0u8; 4];
2760        self.data_file.read_exact(&mut len_bytes).ok()?;
2761        let compressed_len = u32::from_le_bytes(len_bytes) as usize;
2762        if compressed_len > MAX_COMPRESSED_SIZE {
2763            return None;
2764        }
2765        let mut compressed = vec![0u8; compressed_len];
2766        self.data_file.read_exact(&mut compressed).ok()?;
2767        // Use `take(2)` to read only the first 2 bytes of the
2768        // decompressed stream (event_type + source_module), avoiding
2769        // a full decompression just to read the source module byte.
2770        let decoder = ZlibDecoder::new(&compressed[..]);
2771        let mut header = [0u8; 2];
2772        decoder.take(2).read_exact(&mut header).ok()?;
2773        let module = header[1];
2774
2775        // Backfill: write the discovered source_module back into the
2776        // index entry so future calls skip the decompression entirely.
2777        // The source_module field is at offset 61 within the 64-byte
2778        // IndexEntry. On disk, the entry starts at 64 + idx * 64.
2779        if module != 0 {
2780            let disk_offset = 64 + (idx as u64) * (INDEX_ENTRY_SIZE as u64) + 61;
2781            // Write to disk BEFORE updating the in-memory entry. If
2782            // seek fails, we must not write at the wrong position
2783            // (which could corrupt other index entries or the header).
2784            // If write fails, the in-memory entry must not diverge
2785            // from disk. Only update memory after the disk write
2786            // succeeds.
2787            if self.meta_file.seek(SeekFrom::Start(disk_offset)).is_ok()
2788                && self.meta_file.write_all(&[module]).is_ok()
2789            {
2790                self.entries[idx].source_module = module;
2791            }
2792        }
2793        Some(module)
2794    }

2796    /// Retrieve recent episodes, optionally filtered by source_module.
2797    pub fn recent_episodes(&mut self, limit: usize, source_filter: Option<u8>) -> Vec<Episode> {
2798        if limit == 0 {
2799            return Vec::new();
2800        }
2801
2802        let candidate_ids: Vec<u64> = match source_filter {
2803            None => {
2804                // `active_ids` is always in ascending episode-ID order
2805                // (see `recent_episode_ids`), so the most recent IDs
2806                // are the tail. Iterate in reverse and take `limit`
2807                // — O(limit) instead of cloning and sorting the full
2808                // list on every call.
2809                self.active_ids.iter().rev().take(limit).copied().collect()
2810            }
2811            Some(target_module) => {
2812                // Iterate from most recent (tail of active_ids) to oldest.
2813                // This lets us stop early once we have enough matches,
2814                // rather than scanning all 250K+ entries every call.
2815                //
2816                // We use a manual index loop instead of cloning
2817                // active_ids into a reversed Vec. Each field access
2818                // (active_ids[i], id_to_idx.get, entries[idx]) copies
2819                // a Copy value (u64, usize, u8) and releases the
2820                // borrow before source_module_of(&mut self) is called,
2821                // so the borrow checker is satisfied without any
2822                // allocation.
2823                let mut matching: Vec<u64> = Vec::new();
2824                let mut i = self.active_ids.len();
2825                while i > 0 {
2826                    i -= 1;
2827                    let id = self.active_ids[i];
2828                    if let Some(&idx) = self.id_to_idx.get(&id) {
2829                        let module = self.entries[idx].source_module;
2830                        // Use the source_module field from the index entry.
2831                        // For old v2 entries where source_module is 0 (was
2832                        // padding), fall back to decompression + backfill.
2833                        let module = if module != 0 {
2834                            module
2835                        } else {
2836                            self.source_module_of(idx).unwrap_or(0)
2837                        };
2838                        if module == target_module {
2839                            matching.push(id);
2840                            if matching.len() >= limit {
2841                                break;
2842                            }
2843                        }

2888            if let Ok(ep) = self.retrieve_raw(id) {
2889                results.push(ep);

2891        }

3088fn decompress(data: &[u8], expected_len: usize) -> Result<Vec<u8>, LtmError> {
3089    // Cap expected_len at MAX_PAYLOAD_SIZE to prevent OOM from a
3090    // corrupted index entry with a huge uncompressed_len. The caller
3091    // already caps it, but defense-in-depth.
3092    let capped_len = expected_len.min(MAX_PAYLOAD_SIZE);
3093    let decoder = ZlibDecoder::new(data);
3094    // Use `take(capped_len)` to bound the decompressed output. Without
3095    // this, `read_to_end` would ignore `with_capacity` and decompress
3096    // a zlib bomb to many megabytes before the post-read length check
3097    // fires. `take` limits the total bytes read from the decoder to
3098    // `capped_len`, preventing unbounded allocation.
3099    let mut result = Vec::with_capacity(capped_len);
3100    decoder
3101        .take(capped_len as u64 + 1)
3102        .read_to_end(&mut result)
3103        .map_err(|e| LtmError::DecompressFailed(e.to_string()))?;
3104    if expected_len > MAX_PAYLOAD_SIZE || result.len() != expected_len {
3105        return Err(LtmError::DecompressFailed(format!(
3106            "decompressed length {} does not match expected {expected_len} within maximum {MAX_PAYLOAD_SIZE}",
3107            result.len(),
3108        )));
3109    }
3110    Ok(result)
