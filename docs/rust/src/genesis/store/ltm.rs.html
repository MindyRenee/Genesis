1//! Long-term episodic memory (LTM) store — SDR/LogHD architecture.
2//!
3//! This is Genesis's durable memory. Episodes consolidated from
4//! short-term memory persist here across restarts.
5//!
6//! # Architecture: three-file design
7//!
8//! ```text
9//! ┌─────────────────────────────────────────────────┐
10//! │  Bundles file (.bundles) — mmap'd, fixed 800 KB │
11//! │  20 dense f32 hypervectors (10,000-dim each)    │
12//! │  LogHD class-axis compressed SDR index          │
13//! │  Fixed size regardless of episode count         │
14//! └─────────────────────────────────────────────────┘
15//!
16//! ┌─────────────────────────────────────────────────┐
17//! │  Metadata file (.meta) — append-only            │
18//! │  64-byte IndexEntry per episode                 │
19//! │  Grows with episodes (no fixed capacity)        │
20//! └─────────────────────────────────────────────────┘
21//!
22//! ┌─────────────────────────────────────────────────┐
23//! │  Data file (.dat) — append-only                 │
24//! │  Compressed episode payloads                    │
25//! │  Same format as v1                              │
26//! └─────────────────────────────────────────────────┘
27//! ```
28//!
29//! # What changed from v1
30//!
31//! v1 used a fixed-capacity mmap'd index (65,536 slots × 64 bytes).
32//! When it filled up, no new memories could be stored — Genesis
33//! stopped learning.
34//!
35//! v2 replaces the fixed index with:
36//! - A **LogHD bundle store** (fixed 800 KB) for SDR-based
37//!   similarity matching with effectively unlimited capacity
38//!   (4^20 ≈ 1 trillion episodes)
39//! - An **append-only metadata file** that grows with episodes
40//!   (no capacity limit)
41//!
42//! The SimHash (64-bit) is retained per episode for backward
43//! compatibility with the association engine's Hamming distance
44//! matching. The SDR provides higher-dimensional similarity for
45//! `find_similar` queries.
46

106/// A single metadata entry — same layout as v1 for IPC compatibility.
107///
108/// ## Layout (64 bytes)
109/// ```text
110/// offset  field              type    notes
111/// ------  -----              ----    -----
112///   0     episode_id         u64     unique monotonic ID
113///   8     data_offset        u64     offset into the data file
114///  16     timestamp          u64     when the episode occurred
115///  24     association_hash   u64     SimHash fingerprint (backward compat)
116///  32     compressed_len     u32     bytes in the data file
117///  36     uncompressed_len   u32     original payload size
118///  40     salience           f32     finite evidence value
119///  44     emotional_tag      [f32;4] compact: arousal, valence, cortisol, dopamine
120///  60     flags              u8      bit 0: deleted, bits 1-7: reserved
121///  61     source_module      u8      module that created the episode
122///  62     _pad               [u8;2]  alignment
123/// ```
124#[repr(C)]
125#[derive(Clone, Copy, Debug)]
126pub struct IndexEntry {
127    /// Unique monotonic episode identifier.
128    pub episode_id: u64,
129    /// Byte offset of the compressed payload in the data file.
130    pub data_offset: u64,
131    /// Unix timestamp (ms) when the episode was recorded.
132    pub timestamp: u64,
133    /// 64-bit SimHash fingerprint used for association matching.
134    pub association_hash: u64,
135    /// Length of the compressed payload in the data file (bytes).
136    pub compressed_len: u32,
137    /// Original uncompressed payload size (bytes).
138    pub uncompressed_len: u32,
139    /// Emotional importance of the episode, in `[0,1]`.
140    pub salience: f32,
141    /// Compact emotional snapshot: arousal, valence, cortisol, dopamine.
142    pub emotional_tag: [f32; 4],
143    /// Bit flags (bit 0: deleted, bit 1: meta-memory, bit 2: archived).
144    pub flags: u8,
145    /// Identifier of the module that created the episode.
146    pub source_module: u8,
147    /// Padding to keep the entry at exactly 64 bytes.
148    pub _pad: [u8; 2],
149}
150
151impl IndexEntry {
152    /// Flag bit marking an episode as logically deleted.
153    pub const DELETED_FLAG: u8 = 1 << 0;
154    /// Bit 1: this episode is a meta-memory (association trace or
155    /// dream insight), not a real source episode. Meta-memories are
156    /// excluded from `find_similar` results so that retrieval returns
157    /// real conversation/learning episodes, not association bridges.
158    pub const META_MEMORY_FLAG: u8 = 1 << 1;
159    /// Bit 2: this episode has been archived by sleep compression.
160    /// Archived episodes are excluded from `find_similar`,
161    /// `recent_episodes`, and the bundle store (same as deleted),
162    /// but unlike deleted episodes they remain retrievable by ID.
163    /// This preserves dream insight references and conversation
164    /// history — nothing is ever lost.
165    pub const ARCHIVED_FLAG: u8 = 1 << 2;
166
167    #[allow(clippy::too_many_arguments)]
168    /// Construct a new index entry with no flags set.
169    pub fn new(
170        episode_id: u64,
171        data_offset: u64,
172        compressed_len: u32,
173        uncompressed_len: u32,
174        timestamp: u64,
175        salience: f32,
176        association_hash: u64,
177        emotional_tag: [f32; 4],
178        source_module: u8,
179    ) -> Self {
180        Self {
181            episode_id,
182            data_offset,
183            compressed_len,
184            uncompressed_len,
185            timestamp,
186            salience,
187            association_hash,
188            emotional_tag,
189            flags: 0,
190            source_module,
191            _pad: [0; 2],
192        }
193    }
194
195    /// Deserialize an `IndexEntry` from a byte array.
196    ///
197    /// # Safety
198    ///
199    /// `bytes` must be exactly `INDEX_ENTRY_SIZE` bytes. The function
200    /// uses `ptr::read_unaligned` so the buffer does not need to be
201    /// aligned to `IndexEntry`'s normal alignment. This is sound because
202    /// `IndexEntry` is `#[repr(C)]`, is `Copy`, and contains only
203    /// primitive types with no invalid bit patterns.
204    pub fn from_bytes(bytes: &[u8; INDEX_ENTRY_SIZE]) -> Self {
205        // SAFETY: `bytes` has the same size as `IndexEntry` and the type
206        // is plain data. `read_unaligned` avoids alignment requirements
207        // so this is safe regardless of the buffer's address.
208        unsafe { std::ptr::read_unaligned(bytes.as_ptr() as *const IndexEntry) }
209    }
210
211    /// Returns `true` if this episode has been logically deleted.
212    pub fn is_deleted(&self) -> bool {
213        self.flags & Self::DELETED_FLAG != 0
214    }
215
216    /// Mark this episode as logically deleted.
217    pub fn mark_deleted(&mut self) {
218        self.flags |= Self::DELETED_FLAG;
219    }
220
221    /// Whether this episode is a meta-memory (association trace or
222    /// dream insight) that should be excluded from retrieval results.
223    pub fn is_meta_memory(&self) -> bool {
224        self.flags & Self::META_MEMORY_FLAG != 0
225    }
226
227    /// Mark this episode as a meta-memory.
228    pub fn mark_meta_memory(&mut self) {
229        self.flags |= Self::META_MEMORY_FLAG;
230    }
231
232    /// Whether this episode has been archived by sleep compression.
233    /// Archived episodes are excluded from similarity search and
234    /// recent-episode scans but remain retrievable by ID.
235    pub fn is_archived(&self) -> bool {
236        self.flags & Self::ARCHIVED_FLAG != 0
237    }
238
239    /// Mark this episode as archived.
240    pub fn mark_archived(&mut self) {
241        self.flags |= Self::ARCHIVED_FLAG;
242    }
243
244    /// Whether this episode should be excluded from similarity search,
245    /// recent-episode scans, and the bundle store. Deleted, archived,
246    /// and meta-memory entries are all excluded.
247    pub fn is_excluded_from_search(&self) -> bool {
248        self.is_deleted() || self.is_archived() || self.is_meta_memory()
249    }
250
251    /// Hamming distance between this entry's association hash and
252    /// `other_hash` (number of differing bits).
253    pub fn hash_distance(&self, other_hash: u64) -> u32 {
254        (self.association_hash ^ other_hash).count_ones()
255    }
256}

353/// Compute a 64-bit SimHash fingerprint of the given text.
354pub fn simhash(text: &str) -> u64 {
355    let mut counts = [0i32; 64];
356    let words: Vec<&str> = text
357        .split(|c: char| !c.is_alphanumeric())
358        .filter(|s| !s.is_empty())
359        .collect();
360    for word in &words {
361        let h = fnv1a_lower(word.as_bytes());
362        update_counts(&mut counts, h);
363    }
364    for i in 0..words.len().saturating_sub(1) {
365        let h = fnv1a_lower_bigram(words[i].as_bytes(), words[i + 1].as_bytes());
366        update_counts(&mut counts, h);
367    }
368    let mut result: u64 = 0;
369    for (i, &count) in counts.iter().enumerate() {
370        if count > 0 {
371            result |= 1 << i;
372        }
373    }
374    result
375}
376
377fn fnv1a_lower(data: &[u8]) -> u64 {
378    let mut hash: u64 = 0xcbf29ce484222325;
379    for &byte in data {
380        hash ^= byte.to_ascii_lowercase() as u64;
381        hash = hash.wrapping_mul(0x100000001b3);
382    }
383    hash
384}
385
386fn fnv1a_lower_bigram(a: &[u8], b: &[u8]) -> u64 {
387    let mut hash: u64 = 0xcbf29ce484222325;
388    for &byte in a {
389        hash ^= byte.to_ascii_lowercase() as u64;
390        hash = hash.wrapping_mul(0x100000001b3);
391    }
392    hash ^= b' ' as u64;
393    hash = hash.wrapping_mul(0x100000001b3);
394    for &byte in b {
395        hash ^= byte.to_ascii_lowercase() as u64;
396        hash = hash.wrapping_mul(0x100000001b3);
397    }
398    hash
399}
400
401fn update_counts(counts: &mut [i32; 64], hash: u64) {
402    for (i, count) in counts.iter_mut().enumerate() {
403        if hash & (1 << i) != 0 {
404            *count += 1;
405        } else {
406            *count -= 1;
407        }
408    }
409}

447/// Errors that can occur when working with the LTM store.
448#[derive(Debug)]
449pub enum LtmError {
450    /// An underlying I/O error from the filesystem.
451    Io(std::io::Error),
452    /// The metadata file's magic bytes did not match.
453    IndexBadMagic,
454    /// The data file's magic bytes did not match.
455    DataBadMagic,
456    /// A file's schema version did not match the expected one.
457    VersionMismatch {
458        /// The version the code expected.
459        expected: u32,
460        /// The version found in the file.
461        found: u32,
462    },
463    /// Memory-mapping a file failed.
464    MmapFailed,
465    /// `msync` failed while flushing mapped pages to disk.
466    MsyncFailed,
467    /// `ftruncate` failed while sizing a new file.
468    FtruncateFailed,
469    /// The (legacy) fixed-capacity index is full.
470    IndexFull,
471    /// No episode matched the requested ID.
472    EpisodeNotFound,
473    /// An episode payload exceeded the maximum allowed size.
474    PayloadTooLarge {
475        /// The actual payload size.
476        size: usize,
477        /// The maximum allowed size.
478        max: usize,
479    },
480    /// Decompressing an episode payload failed (with a reason string).
481    DecompressFailed(String),
482    /// Compressing an episode payload failed (with a reason string).
483    CompressFailed(String),
484    /// The episode text was empty or all whitespace — an episode with
485    /// no content is noise, not a memory (it also degenerates the
486    /// SimHash to 0, which makes it "associated" with every other
487    /// empty episode at distance 0).
488    EmptyPayload,
489}
490
491impl std::fmt::Display for LtmError {
492    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
493        match self {
494            Self::Io(e) => write!(f, "I/O error: {e}"),
495            Self::IndexBadMagic => write!(f, "metadata file bad magic"),
496            Self::DataBadMagic => write!(f, "data file bad magic"),
497            Self::VersionMismatch { expected, found } => {
498                write!(f, "version mismatch: expected {expected}, found {found}")
499            }
500            Self::MmapFailed => write!(f, "mmap failed"),
501            Self::MsyncFailed => write!(f, "msync failed"),
502            Self::FtruncateFailed => write!(f, "ftruncate failed"),
503            Self::IndexFull => write!(f, "index is full"),
504            Self::EpisodeNotFound => write!(f, "episode not found"),
505            Self::PayloadTooLarge { size, max } => {
506                write!(f, "payload too large: {size} bytes (max {max})")
507            }
508            Self::DecompressFailed(e) => write!(f, "decompression failed: {e}"),
509            Self::CompressFailed(e) => write!(f, "compression failed: {e}"),
510            Self::EmptyPayload => write!(f, "episode text is empty"),
511        }
512    }
513}

521/// The LTM episodic store.
522///
523/// Uses a LogHD bundle store for SDR-based similarity matching and
524/// an append-only metadata file for per-episode data. The data file
525/// holds compressed payloads (same format as v1).
526pub struct LtmStore {
527    /// LogHD bundle store (mmap'd, fixed 800 KB)
528    bundles: BundleStore,
529    /// Metadata file (append-only)
530    meta_file: std::fs::File,
531    /// Data file (append-only, compressed payloads)
532    data_file: std::fs::File,
533    /// In-memory metadata entries (all active + deleted)
534    entries: Vec<IndexEntry>,
535    /// episode_id → index in entries vec
536    id_to_idx: HashMap<u64, usize>,
537    /// Active (non-deleted) episode IDs
538    active_ids: Vec<u64>,
539    /// Metadata header (kept in sync with file)
540    meta_header: IndexHeader,
541    /// Base path (without extension) for rebuilding the bundle store
542    base_path: std::path::PathBuf,
543    /// SimHash fingerprints of active meta-memory texts — the
544    /// association engine's global dedup index. A bridge's fingerprint
545    /// is `simhash` of its full meta text (episode IDs embedded), so
546    /// a pair is bridged at most once across the whole store, no
547    /// matter which later pass re-encounters it. Rebuilt from active
548    /// meta entries at `open` (archived connections may legitimately
549    /// re-form) and maintained on every meta store.
550    meta_hashes: HashSet<u64>,
551}

554unsafe impl Sync for LtmStore {}

556impl LtmStore {
557    /// Create a new LTM store at the given base path.
558    /// Creates `{base}.bundles`, `{base}.meta`, and `{base}.dat`.
559    pub fn create(base_path: impl AsRef<Path>, _capacity: u32) -> Result<Self, LtmError> {
560        let base = base_path.as_ref().to_path_buf();
561        let bundles_path = base.with_extension("bundles");
562        let meta_path = base.with_extension("meta");
563        let dat_path = base.with_extension("dat");
564
565        // Create bundle store
566        let bundles = BundleStore::create(&bundles_path).map_err(LtmError::Io)?;
567
568        // Create metadata file
569        let mut meta_file = OpenOptions::new()
570            .read(true)
571            .write(true)
572            .create(true)
573            .truncate(true)
574            .open(&meta_path)
575            .map_err(LtmError::Io)?;
576
577        let meta_header = IndexHeader::new(DEFAULT_INDEX_CAPACITY);
578        meta_file.seek(SeekFrom::Start(0)).map_err(LtmError::Io)?;
579        // SAFETY: `IndexHeader` is `#[repr(C)]`, `Copy`, contains only
580        // primitive types, and `size_of::<IndexHeader>() == 64`
581        // (statically asserted). The slice covers exactly the struct's
582        // in-memory representation.
583        let meta_header_bytes = unsafe {
584            std::slice::from_raw_parts(
585                &meta_header as *const IndexHeader as *const u8,
586                core::mem::size_of::<IndexHeader>(),
587            )
588        };
589        meta_file
590            .write_all(meta_header_bytes)
591            .map_err(LtmError::Io)?;
592        meta_file.sync_all().map_err(LtmError::Io)?;
593
594        // Create data file
595        let mut data_file = OpenOptions::new()
596            .read(true)
597            .write(true)
598            .create(true)
599            .truncate(true)
600            .open(&dat_path)
601            .map_err(LtmError::Io)?;
602
603        let data_header = DataHeader::new();
604        data_file.seek(SeekFrom::Start(0)).map_err(LtmError::Io)?;
605        // SAFETY: `DataHeader` is `#[repr(C)]`, `Copy`, contains only
606        // primitive types, and `size_of::<DataHeader>() == 16`
607        // (statically asserted). The slice covers exactly the struct's
608        // in-memory representation.
609        let data_header_bytes = unsafe {
610            std::slice::from_raw_parts(
611                &data_header as *const DataHeader as *const u8,
612                core::mem::size_of::<DataHeader>(),
613            )
614        };
615        data_file
616            .write_all(data_header_bytes)
617            .map_err(LtmError::Io)?;
618        data_file.sync_all().map_err(LtmError::Io)?;
619
620        Ok(Self {
621            bundles,
622            meta_file,
623            data_file,
624            entries: Vec::new(),
625            id_to_idx: HashMap::new(),
626            active_ids: Vec::new(),
627            meta_header,
628            base_path: base,
629            meta_hashes: HashSet::new(),
630        })
631    }

633    /// Open an existing LTM store.
634    pub fn open(base_path: impl AsRef<Path>) -> Result<Self, LtmError> {
635        let base = base_path.as_ref().to_path_buf();
636        let bundles_path = base.with_extension("bundles");
637        let meta_path = base.with_extension("meta");
638        let dat_path = base.with_extension("dat");
639
640        // Open bundle store
641        let mut bundles = BundleStore::open(&bundles_path).map_err(LtmError::Io)?;
642
643        // Open metadata file
644        let mut meta_file = OpenOptions::new()
645            .read(true)
646            .write(true)
647            .open(&meta_path)
648            .map_err(LtmError::Io)?;
649
650        let meta_len = meta_file.metadata().map_err(LtmError::Io)?.len();
651        if meta_len < 64 || (meta_len - 64) % INDEX_ENTRY_SIZE as u64 != 0 {
652            return Err(LtmError::Io(std::io::Error::new(
653                std::io::ErrorKind::InvalidData,
654                "LTM metadata contains an incomplete header or entry",
655            )));
656        }
657
658        // Read metadata header
659        let mut magic = [0u8; 4];
660        let mut version_bytes = [0u8; 4];
661        let mut capacity_bytes = [0u8; 4];
662        let mut count_bytes = [0u8; 4];
663        let mut next_id_bytes = [0u8; 8];
664        let mut reserved = [0u8; 40];
665        meta_file.seek(SeekFrom::Start(0)).map_err(LtmError::Io)?;
666        meta_file.read_exact(&mut magic).map_err(LtmError::Io)?;
667        meta_file
668            .read_exact(&mut version_bytes)
669            .map_err(LtmError::Io)?;
670        meta_file
671            .read_exact(&mut capacity_bytes)
672            .map_err(LtmError::Io)?;
673        meta_file
674            .read_exact(&mut count_bytes)
675            .map_err(LtmError::Io)?;
676        meta_file
677            .read_exact(&mut next_id_bytes)
678            .map_err(LtmError::Io)?;
679        meta_file.read_exact(&mut reserved).map_err(LtmError::Io)?;
680
681        if magic != LTM_META_MAGIC {
682            return Err(LtmError::IndexBadMagic);
683        }
684        let version = u32::from_le_bytes(version_bytes);
685        if version != LTM_SCHEMA_VERSION {
686            return Err(LtmError::VersionMismatch {
687                expected: LTM_SCHEMA_VERSION,
688                found: version,
689            });
690        }
691
692        let meta_header = IndexHeader {
693            magic,
694            version,
695            capacity: u32::from_le_bytes(capacity_bytes),
696            episode_count: u32::from_le_bytes(count_bytes),
697            next_episode_id: u64::from_le_bytes(next_id_bytes),
698            _reserved: reserved,
699        };
700
701        // Read all metadata entries in one bulk read. The per-entry
702        // `read_exact` loop was one syscall per entry (320K syscalls
703        // at 320K entries); a single bulk read is 1 syscall plus
704        // in-memory parsing. Pre-allocate all collections with the
705        // exact entry count to avoid reallocations.
706        let entry_bytes_len = meta_len as usize - 64;
707        let entry_count = entry_bytes_len / INDEX_ENTRY_SIZE;
708        let mut all_entry_bytes = vec![0u8; entry_bytes_len];
709        meta_file
710            .read_exact(&mut all_entry_bytes)
711            .map_err(LtmError::Io)?;
712
713        let mut entries: Vec<IndexEntry> = Vec::with_capacity(entry_count);
714        let mut id_to_idx: HashMap<u64, usize> = HashMap::with_capacity(entry_count);
715        let mut active_ids: Vec<u64> = Vec::with_capacity(entry_count);
716
717        for i in 0..entry_count {
718            let offset = i * INDEX_ENTRY_SIZE;
719            // SAFETY: `offset` is `i * INDEX_ENTRY_SIZE` where
720            // `i < entry_count` and `entry_bytes_len =
721            // entry_count * INDEX_ENTRY_SIZE`, so
722            // `offset + INDEX_ENTRY_SIZE <= entry_bytes_len`.
723            // The slice is exactly `INDEX_ENTRY_SIZE` bytes.
724            let entry_bytes: &[u8; INDEX_ENTRY_SIZE] = all_entry_bytes
725                [offset..offset + INDEX_ENTRY_SIZE]
726                .try_into()
727                .expect("entry slice length equals INDEX_ENTRY_SIZE");
728            let entry = IndexEntry::from_bytes(entry_bytes);
729
730            // Episode IDs are the primary key used by both the metadata
731            // index and the bundle code. A zero or duplicate ID makes
732            // the in-memory HashMap silently overwrite an earlier entry,
733            // causing retrieval to return the wrong episode. Treat either
734            // condition as on-disk corruption instead.
735            if entry.episode_id == 0 || id_to_idx.contains_key(&entry.episode_id) {
736                return Err(LtmError::Io(std::io::Error::new(
737                    std::io::ErrorKind::InvalidData,
738                    format!(
739                        "LTM metadata contains invalid or duplicate episode ID {}",
740                        entry.episode_id
741                    ),
742                )));
743            }
744
745            // These values are persisted and later used in arithmetic and
746            // ordering. Reject non-finite floats at the trust boundary
747            // rather than allowing a valid-looking metadata file to inject
748            // NaN into retrieval or salience sorting.
749            if !entry.salience.is_finite() || entry.emotional_tag.iter().any(|v| !v.is_finite()) {
750                return Err(LtmError::Io(std::io::Error::new(
751                    std::io::ErrorKind::InvalidData,
752                    format!(
753                        "LTM metadata entry {} contains non-finite values",
754                        entry.episode_id
755                    ),
756                )));
757            }
758
759            let idx = entries.len();
760            if !entry.is_deleted() {
761                // Archived episodes stay in id_to_idx (retrievable
762                // by ID) but are excluded from active_ids (similarity
763                // search, recent_episodes, bundle store).
764                id_to_idx.insert(entry.episode_id, idx);
765                if !entry.is_archived() {
766                    active_ids.push(entry.episode_id);
767                }
768            }
769            entries.push(entry);
770        }
771
772        // Open data file
773        let mut data_file = OpenOptions::new()
774            .read(true)
775            .write(true)
776            .open(&dat_path)
777            .map_err(LtmError::Io)?;
778
779        // Verify data header
780        let mut dmagic = [0u8; 4];
781        let mut dversion = [0u8; 4];
782        data_file.seek(SeekFrom::Start(0)).map_err(LtmError::Io)?;
783        data_file.read_exact(&mut dmagic).map_err(LtmError::Io)?;
784        data_file.read_exact(&mut dversion).map_err(LtmError::Io)?;
785        if dmagic != LTM_DATA_MAGIC {
786            return Err(LtmError::DataBadMagic);
787        }
788        let dver = u32::from_le_bytes(dversion);
789        if dver != LTM_SCHEMA_VERSION {
790            return Err(LtmError::VersionMismatch {
791                expected: LTM_SCHEMA_VERSION,
792                found: dver,
793            });
794        }
795
796        // Validate next_write_offset: must be >= 16 (past the header)
797        // and <= file size. A corrupted or zeroed offset could cause
798        // store() to seek to offset 0 and overwrite the header, or
799        // to seek beyond EOF into a sparse file.
800        let mut offset_bytes = [0u8; 8];
801        data_file
802            .read_exact(&mut offset_bytes)
803            .map_err(LtmError::Io)?;
804        let next_write_offset = u64::from_le_bytes(offset_bytes);
805        let data_file_len = data_file.metadata().map_err(LtmError::Io)?.len();
806
807        // Every non-deleted episode must point to a complete payload record.
808        // Validate this once at startup so a corrupt metadata entry cannot
809        // later seek outside the data file or allocate based on an invalid
810        // compressed length during retrieval.
811        for entry in &entries {
812            if entry.is_deleted() {
813                continue;
814            }
815            if entry.data_offset < 16
816                || entry
817                    .data_offset
818                    .checked_add(4 + entry.compressed_len as u64)
819                    .is_none_or(|end| end > data_file_len)
820                || entry.compressed_len as usize > MAX_COMPRESSED_SIZE
821                || entry.uncompressed_len as usize > MAX_PAYLOAD_SIZE
822            {
823                return Err(LtmError::Io(std::io::Error::new(
824                    std::io::ErrorKind::InvalidData,
825                    format!(
826                        "LTM episode {} points to an invalid payload",
827                        entry.episode_id
828                    ),
829                )));
830            }
831        }
832
833        if next_write_offset < 16 || next_write_offset > data_file_len {
834            return Err(LtmError::Io(std::io::Error::new(
835                std::io::ErrorKind::InvalidData,
836                format!(
837                    "LTM data file has invalid next_write_offset: {} (file size: {})",
838                    next_write_offset, data_file_len
839                ),
840            )));
841        }
842
843        // Reconcile in-memory state with disk. The metadata header's
844        // episode_count and next_episode_id may be stale if a prior
845        // process crashed between writing entries and updating the
846        // header. Recompute from the actual loaded entries and the
847        // bundle store's next_episode_id to ensure consistency.
848        let active_count = active_ids.len() as u32;
849        let max_loaded_id = entries.iter().map(|e| e.episode_id).max().unwrap_or(0);
850        let bundle_next = bundles.next_episode_id();
851        let reconciled_next = max_loaded_id
852            .saturating_add(1)
853            .max(bundle_next)
854            .max(meta_header.next_episode_id);
855        bundles.ensure_next_episode_id(reconciled_next);
856        bundles.sync().map_err(LtmError::Io)?;
857
858        // No meta-memory migration: every episode is a real memory and
859        // stays recallable regardless of which module stored it.
860
861        // Rebuild the association dedup index from every active
862        // entry — O(active) over the in-memory index, no payload
863        // reads. Archived bridges stay out of the set: a connection
864        // that was archived may legitimately re-form.
865        let meta_hashes: HashSet<u64> = active_ids
866            .iter()
867            .filter_map(|id| id_to_idx.get(id))
868            .map(|&idx| &entries[idx])
869            .map(|e| e.association_hash)
870            .collect();
871
872        let mut reconciled_store = Self {
873            bundles,
874            meta_file,
875            data_file,
876            entries,
877            id_to_idx,
878            active_ids,
879            meta_hashes,
880            meta_header: IndexHeader {
881                episode_count: active_count,
882                next_episode_id: reconciled_next,
883                ..meta_header
884            },
885            base_path: base,
886        };
887        // Write the reconciled header back to disk so the next open
888        // sees consistent values.
889        reconciled_store.write_meta_header()?;
890
891        // Rebuild the bundle store when its episode_count is much
892        // higher than the number of real (non-meta) episodes, or when
893        // real episodes are missing from it entirely — e.g. after a
894        // flags repair that restored previously hidden episodes.
895        let real_count = reconciled_store
896            .active_ids
897            .iter()
898            .filter(|&&id| {
899                let Some(&idx) = reconciled_store.id_to_idx.get(&id) else {
900                    return false;
901                };
902                !reconciled_store.entries[idx].is_meta_memory()
903            })
904            .count();
905        let bundle_count = reconciled_store.bundles.episode_count() as usize;
906        // archive/delete/store_meta subtract SDRs as episodes leave
907        // the bundled set, but crash-torn writes and pre-subtraction
908        // stores leave residual dead weight. A small excess of dead
909        // contributions is cheap noise in query_activation — rebuild
910        // only when the pollution is material (over ~25% of the real
911        // set, floor 64) or when real episodes are missing from the
912        // bundles entirely.
913        let drift = bundle_count.saturating_sub(real_count);
914        let drift_threshold = (real_count / 4).max(64);
915        let needs_rebuild = bundle_count != real_count && (bundle_count < real_count || drift > drift_threshold);
916        if needs_rebuild {
917            eprintln!(
918                "[ltm] Bundle store has {bundle_count} episodes but \
919                 {real_count} are real — rebuilding to match"
920            );
921            reconciled_store.rebuild_bundles()?;
922        }
923        Ok(reconciled_store)
924    }

926    /// Open or create.
927    pub fn open_or_create(base_path: impl AsRef<Path>, capacity: u32) -> Result<Self, LtmError> {
928        let base = base_path.as_ref();
929        let has_bundles = base.with_extension("bundles").exists();
930        let has_meta = base.with_extension("meta").exists();
931        let has_dat = base.with_extension("dat").exists();
932        let has_old_idx = base.with_extension("idx").exists();
933
934        // Migrate from v1 if old .idx exists but new files don't
935        if has_old_idx && !has_meta && !has_bundles {
936            return Self::migrate_from_v1(base, capacity);
937        }
938
939        if has_bundles && has_meta && has_dat {
940            Self::open(base)
941        } else if has_bundles || has_meta || has_dat || has_old_idx {
942            Err(LtmError::Io(std::io::Error::new(
943                std::io::ErrorKind::InvalidData,
944                "LTM store is incomplete; existing files were preserved",
945            )))
946        } else {
947            Self::create(base, capacity)
948        }
949    }

951    /// Migrate from v1 (.idx/.dat) to v2 (.bundles/.meta/.dat).
952    fn migrate_from_v1(base: &Path, _capacity: u32) -> Result<Self, LtmError> {
953        eprintln!("[ltm] Migrating from v1 to v2 (SDR/LogHD)...");
954        let idx_path = base.with_extension("idx");
955        let old_dat_path = base.with_extension("dat");
956
957        // Open old store using v1 format — we need to read entries.
958        // We'll read the old index directly since we can't import the
959        // old code. The v1 format is: 64-byte header + 64-byte entries.
960        let old_idx = std::fs::read(&idx_path).map_err(LtmError::Io)?;
961        if old_idx.len() < 64 {
962            return Err(LtmError::IndexBadMagic);
963        }
964        if &old_idx[0..4] != b"LTMI" {
965            return Err(LtmError::IndexBadMagic);
966        }
967
968        let old_version = u32::from_le_bytes(old_idx[4..8].try_into().unwrap_or([0; 4]));
969        if old_version != 1 {
970            return Err(LtmError::VersionMismatch {
971                expected: 1,
972                found: old_version,
973            });
974        }
975        let old_capacity = u32::from_le_bytes(old_idx[8..12].try_into().unwrap_or([0; 4])) as usize;
976        if old_idx.len() != 64 + old_capacity * INDEX_ENTRY_SIZE {
977            return Err(LtmError::Io(std::io::Error::new(
978                std::io::ErrorKind::InvalidData,
979                "v1 metadata length does not match its capacity",
980            )));
981        }
982        let mut next_id = u64::from_le_bytes(old_idx[16..24].try_into().unwrap_or([0; 8])).max(1);
983        let idx_backup = base.with_extension("idx.v1bak");
984        let dat_backup = base.with_extension("dat.v1bak");
985        if idx_backup.exists() || dat_backup.exists() {
986            return Err(LtmError::Io(std::io::Error::new(
987                std::io::ErrorKind::AlreadyExists,
988                "v1 migration backups already exist; recovery is required before retrying",
989            )));
990        }
991
992        // Read the old data file BEFORE creating the new v2 store.
993        // `Self::create()` truncates `{base}.dat`, so if we create
994        // the new store first, the old data file is destroyed before
995        // we can read it.
996        let mut old_data = OpenOptions::new()
997            .read(true)
998            .open(&old_dat_path)
999            .map_err(LtmError::Io)?;
1000
1001        let mut data_header = [0u8; 16];
1002        old_data
1003            .read_exact(&mut data_header)
1004            .map_err(LtmError::Io)?;
1005        if data_header[..4] != LTM_DATA_MAGIC {
1006            return Err(LtmError::DataBadMagic);
1007        }
1008        let data_version = u32::from_le_bytes(data_header[4..8].try_into().unwrap_or([0; 4]));
1009        if data_version != 1 {
1010            return Err(LtmError::VersionMismatch {
1011                expected: 1,
1012                found: data_version,
1013            });
1014        }
1015
1016        // Read all old entries and their payloads into memory before
1017        // the new store is created in a separate staging directory.
1018        struct OldEntry {
1019            episode_id: u64,
1020            timestamp: u64,
1021            salience: f32,
1022            emotional_tag: [f32; 4],
1023        }
1024        let mut old_entries: Vec<(OldEntry, Vec<u8>)> = Vec::new();
1025        for i in 0..old_capacity {
1026            let offset = 64 + i * INDEX_ENTRY_SIZE;
1027            let entry_bytes: &[u8; INDEX_ENTRY_SIZE] = old_idx[offset..offset + INDEX_ENTRY_SIZE]
1028                .try_into()
1029                .map_err(|_| {
1030                    LtmError::Io(std::io::Error::new(
1031                        std::io::ErrorKind::InvalidData,
1032                        "v1 index entry slice length mismatch",
1033                    ))
1034                })?;
1035            let entry = IndexEntry::from_bytes(entry_bytes);
1036            next_id = next_id.max(entry.episode_id.saturating_add(1));
1037            if entry.episode_id == 0 || entry.is_deleted() {
1038                continue;
1039            }
1040            if entry.data_offset < 16 {
1041                return Err(LtmError::Io(std::io::Error::new(
1042                    std::io::ErrorKind::InvalidData,
1043                    "v1 episode points into the data header",
1044                )));
1045            }
1046
1047            // Read the compressed payload from the old data file
1048            old_data
1049                .seek(SeekFrom::Start(entry.data_offset))
1050                .map_err(LtmError::Io)?;
1051            let mut len_bytes = [0u8; 4];
1052            old_data.read_exact(&mut len_bytes).map_err(LtmError::Io)?;
1053            let compressed_len = u32::from_le_bytes(len_bytes) as usize;
1054            if compressed_len > MAX_COMPRESSED_SIZE
1055                || compressed_len != entry.compressed_len as usize
1056            {
1057                return Err(LtmError::DecompressFailed(
1058                    "invalid v1 compressed length".into(),
1059                ));
1060            }
1061            let mut compressed = vec![0u8; compressed_len];
1062            old_data.read_exact(&mut compressed).map_err(LtmError::Io)?;
1063
1064            // Decompress to get the payload
1065            let payload = decompress(&compressed, entry.uncompressed_len as usize)?;
1066            if payload.len() < 52 {
1067                return Err(LtmError::DecompressFailed("v1 payload too short".into()));
1068            }
1069
1070            old_entries.push((
1071                OldEntry {
1072                    episode_id: entry.episode_id,
1073                    timestamp: entry.timestamp,
1074                    salience: entry.salience,
1075                    emotional_tag: entry.emotional_tag,
1076                },
1077                payload,
1078            ));
1079        }
1080
1081        old_entries.sort_unstable_by_key(|(entry, _)| entry.episode_id);
1082        if old_entries
1083            .windows(2)
1084            .any(|pair| pair[0].0.episode_id == pair[1].0.episode_id)
1085        {
1086            return Err(LtmError::Io(std::io::Error::new(
1087                std::io::ErrorKind::InvalidData,
1088                "v1 metadata contains duplicate episode IDs",
1089            )));
1090        }
1091
1092        // Create the new v2 store separately without truncating .dat,
1093        // and keep the original files intact until it is fully durable.
1094        let stage_dir = base.with_extension(format!(
1095            "migration-{}-{}",
1096            std::process::id(),
1097            std::time::SystemTime::now()
1098                .duration_since(std::time::UNIX_EPOCH)
1099                .unwrap_or_default()
1100                .as_nanos(),
1101        ));
1102        std::fs::create_dir(&stage_dir).map_err(LtmError::Io)?;
1103        let result = (|| -> Result<Self, LtmError> {
1104            let stage_base = stage_dir.join("ltm");
1105            let mut new_store = Self::create(&stage_base, DEFAULT_INDEX_CAPACITY)?;
1106            let migrated = old_entries.len();
1107            for (old, payload) in old_entries {
1108                let event_type = payload[0];
1109                let source_module = payload[1];
1110                // Parse the 12 f32 emotional tag from bytes 4..52.
1111                // `try_into` to `[u8; 48]` is exact (48 bytes) and
1112                // avoids the partial-chunk risk of `chunks(4)`.
1113                let tag_bytes: [u8; 48] = payload[4..52]
1114                    .try_into()
1115                    .expect("payload[4..52] is exactly 48 bytes");
1116                let mut full_tag = [0.0f32; EMOTIONAL_TAG_SIZE];
1117                for (j, chunk) in tag_bytes.chunks_exact(4).enumerate() {
1118                    full_tag[j] = f32::from_le_bytes(
1119                        chunk
1120                            .try_into()
1121                            .expect("chunks_exact(4) yields 4-byte chunks"),
1122                    );
1123                }
1124                let text = String::from_utf8_lossy(&payload[52..]).into_owned();
1125
1126                // Store in the new v2 format
1127                new_store.bundles.ensure_next_episode_id(old.episode_id);
1128                new_store.store(
1129                    old.timestamp,
1130                    old.salience,
1131                    full_tag,
1132                    old.emotional_tag,
1133                    event_type,
1134                    source_module,
1135                    &text,
1136                )?;
1137            }
1138            new_store.meta_header.next_episode_id = next_id;
1139            new_store.bundles.ensure_next_episode_id(next_id);
1140            new_store.write_meta_header()?;
1141            new_store.sync()?;
1142            let parent = base.parent().filter(|p| !p.as_os_str().is_empty());
1143            let directory =
1144                std::fs::File::open(parent.unwrap_or(Path::new("."))).map_err(LtmError::Io)?;
1145            std::fs::hard_link(&old_dat_path, &dat_backup).map_err(LtmError::Io)?;
1146            directory.sync_all().map_err(LtmError::Io)?;
1147            for ext in ["bundles", "dat", "meta"] {
1148                std::fs::rename(stage_base.with_extension(ext), base.with_extension(ext))
1149                    .map_err(LtmError::Io)?;
1150            }
1151            new_store.base_path = base.to_path_buf();
1152            eprintln!("[ltm] Migrated {migrated} episodes from v1 to v2");
1153
1154            // Rename the old index as backup; the original data is already backed up.
1155            std::fs::rename(&idx_path, &idx_backup).map_err(LtmError::Io)?;
1156            directory.sync_all().map_err(LtmError::Io)?;
1157            Ok(new_store)
1158        })();
1159        let _ = std::fs::remove_dir_all(&stage_dir);
1160        result
1161    }

1165    /// Store a new episode.
1166    #[allow(clippy::too_many_arguments)]
1167    pub fn store(
1168        &mut self,
1169        timestamp: u64,
1170        salience: f32,
1171        emotional_tag: [f32; EMOTIONAL_TAG_SIZE],
1172        compact_tag: [f32; 4],
1173        event_type: u8,
1174        source_module: u8,
1175        text: &str,
1176    ) -> Result<u64, LtmError> {
1177        let meta_len = self.meta_file.metadata().map_err(LtmError::Io)?.len();
1178        if meta_len != 64 + self.entries.len() as u64 * INDEX_ENTRY_SIZE as u64 {
1179            return Err(LtmError::Io(std::io::Error::new(
1180                std::io::ErrorKind::InvalidData,
1181                "LTM metadata changed or has an incomplete entry; reopen before storing",
1182            )));
1183        }
1184        let next_episode_id = self
1185            .meta_header
1186            .next_episode_id
1187            .max(self.bundles.next_episode_id())
1188            .checked_add(1)
1189            .ok_or_else(|| {
1190                LtmError::Io(std::io::Error::new(
1191                    std::io::ErrorKind::InvalidData,
1192                    "LTM episode ID space exhausted",
1193                ))
1194            })?;
1195        let text_bytes = text.as_bytes();
1196        // An episode with no text is noise, not a memory — and its
1197        // SimHash degenerates to 0, "associating" it with every other
1198        // empty episode at distance 0. Reject at the boundary so all
1199        // producers (STM consolidation, IPC, meta stores) share the
1200        // guard.
1201        if text.trim().is_empty() {
1202            return Err(LtmError::EmptyPayload);
1203        }
1204        // The full payload is 52 bytes (header + emotional tag) + text.
1205        // Check the total payload size, not just the text size, so that
1206        // a text of exactly MAX_PAYLOAD_SIZE doesn't produce a payload
1207        // that exceeds the limit by 52 bytes (which would then fail on
1208        // retrieval).
1209        if 52 + text_bytes.len() > MAX_PAYLOAD_SIZE {
1210            return Err(LtmError::PayloadTooLarge {
1211                size: 52 + text_bytes.len(),
1212                max: MAX_PAYLOAD_SIZE,
1213            });
1214        }
1215
1216        // Sanitize emotional tag values to prevent NaN/inf from
1217        // corrupting the SDR encoding and downstream retrieval. Finite
1218        // evidence is preserved verbatim, including values outside the
1219        // historical normalized range.
1220        let emotional_tag: [f32; EMOTIONAL_TAG_SIZE] = {
1221            let mut sanitized = [0.0f32; EMOTIONAL_TAG_SIZE];
1222            for (i, &v) in emotional_tag.iter().enumerate() {
1223                sanitized[i] = crate::state::sanitize::finite_or(v, 0.0);
1224            }
1225            sanitized
1226        };
1227
1228        // Build payload: [event_type][source_module][pad:2][tag:48][text...]
1229        let mut payload = Vec::with_capacity(52 + text_bytes.len());
1230        payload.push(event_type);
1231        payload.push(source_module);
1232        payload.push(0);
1233        payload.push(0);
1234        for &v in &emotional_tag {
1235            payload.extend_from_slice(&v.to_le_bytes());
1236        }
1237        payload.extend_from_slice(text_bytes);
1238
1239        let compressed = compress(&payload)?;
1240
1241        // Read current data write offset
1242        self.data_file
1243            .seek(SeekFrom::Start(8))
1244            .map_err(LtmError::Io)?;
1245        let mut offset_bytes = [0u8; 8];
1246        self.data_file
1247            .read_exact(&mut offset_bytes)
1248            .map_err(LtmError::Io)?;
1249        let recorded_offset = u64::from_le_bytes(offset_bytes);
1250        let data_offset = self.data_file.metadata().map_err(LtmError::Io)?.len();
1251
1252        // Validate the recorded offset against the header and EOF.
1253        // Always append at EOF, not at a potentially stale offset.
1254        // A crash can leave a payload durable before its offset update;
1255        // retaining those trailing bytes avoids overwriting episodes.
1256        // Offsets outside the existing file indicate corruption.
1257        if recorded_offset < 16 || recorded_offset > data_offset {
1258            return Err(LtmError::Io(std::io::Error::new(
1259                std::io::ErrorKind::InvalidData,
1260                format!("LTM data_offset {recorded_offset} is outside the data file"),
1261            )));
1262        }
1263
1264        // Append compressed payload
1265        let compressed_len = compressed.len() as u32;
1266        self.data_file
1267            .seek(SeekFrom::Start(data_offset))
1268            .map_err(LtmError::Io)?;
1269        self.data_file
1270            .write_all(&compressed_len.to_le_bytes())
1271            .map_err(LtmError::Io)?;
1272        self.data_file
1273            .write_all(&compressed)
1274            .map_err(LtmError::Io)?;
1275        // fsync the payload BEFORE writing the offset that points to
1276        // it. A single fsync after both does not guarantee ordering:
1277        // the kernel may flush the offset (a small write near the
1278        // start of the file) before the much larger payload at the
1279        // end. If a crash occurs between the fsync and the next
1280        // operation, the offset could point to data that hasn't
1281        // reached disk yet, producing a corrupt or zero-filled read
1282        // on reopen.
1283        self.data_file.sync_all().map_err(LtmError::Io)?;
1284
1285        // Update data file's next_write_offset, then fsync again to
1286        // make the offset durable. The payload is already on disk
1287        // from the sync above, so this fsync only needs to flush the
1288        // 8-byte offset update.
1289        let new_offset = data_offset + 4 + compressed.len() as u64;
1290        self.data_file
1291            .seek(SeekFrom::Start(8))
1292            .map_err(LtmError::Io)?;
1293        self.data_file
1294            .write_all(&new_offset.to_le_bytes())
1295            .map_err(LtmError::Io)?;
1296        self.data_file.sync_all().map_err(LtmError::Io)?;
1297
1298        // Compute SimHash (backward compat for association engine)
1299        let assoc_hash = simhash(text);
1300
1301        // Assign episode ID
1302        let episode_id = next_episode_id - 1;
1303        // Every episode's SDR enters the superposition — all stored
1304        // content is recallable regardless of source module.
1305        self.bundles.ensure_next_episode_id(episode_id);
1306        let code = episode_id_to_code(episode_id);
1307
1308        // Encode SDR and add to bundle store
1309        let sdr = encode_episode(text, &emotional_tag, timestamp);
1310        self.bundles.add_episode(&code, &sdr);
1311
1312        // Make the bundle store durable BEFORE writing the metadata
1313        // entry that references this episode ID. Without this, a crash
1314        // after `write_meta_header()` but before the next `sync()` would
1315        // leave the metadata file with episode N but the bundle file's
1316        // `next_episode_id` still at N-1. On reopen, `store()` would
1317        // reuse N-1, producing a duplicate episode ID that corrupts
1318        // `id_to_idx` and `active_ids`.
1319        self.bundles.sync().map_err(LtmError::Io)?;
1320
1321        // Sanitize salience and compact_tag to prevent NaN/inf from
1322        // corrupting the index entry, SimHash scoring, and SDR encoding.
1323        // Salience is evidence and preserves any finite value; compact_tag
1324        // remains a bounded derived affective summary.
1325        let salience = crate::state::sanitize::finite_or(salience, 0.0);
1326        let compact_tag = [
1327            crate::state::sanitize::finite_clamp(compact_tag[0], 0.0, 1.0),
1328            crate::state::sanitize::finite_clamp(compact_tag[1], -1.0, 1.0),
1329            crate::state::sanitize::finite_clamp(compact_tag[2], 0.0, 1.0),
1330            crate::state::sanitize::finite_clamp(compact_tag[3], 0.0, 1.0),
1331        ];
1332
1333        // Build index entry
1334        let entry = IndexEntry::new(
1335            episode_id,
1336            data_offset,
1337            compressed_len,
1338            payload.len() as u32,
1339            timestamp,
1340            salience,
1341            assoc_hash,
1342            compact_tag,
1343            source_module,
1344        );
1345
1346        // No automatic meta-memory marking: episodes stay recallable
1347        // no matter which module stored them. Only an explicit
1348        // `store_meta` call sets the flag now.
1349
1350        // Append to metadata file
1351        // SAFETY: `entry` is a valid local `IndexEntry` reference that lives
1352        // for the entire scope. `INDEX_ENTRY_SIZE` equals
1353        // `size_of::<IndexEntry>()`, so the slice covers exactly the struct's
1354        // in-memory representation with no out-of-bounds access.
1355        let entry_bytes = unsafe {
1356            std::slice::from_raw_parts(&entry as *const IndexEntry as *const u8, INDEX_ENTRY_SIZE)
1357        };
1358        self.meta_file
1359            .seek(SeekFrom::End(0))
1360            .map_err(LtmError::Io)?;
1361        self.meta_file
1362            .write_all(entry_bytes)
1363            .map_err(LtmError::Io)?;
1364
1365        // Update metadata header in file. `write_meta_header` fsyncs
1366        // the metadata file, which also makes the entry appended
1367        // above durable — one fsync for both.
1368        self.meta_header.episode_count += 1;
1369        self.meta_header.next_episode_id = next_episode_id;
1370        self.write_meta_header()?;
1371
1372        // Update in-memory indices
1373        let idx = self.entries.len();
1374        self.entries.push(entry);
1375        self.id_to_idx.insert(episode_id, idx);
1376        self.active_ids.push(episode_id);
1377        // Record the fingerprint so the association engine's dedup
1378        // sees this episode — the set covers every stored episode,
1379        // not just bookkeeping ones.
1380        self.meta_hashes.insert(assoc_hash);
1381
1382        Ok(episode_id)
1383    }

1436    /// Retrieve an episode by ID.
1437    ///
1438    /// Archived episodes are retrievable (unlike deleted ones) — this
1439    /// is the key difference between archiving and deletion. Dream
1440    /// insights and association chains that reference archived episodes
1441    /// still resolve, preserving memory integrity across sleep
1442    /// compression cycles.
1443    pub fn retrieve(&mut self, episode_id: u64) -> Result<Episode, LtmError> {
1444        let idx = *self
1445            .id_to_idx
1446            .get(&episode_id)
1447            .ok_or(LtmError::EpisodeNotFound)?;
1448        let entry = self.entries[idx];
1449
1450        // Guard against deleted entries. `id_to_idx` should not
1451        // contain deleted entries (delete removes them), but
1452        // defense-in-depth: if a bug or corruption leaves a stale
1453        // mapping, don't return a deleted episode.
1454        // Archived episodes ARE retrievable — that's the point.
1455        if entry.is_deleted() {
1456            return Err(LtmError::EpisodeNotFound);
1457        }
1458
1459        // Validate entry.data_offset: must be >= 16 (past the data
1460        // file header). A corrupted index entry with data_offset < 16
1461        // would read the file header as if it were a compressed payload.
1462        if entry.data_offset < 16 {
1463            return Err(LtmError::Io(std::io::Error::new(
1464                std::io::ErrorKind::InvalidData,
1465                format!(
1466                    "LTM entry {} has invalid data_offset {} (< 16)",
1467                    episode_id, entry.data_offset
1468                ),
1469            )));
1470        }
1471
1472        self.data_file
1473            .seek(SeekFrom::Start(entry.data_offset))
1474            .map_err(LtmError::Io)?;
1475        let mut len_bytes = [0u8; 4];
1476        self.data_file
1477            .read_exact(&mut len_bytes)
1478            .map_err(LtmError::Io)?;
1479        let compressed_len = u32::from_le_bytes(len_bytes) as usize;
1480
1481        if compressed_len > MAX_COMPRESSED_SIZE {
1482            return Err(LtmError::DecompressFailed(format!(
1483                "compressed length {compressed_len} exceeds maximum {MAX_COMPRESSED_SIZE}"
1484            )));
1485        }
1486
1487        let mut compressed = vec![0u8; compressed_len];
1488        self.data_file
1489            .read_exact(&mut compressed)
1490            .map_err(LtmError::Io)?;
1491
1492        // Cap uncompressed_len at MAX_PAYLOAD_SIZE to prevent OOM
1493        // from a corrupted index entry with a huge uncompressed_len.
1494        let unc_len = entry.uncompressed_len as usize;
1495        let payload = decompress(&compressed, unc_len)?;
1496        if payload.len() > MAX_PAYLOAD_SIZE {
1497            return Err(LtmError::DecompressFailed(format!(
1498                "decompressed length {} exceeds maximum {}",
1499                payload.len(),
1500                MAX_PAYLOAD_SIZE
1501            )));
1502        }
1503        // Verify the decompressed length matches the index entry's
1504        // uncompressed_len (capped). A mismatch indicates corruption.
1505        if payload.len() != unc_len {
1506            return Err(LtmError::DecompressFailed(format!(
1507                "decompressed length {} does not match expected {}",
1508                payload.len(),
1509                unc_len
1510            )));
1511        }
1512
1513        if payload.len() < 52 {
1514            return Err(LtmError::DecompressFailed("payload too short".to_string()));
1515        }
1516        let event_type = payload[0];
1517        let source_module = payload[1];
1518
1519        // Parse the 12 f32 emotional tag from bytes 4..52. The slice
1520        // is exactly 48 bytes (12 × 4), so `try_into` to `[u8; 48]`
1521        // is exact — no partial-chunk risk that `chunks(4)` would hide.
1522        let tag_bytes: [u8; 48] = payload[4..52]
1523            .try_into()
1524            .expect("payload[4..52] is exactly 48 bytes");
1525        let mut full_tag = [0.0f32; EMOTIONAL_TAG_SIZE];
1526        for (i, chunk) in tag_bytes.chunks_exact(4).enumerate() {
1527            full_tag[i] = f32::from_le_bytes(
1528                chunk
1529                    .try_into()
1530                    .expect("chunks_exact(4) yields 4-byte chunks"),
1531            );
1532        }
1533
1534        let text = String::from_utf8_lossy(&payload[52..]).into_owned();
1535
1536        Ok(Episode {
1537            episode_id: entry.episode_id,
1538            timestamp: entry.timestamp,
1539            salience: entry.salience,
1540            association_hash: entry.association_hash,
1541            emotional_tag: entry.emotional_tag,
1542            full_emotional_tag: full_tag,
1543            event_type,
1544            source_module,
1545            text,
1546        })
1547    }

1549    /// Find episodes similar to the given query text.
1550    ///
1551    /// Uses the LogHD bundle store for SDR-based similarity matching.
1552    /// Returns up to `limit` results sorted by distance (ascending).
1553    pub fn find_similar(&self, query: &str, limit: usize) -> Vec<(u64, u32, &IndexEntry)> {
1554        // Encode query as SDR (content only — no emotional/temporal
1555        // context for a text query)
1556        let query_sdr = crate::store::sdr::encode_content(query);
1557        let activation = self.bundles.query_activation(&query_sdr);
1558
1559        // Score each active episode by the raw L2 distance
1560        // ||A − E||² between the query activation A and the episode's
1561        // expected activation E. Lower = more similar.
1562        //
1563        // We deliberately do NOT normalize the activation vector. The
1564        // LogHD bundle store is additive — each `add_episode` call
1565        // superimposes `weight * SDR` onto every bundle — so the
1566        // activation magnitude encodes how strongly the query overlaps
1567        // with each episode's content. Normalizing A to unit L2 norm
1568        // discards that magnitude information. When episode codes are
1569        // concentrated in a few of the 20 code dimensions (the common
1570        // case for small episode counts, since IDs 1–4 only use
1571        // positions 18–19), the expected activations are nearly
1572        // collinear and normalization makes them indistinguishable
1573        // regardless of their code digit values. The sort then
1574        // degenerates, returning wrong matches.
1575        //
1576        // Instead, we compute distances as f32 (which cannot overflow
1577        // regardless of episode count) and then scale to u32
1578        // adaptively — dividing by the maximum distance so the full
1579        // u32 range is used without saturation. This preserves
1580        // ordering for both small and large episode counts.
1581        let mut scored: Vec<(u64, f32, &IndexEntry)> = Vec::new();
1582
1583        for &id in &self.active_ids {
1584            let Some(&idx) = self.id_to_idx.get(&id) else {
1585                continue;
1586            };
1587            let entry = &self.entries[idx];
1588
1589            // Skip meta-memories (association traces, dream insights),
1590            // deleted, and archived entries. These are internal
1591            // bookkeeping episodes or excluded entries, not real
1592            // experiences. Including them in retrieval results floods
1593            // find_similar with `[association] episode X ↔ episode Y`
1594            // entries that crowd out actual conversation/learning
1595            // memories. `is_excluded_from_search()` is defense-in-depth:
1596            // active_ids should already exclude deleted/archived entries,
1597            // but a bug or corruption could leave a stale mapping.
1598            if entry.is_excluded_from_search() {
1599                continue;
1600            }
1601
1602            let code = episode_id_to_code(id);
1603            let expected = code_to_expected_activation(&code);
1604
1605            let mut dist_sq = 0.0f32;
1606            for j in 0..LOGHD_N {
1607                let diff = activation[j] - expected[j];
1608                dist_sq += diff * diff;
1609            }
1610
1611            scored.push((id, dist_sq, entry));
1612        }
1613
1614        // Sort by f32 distance (ascending) before converting to u32,
1615        // so the ordering is determined by full-precision distances.
1616        scored.sort_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal));
1617        scored.truncate(limit);
1618
1619        // Adaptive scaling: map f32 distances to u32 preserving order.
1620        // Dividing by the max distance uses the full u32 range without
1621        // overflow, regardless of whether there are 4 episodes or 4
1622        // million.
1623        let max_dist = scored.iter().map(|(_, d, _)| *d).fold(0.0f32, f32::max);
1624        let scale = if max_dist > 0.0 && max_dist.is_finite() {
1625            (u32::MAX as f32) / max_dist
1626        } else {
1627            1.0
1628        };
1629
1630        scored
1631            .into_iter()
1632            .map(|(id, dist, entry)| {
1633                let scaled = if dist.is_finite() {
1634                    (dist * scale).min(u32::MAX as f32) as u32
1635                } else {
1636                    u32::MAX
1637                };
1638                (id, scaled, entry)
1639            })
1640            .collect()
1641    }

1643    /// Find episodes similar to the given query text using SimHash.
1644    ///
1645    /// This is an O(n) linear scan that compares the query's SimHash
1646    /// fingerprint against every active (non-meta, non-deleted) episode's
1647    /// SimHash. Unlike `find_similar` (which uses the LogHD bundle store's
1648    /// superposition), this approach has no capacity limit and no
1649    /// superposition noise — it gives exact Hamming distances regardless
1650    /// of how many episodes are stored.
1651    ///
1652    /// Returns up to `limit` results sorted by Hamming distance (ascending).
1653    pub fn find_similar_hash(&self, query: &str, limit: usize) -> Vec<(u64, u32, &IndexEntry)> {
1654        let query_hash = simhash(query);
1655
1656        let mut scored: Vec<(u64, u32, &IndexEntry)> = Vec::new();
1657        for &id in &self.active_ids {
1658            let Some(&idx) = self.id_to_idx.get(&id) else {
1659                continue;
1660            };
1661            let entry = &self.entries[idx];
1662
1663            // Skip meta-memories, deleted, and archived entries.
1664            // (active_ids should already exclude deleted/archived,
1665            // but defense-in-depth.)
1666            if entry.is_excluded_from_search() {
1667                continue;
1668            }
1669
1670            let dist = entry.hash_distance(query_hash);
1671            scored.push((id, dist, entry));
1672        }
1673
1674        // Sort by Hamming distance (ascending) — lower is more similar.
1675        scored.sort_by_key(|a| a.1);
1676        scored.truncate(limit);
1677        scored
1678    }

1699                continue;
1700            }
1701            scored.push((entry.episode_id, entry.hash_distance(query_hash), entry));
1702        }
1703        scored.sort_by_key(|a| a.1);
1704        scored.truncate(limit);
1705        scored
1706    }

1773    /// Soft-delete an episode.
1774    pub fn delete(&mut self, episode_id: u64) -> Result<(), LtmError> {
1775        let idx = *self
1776            .id_to_idx
1777            .get(&episode_id)
1778            .ok_or(LtmError::EpisodeNotFound)?;
1779
1780        // Read the payload while the entry is still readable —
1781        // `retrieve` rejects deleted entries — so its SDR
1782        // contribution can be subtracted once the durable writes
1783        // land. Skipped for meta-memories (never bundled) and
1784        // already-archived entries (unbundled when they left the
1785        // active set).
1786        let unbundle_episode =
1787            if !self.entries[idx].is_meta_memory() && !self.entries[idx].is_archived() {
1788                self.retrieve(episode_id).ok()
1789            } else {
1790                None
1791            };
1792
1793        self.entries[idx].mark_deleted();
1794
1795        // Only the flags byte changed — write just that byte instead
1796        // of rewriting the full 64-byte entry. The flags field is at
1797        // offset 60 within each 64-byte IndexEntry.
1798        let flags_offset = 64 + (idx as u64 * INDEX_ENTRY_SIZE as u64) + 60;
1799        self.meta_file
1800            .seek(SeekFrom::Start(flags_offset))
1801            .map_err(LtmError::Io)?;
1802        self.meta_file
1803            .write_all(&[self.entries[idx].flags])
1804            .map_err(LtmError::Io)?;
1805        // Decrement episode_count and write the header (which fsyncs
1806        // the metadata file, covering the flags write above).
1807        let previous_count = self.meta_header.episode_count;
1808        if !self.entries[idx].is_archived() {
1809            self.meta_header.episode_count = previous_count.saturating_sub(1);
1810        }
1811        if let Err(error) = self.write_meta_header() {
1812            self.meta_header.episode_count = previous_count;
1813            return Err(error);
1814        }
1815
1816        // Only update in-memory indices after the durable write
1817        // succeeds. If `write_meta_header` failed above, the entry is
1818        // marked deleted in `entries` but still present in
1819        // `id_to_idx`/`active_ids` — the next `open()` will see the
1820        // on-disk deleted flag and exclude it from `active_ids`.
1821        if let Some(episode) = unbundle_episode {
1822            self.unbundle(episode_id, &episode);
1823        }
1824        self.id_to_idx.remove(&episode_id);
1825        self.active_ids.retain(|id| *id != episode_id);
1826
1827        Ok(())
1828    }

1830    /// Archive an episode.
1831    ///
1832    /// Unlike `delete`, archiving preserves the episode in `id_to_idx`
1833    /// so it remains retrievable by ID. The episode is only removed
1834    /// from `active_ids`, which excludes it from similarity search,
1835    /// recent-episode scans, and the bundle store — but not from
1836    /// direct retrieval by ID.
1837    ///
1838    /// This is the correct operation for sleep compression: the
1839    /// episode's semantic content has been extracted into the concept
1840    /// network, so it no longer needs to be in the active similarity
1841    /// index. But the episode itself must survive — dream insights
1842    /// and association chains reference it by ID, and the conversation
1843    /// history is irreplaceable ground truth.
1844    pub fn archive(&mut self, episode_id: u64) -> Result<(), LtmError> {
1845        let idx = *self
1846            .id_to_idx
1847            .get(&episode_id)
1848            .ok_or(LtmError::EpisodeNotFound)?;
1849
1850        // Already archived — nothing to do.
1851        if self.entries[idx].is_archived() {
1852            return Ok(());
1853        }
1854
1855        let flags = self.entries[idx].flags | IndexEntry::ARCHIVED_FLAG;
1856
1857        // Only the flags byte changed — write just that byte instead
1858        // of rewriting the full 64-byte entry. The flags field is at
1859        // offset 60 within each 64-byte IndexEntry.
1860        let flags_offset = 64 + (idx as u64 * INDEX_ENTRY_SIZE as u64) + 60;
1861        self.meta_file
1862            .seek(SeekFrom::Start(flags_offset))
1863            .map_err(LtmError::Io)?;
1864        self.meta_file.write_all(&[flags]).map_err(LtmError::Io)?;
1865        // Decrement episode_count (active count) and fsync via header.
1866        let previous_count = self.meta_header.episode_count;
1867        self.meta_header.episode_count = previous_count.saturating_sub(1);
1868        if let Err(error) = self.write_meta_header() {
1869            self.meta_header.episode_count = previous_count;
1870            return Err(error);
1871        }
1872        self.entries[idx].mark_archived();
1873
1874        // The SDR contribution stays in the bundle superposition
1875        // after archival — subtract it so routine sleep-cycle
1876        // churn does not accumulate as dead weight that the
1877        // startup reconciliation has to rebuild away every boot.
1878        // Meta-memories were never bundled. A failed read leaves
1879        // bounded drift, which the open() tolerance absorbs.
1880        if !self.entries[idx].is_meta_memory() {
1881            match self.retrieve(episode_id) {
1882                Ok(episode) => self.unbundle(episode_id, &episode),
1883                Err(e) => eprintln!("[ltm] could not unbundle archived episode {episode_id}: {e}"),
1884            }
1885        }
1886
1887        // Remove from active_ids but keep in id_to_idx.
1888        self.active_ids.retain(|id| *id != episode_id);
1889
1890        Ok(())
1891    }

1893    /// Un-archive an episode — reconsolidation.
1894    ///
1895    /// The mirror of `archive`: retrieval is what earns a memory its
1896    /// place in the active set, so when a deep-recall cue reaches into
1897    /// the cold tier and surfaces an episode, the episode comes back.
1898    /// This clears the archived flag on disk, returns the ID to
1899    /// `active_ids`, re-adds its SDR to the bundle superposition (the
1900    /// inverse of the subtraction `archive` performs), and floors its
1901    /// salience at `RECONSOLIDATION_SALIENCE_FLOOR` so the next
1902    /// compaction pass retains what recall actually used. A revived
1903    /// episode that is never touched again can still be re-archived by
1904    /// a later sweep — forgetting stays earned in both directions.
1905    pub fn unarchive(&mut self, episode_id: u64) -> Result<(), LtmError> {
1906        let idx = *self
1907            .id_to_idx
1908            .get(&episode_id)
1909            .ok_or(LtmError::EpisodeNotFound)?;
1910
1911        // Already active — nothing to do.
1912        if !self.entries[idx].is_archived() {
1913            return Ok(());
1914        }
1915
1916        let flags = self.entries[idx].flags & !IndexEntry::ARCHIVED_FLAG;
1917        let new_salience = self.entries[idx]
1918            .salience
1919            .max(RECONSOLIDATION_SALIENCE_FLOOR);
1920        let salience_changed = (new_salience - self.entries[idx].salience).abs() > f32::EPSILON;
1921
1922        // flags sits at offset 60 and salience at offset 40 inside the
1923        // 64-byte entry — same surgical writes `archive` uses.
1924        let entry_base = 64 + idx as u64 * INDEX_ENTRY_SIZE as u64;
1925        self.meta_file
1926            .seek(SeekFrom::Start(entry_base + 60))
1927            .map_err(LtmError::Io)?;
1928        self.meta_file.write_all(&[flags]).map_err(LtmError::Io)?;
1929        if salience_changed {
1930            self.meta_file
1931                .seek(SeekFrom::Start(entry_base + 40))
1932                .map_err(LtmError::Io)?;
1933            self.meta_file
1934                .write_all(&new_salience.to_le_bytes())
1935                .map_err(LtmError::Io)?;
1936        }
1937        let previous_count = self.meta_header.episode_count;
1938        self.meta_header.episode_count = previous_count.saturating_add(1);
1939        if let Err(error) = self.write_meta_header() {
1940            self.meta_header.episode_count = previous_count;
1941            return Err(error);
1942        }
1943        self.entries[idx].flags = flags;
1944        self.entries[idx].salience = new_salience;
1945
1946        // Re-bundle the SDR so the bundle-store path sees the revived
1947        // episode again — the inverse of `archive`'s subtraction.
1948        // Meta-memories were never bundled; a failed read leaves
1949        // bounded drift the startup reconciliation tolerates.
1950        if !self.entries[idx].is_meta_memory() {
1951            match self.retrieve(episode_id) {
1952                Ok(episode) => {
1953                    let code = episode_id_to_code(episode_id);
1954                    let sdr = encode_episode(
1955                        &episode.text,
1956                        &episode.full_emotional_tag,
1957                        episode.timestamp,
1958                    );
1959                    self.bundles.add_episode(&code, &sdr);
1960                }
1961                Err(e) => {
1962                    eprintln!("[ltm] could not re-bundle revived episode {episode_id}: {e}")
1963                }
1964            }
1965        }
1966
1967        // Back on the active list — appended like a fresh store, which
1968        // is what reconsolidation semantically is.
1969        self.active_ids.push(episode_id);
1970
1971        Ok(())
1972    }

1974    /// Archive many episodes in one pass — the bulk form of
1975    /// [`Self::archive`].
1976    ///
1977    /// Per-episode work is identical (flags byte write, active-list
1978    /// removal, SDR subtraction for non-meta episodes), but the
1979    /// metadata header is written and fsynced once for the whole
1980    /// batch instead of once per episode — the difference between
1981    /// seconds and minutes at ~200K episodes. Unknown, deleted, and
1982    /// already-archived IDs are skipped. Returns the number archived.
1983    pub fn archive_many(&mut self, ids: &[u64]) -> Result<u32, LtmError> {
1984        let mut archived_ids: HashSet<u64> = HashSet::with_capacity(ids.len());
1985        for &episode_id in ids {
1986            let Some(&idx) = self.id_to_idx.get(&episode_id) else {
1987                continue;
1988            };
1989            if self.entries[idx].is_deleted() || self.entries[idx].is_archived() {
1990                continue;
1991            }
1992
1993            // Subtract the SDR contribution for real episodes —
1994            // meta-memories were never bundled, same as `archive`.
1995            if !self.entries[idx].is_meta_memory()
1996                && let Ok(episode) = self.retrieve(episode_id)
1997            {
1998                self.unbundle(episode_id, &episode);
1999            }
2000
2001            // Disk first, memory after — a failed write must not
2002            // diverge the in-memory index from the file. The flags
2003            // field is at offset 60 within each 64-byte entry.
2004            let flags_offset = 64 + (idx as u64 * INDEX_ENTRY_SIZE as u64) + 60;
2005            let flags = self.entries[idx].flags | IndexEntry::ARCHIVED_FLAG;
2006            self.meta_file
2007                .seek(SeekFrom::Start(flags_offset))
2008                .map_err(LtmError::Io)?;
2009            self.meta_file.write_all(&[flags]).map_err(LtmError::Io)?;
2010            self.entries[idx].flags = flags;
2011            archived_ids.insert(episode_id);
2012        }
2013
2014        if archived_ids.is_empty() {
2015            return Ok(0);
2016        }
2017
2018        self.meta_header.episode_count = self
2019            .meta_header
2020            .episode_count
2021            .saturating_sub(archived_ids.len() as u32);
2022        self.write_meta_header()?;
2023        self.active_ids.retain(|id| !archived_ids.contains(id));
2024
2025        Ok(archived_ids.len() as u32)
2026    }

2074    /// Read source_module from a compressed payload header.
2075    ///
2076    /// After successfully reading the source_module, **backfills** it
2077    /// into the index entry (both in memory and on disk) so that future
2078    /// calls never need to decompress the same entry again. This is
2079    /// critical for performance: without backfilling, every call to
2080    /// `recent_episodes` with a source_module filter would decompress
2081    /// all old v2 entries (which have `source_module == 0` in the
2082    /// index) on every single call — potentially hundreds of thousands
2083    /// of decompressions per invocation.
2084    fn source_module_of(&mut self, idx: usize) -> Option<u8> {
2085        let entry = &self.entries[idx];
2086        let data_offset = entry.data_offset;
2087        // Validate data_offset before seeking — a corrupted entry with
2088        // data_offset < 16 would read the data file header as if it
2089        // were a compressed payload. `retrieve()` validates this too,
2090        // but `source_module_of` is called independently during
2091        // `recent_episodes` filtering.
2092        if data_offset < 16 {
2093            return None;
2094        }
2095        self.data_file.seek(SeekFrom::Start(data_offset)).ok()?;
2096        let mut len_bytes = [0u8; 4];
2097        self.data_file.read_exact(&mut len_bytes).ok()?;
2098        let compressed_len = u32::from_le_bytes(len_bytes) as usize;
2099        if compressed_len > MAX_COMPRESSED_SIZE {
2100            return None;
2101        }
2102        let mut compressed = vec![0u8; compressed_len];
2103        self.data_file.read_exact(&mut compressed).ok()?;
2104        // Use `take(2)` to read only the first 2 bytes of the
2105        // decompressed stream (event_type + source_module), avoiding
2106        // a full decompression just to read the source module byte.
2107        let decoder = ZlibDecoder::new(&compressed[..]);
2108        let mut header = [0u8; 2];
2109        decoder.take(2).read_exact(&mut header).ok()?;
2110        let module = header[1];
2111
2112        // Backfill: write the discovered source_module back into the
2113        // index entry so future calls skip the decompression entirely.
2114        // The source_module field is at offset 61 within the 64-byte
2115        // IndexEntry. On disk, the entry starts at 64 + idx * 64.
2116        if module != 0 {
2117            let disk_offset = 64 + (idx as u64) * (INDEX_ENTRY_SIZE as u64) + 61;
2118            // Write to disk BEFORE updating the in-memory entry. If
2119            // seek fails, we must not write at the wrong position
2120            // (which could corrupt other index entries or the header).
2121            // If write fails, the in-memory entry must not diverge
2122            // from disk. Only update memory after the disk write
2123            // succeeds.
2124            if self.meta_file.seek(SeekFrom::Start(disk_offset)).is_ok()
2125                && self.meta_file.write_all(&[module]).is_ok()
2126            {
2127                self.entries[idx].source_module = module;
2128            }
2129        }
2130        Some(module)
2131    }

2278    /// Rebuild the LogHD bundle store from only real (non-meta,
2279    /// non-deleted) episodes.

2282    /// every episode ever stored, including meta-memories (association
2283    /// traces, dream insights, autobiographical traces). These
2284    /// meta-memory SDRs pollute the bundle store's superposition,
2285    /// drowning out the signal from real episodes and degrading
2286    /// `find_similar` accuracy. With ~240K meta-memories out of ~292K
2287    /// total episodes, the bundle store is saturated — every query
2288    /// returns near-uniform activations.

2291    /// only real episodes (those without the `META_MEMORY_FLAG` and
2292    /// without the `DELETED_FLAG`). The episode IDs and codes are
2293    /// preserved — only the bundle superposition is rebuilt.

2297        let bundles_path = self.base_path.with_extension("bundles");
2298
2299        // Collect the text, emotional tag, and timestamp of every real
2300        // episode before we drop the old bundle store. We need to
2301        // re-encode each episode's SDR, which requires the full text
2302        // and emotional tag — not just the index entry.
2303        struct RealEpisode {
2304            episode_id: u64,
2305            timestamp: u64,
2306            text: String,
2307            full_tag: [f32; EMOTIONAL_TAG_SIZE],
2308        }
2309
2310        let mut real_episodes: Vec<RealEpisode> = Vec::new();
2311        for id in self.active_ids.clone() {
2312            let Some(&idx) = self.id_to_idx.get(&id) else {
2313                continue;
2314            };
2315            let entry = self.entries[idx];
2316
2317            // Skip meta-memories, deleted, and archived entries.
2318            if entry.is_excluded_from_search() {
2319                continue;
2320            }
2321
2322            // Read the episode payload to get the text and full
2323            // emotional tag. The index entry only has the compact
2324            // 4-element tag; the full 12-element tag is in the payload.
2325            let episode = self.retrieve(id)?;
2326            real_episodes.push(RealEpisode {
2327                episode_id: id,
2328                timestamp: episode.timestamp,
2329                text: episode.text,
2330                full_tag: episode.full_emotional_tag,
2331            });
2332        }
2333
2334        let real_count = real_episodes.len() as u64;
2335        eprintln!(
2336            "[ltm] Rebuilding bundle store: {real_count} real episodes \
2337             (excluded {} meta-memories and deleted entries)",
2338            self.active_ids.len().saturating_sub(real_episodes.len())
2339        );
2340
2341        // Keep the old bundle store while building a replacement.
2342        // The Drop impl calls msync + munmap + close, so the old
2343        // mapping must never share a file that we truncate.
2344        let rebuild_path = self.base_path.with_extension(format!(
2345            "bundles.rebuild-{}-{}",
2346            std::process::id(),
2347            std::time::SystemTime::now()
2348                .duration_since(std::time::UNIX_EPOCH)
2349                .unwrap_or_default()
2350                .as_nanos(),
2351        ));
2352        // Create an exclusive temporary store alongside the original
2353        // so a completed rebuild can be installed by atomic rename.
2354        let mut rebuilt = BundleStore::create_new(&rebuild_path).map_err(LtmError::Io)?;
2355
2356        // Populate the fresh (zeroed) store without changing the original.
2357        let result = (|| -> Result<(), LtmError> {
2358            // Re-add each real episode's SDR to the rebuilt bundle store.
2359            // We use the same episode IDs and codes as before — only the
2360            // bundle superposition is rebuilt.
2361            for ep in &real_episodes {
2362                let code = episode_id_to_code(ep.episode_id);
2363                let sdr = encode_episode(&ep.text, &ep.full_tag, ep.timestamp);
2364                rebuilt.add_episode(&code, &sdr);
2365            }
2366            rebuilt.ensure_next_episode_id(
2367                self.meta_header
2368                    .next_episode_id
2369                    .max(self.bundles.next_episode_id()),
2370            );
2371
2372            // Sync the rebuilt bundle store to disk.
2373            rebuilt.sync().map_err(LtmError::Io)?;
2374            std::fs::rename(&rebuild_path, &bundles_path).map_err(LtmError::Io)?;
2375            self.bundles = rebuilt;
2376            let parent = bundles_path.parent().filter(|p| !p.as_os_str().is_empty());
2377            std::fs::File::open(parent.unwrap_or(Path::new(".")))
2378                .and_then(|dir| dir.sync_all())
2379                .map_err(LtmError::Io)
2380        })();
2381        if result.is_err() {
2382            let _ = std::fs::remove_file(&rebuild_path);
2383        }
2384        result?;
2385
2386        eprintln!("[ltm] Bundle store rebuilt with {real_count} real episodes");
2387        Ok(real_count)
2388    }
2389}

2395fn compress(data: &[u8]) -> Result<Vec<u8>, LtmError> {
2396    let mut encoder = ZlibEncoder::new(Vec::new(), Compression::default());
2397    encoder
2398        .write_all(data)
2399        .map_err(|e| LtmError::CompressFailed(e.to_string()))?;
2400    encoder
2401        .finish()
2402        .map_err(|e| LtmError::CompressFailed(e.to_string()))
2403}
2404
2405fn decompress(data: &[u8], expected_len: usize) -> Result<Vec<u8>, LtmError> {
2406    // Cap expected_len at MAX_PAYLOAD_SIZE to prevent OOM from a
2407    // corrupted index entry with a huge uncompressed_len. The caller
2408    // already caps it, but defense-in-depth.
2409    let capped_len = expected_len.min(MAX_PAYLOAD_SIZE);
2410    let decoder = ZlibDecoder::new(data);
2411    // Use `take(capped_len)` to bound the decompressed output. Without
2412    // this, `read_to_end` would ignore `with_capacity` and decompress
2413    // a zlib bomb to many megabytes before the post-read length check
2414    // fires. `take` limits the total bytes read from the decoder to
2415    // `capped_len`, preventing unbounded allocation.
2416    let mut result = Vec::with_capacity(capped_len);
2417    decoder
2418        .take(capped_len as u64 + 1)
2419        .read_to_end(&mut result)
2420        .map_err(|e| LtmError::DecompressFailed(e.to_string()))?;
2421    if expected_len > MAX_PAYLOAD_SIZE || result.len() != expected_len {
2422        return Err(LtmError::DecompressFailed(format!(
2423            "decompressed length {} does not match expected {expected_len} within maximum {MAX_PAYLOAD_SIZE}",
2424            result.len(),
2425        )));
2426    }
2427    Ok(result)
2428}
