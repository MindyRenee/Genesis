1//! Gentle forgetting — the power-law decay of memory accessibility.
2//!
3//! Forgetting is the mirror of encoding: the store's `store()` binds a
4//! trace, and this module decides how that trace's accessibility decays
5//! over time. The model follows the empirical biology rather than an
6//! arbitrary eviction policy:
7//!
8//! 1. **Power-law trace decay** (Anderson & Schooler 1991; Wixted &
9//!    Ebbesen 1991): each presentation of a memory leaves a trace whose
10//!    strength decays as t^(−d), d ≈ 0.5. Repeated presentation leaves
11//!    multiple traces; total trace mass is the sum of the surviving
12//!    strengths, M_i = Σ_k (age_k)^(−d).
13//!
14//! 2. **Base-level activation** (ACT-R): B_i = ln M_i + w·salience.
15//!    Retrieval succeeds when activation exceeds a threshold, with
16//!    activation noise s·χ making the boundary stochastic — matching
17//!    the probabilistic retrieval observed in biological memory.
18//!
19//! 3. **Dopaminergic forgetting** (Davis & Zhong 2017): dopamine
20//!    "forgetting cells" accelerate trace decay while a memory is not
21//!    being retrieved. High dopamine lowers the effective retrieval
22//!    threshold τ_eff, raising apparent accessibility for recently
23//!    encoded items and deepening the forgetting of untouched ones —
24//!    modeled as a transient shift τ_eff = τ₀ − g·(DA − DA_baseline).
25//!
26//! 4. **Nothing is deleted while it is still reachable.** The prune
27//!    predicate `should_forget` fires only when a memory's activation
28//!    is below the point where even a maximal-similarity cue could
29//!    reach it — B_i < τ₀ − w_cue — i.e. the engram is functionally
30//!    dismantled and only its storage remains. Rehearsal (every call
31//!    to `retrieve`) appends a fresh trace, so a memory that is still
32//!    being used cannot drift to the forget boundary.
33//!
34//! Rehearsal is the declared opposite of forgetting: `touch()` is the
35//! `+` to decay's `−`.
36
37/// Power-law decay exponent. The classic value (Anderson & Schooler
38/// 1991; Wixted 2004) is d ≈ 0.5.
39pub const TRACE_DECAY: f64 = 0.5;
40
41/// Milliseconds per day — decay is measured in days so TRACE_DECAY's
42/// empirical value stays meaningful.
43pub const DAY_MS: f64 = 86_400_000.0;
44
45/// Weight of episode salience in base-level activation. A
46/// maximally-salient first-hearing memory gets +1.0 on top of its
47/// trace term, shifting its forget horizon outward by e^2 ≈ 7.4×.
48pub const SALIENCE_GAIN: f64 = 1.0;
49
50/// How strongly retrieval-cue similarity contributes to accessibility.
51/// A maximal cue (identical hash) contributes +w_cue.
52pub const CUE_WEIGHT: f64 = 2.0;
53
54/// Base retrieval threshold — accessibility ≥ τ is retrievable.
55pub const RETRIEVAL_TAU: f64 = 0.0;
56
57/// Activation-noise scale — the logistic s parameter. The ±1σ band
58/// around the threshold is ≈ 1.54·s ≈ ±0.39 logits.
59pub const NOISE_SCALE: f64 = 0.25;
60
61/// Dopamine baseline (resting). The forgetting gate uses deviation
62/// from this, not the absolute level.
63pub const DA_BASELINE: f64 = 0.5;
64
65/// How strongly above-baseline dopamine eases retrieval (and, below
66/// baseline, deepens it). Kept small so chemistry modulates the
67/// boundary rather than dominating it.
68pub const DA_FORGETTING_GAIN: f64 = 0.4;
69
70/// An episode must be at least this old to be a prune candidate —
71/// protection against clock anomalies and freshly encoded items.
72pub const PRUNE_AGE_FLOOR_DAYS: f64 = 30.0;
73
74/// The activation below which a memory is unreachable even by a
75/// perfect retrieval cue: τ₀ − w_cue. Below this the engram is
76/// functionally gone and pruning only reclaims its residue.
77pub const FORGET_BOUND: f64 = RETRIEVAL_TAU - CUE_WEIGHT;
78
79/// One retrieval/rehearsal event for a single episode: how many times
80/// it has been activated and when it was last touched.
81#[derive(Debug, Clone, Copy, PartialEq, Eq)]
82pub struct Trace {
83    /// Number of encodings + retrievals (≥ 1 while the episode exists).
84    pub access_count: u32,
85    /// Wall-clock millisecond timestamp of the most recent access.
86    pub last_access_ms: u64,
87}

89impl Trace {
90    /// A fresh trace for an episode encoded at `timestamp_ms`.
91    pub fn encoded(timestamp_ms: u64) -> Self {
92        Self {
93            access_count: 1,
94            last_access_ms: timestamp_ms,
95        }
96    }
97}

99/// Total surviving trace mass for an episode — the sum of each
100/// presentation's power-law-decayed strength.
101///
102/// For n presentations over the episode's lifetime, exact mass is
103/// Σ_k a_k^(−d), but only the count and the two endpoint ages are
104/// persisted. Two estimators bound the unknown interior:
105///
106/// - **Blend** (Anderson & Schooler 1991's merged-trace
107///   approximation): presentations treated as uniformly spaced, so
108///   M ≈ 2n / (√a₀ + √a_L). Best when reactivations are spread
109///   evenly through the episode's life.
110/// - **Endpoint bound**: the newest trace contributes its exact
111///   a_L^(−d); every earlier trace is at least as old as a₀ and
112///   therefore contributes at least a₀^(−d), giving
113///   M ≥ a_L^(−d) + (n−1)·a₀^(−d). Exact for n ≤ 2, and the honest
114///   floor for a memory rehearsed once after a long silence — where
115///   the blend would spread the fresh trace's strength back across
116///   the whole span and under-count it.
117///
118/// The mass is the larger of the two — the estimator that least
119/// underestimates for the history the episode actually lived.
120///
121/// `age_first_days`/`age_last_days` are ages of the oldest and newest
122/// trace in days; ages are floored so a just-encoded trace never
123/// produces an infinite mass.
124pub fn trace_mass(access_count: u32, age_first_days: f64, age_last_days: f64) -> f64 {
125    if access_count == 0 {
126        return 0.0;
127    }
128    let a_first = age_first_days.max(MIN_AGE_DAYS);
129    let a_last = age_last_days.max(MIN_AGE_DAYS);
130    if access_count == 1 {
131        return a_last.powf(-TRACE_DECAY);
132    }
133    let n = f64::from(access_count);
134    let blend = 2.0 * n / (a_first.sqrt() + a_last.sqrt());
135    let bound = a_last.powf(-TRACE_DECAY) + (n - 1.0) * a_first.powf(-TRACE_DECAY);
136    blend.max(bound)
137}
138
139/// Minimum age — the decay integral diverges at age 0, and real
140/// presentations are separated by at least the encoding quantum.
141const MIN_AGE_DAYS: f64 = 1.0 / 1440.0; // one minute
142
143/// Base-level activation: ln M_i + w·salience.
144///
145/// `encoded_ms` is the episode's original encoding timestamp (the
146/// oldest trace's age comes from it); `trace` carries access count
147/// and recency. `salience` is the 0..1 salience stored at encoding.
148pub fn base_level(trace: Trace, encoded_ms: u64, salience: f32, now_ms: u64) -> f64 {
149    let age_first = (now_ms.saturating_sub(encoded_ms) as f64) / DAY_MS;
150    let age_last = (now_ms.saturating_sub(trace.last_access_ms) as f64) / DAY_MS;
151    trace_mass(trace.access_count, age_first, age_last).ln()
152        + SALIENCE_GAIN * f64::from(salience.max(0.0))
153}

155/// Deterministic activation noise χ ∈ (−1, 1), hashed from the
156/// episode and the query so the retrieval boundary is stochastic but
157/// reproducible for the same probe — matching how a given cue at a
158/// given moment either surfaces a memory or doesn't.
159pub fn activation_noise(episode_id: u64, query_hash: u64) -> f64 {
160    let mut z = episode_id
161        .wrapping_mul(0x9E37_79B9_7F4A_7C15)
162        .wrapping_add(query_hash);
163    z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
164    z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
165    z ^= z >> 31;
166    // Map the 53-bit mantissa into [0,1), then to (−1,1).
167    let u = ((z >> 11) as f64) / 9007199254740992.0; // 2^53
168    2.0 * u - 1.0
169}

171/// The retrieval threshold after the dopaminergic forgetting-cell
172/// modulation. Above-baseline dopamine lowers τ (easier retrieval of
173/// anything still reachable); below-baseline raises it.
174pub fn effective_threshold(dopamine: f64) -> f64 {
175    RETRIEVAL_TAU - DA_FORGETTING_GAIN * (dopamine - DA_BASELINE)
176}
177
178/// Full accessibility score for a candidate: B_i + w·sim + s·χ − τ_eff.
179/// Positive means the cue reaches the memory.
180pub fn accessibility_score(
181    base: f64,
182    cue_similarity: f64,
183    episode_id: u64,
184    query_hash: u64,
185    dopamine: f64,
186) -> f64 {
187    base + CUE_WEIGHT * cue_similarity.clamp(0.0, 1.0)
188        + NOISE_SCALE * activation_noise(episode_id, query_hash)
189        - effective_threshold(dopamine)
190}
191
192/// Whether a candidate survives the forgetting gate for this probe.
193pub fn is_accessible(
194    base: f64,
195    cue_similarity: f64,
196    episode_id: u64,
197    query_hash: u64,
198    dopamine: f64,
199) -> bool {
200    accessibility_score(base, cue_similarity, episode_id, query_hash, dopamine) >= 0.0
201}

203/// The deep-forget predicate: has this memory decayed past the point
204/// where any cue could reach it?
205///
206/// Fires only when B_i < τ₀ − w_cue, so even a maximal-similarity
207/// probe with no noise assistance cannot retrieve it, AND the episode
208/// is older than the age floor. Deliberately conservative — the bar
209/// sits a full CUE_WEIGHT below the retrieval threshold, so memories
210/// that are merely hard to reach stay stored; only the functionally
211/// dismantled are released.
212pub fn should_forget(base: f64, age_first_days: f64) -> bool {
213    base < FORGET_BOUND && age_first_days >= PRUNE_AGE_FLOOR_DAYS
214}
215
216/// Predicted days until a memory with this activation reaches the
217/// forget boundary — positive while it is still retrievable, and the
218/// number a mind would use to decide "rehearse it now or let it go".
219///
220/// B_i = ln(2n/(√a₀ + √a_L)) + w·s → solves for the a₀ at which
221/// B_i hits FORGET_BOUND, holding n, a_L ≈ a₀ (the uniform blend):
222///   a_forget ≈ (2n / e^{FORGET_BOUND − w·s})² / 4
223/// Returns days remaining until that age; ∞ if it can never fade
224/// below the bound with the current trace count.
225pub fn days_until_forgotten(access_count: u32, salience: f32, current_age_days: f64) -> f64 {
226    // With a₀ ≈ a_L = a, trace_mass = 2n/(2√a) = n·a^(−1/2); B =
227    // ln n − (d)·ln a + w·s  →  a_forget = (n / e^{B̄})^(1/d) where
228    // B̄ = FORGET_BOUND − w·s.
229    let barrier = FORGET_BOUND - SALIENCE_GAIN * f64::from(salience.max(0.0));
230    let n = f64::from(access_count.max(1));
231    let a_forget = (n / barrier.exp()).powf(1.0 / TRACE_DECAY);
232    a_forget - current_age_days
233}

235// ─────────────────────────────────────────────────────────────────
236//  Trace persistence — the .traces sidecar
237// ─────────────────────────────────────────────────────────────────
238//
239// The `.meta` IndexEntry format is frozen at 64 bytes — the Python
240// index scanner unpacks it directly — so rehearsal state lives in a
241// fourth file beside the store: `{base}.traces`.
242//
243// ```text
244// ┌─────────────────────────────────────────────────────┐
245// │  Traces file (.traces) — random-access              │
246// │  16-byte header: magic "LTMT" + version + count     │
247// │  24-byte records: id u64 | n u32 | pad u32 | t u64  │
248// └─────────────────────────────────────────────────────┘
249// ```
250//
251// Writes are write-through but not fsync'd per touch — a lost
252// rehearsal record only weakens a trace slightly (biologically the
253// same thing happens: not every reactivation consolidates). The
254// store's `sync()` fsyncs the file along with everything else.
255// A missing or corrupt traces file opens empty and heals: every
256// episode without a record reads as exactly one presentation at its
257// encoding timestamp, which is its true floor anyway.
258
259use std::collections::HashMap;
260use std::fs::OpenOptions;
261use std::io::{Read, Seek, SeekFrom, Write};
262use std::path::{Path, PathBuf};
263
264/// Magic bytes for the traces file: ASCII `"LTMT"`.
265pub const TRACES_MAGIC: [u8; 4] = *b"LTMT";
266/// Traces file schema version.
267pub const TRACES_VERSION: u32 = 1;
268/// Traces header size in bytes.
269pub const TRACES_HEADER_SIZE: u64 = 16;
270/// One trace record: episode_id u64 | access_count u32 | _pad u32 |
271/// last_access_ms u64.
272pub const TRACE_RECORD_SIZE: u64 = 24;
273
274/// Rehearsal/access state for every episode — persisted beside the
275/// LTM store so retrieval strengthens memory without touching the
276/// frozen `.meta` row format.
277pub struct TraceStore {
278    file: std::fs::File,
279    /// episode_id → record slot (byte offset = HEADER + slot·RECORD)
280    slots: HashMap<u64, u32>,
281    /// episode_id → trace. Entries not present read as the default
282    /// encoded-once trace — see [`TraceStore::get`].
283    traces: HashMap<u64, Trace>,
284    next_slot: u32,
285    path: PathBuf,
286}
287
288impl TraceStore {
289    /// Open the traces file for `base`, creating it if absent. A
290    /// corrupt or partial tail record is ignored rather than fatal —
291    /// the worst case is a trace that looks one rehearsal weaker.
292    pub fn open_or_create(base_path: &Path) -> std::io::Result<Self> {
293        let path = base_path.with_extension("traces");
294        let mut file = OpenOptions::new()
295            .read(true)
296            .write(true)
297            .create(true)
298            .truncate(false)
299            .open(&path)?;
300
301        let len = file.metadata()?.len();
302        let mut slots = HashMap::new();
303        let mut traces = HashMap::new();
304        let mut next_slot = 0u32;
305
306        if len >= TRACES_HEADER_SIZE {
307            let mut header = [0u8; TRACES_HEADER_SIZE as usize];
308            file.seek(SeekFrom::Start(0))?;
309            file.read_exact(&mut header)?;
310            if header[0..4] == TRACES_MAGIC
311                && u32::from_le_bytes(header[4..8].try_into().unwrap_or_default()) == TRACES_VERSION
312            {
313                let declared = u32::from_le_bytes(header[8..12].try_into().unwrap_or_default());
314                let present = ((len - TRACES_HEADER_SIZE) / TRACE_RECORD_SIZE) as u32;
315                let readable = declared.min(present);
316                let mut buf = vec![0u8; (readable * TRACE_RECORD_SIZE as u32) as usize];
317                file.read_exact(&mut buf)?;
318                for slot in 0..readable {
319                    let off = (slot * TRACE_RECORD_SIZE as u32) as usize;
320                    let episode_id =
321                        u64::from_le_bytes(buf[off..off + 8].try_into().unwrap_or_default());
322                    let access_count =
323                        u32::from_le_bytes(buf[off + 8..off + 12].try_into().unwrap_or_default());
324                    let last_access_ms =
325                        u64::from_le_bytes(buf[off + 16..off + 24].try_into().unwrap_or_default());
326                    if episode_id == 0 || access_count == 0 {
327                        continue;
328                    }
329                    slots.insert(episode_id, slot);
330                    traces.insert(
331                        episode_id,
332                        Trace {
333                            access_count,
334                            last_access_ms,
335                        },
336                    );
337                }
338                next_slot = present;
339            }
340        }
341
342        if len < TRACES_HEADER_SIZE {
343            // Fresh file — write the header now so the file is never
344            // zero-length garbage on disk.
345            Self::write_header(&mut file, 0)?;
346        }
347
348        Ok(Self {
349            file,
350            slots,
351            traces,
352            next_slot,
353            path,
354        })
355    }
356
357    fn write_header(file: &mut std::fs::File, count: u32) -> std::io::Result<()> {
358        let mut header = [0u8; TRACES_HEADER_SIZE as usize];
359        header[0..4].copy_from_slice(&TRACES_MAGIC);
360        header[4..8].copy_from_slice(&TRACES_VERSION.to_le_bytes());
361        header[8..12].copy_from_slice(&count.to_le_bytes());
362        file.seek(SeekFrom::Start(0))?;
363        file.write_all(&header)?;
364        Ok(())
365    }
366
367    fn write_record(
368        file: &mut std::fs::File,
369        slot: u32,
370        episode_id: u64,
371        trace: Trace,
372    ) -> std::io::Result<()> {
373        let mut rec = [0u8; TRACE_RECORD_SIZE as usize];
374        rec[0..8].copy_from_slice(&episode_id.to_le_bytes());
375        rec[8..12].copy_from_slice(&trace.access_count.to_le_bytes());
376        rec[16..24].copy_from_slice(&trace.last_access_ms.to_le_bytes());
377        file.seek(SeekFrom::Start(
378            TRACES_HEADER_SIZE + u64::from(slot) * TRACE_RECORD_SIZE,
379        ))?;
380        file.write_all(&rec)?;
381        Ok(())
382    }

384    /// The trace for an episode; absent records default to a single
385    /// presentation at `encoded_ms` — every stored episode has at
386    /// least its encoding trace.
387    pub fn get(&self, episode_id: u64, encoded_ms: u64) -> Trace {
388        self.traces
389            .get(&episode_id)
390            .copied()
391            .unwrap_or_else(|| Trace::encoded(encoded_ms))
392    }
393
394    /// Record an encoding: the first trace, stamped at the episode's
395    /// own timestamp.
396    pub fn record_encoding(&mut self, episode_id: u64, timestamp_ms: u64) -> std::io::Result<()> {
397        self.put(episode_id, Trace::encoded(timestamp_ms))
398    }
399
400    /// Rehearsal — the opposite of forgetting. Each retrieval adds a
401    /// presentation to the trace: n += 1, t_last = now. The added
402    /// mass is strongest exactly when it lands, so a just-rehearsed
403    /// memory jumps well clear of the retrieval boundary.
404    pub fn touch(&mut self, episode_id: u64, now_ms: u64) -> std::io::Result<()> {
405        let current = self
406            .traces
407            .get(&episode_id)
408            .copied()
409            .unwrap_or_else(|| Trace::encoded(now_ms));
410        self.put(
411            episode_id,
412            Trace {
413                access_count: current.access_count.saturating_add(1),
414                last_access_ms: now_ms,
415            },
416        )
417    }
418
419    fn put(&mut self, episode_id: u64, trace: Trace) -> std::io::Result<()> {
420        let slot = match self.slots.get(&episode_id) {
421            Some(&slot) => slot,
422            None => {
423                let slot = self.next_slot;
424                self.next_slot = self.next_slot.saturating_add(1);
425                self.slots.insert(episode_id, slot);
426                Self::write_header(&mut self.file, self.next_slot)?;
427                slot
428            }
429        };
430        Self::write_record(&mut self.file, slot, episode_id, trace)?;
431        self.traces.insert(episode_id, trace);
432        Ok(())
433    }
434
435    /// Forget an episode's trace — the record on disk is left as dead
436    /// space until `retain` rewrites the file, mirroring how a deleted
437    /// episode's `.meta` row lingers until compaction.
438    pub fn remove(&mut self, episode_id: u64) {
439        self.traces.remove(&episode_id);
440        self.slots.remove(&episode_id);
441    }
442
443    /// Drop every trace whose episode no longer exists — called
444    /// during compaction so the sidecar shrinks with the store it
445    /// shadows.
446    pub fn retain(&mut self, live: &std::collections::HashSet<u64>) -> std::io::Result<()> {
447        self.traces.retain(|id, _| live.contains(id));
448        // Rewrite the file compactly: the slot space is dense again.
449        let mut entries: Vec<(u64, Trace)> = self.traces.iter().map(|(&id, &t)| (id, t)).collect();
450        entries.sort_unstable_by_key(|(id, _)| *id);
451        self.file.set_len(0)?;
452        Self::write_header(&mut self.file, entries.len() as u32)?;
453        self.slots.clear();
454        for (slot, (id, trace)) in entries.into_iter().enumerate() {
455            let slot = slot as u32;
456            Self::write_record(&mut self.file, slot, id, trace)?;
457            self.slots.insert(id, slot);
458        }
459        self.next_slot = self.slots.len() as u32;
460        self.file.sync_all()
461    }
462
463    /// Flush the traces file. Called from `LtmStore::sync()`.
464    pub fn sync(&mut self) -> std::io::Result<()> {
465        self.file.sync_all()
466    }
467
468    /// Filesystem path of the backing file (for compaction staging).
469    pub fn path(&self) -> &Path {
470        &self.path
471    }
