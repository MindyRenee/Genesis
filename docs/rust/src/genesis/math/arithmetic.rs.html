1//! Arithmetic — exact expression evaluation and spoken-number primitives.
2//!
3//! Machine-side counterpart to `python/parietal_lobe/math/arithmetic.py`.
4//! Pure deterministic evaluation over binary operations, integer powers,
5//! Euclidean division/modulo, and real square roots without host `eval`.
6
7/// Binary operations supported by the exact arithmetic engine.
8#[derive(Clone, Copy, Debug, PartialEq, Eq)]
9pub enum ArithOp {
10    Add,
11    Sub,
12    Mul,
13    Div,
14    FloorDiv,
15    Mod,
16    Pow,
17}
18
19const MAX_POW_ABS: f64 = 10_000.0;
20
21/// Evaluate one binary arithmetic step; returns `None` on division by zero,
22/// exponent overflow, or non-real/non-finite results.
23pub fn apply_op(op: ArithOp, lhs: f64, rhs: f64) -> Option<f64> {
24    if !lhs.is_finite() || !rhs.is_finite() {
25        return None;
26    }
27    let out = match op {
28        ArithOp::Add => lhs + rhs,
29        ArithOp::Sub => lhs - rhs,
30        ArithOp::Mul => lhs * rhs,
31        ArithOp::Div => {
32            if rhs == 0.0 {
33                return None;
34            }
35            lhs / rhs
36        }
37        ArithOp::FloorDiv => {
38            if rhs == 0.0 {
39                return None;
40            }
41            (lhs / rhs).floor()
42        }
43        ArithOp::Mod => {
44            if rhs == 0.0 {
45                return None;
46            }
47            lhs.rem_euclid(rhs)
48        }
49        ArithOp::Pow => {
50            if rhs.abs() > MAX_POW_ABS {
51                return None;
52            }
53            if lhs < 0.0 && rhs.fract() != 0.0 {
54                return None;
55            }
56            lhs.powf(rhs)
57        }
58    };
59    out.is_finite().then_some(out)
60}
61
62/// Principal real square root; `None` for negative or non-finite inputs.
63pub fn real_sqrt(x: f64) -> Option<f64> {
64    if !x.is_finite() || x < 0.0 {
65        return None;
66    }
67    Some(x.sqrt())
68}

70#[cfg(test)]
71mod tests {
72    use super::*;
73
74    #[test]
75    fn basic_ops_and_guards() {
76        assert_eq!(apply_op(ArithOp::Add, 7.0, 5.0), Some(12.0));
77        assert_eq!(apply_op(ArithOp::Div, 1.0, 0.0), None);
78        assert_eq!(apply_op(ArithOp::Pow, 2.0, 10.0), Some(1024.0));
79        assert_eq!(apply_op(ArithOp::Pow, -1.0, 0.5), None);
80        assert_eq!(real_sqrt(144.0), Some(12.0));
81        assert_eq!(real_sqrt(-1.0), None);
82    }
