1//! Trigonometry — the circle's arithmetic.
2//!
3//! Angles, the unit circle, and the triangle laws. Everything here
4//! is a function over `f64`: angles in radians unless a name says
5//! degrees. No parsing, no vocabulary — the machine side only
6//! measures.
7
8use std::f64::consts::{PI, TAU};
9
10/// Two full turns in radians.
11pub const FULL_TURN: f64 = TAU;
12
13/// Degrees to radians.
14pub fn deg(degrees: f64) -> f64 {
15    degrees * PI / 180.0
16}
17
18/// Radians to degrees.
19pub fn rad(radians: f64) -> f64 {
20    radians * 180.0 / PI
21}
22
23/// Wrap an angle into (-pi, pi] — the canonical turn.
24pub fn normalize(angle: f64) -> f64 {
25    let mut a = angle % TAU;
26    if a <= -PI {
27        a += TAU;
28    } else if a > PI {
29        a -= TAU;
30    }
31    a
32}
33
34/// Point on the unit circle at `angle`.
35pub fn unit_point(angle: f64) -> (f64, f64) {
36    (angle.cos(), angle.sin())
37}
38
39/// Signed angle between two unit-circle bearings, in (-pi, pi].
40pub fn angle_between(from: f64, to: f64) -> f64 {
41    normalize(to - from)
42}
43
44/// The internal angle at vertex B for triangle side lengths
45/// (a = |BC|, b = |CA|, c = |AB|). Law of cosines, inverted.
46pub fn angle_from_sides(adjacent_a: f64, opposite_b: f64, adjacent_c: f64) -> Option<f64> {
47    if adjacent_a <= 0.0 || adjacent_c <= 0.0 {
48        return None;
49    }
50    let cos_b = (adjacent_a * adjacent_a + adjacent_c * adjacent_c - opposite_b * opposite_b)
51        / (2.0 * adjacent_a * adjacent_c);
52    if (-1.0..=1.0).contains(&cos_b) {
53        Some(cos_b.clamp(-1.0, 1.0).acos())
54    } else {
55        None // no triangle has these sides
56    }
57}
58
59/// Side opposite the given angle from two adjacent sides.
60/// Law of cosines: c² = a² + b² - 2ab·cos(C).
61pub fn side_from_angle(a: f64, b: f64, angle_c: f64) -> Option<f64> {
62    if a <= 0.0 || b <= 0.0 {
63        return None;
64    }
65    Some((a * a + b * b - 2.0 * a * b * angle_c.cos()).sqrt())
66}
67
68/// All three angles of a triangle from its sides (a, b, c).
69/// Returns (A, B, C) opposite each side, or None when the
70/// sides cannot form a triangle.
71pub fn triangle_angles(a: f64, b: f64, c: f64) -> Option<(f64, f64, f64)> {
72    if a <= 0.0 || b <= 0.0 || c <= 0.0 || a + b <= c || b + c <= a || c + a <= b {
73        return None;
74    }
75    let big_a = angle_from_sides(b, a, c)?;
76    let big_b = angle_from_sides(a, b, c)?;
77    Some((big_a, big_b, PI - big_a - big_b))
78}
79
80/// Heron's area from three side lengths.
81pub fn area_heron(a: f64, b: f64, c: f64) -> Option<f64> {
82    triangle_angles(a, b, c)?;
83    let s = (a + b + c) / 2.0;
84    Some((s * (s - a) * (s - b) * (s - c)).sqrt())
85}

87/// Area from two sides and the included angle.
88pub fn area_sas(a: f64, b: f64, angle_c: f64) -> f64 {
89    0.5 * a * b * angle_c.sin()
90}

92/// Law of sines: given side `a` opposite angle `a_ang`, the side
93/// opposite `b_ang`.
94pub fn side_law_of_sines(a: f64, a_ang: f64, b_ang: f64) -> Option<f64> {
95    let sa = a_ang.sin();
96    if sa.abs() < f64::EPSILON {
97        return None;
98    }
99    Some(a * b_ang.sin() / sa)
100}

102/// The six trigonometric ratios at an angle, gathered.
103/// Returns (sin, cos, tan, csc, sec, cot) with None where the
104/// ratio diverges.
105pub fn ratios(angle: f64) -> (f64, f64, Option<f64>, Option<f64>, Option<f64>, Option<f64>) {
106    let (s, c) = (angle.sin(), angle.cos());
107    let tan = if c.abs() > 1e-15 { Some(s / c) } else { None };
108    let csc = if s.abs() > 1e-15 { Some(1.0 / s) } else { None };
109    let sec = if c.abs() > 1e-15 { Some(1.0 / c) } else { None };
110    let cot = if s.abs() > 1e-15 { Some(c / s) } else { None };
111    (s, c, tan, csc, sec, cot)
112}

114/// The identity sin²θ + cos²θ = 1, checked numerically.
115/// The machine verifies what it cannot prove.
116pub fn pythagorean_holds(angle: f64, tolerance: f64) -> bool {
117    let (s, c) = (angle.sin(), angle.cos());
118    (s * s + c * c - 1.0).abs() <= tolerance
119}
120
121/// Sum-to-product check: sin(a+b) ≈ sin a cos b + cos a sin b.
122pub fn angle_sum_holds(a: f64, b: f64, tolerance: f64) -> bool {
123    ((a + b).sin() - (a.sin() * b.cos() + a.cos() * b.sin())).abs() <= tolerance
124}

131    fn normalize_wraps() {
132        assert!((normalize(3.0 * PI) - PI).abs() < 1e-12);

134    }

138        let (x, y) = unit_point(0.0);

140    }
141
142    #[test]
143    fn triangle_345() {
144        let (a, b, c) = triangle_angles(3.0, 4.0, 5.0).unwrap();
145        assert!((a + b + c - PI).abs() < 1e-12);
146        assert!((c - PI / 2.0).abs() < 1e-12);
147        assert!((area_heron(3.0, 4.0, 5.0).unwrap() - 6.0).abs() < 1e-12);
148    }
149
150    #[test]
151    fn identities_hold() {
152        for k in 0..8 {
153            let t = k as f64 * 0.7 - 2.1;
