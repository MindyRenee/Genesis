1//! Minimal RealtimeKit client over the system bus — just enough
2//! D-Bus to request and *hold* a scheduling grant.
3//!
4//! rtkit revokes an elevation when the bus client that requested it
5//! disappears: a one-shot `dbus-send` is rolled back seconds after
6//! it exits (measured on this host: the grant to a probe thread was
7//! gone within ~45 s while the identical request from a client that
8//! stayed connected held indefinitely). PulseAudio's grants survive
9//! because it keeps a persistent connection — this module does the
10//! same, speaking the wire protocol directly so the daemon grows no
11//! dependencies: SASL EXTERNAL auth, the required Hello, one
12//! METHOD_CALL, then the socket stays open for as long as the
13//! caller keeps the returned stream.
14//!
15//! Everything here is best-effort: an absent bus, a refused auth,
16//! or a denied grant all surface as `Err` and the caller continues
17//! at default scheduling — degraded jitter, never a failure.

25/// Levels tried strictest-first — rtkit clamps or denies a request
26/// below its advertised MinNiceLevel, so walking down covers daemons
27/// configured tighter than -15.
28const NICE_LEVELS: [i32; 3] = [-15, -10, -5];

30fn align_up(n: usize, a: usize) -> usize {
31    (n + a - 1) & !(a - 1)
32}
33
34fn pad(buf: &mut Vec<u8>, align: usize) {
35    while buf.len() % align != 0 {
36        buf.push(0);
37    }
38}

40/// `"s"`/`"o"` wire form: u32 length + bytes + NUL.
41fn enc_str(s: &str) -> Vec<u8> {
42    let mut v = Vec::with_capacity(s.len() + 5);
43    v.extend_from_slice(&(s.len() as u32).to_le_bytes());
44    v.extend_from_slice(s.as_bytes());
45    v.push(0);
46    v
47}
48
49/// `"g"` wire form: u8 length + bytes + NUL.
50fn enc_sig(s: &str) -> Vec<u8> {
51    let mut v = Vec::with_capacity(s.len() + 2);
52    v.push(s.len() as u8);
53    v.extend_from_slice(s.as_bytes());
54    v.push(0);
55    v
56}

58/// One header field — an 8-aligned struct of a byte code and a
59/// variant. The `fields` buffer begins at absolute offset 16 (itself
60/// 8-aligned), so buffer-relative padding is wire-correct.
61fn marshal_field(buf: &mut Vec<u8>, code: u8, sig: &str, value: &[u8], val_align: usize) {
62    pad(buf, 8);
63    buf.push(code);
64    buf.push(sig.len() as u8);
65    buf.extend_from_slice(sig.as_bytes());
66    buf.push(0);
67    pad(buf, val_align);
68    buf.extend_from_slice(value);
69}

71/// METHOD_CALL in little-endian framing. Field codes: 1 PATH,
72/// 2 INTERFACE, 3 MEMBER, 6 DESTINATION, 8 SIGNATURE. The bus
73/// fills in SENDER; we never set it.
74fn marshal_call(
75    serial: u32,
76    path: &str,
77    iface: &str,
78    member: &str,
79    dest: &str,
80    sig: &str,
81    body: &[u8],
82) -> Vec<u8> {
83    let mut fields = Vec::new();
84    marshal_field(&mut fields, 1, "o", &enc_str(path), 4);
85    marshal_field(&mut fields, 2, "s", &enc_str(iface), 4);
86    marshal_field(&mut fields, 3, "s", &enc_str(member), 4);
87    marshal_field(&mut fields, 6, "s", &enc_str(dest), 4);
88    if !sig.is_empty() {
89        marshal_field(&mut fields, 8, "g", &enc_sig(sig), 1);
90    }
91
92    let mut m = Vec::with_capacity(24 + fields.len() + body.len());
93    m.extend_from_slice(b"l\x01\x00\x01");
94    m.extend_from_slice(&(body.len() as u32).to_le_bytes());
95    m.extend_from_slice(&serial.to_le_bytes());
96    m.extend_from_slice(&(fields.len() as u32).to_le_bytes());
97    pad(&mut m, 8);
98    m.extend_from_slice(&fields);
99    pad(&mut m, 8);
100    m.extend_from_slice(body);
101    m
102}

104struct Msg {
105    mtype: u8,
106    reply_serial: Option<u32>,
107    error_name: Option<String>,
108}
109
110/// Read one full message: 16-byte fixed header, the header-fields
111/// array, then the body. Only the pieces we need are decoded —
112/// type, REPLY_SERIAL (code 5, "u"), and ERROR_NAME (code 4, "s").
113fn read_message(s: &mut UnixStream) -> io::Result<Msg> {
114    let debug = std::env::var_os("GENESIS_DBUS_DEBUG").is_some();
115    let mut fixed = [0u8; 16];
116    s.read_exact(&mut fixed)?;
117    // Fixed header: 0 endianness, 1 type, 2 flags, 3 version,
118    // 4-7 body length, 8-11 serial, 12-15 header-fields array len.
119    let body_len = u32::from_le_bytes([fixed[4], fixed[5], fixed[6], fixed[7]]) as usize;
120    let fields_len = u32::from_le_bytes([fixed[12], fixed[13], fixed[14], fixed[15]]) as usize;
121    if debug {
122        eprintln!(
123            "[dbus] msg type={} blen={} flen={} hdr={:02x?}",
124            fixed[1], body_len, fields_len, fixed
125        );
126    }
127    if fields_len > (1 << 20) || body_len > (1 << 20) {
128        return Err(io::Error::new(
129            io::ErrorKind::InvalidData,
130            format!("oversized dbus frame: hdr={:02x?}", fixed),
131        ));
132    }
133    let mut rest = vec![0u8; align_up(fields_len, 8) + body_len];
134    s.read_exact(&mut rest)?;
135    let fields = &rest[..fields_len];
136
137    let mut reply_serial = None;
138    let mut error_name = None;
139    let mut i = 0usize;
140    while i + 2 <= fields.len() {
141        let code = fields[i];
142        i += 1;
143        let sl = fields[i] as usize;
144        i += 1;
145        if i + sl + 1 > fields.len() {
146            break;
147        }
148        let vsig = &fields[i..i + sl];
149        i += sl + 1;
150        match vsig {
151            b"o" | b"s" => {
152                i = align_up(i, 4);
153                if i + 4 > fields.len() {
154                    break;
155                }
156                let l = u32::from_le_bytes([fields[i], fields[i + 1], fields[i + 2], fields[i + 3]])
157                    as usize;
158                let start = i + 4;
159                if start + l + 1 > fields.len() {
160                    break;
161                }
162                if code == 4 {
163                    error_name =
164                        Some(String::from_utf8_lossy(&fields[start..start + l]).into_owned());
165                }
166                i = start + l + 1;
167            }
168            b"g" => {
169                if i >= fields.len() {
170                    break;
171                }
172                let l = fields[i] as usize;
173                i += 1 + l + 1;
174            }
175            b"u" => {
176                i = align_up(i, 4);
177                if i + 4 > fields.len() {
178                    break;
179                }
180                if code == 5 {
181                    reply_serial = Some(u32::from_le_bytes([
182                        fields[i],
183                        fields[i + 1],
184                        fields[i + 2],
185                        fields[i + 3],
186                    ]));
187                }
188                i += 4;
189            }
190            _ => break,
191        }
192        while i % 8 != 0 && i < fields.len() {
193            i += 1;
194        }
195    }
196    Ok(Msg {
197        mtype: fixed[1],
198        reply_serial,
199        error_name,
200    })
201}

203enum Reply {
204    Return,
205    Error(String),
206}

208/// A D-Bus method target — everything marshal_call needs that isn't
209/// the serial or the body.
210struct CallSpec<'a> {
211    path: &'a str,
212    iface: &'a str,
213    member: &'a str,
214    dest: &'a str,
215    sig: &'a str,
216    body: Vec<u8>,
217}
218
219/// Send one METHOD_CALL and read until its reply, skipping signals
220/// (e.g. the bus's own NameAcquired for our unique name).
221fn dbus_call(s: &mut UnixStream, serial: &mut u32, c: &CallSpec) -> io::Result<Reply> {
222    *serial += 1;
223    let mine = *serial;
224    s.write_all(&marshal_call(
225        mine, c.path, c.iface, c.member, c.dest, c.sig, &c.body,
226    ))?;
227    loop {
228        let m = read_message(s)?;
229        match (m.mtype, m.reply_serial) {
230            (2, Some(rs)) if rs == mine => return Ok(Reply::Return),
231            (3, Some(rs)) if rs == mine => {
232                return Ok(Reply::Error(
233                    m.error_name.unwrap_or_else(|| "dbus error".into()),
234                ));
235            }
236            _ => {}
237        }
238    }
239}

241/// SASL EXTERNAL over the unix socket: the one leading NUL byte,
242/// then the uid's decimal string hex-encoded.
243fn sasl_auth(s: &mut UnixStream) -> io::Result<()> {
244    // SAFETY: getuid takes no arguments, touches no memory, and
245    // cannot fail — it always returns the caller's real uid.
246    let uid = unsafe { libc::getuid() };
247    let hex: String = uid
248        .to_string()
249        .bytes()
250        .map(|b| format!("{b:02x}"))
251        .collect();
252    s.write_all(b"\0")?;
253    s.write_all(format!("AUTH EXTERNAL {hex}\r\n").as_bytes())?;
254    let mut line = Vec::with_capacity(64);
255    let mut byte = [0u8; 1];
256    while line.len() < 4096 {
257        s.read_exact(&mut byte)?;
258        if byte[0] == b'\n' {
259            break;
260        }
261        line.push(byte[0]);
262    }
263    let line = String::from_utf8_lossy(&line);
264    if !line.starts_with("OK") {
265        return Err(io::Error::new(
266            io::ErrorKind::PermissionDenied,
267            format!("system bus auth: {}", line.trim()),
268        ));
269    }
270    s.write_all(b"BEGIN\r\n")
271}

273fn connect() -> io::Result<UnixStream> {
274    let addr = std::env::var("DBUS_SYSTEM_BUS_ADDRESS").unwrap_or_default();
275    let path = addr.strip_prefix("unix:path=").unwrap_or(SYSTEM_BUS_PATH);
276    let s = UnixStream::connect(path)?;
277    s.set_read_timeout(Some(Duration::from_secs(10)))?;
278    Ok(s)
279}

281/// Ask rtkit for high priority (nice) on `tid` of `pid`. Returns
282/// the granted level *and the open socket* — the grant is owned by
283/// this bus client, so the caller must keep the stream alive for
284/// as long as the elevation should hold (rtkit revokes on client
285/// disconnect). `Err` covers every failure: no bus, no rtkit, or a
286/// denial at all levels — the caller continues unboosted.
287pub fn grant_high_priority(pid: u64, tid: u64) -> io::Result<(i32, UnixStream)> {
288    let mut s = connect()?;
289    sasl_auth(&mut s)?;
290    let mut serial = 0u32;
291    // Hello is mandatory before any real call — it registers our
292    // unique name, the identity rtkit watches.
293    match dbus_call(
294        &mut s,
295        &mut serial,
296        &CallSpec {
297            path: "/org/freedesktop/DBus",
298            iface: "org.freedesktop.DBus",
299            member: "Hello",
300            dest: "org.freedesktop.DBus",
301            sig: "",
302            body: Vec::new(),
303        },
304    )? {
305        Reply::Return => {}
306        Reply::Error(e) => {
307            return Err(io::Error::other(format!("Hello refused: {e}")));
308        }
309    }
310    let mut denied = String::new();
311    for nice in NICE_LEVELS {
312        // Signature "tti": u64 process, u64 thread, i32 level — each
313        // aligned to its own size within the body.
314        let mut body = Vec::with_capacity(24);
315        pad(&mut body, 8);
316        body.extend_from_slice(&pid.to_le_bytes());
317        pad(&mut body, 8);
318        body.extend_from_slice(&tid.to_le_bytes());
319        pad(&mut body, 4);
320        body.extend_from_slice(&nice.to_le_bytes());
321        match dbus_call(
322            &mut s,
323            &mut serial,
324            &CallSpec {
325                path: "/org/freedesktop/RealtimeKit1",
326                iface: "org.freedesktop.RealtimeKit1",
327                member: "MakeThreadHighPriorityWithPID",
328                dest: "org.freedesktop.RealtimeKit1",
329                sig: "tti",
330                body,
331            },
332        )? {
333            Reply::Return => return Ok((nice, s)),
334            Reply::Error(e) => denied = e,
335        }
336    }
337    Err(io::Error::new(io::ErrorKind::PermissionDenied, denied))
338}
