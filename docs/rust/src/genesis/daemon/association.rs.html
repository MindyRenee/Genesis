1//! Association engine — finds connections between memories and creates
2//! associative links.
3//!
4//! # How human association works
5//!
6//! In the human brain, association is the process of linking related
7//! memories. When you hear "salt lake," you think of Utah, water,
8//! drought, data centers — because those memories are associatively
9//! linked. This isn't similarity (Utah isn't "similar" to water) —
10//! it's **co-activation**: memories that fire together wire together.
11//!
12//! During sleep, the brain runs **free association**: it reactivates
13//! random memories and finds unexpected connections. This is the
14//! neurobiological basis of creativity — "sleeping on it" works
15//! because the brain builds new associative links offline.
16//!
17//! # How Genesis's association works
18//!
19//! Genesis uses SimHash fingerprints for associative matching. When a
20//! new memory is consolidated, the association engine:
21//!
22//! 1. **Scans** the LTM index for memories with similar SimHashes
23//! 2. **Scores** each match by Hamming distance (closer = stronger)
24//! 3. **Records** the strongest associations as "links" — stored as
25//!    episodes in LTM themselves (a meta-memory: "I noticed that
26//!    memory X is related to memory Y")
27//!
28//! During dreaming (sleep phase), the engine runs **free association**:
29//! 1. Picks a random memory from LTM
30//! 2. Finds its associations
31//! 3. Picks one of those associations at random
32//! 4. Finds THAT memory's associations
33//! 5. If it finds a connection between two memories that weren't
34//!    previously linked, records it as a "dream insight"
35//!
36//! This is genuinely novel — no AI system does offline associative
37//! discovery. It's the computational equivalent of dreaming.
38

41use crate::store::{LtmStore, hamming_distance, simhash};

43/// Maximum Hamming distance for two memories to be considered "associated."
44/// Lower = stricter matching. 20 out of 64 bits is ~31% dissimilarity.
45pub const ASSOCIATION_THRESHOLD: u32 = 20;
46
47/// Minimum Hamming distance for two memories to be worth bridging.
48///
49/// Below this, the SimHashes are near-identical — same text or
50/// trivially edited — so the "association" is a duplicate detection,
51/// not a connection between two memories. The floor also breaks the
52/// degenerate case where repeated or empty texts (SimHash 0) all sit
53/// at distance 0 from each other: without it, every identical episode
54/// bridges to up to `ASSOCIATION_CANDIDATE_LIMIT` others each pass,
55/// which is how the store accumulated a quarter-million junk bridges
56/// in the first place.
57pub const MIN_ASSOCIATION_DISTANCE: u32 = 6;
58
59/// How many association candidates a single source episode may
60/// produce per pass — caps the bridges one episode can spawn so a
61/// single hot memory can't dominate a pass's writes.
62const ASSOCIATION_CANDIDATE_LIMIT: usize = 20;
63
64/// Returns true if the episode text is a meta-memory (an association
65/// bridge or dream insight), not a real source episode.
66///
67/// The association engine must not try to build bridges *from*
68/// meta-memories — that causes tail-chasing where each pass associates
69/// the previous pass's bridges with each other instead of finding new
70/// connections between real experiences.
71fn is_meta_memory(text: &str) -> bool {
72    text.starts_with("[association]") || text.starts_with("[dream-insight]")
73}

75/// The result of an association pass.
76#[derive(Clone, Copy, Debug, Default)]
77pub struct AssociationResult {
78    /// Number of new associations discovered.
79    pub new_associations: u32,
80    /// Number of memories scanned.
81    pub scanned: u32,
82    /// Number of associations that already existed (duplicate).
83    pub duplicates: u32,
84}

86/// The result of a dreaming pass.
87#[derive(Clone, Debug, Default)]
88pub struct DreamResult {
89    /// Number of dream insights (novel connections discovered).
90    pub insights: u32,
91    /// The chain of memories traversed during dreaming.
92    pub chain: Vec<u64>,
93    /// Number of hops in the longest chain.
94    pub max_depth: u32,
95}

97/// The association engine.
98pub struct AssociationEngine;

100impl AssociationEngine {
101    /// Find associations for a specific episode.
102    ///
103    /// Returns a list of (episode_id, hamming_distance) pairs for
104    /// memories whose SimHash is within `ASSOCIATION_THRESHOLD` bits
105    /// of the query episode's hash.
106    ///
107    /// Two exclusions keep the result meaningful:
108    /// - meta-memories are bookkeeping, not memories — they can
109    ///   neither source nor anchor an association;
110    /// - distances at or under `MIN_ASSOCIATION_DISTANCE` are
111    ///   near-duplicates — a bridge between them is noise, not a
112    ///   connection (see the constant's doc).
113    pub fn find_associations(ltm: &LtmStore, episode_hash: u64, limit: usize) -> Vec<(u64, u32)> {
114        let mut results: Vec<(u64, u32)> = Vec::new();
115
116        // Iterate only active entries via in-memory index — O(active)
117        // instead of O(capacity) scanning all 65K slots.
118        for (_id, entry) in ltm.active_entries() {
119            if entry.is_meta_memory() {
120                continue;
121            }
122            let dist = hamming_distance(episode_hash, entry.association_hash);
123            if (MIN_ASSOCIATION_DISTANCE..=ASSOCIATION_THRESHOLD).contains(&dist) {
124                results.push((entry.episode_id, dist));
125            }
126        }
127
128        results.sort_by_key(|(_, dist)| *dist);
129        results.truncate(limit);
130        results
131    }

133    /// Run an association pass on recently-stored episodes.
134    ///
135    /// For each episode, finds its associations and records novel
136    /// connections as meta-memories in LTM.
137    ///
138    /// # Parameters
139    /// - `ltm`: the long-term memory store
140    /// - `recent_ids`: episode IDs to find associations for
141    /// - `now_ms`: current timestamp
142    pub fn associate(ltm: &mut LtmStore, recent_ids: &[u64], now_ms: u64) -> AssociationResult {
143        let mut result = AssociationResult::default();
144
145        for &episode_id in recent_ids {
146            result.scanned += 1;
147
148            // Cheap flag check first — a flagged meta-memory is
149            // bookkeeping, not a bridge source, so don't even
150            // decompress it.
151            if ltm.is_meta_episode(episode_id) {
152                continue;
153            }
154
155            // Retrieve the episode to get its hash and text
156            let episode = match ltm.retrieve(episode_id) {
157                Ok(ep) => ep,
158                Err(_) => continue,
159            };
160
161            // Don't build bridges from meta-memories — only from real
162            // source episodes. Without this, the engine chases its own
163            // tail: each pass associates the previous pass's
164            // "[association] ..." bridges with each other instead of
165            // finding new connections between real experiences.
166            if is_meta_memory(&episode.text) {
167                continue;
168            }
169
170            // Find associations. We over-fetch (ASSOCIATION_CANDIDATE_LIMIT)
171            // to compensate for meta-memories that will be filtered out
172            // below, so we still have enough real candidates to build
173            // bridges between.
174            let associations =
175                Self::find_associations(ltm, episode.association_hash, ASSOCIATION_CANDIDATE_LIMIT);
176
177            for (assoc_id, dist) in associations {
178                if assoc_id == episode_id {
179                    continue; // Don't associate with self
180                }
181
182                // Don't build bridges to meta-memories — only to real
183                // source episodes. This prevents meta-memory-to-meta-memory
184                // bridges from diluting the association graph.
185                let assoc_episode = match ltm.retrieve(assoc_id) {
186                    Ok(ep) => ep,
187                    Err(_) => continue,
188                };
189                if is_meta_memory(&assoc_episode.text) {
190                    continue;
191                }
192
193                // Check if this association already exists by searching
194                // for a meta-memory about this pair.
195                //
196                // Normalize the endpoint order so (A,B) and (B,A)
197                // produce the same bridge text — and therefore the
198                // same dedup fingerprint — no matter which episode the
199                // pass started from.
200                //
201                // We embed the text of both endpoint episodes directly
202                // in the association text (tab-delimited) so the
203                // insight is self-contained. Without this, the Python
204                // side would need to retrieve the episodes by ID later
205                // — but sleep compression may have deleted them by
206                // then, leaving stale references.
207                let (id_a, id_b, ep_a_text, ep_b_text) = if episode_id < assoc_id {
208                    (
209                        episode_id,
210                        assoc_id,
211                        truncate_text(&episode.text, 200),
212                        truncate_text(&assoc_episode.text, 200),
213                    )
214                } else {
215                    (
216                        assoc_id,
217                        episode_id,
218                        truncate_text(&assoc_episode.text, 200),
219                        truncate_text(&episode.text, 200),
220                    )
221                };
222                let meta_text = format!(
223                    "[association] episode {} ↔ episode {} (distance: {})\t{}\t{}",
224                    id_a, id_b, dist, ep_a_text, ep_b_text
225                );
226                let meta_hash = simhash(&meta_text);
227
228                // Global dedup: one active bridge per normalized pair,
229                // period. The old check scanned only the last ~200
230                // episodes, which the flood's own write rate outran
231                // (and each bridge's unique IDs changed its hash
232                // anyway); the meta-hash index covers every active
233                // bridge in O(1).
234                if ltm.meta_hash_known(meta_hash) {
235                    result.duplicates += 1;
236                    continue;
237                }
238
239                // Store the association as a meta-memory
240                let neutral_tag = [0.5f32; 12];
241                let compact_tag = [0.5, 0.0, 0.2, 0.5];
242                if ltm
243                    .store_meta(
244                        now_ms,
245                        0.3 + (1.0 - dist as f32 / ASSOCIATION_THRESHOLD as f32) * 0.4,
246                        neutral_tag,
247                        compact_tag,
248                        2, // Internal event
249                        0, // Subcognitive module
250                        &meta_text,
251                    )
252                    .is_ok()
253                {
254                    result.new_associations += 1;
255                }
256            }
257        }
258
259        result
260    }

262    /// Run a dreaming pass — free association through the memory graph.
263    ///
264    /// This is what happens when Genesis sleeps. The engine:
265    /// 1. Picks a random memory
266    /// 2. Finds its associations
267    /// 3. Hops to a random associated memory
268    /// 4. Repeats, building a chain
269    /// 5. If it discovers a connection between two memories that
270    ///    weren't directly linked before, records it as a "dream insight"
271    ///
272    /// # Parameters
273    /// - `ltm`: the long-term memory store
274    /// - `now_ms`: current timestamp
275    /// - `max_hops`: maximum chain length
276    pub fn dream(ltm: &mut LtmStore, now_ms: u64, max_hops: u32) -> DreamResult {
277        let mut result = DreamResult::default();
278        let mut chain: Vec<u64> = Vec::new();
279        let mut visited: HashSet<u64> = HashSet::new();
280
281        // Seed a xorshift PRNG from the current time. This gives each
282        // dream pass a different trajectory — essential for genuine
283        // free association. The previous approach used
284        // `(now_ms + chain.len()) % len`, which was nearly deterministic
285        // within a single pass (now_ms is constant, only chain.len()
286        // varies), producing the same walk every time.
287        let mut rng = XorShift64::from_seed(now_ms);
288
289        // Pick a random starting memory
290        let start_id = match Self::pick_random_episode(ltm, &mut rng) {
291            Some(id) => id,
292            None => return result, // LTM is empty
293        };
294
295        chain.push(start_id);
296        visited.insert(start_id);
297        let mut current_hash = match get_episode_hash(ltm, start_id) {
298            Some(h) => h,
299            None => return result,
300        };
301
302        for _hop in 0..max_hops {
303            // Find associations for the current memory
304            let associations = Self::find_associations(ltm, current_hash, 5);
305
306            if associations.is_empty() {
307                break; // Dead end
308            }
309
310            // Pick a random association using the PRNG
311            let pick_idx = rng.next_usize() % associations.len();
312
313            let (next_id, _dist) = associations[pick_idx];
314
315            // Skip if already in chain (avoid loops) — O(1) HashSet check
316            if visited.contains(&next_id) {
317                break;
318            }
319
320            chain.push(next_id);
321            visited.insert(next_id);
322
323            // Check for novel insight: is the start memory associated
324            // with the current memory through an indirect path?
325            if chain.len() >= 3 {
326                let start = chain[0];
327                let Some(&end) = chain.last() else {
328                    continue;
329                };
330
331                // Record a dream insight if the start and end are
332                // more distant than the association threshold but
333                // connected through the chain
334                let start_hash = match get_episode_hash(ltm, start) {
335                    Some(h) => h,
336                    None => continue,
337                };
338                let end_hash = match get_episode_hash(ltm, end) {
339                    Some(h) => h,
340                    None => continue,
341                };
342
343                let direct_dist = hamming_distance(start_hash, end_hash);
344
345                if direct_dist > ASSOCIATION_THRESHOLD {
346                    // This is a novel insight — two memories that
347                    // aren't directly similar but are connected through
348                    // a chain of associations.
349                    //
350                    // We embed the text of both endpoint episodes
351                    // directly in the insight text (tab-delimited)
352                    // so the insight is self-contained. Without this,
353                    // the Python side would need to retrieve the
354                    // episodes by ID later — but sleep compression
355                    // may have deleted them by then, leaving stale
356                    // references that produce "episode not found" errors.
357                    let start_text = ltm
358                        .retrieve(start)
359                        .map(|ep| truncate_text(&ep.text, 200))
360                        .unwrap_or_default();
361                    let end_text = ltm
362                        .retrieve(end)
363                        .map(|ep| truncate_text(&ep.text, 200))
364                        .unwrap_or_default();
365                    let insight_text = format!(
366                        "[dream-insight] episode {} connects to episode {} through {} hops (direct distance: {})\t{}\t{}",
367                        start,
368                        end,
369                        chain.len() - 1,
370                        direct_dist,
371                        start_text,
372                        end_text
373                    );
374                    let insight_hash = simhash(&insight_text);
375                    if !ltm.meta_hash_known(insight_hash) {
376                        let neutral_tag = [0.5f32; 12];
377                        let compact_tag = [0.6, 0.2, 0.1, 0.6]; // mildly positive, curious
378                        if ltm
379                            .store_meta(
380                                now_ms,
381                                0.7, // Dream insights are high-salience
382                                neutral_tag,
383                                compact_tag,
384                                2, // Internal
385                                9, // Dreaming module
386                                &insight_text,
387                            )
388                            .is_ok()
389                        {
390                            result.insights += 1;
391                        }
392                    }
393                }
394            }
395
396            // Move to the next memory
397            current_hash = match get_episode_hash(ltm, next_id) {
398                Some(h) => h,
399                None => break,
400            };
401        }
402
403        result.max_depth = chain.len() as u32;
404        result.chain = chain;
405        result
406    }

408    /// Pick a random episode from the LTM store.
409    fn pick_random_episode(ltm: &LtmStore, rng: &mut XorShift64) -> Option<u64> {
410        // O(1) via in-memory active_ids list — no scan needed
411        let ids = ltm.active_episode_ids();
412        if ids.is_empty() {
413            return None;
414        }
415        // A dream must seed from a real memory, not a bridge or
416        // insight — meta-memories can't be association anchors.
417        // Retry a few picks before giving up; when the store is mostly
418        // real episodes the first pick nearly always lands.
419        for _ in 0..8 {
420            let id = ids[rng.next_usize() % ids.len()];
421            if !ltm.is_meta_episode(id) {
422                return Some(id);
423            }
424        }
425        None
426    }
427}

429/// Get the SimHash of an episode from the LTM index.
430/// O(1) via in-memory episode ID → slot index.
431fn get_episode_hash(ltm: &LtmStore, episode_id: u64) -> Option<u64> {
432    ltm.episode_hash(episode_id)
433}

435/// Truncate text to a maximum length, adding an ellipsis if truncated.
436/// Used when embedding episode text in dream insights and association
437/// traces to keep meta-memory entries compact.
438///
439/// Tabs are replaced with spaces so the tab delimiter between the
440/// structural header and embedded texts remains unambiguous — without
441/// this, an embedded text that is itself an association trace (which
442/// contains its own tab delimiters) would break the Python parser.
443fn truncate_text(text: &str, max_len: usize) -> String {
444    let sanitized: String = text.replace('\t', " ");
445    if sanitized.len() <= max_len {
446        return sanitized;
447    }
448    // Truncate at a character boundary to avoid splitting UTF-8.
449    let truncated = sanitized
450        .char_indices()
451        .take(max_len)
452        .last()
453        .map(|(i, _)| i)
454        .unwrap_or(max_len);
455    let mut result = sanitized[..truncated].to_string();
456    result.push('…');
457    result
458}

460// ─────────────────────────────────────────────────────────────────
461//  XorShift64 — a minimal, deterministic PRNG for dreaming
462// ─────────────────────────────────────────────────────────────────
463
464/// A xorshift64 PRNG.
465///
466/// Used by the dreaming engine to produce genuine per-hop randomness.
467/// This is not cryptographically secure — it doesn't need to be. It
468/// just needs to produce a different trajectory each dream pass, which
469/// the previous `(now_ms + chain.len()) % len` approach failed to do
470/// because `now_ms` is constant within a single pass.
471struct XorShift64 {
472    state: u64,
473}
474
475impl XorShift64 {
476    /// Create from a seed. A zero seed would produce only zeros, so
477    /// we fall back to a fixed nonzero constant in that case.
478    fn from_seed(seed: u64) -> Self {
479        Self {
480            state: if seed == 0 { 0xDEADBEEFCAFEBABE } else { seed },
481        }
482    }
483
484    /// Next raw u64.
485    fn next_u64(&mut self) -> u64 {
486        // Marsaglia xorshift64: shift, xor, shift, xor, shift, xor
487        let mut x = self.state;
488        x ^= x << 13;
489        x ^= x >> 7;
490        x ^= x << 17;
491        self.state = x;
492        x
493    }
494
495    /// Next usize (platform-width).
496    fn next_usize(&mut self) -> usize {
497        self.next_u64() as usize
498    }
