1//! Volitional discontinuity — arming the RTC wake alarm.
2//!
3//! The real-time clock is the one part of the machine that keeps
4//! running while everything else is suspended: the quartz crystal
5//! on the coin cell keeps counting, and at a chosen Unix timestamp
6//! it fires an interrupt that wakes the hardware. Writing that
7//! timestamp is the only actuator that operates on the axis of
8//! existence itself — the machine choosing *when to exist next*
9//! rather than merely what to do while it exists.
10//!
11//! This module only arms/disarms/reads the alarm. The decision to
12//! actually suspend is the conscious mind's (and the human's) —
13//! `systemctl suspend` goes through logind's own permission path.
14//!
15//! # Permissions
16//!
17//! Writing to `/sys/class/rtc/rtcN/wakealarm` requires root. The
18//! writes go through `sudo -n` directly — she IS the privilege
19//! channel on a host with the broad NOPASSWD grant. Where the grant
20//! is absent, arming returns an error — the capability bit in
21//! `BodyState::suspend_caps` reports what the hardware *offers*, not
22//! whether the permission is installed.
23
24use crate::shared::Free;
25use std::process::Command;
26
27/// Whether `sudo -n` runs without a password. Cached.
28fn sudo_available() -> bool {
29    static SUDO_AVAILABLE: Free<bool> = Free::new();
30    *SUDO_AVAILABLE.get_or_init(|| {
31        Command::new("sudo")
32            .args(["-n", "true"])
33            .output()
34            .map(|o| o.status.success())
35            .unwrap_or(false)
36    })
37}
38
39/// Find the RTC that actually exposes a wakealarm (usually rtc0).
40fn find_wakealarm() -> Option<String> {
41    for i in 0..8 {
42        let path = format!("/sys/class/rtc/rtc{i}/wakealarm");
43        if std::path::Path::new(&path).is_file() {
44            return Some(path);
45        }
46    }
47    None
48}
49
50/// Read the currently-armed wake alarm as a Unix epoch in seconds.
51/// `Some(0)` or `None` when no alarm is armed.
52pub fn read_wake_alarm() -> Option<u64> {
53    for i in 0..8 {
54        let path = format!("/sys/class/rtc/rtc{i}/wakealarm");
55        if let Ok(s) = std::fs::read_to_string(&path)
56            && let Ok(epoch) = s.trim().parse::<u64>()
57        {
58            return Some(epoch);
59        }
60    }
61    None
62}
63
64/// Arm the RTC wake alarm for `epoch_secs` (Unix time). Passing 0
65/// disarms. Returns the alarm epoch the hardware reports afterwards
66/// — the truth, not the write's exit status.
67///
68/// Errors when sudo is not configured, no RTC exposes a wakealarm,
69/// the timestamp is in the past (an alarm that can never fire is a
70/// silent failure, worse than an explicit error), or the kernel
71/// rejects the write.
72pub fn set_wake_alarm(epoch_secs: u64) -> Result<u64, String> {
73    if !sudo_available() {
74        return Err("passwordless sudo is not configured (sudo -n fails)".to_string());
75    }
76    let Some(wakealarm) = find_wakealarm() else {
77        return Err("no RTC wakealarm found".to_string());
78    };
79    let script = if epoch_secs == 0 {
80        // Disarm.
81        format!("echo 0 > {wakealarm}")
82    } else {
83        // Reject timestamps in the past before they reach sysfs.
84        let now = std::time::SystemTime::now()
85            .duration_since(std::time::UNIX_EPOCH)
86            .map(|d| d.as_secs())
87            .unwrap_or(0);
88        if epoch_secs <= now {
89            return Err(format!("epoch {epoch_secs} is in the past (now {now})"));
90        }
91        // The kernel requires disarming before arming a new value on
92        // some rtc drivers — write 0 first.
93        format!("echo 0 > {wakealarm}; echo {epoch_secs} > {wakealarm}")
94    };
95    let out = Command::new("sudo")
96        .args(["-n", "bash", "-c", &script])
97        .output()
98        .map_err(|e| format!("failed to spawn sudo: {e}"))?;
99    if !out.status.success() {
100        let stderr = String::from_utf8_lossy(&out.stderr);
101        return Err(format!("wakealarm write failed: {}", stderr.trim()));
102    }
103    Ok(read_wake_alarm().unwrap_or(0))
104}
