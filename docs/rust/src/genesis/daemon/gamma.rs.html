1//! Gamma-band phase binding — the synchrony layer that forms proto-objects.
2//!
3//! The brain does not *compute* grouping; it *resonates* it. In V1/V2,
4//! features that belong to the same object fire in gamma phase-lock while
5//! features of different objects desynchronize (binding-by-synchrony,
6//! Singer & von der Malsburg). A proto-object is literally a gamma-coherent
7//! assembly: an internal token the system can attend, track, and reason
8//! about before any recognition happens.
9//!
10//! The math is Kuramoto: each visible cell carries a phase oscillator whose
11//! natural frequency is seeded by colour; coupling pulls phase-locked
12//! neighbours together (contiguity, the dominant Gestalt term) and adds a
13//! weak long-range pull between same-colour cells inside a receptive window
14//! (similarity, the second term). After integration, cells sharing phase
15//! and linkage are emitted as [`ProtoObject`] assemblies — same-colour
16//! components whose collective phases locked get bound even when not
17//! adjacent, which is the Gestalt capability a plain flood-fill lacks.
18//!
19//! Discipline: this layer forms objects; it never recognizes them. And a
20//! frame it cannot bind reports absence — an empty scene, not a fabricated
21//! object.
22
23const TWO_PI: f64 = std::f64::consts::TAU;
24
25/// One bound assembly — a gamma-coherent proto-object.
26#[derive(Debug, Clone)]
27pub struct ProtoObject {
28    /// Palette index (0–15) the assembly was bound under.
29    pub color: u8,
30    /// Number of member cells.
31    pub size: usize,
32    /// Member cells as flat indices (row * width + col).
33    pub cells: Vec<usize>,
34    /// Bounding box: (min_row, min_col, max_row, max_col).
35    pub bbox: (usize, usize, usize, usize),
36    /// Centroid (row, col).
37    pub centroid: (f64, f64),
38    /// Mean phase the assembly locked to — its binding signature.
39    pub phase: f64,
40    /// Mean-field coherence 0..1 of the member oscillators.
41    pub coherence: f64,
42}
43
44/// The V2 grouping field: Kuramoto phases bound to an optic frame.
45pub struct GammaField {
46    width: usize,
47    height: usize,
48    /// Per-cell phase after integration; `f64::NAN` where background.
49    phases: Vec<f64>,
50    /// Component id per cell; `usize::MAX` where background.
51    labels: Vec<usize>,
52}
53
54impl GammaField {
55    /// Bind a colour grid into proto-object assemblies.
56    ///
57    /// `grid` is `height * width` palette indices; `background` is the
58    /// palette index treated as ground (no oscillators form there).
59    /// `steps` is the Kuramoto integration horizon — ~40 iterations
60    /// settles most fields.
61    pub fn bind(
62        grid: &[u8],
63        width: usize,
64        height: usize,
65        background: u8,
66        steps: usize,
67    ) -> Vec<ProtoObject> {
68        let n = width * height;
69        if grid.len() != n || n == 0 {
70            return Vec::new();
71        }
72        let mut field = Self::segment(grid, width, height, background);
73        field.integrate(grid, steps);
74        field.assemblies(grid)
75    }
76
77    /// First pass — contiguity: label 4-connected same-colour components.
78    /// This is the contour-integration stage; grouping is seeded by
79    /// adjacency before similarity coupling acts.
80    fn segment(grid: &[u8], width: usize, height: usize, background: u8) -> Self {
81        let n = width * height;
82        let mut labels = vec![usize::MAX; n];
83        let phases = vec![f64::NAN; n];
84        let mut next_label = 0usize;
85        let mut stack = Vec::new();
86        for seed in 0..n {
87            if grid[seed] == background || labels[seed] != usize::MAX {
88                continue;
89            }
90            let colour = grid[seed];
91            labels[seed] = next_label;
92            stack.push(seed);
93            while let Some(idx) = stack.pop() {
94                let (r, c) = (idx / width, idx % width);
95                for (nr, nc) in [
96                    (r.wrapping_sub(1), c),
97                    (r + 1, c),
98                    (r, c.wrapping_sub(1)),
99                    (r, c + 1),
100                ] {
101                    if nr < height && nc < width {
102                        let nidx = nr * width + nc;
103                        if grid[nidx] == colour && labels[nidx] == usize::MAX {
104                            labels[nidx] = next_label;
105                            stack.push(nidx);
106                        }
107                    }
108                }
109            }
110            next_label += 1;
111        }
112        Self {
113            width,
114            height,
115            phases,
116            labels,
117        }
118    }
119
120    /// Kuramoto integration — natural frequency seeded by colour, strong
121    /// coupling to same-component neighbours, weak similarity coupling to
122    /// same-colour cells in a local window (the Gestalt similarity term
123    /// that lets separated fragments of one colour bind).
124    fn integrate(&mut self, grid: &[u8], steps: usize) {
125        const K_LOCAL: f64 = 0.7;
126        const K_SIMILAR: f64 = 0.05;
127        const WINDOW: usize = 8;
128        let (w, h) = (self.width, self.height);
129        for (i, &lab) in self.labels.iter().enumerate() {
130            if lab != usize::MAX {
131                // Colour seeds the natural frequency — distinct hues enter
132                // distinct phase basins, so binding separates at colour.
133                self.phases[i] = (grid[i] as f64 + 1.0) * 0.5;
134            }
135        }
136        for _ in 0..steps {
137            let cur = self.phases.clone();
138            for idx in 0..cur.len() {
139                if cur[idx].is_nan() {
140                    continue;
141                }
142                let lab = self.labels[idx];
143                let (r, c) = (idx / w, idx % w);
144                let mut drive = 0.0;
145                let mut count = 0.0;
146                // Strong: same-component 4-neighbours (contiguity).
147                for (nr, nc) in [
148                    (r.wrapping_sub(1), c),
149                    (r + 1, c),
150                    (r, c.wrapping_sub(1)),
151                    (r, c + 1),
152                ] {
153                    if nr < h && nc < w {
154                        let nidx = nr * w + nc;
155                        if self.labels[nidx] == lab {
156                            drive += (cur[nidx] - cur[idx]).sin();
157                            count += 1.0;
158                        }
159                    }
160                }
161                let mut delta = if count > 0.0 {
162                    K_LOCAL * drive / count
163                } else {
164                    0.0
165                };
166                // Weak: same-colour cells in a window (similarity Gestalt —
167                // remote fragments of one hue pull toward shared phase).
168                let mut sim = 0.0;
169                let mut scount = 0.0;
170                let r0 = r.saturating_sub(WINDOW);
171                let r1 = (r + WINDOW).min(h - 1);
172                let c0 = c.saturating_sub(WINDOW);
173                let c1 = (c + WINDOW).min(w - 1);
174                for nr in r0..=r1 {
175                    for nc in c0..=c1 {
176                        let nidx = nr * w + nc;
177                        if nidx != idx && self.labels[nidx] != lab && grid[nidx] == grid[idx] {
178                            sim += (cur[nidx] - cur[idx]).sin();
179                            scount += 1.0;
180                        }
181                    }
182                }
183                if scount > 0.0 {
184                    delta += K_SIMILAR * sim / scount;
185                }
186                self.phases[idx] = (cur[idx] + delta).rem_euclid(TWO_PI);
187            }
188        }
189    }

191    /// Read out assemblies: contiguous components, merged when their mean
192    /// phases locked within tolerance — the similarity binding. Each object
193    /// carries its locked phase and mean-field coherence.
194    fn assemblies(&self, grid: &[u8]) -> Vec<ProtoObject> {
195        const PHASE_TOL: f64 = 0.35;
196        let mut comps: std::collections::BTreeMap<usize, Vec<usize>> = Default::default();
197        for (i, &lab) in self.labels.iter().enumerate() {
198            if lab != usize::MAX {
199                comps.entry(lab).or_default().push(i);
200            }
201        }
202        let mut stats: Vec<(usize, f64, f64)> = Vec::new(); // (label, mean_phase, coherence)
203        for (&lab, cells) in &comps {
204            let (mut sx, mut sy) = (0.0, 0.0);
205            for &i in cells {
206                sx += self.phases[i].cos();
207                sy += self.phases[i].sin();
208            }
209            let m = cells.len() as f64;
210            let coherence = (sx * sx + sy * sy).sqrt() / m;
211            stats.push((lab, sy.atan2(sx), coherence));
212        }
213        // Merge components whose collective phases locked (phase-tolerant
214        // union — non-adjacent same-phase fragments form one assembly).
215        let mut parent: Vec<usize> = (0..stats.len()).collect();
216        fn find(p: &mut Vec<usize>, x: usize) -> usize {
217            if p[x] != x {
218                p[x] = find(p, p[x]);
219            }
220            p[x]
221        }
222        for a in 0..stats.len() {
223            for b in (a + 1)..stats.len() {
224                let d = (stats[a].1 - stats[b].1).abs();
225                let d = d.min(TWO_PI - d);
226                if d < PHASE_TOL && grid[comps[&stats[a].0][0]] == grid[comps[&stats[b].0][0]] {
227                    let (ra, rb) = (find(&mut parent, a), find(&mut parent, b));
228                    if ra != rb {
229                        parent[rb] = ra;
230                    }
231                }
232            }
233        }
234        let mut merged: std::collections::BTreeMap<usize, Vec<usize>> = Default::default();
235        for (i, &(lab, phase, coh)) in stats.iter().enumerate() {
236            merged
237                .entry(find(&mut parent, i))
238                .or_default()
239                .extend(comps[&lab].iter().copied());
240            let _ = (phase, coh);
241        }
242        let mut out = Vec::new();
243        for (_root, cells) in merged {
244            let n = cells.len();
245            let (mut rmin, mut rmax, mut cmin, mut cmax) = (usize::MAX, 0, usize::MAX, 0);
246            let (mut sr, mut sc) = (0.0, 0.0);
247            let (mut sx, mut sy) = (0.0, 0.0);
248            for &i in &cells {
249                let (r, c) = (i / self.width, i % self.width);
250                rmin = rmin.min(r);
251                rmax = rmax.max(r);
252                cmin = cmin.min(c);
253                cmax = cmax.max(c);
254                sr += r as f64;
255                sc += c as f64;
256                sx += self.phases[i].cos();
257                sy += self.phases[i].sin();
258            }
259            let first = cells[0];
260            out.push(ProtoObject {
261                color: grid[first],
262                size: n,
263                cells,
264                bbox: (rmin, cmin, rmax, cmax),
265                centroid: (sr / n as f64, sc / n as f64),
266                phase: sy.atan2(sx),
267                coherence: (sx * sx + sy * sy).sqrt() / n as f64,
268            });
269        }
270        out
271    }
