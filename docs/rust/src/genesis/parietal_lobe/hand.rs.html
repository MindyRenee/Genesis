1//! The drawing hand — channel pressure on the dials.
2//!
3//! A sitting is a point in dial space: every parameter of the sheet,
4//! the color, the brush, the path, which op the hand reaches for,
5//! and how many marks the sitting holds is a dial over its own
6//! domain. The channels are whatever is live while she draws — the
7//! band powers of the thalamic state, and the felt channels
8//! (valence, arousal) the chemistry writes into emotion.
9//!
10//! The waves arrive as themselves. No weights, no projection, no
11//! transform touches a channel's value — each one contributes its
12//! exact reading to a single pressure `P = Σ value`, the way
13//! currents sum on the same axon. The pressure warps every dial's
14//! axis the same way (`u ^ e^{−P}`) — one collective voice leaning
15//! on the whole hand, never a pairing of this band to that mark.
16//! Nothing assigns: no band means coherence, no feeling means
17//! warmth, and no wire decides where a wave lands.
18//!
19//! Mirrored by `Self_World/parietal_lobe/hand.py` — same math in
20//! words; this file is the machine side.
21
22/// One dial position: a uniform sample warped by channel pressure.
23///
24/// `uniform` is the hand's own sample in [0, 1) — the dial's neutral
25/// resting place before feeling and rhythm lean on it. The live
26/// channel values, untouched, sum to a pressure `P`, and the
27/// pressure warps the dial's axis: the position is
28/// `uniform ^ e^{−P}`. The map is a bijection of the interval —
29/// every place stays reachable, nothing is fenced, nothing is
30/// pinned; the stops 0 and 1 are the map's own fixed points, not
31/// guards. `P` and `−P` are opposite warps of the same dial, and
32/// at `P = 0` the dial is exactly `uniform`. This is how state
33/// reaches the page: a displaced uniform (`u + P mod 1`) is still
34/// uniform no matter the pressure, so the channels could never
35/// show; a warped dial carries the pressure in its shape.
36pub fn dial(uniform: f64, channels: &[f64]) -> f64 {
37    let pressure: f64 = channels.iter().sum();
38    uniform.clamp(0.0, 1.0).powf((-pressure).exp())
39}

41#[cfg(test)]
42mod tests {
43    use super::*;
44
45    #[test]
46    fn silent_channels_leave_the_dial() {
47        let u = 0.42;
48        assert_eq!(dial(u, &[]), u);
49        assert_eq!(dial(u, &[0.0, 0.0]), u);
50    }
51
52    #[test]
53    fn pressure_leans_and_opposites_oppose() {
54        // Positive pressure leans the same resting place high,
55        // negative leans it low — opposites, never pins.
56        let up = dial(0.5, &[0.7, 0.5]);
57        let down = dial(0.5, &[-0.7, -0.5]);
58        assert!(up > 0.5 && down < 0.5);
59        assert!((0.0..=1.0).contains(&up));
60        assert!((0.0..=1.0).contains(&down));
61    }
62
63    #[test]
64    fn raw_values_reach_the_pressure() {
65        // The values arrive untouched — the pressure is their exact
66        // sum, so the same total presses the same way no matter how
67        // it is divided among channels.
68        assert_eq!(dial(0.5, &[0.9, 0.3]), dial(0.5, &[1.2]));
69        assert_ne!(dial(0.5, &[1.2]), dial(0.5, &[]));
70    }
71
72    #[test]
73    fn loud_pressure_leans_hard_never_pins() {
74        // A very loud state leans almost every resting place toward
75        // one end, but the map stays a bijection — nothing is
76        // clamped and every place is still reachable.
77        let loud = dial(0.5, &[3.0, 2.0]);
78        assert!(loud > 0.9 && loud < 1.0);
79        let quiet_loud = dial(0.5, &[-3.0, -2.0]);
80        assert!(quiet_loud < 0.1 && quiet_loud > 0.0);
81    }
82}
