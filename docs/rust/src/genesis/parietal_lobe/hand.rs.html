1//! The drawing hand — the channel→dial wiring.
2//!
3//! A sitting is a point in dial space: every parameter of the sheet,
4//! the color, the brush, the path, which op the hand reaches for,
5//! and how many marks the sitting holds is a dial over its own
6//! domain. Nothing about the dials is assigned — no band means
7//! coherence, no feeling means warmth — because the wiring itself
8//! carries no meaning. Each (`channel`, `dial`) pair is bound by a
9//! hash-seeded weight in [−1, 1]: fixed, arbitrary, stable across
10//! processes, as her wiring is.
11//!
12//! The channels are whatever is live while she draws — the band
13//! powers of the thalamic state, and the felt channels (valence,
14//! arousal) the chemistry writes into emotion. Every channel pushes
15//! every dial; the dial's resting position is a uniform sample, and
16//! the channels displace it. Two modalities, one mechanism: waves
17//! influence creativity the way chemicals influence emotion —
18//! continuously, without interpretation, and only while the hand is
19//! working.
20//!
21//! Mirrored by `Self_World/parietal_lobe/hand.py` — same math in
22//! words; this file is the machine side.
23
24/// FNV-1a over the `channel:dial` pair — the deterministic seed.
25fn pair_hash(channel: &str, dial: &str) -> u64 {
26    let mut h: u64 = 0xcbf2_9ce4_8422_2325;
27    for &b in channel.as_bytes().iter().chain(b":").chain(dial.as_bytes()) {
28        h ^= u64::from(b);
29        h = h.wrapping_mul(0x0000_0100_0000_01b3);
30    }
31    h
32}
33
34/// SplitMix64 finalize — the seed becomes a well-mixed u64.
35fn mix64(mut z: u64) -> u64 {
36    z = (z ^ (z >> 30)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
37    z = (z ^ (z >> 27)).wrapping_mul(0x94d0_49bb_1331_11eb);
38    z ^ (z >> 31)
39}

41/// The weight binding `channel` to `dial`, in [−1, 1).
42///
43/// Deterministic: the same pair always wires the same way, on any
44/// host, in any process. The top 53 bits of the mixed hash map onto
45/// the unit interval and the interval is centered at zero — a
46/// channel is as likely to push a dial down as up; the direction is
47/// the pair's own, not ours.
48pub fn channel_weight(channel: &str, dial: &str) -> f64 {
49    let u = (mix64(pair_hash(channel, dial)) >> 11) as f64 / (1u64 << 53) as f64;
50    2.0 * u - 1.0
51}
52
53/// One dial position: a uniform resting place displaced by channels.
54///
55/// `uniform` is the hand's own sample in [0, 1) — the dial's neutral
56/// position before feeling and rhythm push it. Each live channel
57/// shifts the dial by `value × channel_weight(channel, dial)`; the
58/// result wraps the dial's circle so no push is ever clamped out.
59/// Returns a position in [0, 1).
60pub fn dial(uniform: f64, channels: &[(String, f64)], dial: &str) -> f64 {
61    let u = channels.iter().fold(uniform, |u, (name, value)| {
62        u + value * channel_weight(name, dial)
63    });
64    u.rem_euclid(1.0)
65}

67#[cfg(test)]
68mod tests {
69    use super::*;
70
71    #[test]
72    fn weight_is_deterministic_and_bounded() {
73        for channel in ["gamma", "theta", "valence", "arousal"] {
74            for dial in ["size", "hue", "op", "strokes"] {
75                let w = channel_weight(channel, dial);
76                assert!((-1.0..1.0).contains(&w));
77                assert_eq!(w, channel_weight(channel, dial));
78            }
79        }
80    }
81
82    #[test]
83    fn different_pairs_wire_differently() {
84        assert_ne!(
85            channel_weight("gamma", "size"),
86            channel_weight("gamma", "hue")
87        );
88        assert_ne!(
89            channel_weight("theta", "size"),
90            channel_weight("gamma", "size")
91        );
92    }
93
94    #[test]
95    fn silent_channels_leave_the_dial() {
96        let u = 0.42;
97        assert_eq!(dial(u, &[], "size"), u);
98        let silent = vec![("gamma".to_string(), 0.0)];
99        assert_eq!(dial(u, &silent, "size"), u);
100    }
101
102    #[test]
103    fn live_channels_move_the_dial_and_wrap() {
104        let live = vec![
105            ("gamma".to_string(), 0.7),
106            ("theta".to_string(), 0.5),
107            ("valence".to_string(), -0.3),
108        ];
109        let moved = dial(0.5, &live, "size");
110        assert!((0.0..1.0).contains(&moved));
111        assert_ne!(moved, 0.5);
112        // A large push wraps the dial rather than pinning it.
113        let loud = vec![("gamma".to_string(), 9.0)];
114        assert!((0.0..1.0).contains(&dial(0.5, &loud, "size")));
115    }
116}
