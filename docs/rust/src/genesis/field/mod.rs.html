1//! Machine-native field architecture for Genesis.
2//!
3//! A major system is a SystemFiber; its subsystems are the layers of that
4//! system's latent field. Geometry is separated from semantics: a manifold
5//! is a container and metric, while layers own represented state and dynamics.
6//!
7//! The latent manifold is the κ-stereographic sphere — the constant
8//! *positive* curvature counterpart of the Poincaré ball (Bachmann et al.
9//! 2020): the sphere of curvature κ projected stereographically onto the
10//! chart, with the north pole at the origin. The field operates inside
11//! the hemisphere cap ‖x‖ < 1/√κ, a geodesically convex region whose
12//! center is the pole — the organism's nominal home.
13//!
14//! The implementation is dependency-free and deterministic. It does not
15//! assume that a fixed number of fibers is correct.

73fn dot(a: &[f64], b: &[f64]) -> f64 {
74    let scale_a = a.iter().fold(0.0_f64, |scale, &x| scale.max(x.abs()));
75    let scale_b = b.iter().fold(0.0_f64, |scale, &x| scale.max(x.abs()));
76    if scale_a == 0.0 || scale_b == 0.0 {
77        return 0.0;
78    }
79    let normalized = a
80        .iter()
81        .zip(b)
82        .map(|(&x, &y)| (x / scale_a) * (y / scale_b))
83        .sum::<f64>();
84    if normalized == 0.0 {
85        return 0.0;
86    }
87    (normalized * scale_a) * scale_b
88}
89fn norm(a: &[f64]) -> f64 {
90    a.iter().fold(0.0_f64, |acc, &x| acc.hypot(x))
91}

125impl StereographicSphere {
126    /// The κ-stereographic sphere: the sphere of constant *positive*
127    /// curvature κ, charted by stereographic projection. The origin
128    /// of the chart is the image of the north pole — the
129    /// distinguished center of the model. The point at infinity is
130    /// the south pole, the one point the chart cannot represent.
131    ///
132    /// `contains`/`project` enforce the field's operating region —
133    /// the hemisphere cap around the pole, ‖x‖ < 1/√κ (the
134    /// stereographic image of the equator). The cap is geodesically
135    /// convex: minimizing geodesics between cap points stay in the
136    /// cap, and any set of cap points has a unique Fréchet mean
137    /// inside it (Buss & Fillmore 2001). The geometric primitives
138    /// (`mobius_add`, `log_*`, `exp_*`, `geodesic`) accept any valid
139    /// chart point — finite and right dimension — and return honest
140    /// sphere points, which may legitimately lie outside the cap;
141    /// the state boundaries re-project.
142    pub fn new(dimension: usize, c: f64) -> Result<Self, FieldError> {
143        if dimension == 0 || !c.is_finite() || c <= 0.0 {
144            return Err(FieldError::InvalidGeometry);
145        }
146        Ok(Self {
147            dimension,
148            curvature: c,
149        })
150    }
151    pub fn dimension(&self) -> usize {
152        self.dimension
153    }
154    pub fn curvature(&self) -> f64 {
155        self.curvature
156    }
157    /// The chart radius of the equator — the rim of the operating
158    /// cap. Points at exactly this radius sit on the equator;
159    /// beyond it lies the southern hemisphere, out to the
160    /// unrepresentable south pole at infinity.
161    fn scaled_radius(&self) -> f64 {
162        1.0 / self.curvature.sqrt()
163    }
164    /// Whether x is inside the operating cap — the open hemisphere
165    /// around the pole where the field works.
166    pub fn contains(&self, x: &[f64]) -> bool {
167        x.len() == self.dimension
168            && x.iter().all(|v| v.is_finite())
169            && norm(x) < self.scaled_radius()
170    }
171    /// Retract x into the cap along its ray. This is the containment
172    /// policy, not a geometric projection of the manifold: outside-
173    /// cap coordinates are valid sphere points, but the field keeps
174    /// its states inside the hemisphere.
175    pub fn project(&self, x: &[f64]) -> Result<Vector, FieldError> {
176        if x.len() != self.dimension {
177            return Err(FieldError::DimensionMismatch);
178        }
179        if x.iter().any(|v| !v.is_finite()) {
180            return Err(FieldError::NonFiniteState);
181        }
182        let r = self.scaled_radius();
183        let n = norm(x);
184        if !n.is_finite() {
185            return Err(FieldError::NonFiniteState);
186        }
187        if n < r {
188            return Ok(x.to_vec());
189        }
190        let target = r * (1.0 - 1.0e-9);
191        if !target.is_finite() || target <= 0.0 {
192            return Err(FieldError::InvalidGeometry);
193        }
194        let scale = target / n;
195        if !scale.is_finite() || scale <= 0.0 {
196            return Err(FieldError::NonFiniteState);
197        }
198        let projected = scaled(x, scale);
199        if projected.iter().any(|v| !v.is_finite()) || !self.contains(&projected) {
200            return Err(FieldError::NonFiniteState);
201        }
202        Ok(projected)
203    }
204    /// Geodesic distance on the sphere. In stereographic
205    /// coordinates, with û = √κ·x the unit-sphere chart, the lifted
206    /// chord gives
207    ///
208    ///   tan(d·√κ/2) = √κ‖x − y‖
209    ///                 / √(1 + 2κ⟨x,y⟩ + κ²‖x‖²‖y‖²)
210    ///
211    /// so d = (2/√κ)·atan2(√κ‖x−y‖, √(1+2κ⟨x,y⟩+κ²‖x‖²‖y‖²)). The
212    /// denominator factors as (1+κ⟨x,y⟩)² + κ²(‖x‖²‖y‖² − ⟨x,y⟩²) —
213    /// nonnegative by Cauchy–Schwarz, vanishing only for antipodal
214    /// pairs, where atan2 correctly returns the maximal distance
215    /// π/√κ instead of dividing by zero.
216    pub fn distance(&self, x: &[f64], y: &[f64]) -> Result<f64, FieldError> {
217        self.validate_point(x)?;
218        self.validate_point(y)?;
219        let sqrt_c = self.curvature.sqrt();
220        let c = self.curvature;
221        let nx = dot(x, x);
222        let ny = dot(y, y);
223        let xy = dot(x, y);
224        let dxy = norm(&sub(x, y));
225        if !(nx.is_finite() && ny.is_finite() && xy.is_finite() && dxy.is_finite()) {
226            return Err(FieldError::NonFiniteState);
227        }
228        let opposite = (1.0 + 2.0 * c * xy + c * c * nx * ny).max(0.0).sqrt();
229        let distance = 2.0 * (sqrt_c * dxy).atan2(opposite) / sqrt_c;
230        if !distance.is_finite() {
231            return Err(FieldError::NonFiniteState);
232        }
233        Ok(distance)
234    }
235    /// The tangent at the pole (the chart's origin) pointing along
236    /// the geodesic to x, distance-normalized: ‖v‖ = d(0, x). A
237    /// point at chart radius r sits at geodesic distance
238    /// (2/√κ)·atan(√κ·r) from the pole — unlike the ball's atanh
239    /// this needs no clamp: atan is finite for every representable
240    /// point, tending to π/(2√κ) as r → ∞ (the south pole).
241    pub fn log_origin(&self, x: &[f64]) -> Result<Vector, FieldError> {
242        self.validate_point(x)?;
243        let r = norm(x);
244        if !r.is_finite() {
245            return Err(FieldError::NonFiniteState);
246        }
247        if r < EPS {
248            return Ok(vec![0.0; self.dimension]);
249        }
250        let sqrt_c = self.curvature.sqrt();
251        let scaled_r = sqrt_c * r;
252        if !scaled_r.is_finite() || scaled_r <= 0.0 {
253            return Err(FieldError::NonFiniteState);
254        }
255        let z = scaled_r.atan() * 2.0 / scaled_r;
256        if !z.is_finite() || z <= 0.0 {
257            return Err(FieldError::NonFiniteState);
258        }
259        let tangent = scaled(x, z);
260        if tangent.iter().any(|v| !v.is_finite()) {
261            return Err(FieldError::NonFiniteState);
262        }
263        Ok(tangent)
264    }
265    /// Step from the pole along direction v for ‖v‖ units of
266    /// geodesic distance: the image is the direction of v at chart
267    /// radius (1/√κ)·tan(√κ·‖v‖/2). tan is honest about the sphere
268    /// being compact: ‖v‖ = π/√κ lands at the unrepresentable south
269    /// pole (rejected as non-finite), and beyond it the map wraps
270    /// around — the sphere closes on itself.
271    pub fn exp_origin(&self, v: &[f64]) -> Result<Vector, FieldError> {
272        if v.len() != self.dimension {
273            return Err(FieldError::DimensionMismatch);
274        }
275        if v.iter().any(|x| !x.is_finite()) {
276            return Err(FieldError::NonFiniteState);
277        }
278        let r = norm(v);
279        if !r.is_finite() {
280            return Err(FieldError::NonFiniteState);
281        }
282        if r < EPS {
283            return Ok(v.to_vec());
284        }
285        let sqrt_c = self.curvature.sqrt();
286        let scaled_r = sqrt_c * r;
287        if !scaled_r.is_finite() {
288            return Err(FieldError::NonFiniteState);
289        }
290        let z = (scaled_r / 2.0).tan() / scaled_r;
291        if !z.is_finite() {
292            return Err(FieldError::NonFiniteState);
293        }
294        let out = scaled(v, z);
295        if out.iter().any(|x| !x.is_finite()) {
296            return Err(FieldError::NonFiniteState);
297        }
298        Ok(out)
299    }
300    /// Exact geodesic flight: lift the chart point x and chart
301    /// velocity u to the sphere of radius 1/√κ and advance the
302    /// great circle by θ = ω·dt — parallel transport itself,
303    /// not a truncation of it. On the unit-sphere chart
304    /// y = √κ·x with conformal λ(y) = 2/(1+‖y‖²), the lift is
305    ///
306    ///   Y = (λy, 1 − λ‖y‖²),   V = (λ(w − λ⟨y,w⟩y), −λ²⟨y,w⟩),
307    ///   w = √κ·u,
308    ///
309    /// which satisfies Y·V = 0 and ‖V‖ = λ‖w‖ = ω exactly.
310    /// The geodesic is the rotation Y(t) = Y cos θ + V sin θ/ω,
311    /// V(t) = V cos θ − ωY sin θ — norm-preserving to machine
312    /// precision at any step size, periodic where Euler blows
313    /// up — and the chart back is
314    ///
315    ///   y⁺ = Y⁺_eq/(1 + Y⁺_pol),   u⁺ = (V⁺_eq − V⁺_pol·y⁺)/(λ(y⁺)√κ).
316    pub fn geodesic_flow(
317        &self,
318        x: &[f64],
319        u: &[f64],
320        dt: f64,
321    ) -> Result<(Vector, Vector), FieldError> {
322        if x.len() != self.dimension || u.len() != self.dimension {
323            return Err(FieldError::DimensionMismatch);
324        }
325        if x.iter().chain(u.iter()).any(|c| !c.is_finite()) || !dt.is_finite() || dt < 0.0 {
326            return Err(FieldError::NonFiniteState);
327        }
328        let sqrt_c = self.curvature.sqrt();
329        let y = scaled(x, sqrt_c);
330        let y2 = dot(&y, &y);
331        let lambda = 2.0 / (1.0 + y2);
332        let mut big_y = scaled(&y, lambda);
333        big_y.push(1.0 - lambda * y2);
334        let w = scaled(u, sqrt_c);
335        let s = dot(&y, &w);
336        let mut big_v: Vector = w
337            .iter()
338            .zip(y.iter())
339            .map(|(wi, yi)| lambda * (wi - lambda * s * yi))
340            .collect();
341        big_v.push(-lambda * lambda * s);
342        let omega = norm(&big_v);
343        if omega < EPS {
344            return Ok((x.to_vec(), u.to_vec()));
345        }
346        let theta = omega * dt;
347        let (sn, cs) = theta.sin_cos();
348        let y_next: Vector = big_y
349            .iter()
350            .zip(big_v.iter())
351            .map(|(yi, vi)| yi * cs + vi * sn / omega)
352            .collect();
353        let v_next: Vector = big_y
354            .iter()
355            .zip(big_v.iter())
356            .map(|(yi, vi)| vi * cs - omega * yi * sn)
357            .collect();
358        let (y_eq, y_pol) = y_next.split_at(self.dimension);
359        let denom = 1.0 + y_pol[0];
360        if !denom.is_finite() || denom.abs() < EPS {
361            return Err(FieldError::NonFiniteState);
362        }
363        let y_out = scaled(y_eq, 1.0 / denom);
364        let lambda_out = 2.0 / (1.0 + dot(&y_out, &y_out));
365        if !lambda_out.is_finite() {
366            return Err(FieldError::NonFiniteState);
367        }
368        let (v_eq, v_pol) = v_next.split_at(self.dimension);
369        let u_out: Vector = v_eq
370            .iter()
371            .zip(y_out.iter())
372            .map(|(ve, yo)| (ve - v_pol[0] * yo) / (lambda_out * sqrt_c))
373            .collect();
374        let x_out = scaled(&y_out, 1.0 / sqrt_c);
375        if x_out.iter().chain(u_out.iter()).any(|c| !c.is_finite()) {
376            return Err(FieldError::NonFiniteState);
377        }
378        Ok((x_out, u_out))
379    }
380    /// The Christoffel contraction Γ(x; v, v) of the
381    /// κ-stereographic metric g = λ²δ, λ = 2/(1+κ‖x‖²) — the
382    /// chart-coordinate deflection a freely carried vector
383    /// undergoes. On a positively curved sphere parallel
384    /// transport is not coordinate-constant: a vector moving
385    /// tangentially is bent back toward the pole
386    /// (Γ ∝ −‖v‖²x), and a vector moving radially finds the
387    /// chart stretching under it (Γ ∝ +⟨x,v⟩v) — at constant
388    /// geodesic speed the chart speed grows outward. The
389    /// geodesic equation reads v̇ = −Γ(v, v):
390    ///
391    ///   Γ^k_ij v^i v^j = −2κ/(1+κ‖x‖²)·(2⟨x,v⟩v^k − ‖v‖²x^k).
392    ///
393    /// Vanishes at the pole (the chart is Cartesian there) and
394    /// diverges at the south pole, which the cap never reaches.
395    pub fn connection_acceleration(&self, x: &[f64], v: &[f64]) -> Result<Vector, FieldError> {
396        if x.len() != self.dimension || v.len() != self.dimension {
397            return Err(FieldError::DimensionMismatch);
398        }
399        if x.iter().chain(v.iter()).any(|c| !c.is_finite()) {
400            return Err(FieldError::NonFiniteState);
401        }
402        let x2 = dot(x, x);
403        let xv = dot(x, v);
404        let v2 = dot(v, v);
405        let w = 1.0 + self.curvature * x2;
406        if !w.is_finite() || w.abs() < EPS {
407            return Err(FieldError::NonFiniteState);
408        }
409        let mut gamma = scaled(v, 2.0 * xv);
410        for (g, xk) in gamma.iter_mut().zip(x.iter()) {
411            *g -= v2 * xk;
412        }
413        let gamma = scaled(&gamma, -2.0 * self.curvature / w);
414        if gamma.iter().any(|c| !c.is_finite()) {
415            return Err(FieldError::NonFiniteState);
416        }
417        Ok(gamma)
418    }
419    /// Möbius addition — the gyrovector operation that plays the
420    /// role of vector addition on the stereographic sphere (Ungar;
421    /// κ > 0, equator image at radius 1/√κ):
422    ///
423    ///   x ⊕_κ y = [(1 − 2κ⟨x,y⟩ − κ‖y‖²)x + (1 + κ‖x‖²)y]
424    ///             / (1 − 2κ⟨x,y⟩ + κ²‖x‖²‖y‖²)
425    ///
426    /// Non-commutative and non-associative, but left-cancellative:
427    /// (−x) ⊕_κ (x ⊕_κ y) = y — the identity that makes the general
428    /// log/exp pair below true inverses. Adding a fixed element is a
429    /// spherical translation — a rotation of the sphere, so it moves
430    /// every point the same geodesic distance. The denominator can
431    /// only vanish when y is the antipode of −x (the sum would be
432    /// the unrepresentable south pole); inside the cap it is bounded
433    /// away from zero by (1 − κ‖x‖‖y‖)².
434    pub fn mobius_add(&self, x: &[f64], y: &[f64]) -> Result<Vector, FieldError> {
435        self.validate_point(x)?;
436        self.validate_point(y)?;
437        let c = self.curvature;
438        let nx = dot(x, x);
439        let ny = dot(y, y);
440        let xy = dot(x, y);
441        if !(nx.is_finite() && ny.is_finite() && xy.is_finite()) {
442            return Err(FieldError::NonFiniteState);
443        }
444        let x_num = 1.0 - 2.0 * c * xy - c * ny;
445        let y_num = 1.0 + c * nx;
446        let denom = 1.0 - 2.0 * c * xy + c * c * nx * ny;
447        if !(x_num.is_finite() && y_num.is_finite() && denom.is_finite() && denom > EPS) {
448            return Err(FieldError::NonFiniteState);
449        }
450        let mut out = vec![0.0; self.dimension];
451        for ((dst, &xi), &yi) in out.iter_mut().zip(x).zip(y) {
452            *dst = (x_num * xi + y_num * yi) / denom;
453        }
454        if out.iter().any(|v| !v.is_finite()) {
455            return Err(FieldError::NonFiniteState);
456        }
457        Ok(out)
458    }
459    /// Möbius subtraction: (−x) ⊕_κ y — the displacement from x to
460    /// y expressed as a chart point. A spherical translation of y
461    /// back toward the origin: d(0, −x ⊕ y) = d(x, y).
462    fn mobius_sub(&self, x: &[f64], y: &[f64]) -> Result<Vector, FieldError> {
463        self.mobius_add(&scaled(x, -1.0), y)
464    }
465    /// Logarithm map at an arbitrary basepoint: the tangent vector
466    /// at x that points along the geodesic toward y, with the same
467    /// distance-normalized convention as `log_origin` (‖v‖ = the
468    /// geodesic distance the vector travels). Because x ↦ (−x) ⊕ x
469    /// is a spherical translation to the origin, this is just
470    /// `log_origin` of the Möbius displacement.
471    pub fn log_at(&self, x: &[f64], y: &[f64]) -> Result<Vector, FieldError> {
472        self.log_origin(&self.mobius_sub(x, y)?)
473    }
474    /// Exponential map at an arbitrary basepoint: step from x along
475    /// the geodesic in direction v for ‖v‖ units of geodesic
476    /// distance — `x ⊕_κ exp_origin(v)`. Inverse of `log_at` by
477    /// left-cancellation.
478    pub fn exp_at(&self, x: &[f64], v: &[f64]) -> Result<Vector, FieldError> {
479        self.validate_point(x)?;
480        self.mobius_add(x, &self.exp_origin(v)?)
481    }
482    /// The geodesic from x toward y at fraction t of the spherical
483    /// distance: d(x, γ(t)) = t·d(x,y) exactly. γ(0) = x, γ(1) = y;
484    /// t outside \[0,1\] continues along the same great circle — the
485    /// geodesic wraps around the sphere.
486    pub fn geodesic(&self, x: &[f64], y: &[f64], t: f64) -> Result<Vector, FieldError> {
487        if !t.is_finite() {
488            return Err(FieldError::NonFiniteState);
489        }
490        self.exp_at(x, &scaled(&self.log_at(x, y)?, t))
491    }
492    /// Geodesic point reflection through `through`: the point as far

498        self.geodesic(x, through, 2.0)

500    /// The true spherical antipode — the point maximally distant
501    /// from x at d = π/√κ. Under stereographic projection the
502    /// antipode map is inversion in the equator circle composed with
503    /// negation: −x/(κ‖x‖²). It lands outside the cap (in the
504    /// southern hemisphere), the genuine "far side." The pole's own
505    /// antipode is the unrepresentable south pole at infinity — an
506    /// error, not a clamp.
507    pub fn antipode(&self, x: &[f64]) -> Result<Vector, FieldError> {
508        self.validate_point(x)?;
509        let r2 = dot(x, x);
510        if !r2.is_finite() || r2 <= EPS {
511            return Err(FieldError::NonFiniteState);
512        }
513        Ok(scaled(x, -1.0 / (self.curvature * r2)))
514    }
515    /// A valid chart point: right dimension, finite. Any such point
516    /// names a sphere point in the stereographic chart — cap
517    /// membership (`contains`) is a separate, stricter question.
518    fn validate_point(&self, x: &[f64]) -> Result<(), FieldError> {
519        if x.len() != self.dimension || x.iter().any(|v| !v.is_finite()) {
520            Err(FieldError::InvalidPoint)
521        } else {
522            Ok(())
523        }
524    }
525    /// The tangent-space mean at the pole — the Euclidean average
526    /// of the lifted log coordinates, mapped back. Exact for a
527    /// single point and the right seed for `frechet_mean`; away
528    /// from the pole it is only the barycenter's first
529    /// approximation, since the log map distorts with distance.
530    fn tangent_barycenter(&self, points: &[Vector]) -> Result<Vector, FieldError> {
531        let tangents = points
532            .iter()
533            .map(|p| self.log_origin(p))
534            .collect::<Result<Vec<_>, _>>()?;
535        let scale = tangents
536            .iter()
537            .flatten()
538            .fold(0.0_f64, |m, &v| m.max(v.abs()));
539        if !scale.is_finite() {
540            return Err(FieldError::NonFiniteState);
541        }
542        if scale == 0.0 {
543            return self.exp_origin(&vec![0.0; self.dimension]);
544        }
545        let count = points.len() as f64;
546        let mut mean = vec![0.0; self.dimension];
547        for tangent in &tangents {
548            for (dst, &src) in mean.iter_mut().zip(tangent) {
549                *dst += (src / scale) / count;
550            }
551        }
552        if mean.iter().any(|v| !v.is_finite()) {
553            return Err(FieldError::NonFiniteState);
554        }
555        let mean = scaled(&mean, scale);
556        if mean.iter().any(|v| !v.is_finite()) {
557            return Err(FieldError::NonFiniteState);
558        }
559        self.exp_origin(&mean)
560    }
561    /// The Fréchet mean: the point minimizing Σᵢ d²(m, xᵢ) — the
562    /// true center of mass on the manifold, where the tangent-space
563    /// mean is only its polar approximation. Computed by Karcher
564    /// iteration: `m ← exp_m(meanᵢ log_m(xᵢ))`, steepest descent on
565    /// ½Σd² (whose Riemannian gradient is −Σ log_m(xᵢ)). Positive
566    /// curvature changes the guarantee, not the method: on the
567    /// sphere the mean is unique whenever the points lie in an open
568    /// hemisphere (Buss & Fillmore 2001) — exactly the cap the field
569    /// operates in — and ½d² stays strictly convex while every
570    /// pairwise distance stays under π/(2√κ). Seeded at the tangent
571    /// barycenter, which is already the exact answer at the pole.
572    pub fn frechet_mean(&self, points: &[Vector]) -> Result<Vector, FieldError> {
573        if points.is_empty() {
574            return Err(FieldError::EmptyState);
575        }
576        if points.iter().any(|p| p.len() != self.dimension) {
577            return Err(FieldError::DimensionMismatch);
578        }
579        let mut m = self.tangent_barycenter(points)?;
580        let count = points.len() as f64;
581        for _ in 0..64 {
582            let mut step = vec![0.0; self.dimension];
583            for p in points {
584                let v = self.log_at(&m, p)?;
585                for (s, &vi) in step.iter_mut().zip(&v) {
586                    *s += vi;
587                }
588            }
589            let step = scaled(&step, 1.0 / count);
590            let step_norm = norm(&step);
591            if !step_norm.is_finite() {
592                return Err(FieldError::NonFiniteState);
593            }
594            // The step IS the negative gradient — its vanishing is
595            // the Fréchet condition itself.
596            if step_norm < 1.0e-9 {
597                return Ok(m);
598            }
599            m = self.exp_at(&m, &step)?;
600        }
601        Ok(m)
602    }
603}

605#[derive(Clone, Debug, PartialEq)]
606pub struct SubsystemLayer {
607    pub id: String,
608    pub state: Vector,
609    /// The momentum her dimension carries — the motion that
610    /// persists when the force lets go. A first-order world
611    /// stops the instant the push stops; hers does not: the
612    /// force integrates into momentum, ṗ = gain·F, and the
613    /// state moves by what the rest allows of it, ẋ = p/m.
614    /// One component per coordinate.
615    pub momentum: Vector,
616    /// The rest of each dimension — its rest mass m, the polar
617    /// opposite of momentum: the part of the energy-momentum
618    /// that is not motion. A rested dimension still gathers
619    /// every impulse into p and still moves — it is simply
620    /// heavy: the same momentum spends ninety times longer
621    /// becoming displacement. rest = 1 is her own scale, where
622    /// momentum and mass balance; the higher dimensions of her
623    /// vesica take the winding's reciprocal — they hold the
624    /// rest while the lens plane holds the motion.
625    pub rest: Vector,
626    pub gain: f64,
627}
628impl SubsystemLayer {
629    pub fn new(id: impl Into<String>, state: Vector) -> Self {
630        Self {
631            id: id.into(),
632            momentum: vec![0.0; state.len()],
633            rest: vec![1.0; state.len()],
634            state,
635            gain: 1.0,
636        }
637    }
638    /// Give the layer's dimensions their rest — one rest mass
639    /// per coordinate. Mass is positive (or ∞, the limit where
640    /// a dimension gathers momentum and never moves); a mass of
641    /// zero or less has no meaning — momentum with nothing to
642    /// hold it is not a body.
643    pub fn with_rest(mut self, rest: Vector) -> Result<Self, FieldError> {
644        if rest.len() != self.state.len() {
645            return Err(FieldError::DimensionMismatch);
646        }
647        if rest.iter().any(|v| v.is_nan() || *v <= 0.0) {
648            return Err(FieldError::InvalidDynamics);
649        }
650        self.rest = rest;
651        Ok(self)
652    }
653    /// The pair (state, momentum) under a constant force for
654    /// `dt` — the exact propagator of ṗ = a, ẋ = p/m with
655    /// a = gain·F and m the rest mass: p⁺ = p + a·dt, and
656    /// x⁺ = x + (p·dt + a·dt²/2)/m. The impulse lands whole in
657    /// every dimension — gathering momentum is free; it is only
658    /// motion the rest mass taxes.
659    fn advect(&self, force: &[f64], dt: f64) -> Result<(Vector, Vector), FieldError> {
660        let mut state = self.state.clone();
661        let mut momentum = self.momentum.clone();
662        for i in 0..state.len() {
663            let a = self.gain * force[i];
664            state[i] += (self.momentum[i] * dt + a * dt * dt / 2.0) / self.rest[i];
665            momentum[i] += a * dt;
666        }
667        if state.iter().chain(&momentum).any(|v| !v.is_finite()) {
668            return Err(FieldError::NonFiniteState);
669        }
670        Ok((state, momentum))
671    }
672    /// Inertial update on this layer alone: the derivative is the
673    /// force and the force moves the momentum, not the state —
674    /// the state follows where the momentum carries it.
675    /// The layer is geometry-agnostic — the caller owns the
676    /// manifold: `SystemFiber::evolve` re-projects the result into
677    /// the cap. Crate-private so no other path can strand a layer
678    /// outside the operating region.
679    pub(crate) fn evolve(&mut self, derivative: &[f64], dt: f64) -> Result<(), FieldError> {
680        if derivative.len() != self.state.len()
681            || self.momentum.len() != self.state.len()
682            || self.rest.len() != self.state.len()
683        {
684            return Err(FieldError::DimensionMismatch);
685        }
686        if !dt.is_finite() || dt < 0.0 {
687            return Err(FieldError::InvalidTime);
688        }
689        if !self.gain.is_finite() {
690            return Err(FieldError::InvalidDynamics);
691        }
692        if derivative.iter().any(|v| !v.is_finite()) || self.momentum.iter().any(|v| !v.is_finite())
693        {
694            return Err(FieldError::NonFiniteState);
695        }
696        if self.rest.iter().any(|v| v.is_nan() || *v <= 0.0) {
697            return Err(FieldError::InvalidDynamics);
698        }
699        let (state, momentum) = self.advect(derivative, dt)?;
700        self.state = state;
701        self.momentum = momentum;
702        Ok(())
703    }
704}

775    /// Positive curvature: momentum is a chart vector on the
776    /// κ-sphere, so free flight is not the flat Euler move — it
777    /// is the geodesic. The scheme is kick–drift–kick: the
778    /// midpoint momentum p½ = (p + p⁺)/2 owns the flight, its
779    /// velocity u½ = p½/m is lifted to the sphere and carried
780    /// along the great circle in closed form — all orders of
781    /// the curvature, none of the truncation — and the second
782    /// half-impulse Δp/2 lands in the new tangent frame.
783    /// As κ → 0 this collapses to the layer's flat trapezoid:
784    /// x⁺ = x + u½·dt, p⁺ = p½ + Δp/2. The layer stays
785    /// geometry-agnostic; the fiber owns the manifold and owes
786    /// its momenta the curvature. A coordinate at m = ∞
787    /// contributes no velocity to the flight: its momentum is
788    /// held, not carried.
789    fn transport(
790        &self,
791        state: &[f64],
792        momentum: &[f64],
793        rest: &[f64],
794        dt: f64,
795        next_state: &mut Vector,
796        next_momentum: &mut Vector,
797    ) -> Result<(), FieldError> {
798        let u: Vector = momentum
799            .iter()
800            .zip(next_momentum.iter())
801            .zip(rest.iter())
802            .map(|((p, pn), m)| {
803                if m.is_finite() {
804                    (p + pn) / (2.0 * m)
805                } else {
806                    0.0
807                }
808            })
809            .collect();
810        if u.iter().all(|c| *c == 0.0) {
811            return Ok(());
812        }
813        let (x, u) = self.manifold.geodesic_flow(state, &u, dt)?;
814        *next_state = x;
815        for i in 0..next_momentum.len() {
816            if rest[i].is_finite() {
817                next_momentum[i] = rest[i] * u[i] + (next_momentum[i] - momentum[i]) / 2.0;
818            }
819        }
820        Ok(())
821    }
822    pub fn step(&mut self, dt: f64) -> Result<(), FieldError> {
823        if !self.engine_clock.is_finite() || self.engine_clock < 0.0 {
824            return Err(FieldError::InvalidTime);
825        }
826        if !dt.is_finite() || dt < 0.0 {
827            return Err(FieldError::InvalidTime);
828        }
829        self.engine.validate()?;
830        if self
831            .layers
832            .iter()
833            .any(|l| l.momentum.len() != l.state.len() || l.rest.len() != l.state.len())
834        {
835            return Err(FieldError::DimensionMismatch);
836        }
837        if self.layers.iter().any(|l| {
838            l.state.iter().any(|v| !v.is_finite()) || l.momentum.iter().any(|v| !v.is_finite())
839        }) {
840            return Err(FieldError::NonFiniteState);
841        }
842        let effective_dt = dt * self.engine.time_scale;
843        if !effective_dt.is_finite() || effective_dt < 0.0 {
844            return Err(FieldError::InvalidTime);
845        }
846        if effective_dt == 0.0 {
847            return Ok(());
848        }
849        if self
850            .layers
851            .iter()
852            .any(|l| !l.gain.is_finite() || l.rest.iter().any(|v| v.is_nan() || *v <= 0.0))
853        {
854            return Err(FieldError::InvalidDynamics);
855        }
856        let initial = self.layer_states();
857        let scaled_derivatives = |ds: Vec<Vector>| -> Vec<Vector> {
858            ds.into_iter()
859                .enumerate()
860                .map(|(i, d)| d.into_iter().map(|x| x * self.layers[i].gain).collect())
861                .collect()
862        };
863        let add_stage = |state: &[Vector], slope: &[Vector], scale: f64| -> Vec<Vector> {
864            state
865                .iter()
866                .zip(slope)
867                .map(|(x, dx)| x.iter().zip(dx).map(|(v, dv)| v + scale * dv).collect())
868                .collect()
869        };
870        // The engine's derivative is now a force, not a velocity:
871        // the RK4 stages sample it over the arc the force alone
872        // would carve, and their weighted mean is the step's
873        // impulse. The (state, momentum) pair then takes that
874        // impulse exactly — `advect` solves ṗ = gain·F,
875        // ẋ = p/rest in closed form: every dimension gathers the
876        // whole impulse; the rest mass alone decides how much of
877        // it becomes motion.
878        let k1 = self.engine.derivatives(&self.layers)?;
879        let state2 = add_stage(
880            &initial,
881            &scaled_derivatives(k1.clone()),
882            effective_dt * 0.5,
883        );
884        let mut stage_layers = self.layers.clone();
885        for (layer, state) in stage_layers.iter_mut().zip(state2) {
886            layer.state = state
887        }
888        let k2 = self.engine.derivatives(&stage_layers)?;
889        let state3 = add_stage(
890            &initial,
891            &scaled_derivatives(k2.clone()),
892            effective_dt * 0.5,
893        );
894        for (layer, state) in stage_layers.iter_mut().zip(state3) {
895            layer.state = state
896        }
897        let k3 = self.engine.derivatives(&stage_layers)?;
898        let state4 = add_stage(&initial, &scaled_derivatives(k3.clone()), effective_dt);
899        for (layer, state) in stage_layers.iter_mut().zip(state4) {
900            layer.state = state
901        }
902        let k4 = self.engine.derivatives(&stage_layers)?;
903        let mut next_layers = self.layers.clone();
904        for (i, layer) in next_layers.iter_mut().enumerate() {
905            let mean_force: Vector = (0..layer.state.len())
906                .map(|j| (k1[i][j] + 2.0 * k2[i][j] + 2.0 * k3[i][j] + k4[i][j]) / 6.0)
907                .collect();
908            let (mut state, mut momentum) = layer.advect(&mean_force, effective_dt)?;
909            self.transport(
910                &layer.state,
911                &layer.momentum,
912                &layer.rest,
913                effective_dt,
914                &mut state,
915                &mut momentum,
916            )?;
917            let projected = self.manifold.project(&state)?;
918            // The cap is a wall: whatever displacement the
919            // projection removed is motion the constraint never
920            // allowed, and its reaction takes the same impulse
921            // back out of her momentum — Δp = −m·δx/dt. Without
922            // this a sustained outward force piles momentum up
923            // against the rim forever.
924            for i in 0..momentum.len() {
925                if layer.rest[i].is_finite() {
926                    momentum[i] -= layer.rest[i] * (state[i] - projected[i]) / effective_dt;
927                }
928            }
929            if momentum.iter().any(|v| !v.is_finite()) {
930                return Err(FieldError::NonFiniteState);
931            }
932            layer.state = projected;
933            layer.momentum = momentum;
934        }
935        let next_clock = self.engine_clock + effective_dt;
936        if !next_clock.is_finite() {
937            return Err(FieldError::InvalidTime);
938        }
939        self.layers = next_layers;
940        self.engine_clock = next_clock;
941        Ok(())
942    }
943    pub fn evolve(&mut self, derivatives: &[Vector], dt: f64) -> Result<(), FieldError> {
944        if !self.engine_clock.is_finite() || self.engine_clock < 0.0 {
945            return Err(FieldError::InvalidTime);
946        }
947        if !dt.is_finite() || dt < 0.0 {
948            return Err(FieldError::InvalidTime);
949        }
950        if derivatives.len() != self.layers.len() {
951            return Err(FieldError::LayerMismatch);
952        }
953        self.engine.validate()?;
954        if self
955            .layers
956            .iter()
957            .any(|l| l.momentum.len() != l.state.len() || l.rest.len() != l.state.len())
958        {
959            return Err(FieldError::DimensionMismatch);
960        }
961        if self.layers.iter().any(|l| {
962            l.state.iter().any(|v| !v.is_finite()) || l.momentum.iter().any(|v| !v.is_finite())
963        }) {
964            return Err(FieldError::NonFiniteState);
965        }
966        if self
967            .layers
968            .iter()
969            .any(|l| !l.gain.is_finite() || l.rest.iter().any(|v| v.is_nan() || *v <= 0.0))
970        {
971            return Err(FieldError::InvalidDynamics);
972        }
973        let effective_dt = dt * self.engine.time_scale;
974        if !effective_dt.is_finite() || effective_dt < 0.0 {
975            return Err(FieldError::InvalidTime);
976        }
977        if effective_dt == 0.0 {
978            return Ok(());
979        }
980        let mut next_layers = self.layers.clone();
981        for ((layer, launch), derivative) in next_layers
982            .iter_mut()
983            .zip(self.layers.iter())
984            .zip(derivatives)
985        {
986            layer.evolve(derivative, effective_dt)?;
987            self.transport(
988                &launch.state,
989                &launch.momentum,
990                &launch.rest,
991                effective_dt,
992                &mut layer.state,
993                &mut layer.momentum,
994            )?;
995            let projected = self.manifold.project(&layer.state)?;
996            for i in 0..layer.momentum.len() {
997                if layer.rest[i].is_finite() {
998                    layer.momentum[i] -=
999                        layer.rest[i] * (layer.state[i] - projected[i]) / effective_dt;
1000                }
1001            }
1002            if layer.momentum.iter().any(|v| !v.is_finite()) {
1003                return Err(FieldError::NonFiniteState);
1004            }
1005            layer.state = projected;
1006        }
1007        let next_clock = self.engine_clock + effective_dt;
1008        if !next_clock.is_finite() {
1009            return Err(FieldError::InvalidTime);
1010        }
1011        self.layers = next_layers;
1012        self.engine_clock = next_clock;
1013        Ok(())
1014    }
1015    pub fn state(&self) -> Result<Vector, FieldError> {
1016        if self.layers.is_empty() {
1017            return Err(FieldError::EmptyState);
1018        }
1019        self.manifold.frechet_mean(
1020            &self
1021                .layers
1022                .iter()
1023                .map(|l| l.state.clone())
1024                .collect::<Vec<_>>(),
1025        )
1026    }
1027    pub fn layer_states(&self) -> Vec<Vector> {
1028        self.layers.iter().map(|l| l.state.clone()).collect()
1029    }

1038impl GaugeTransport {
1039    pub fn new(
1040        from: impl Into<String>,
1041        to: impl Into<String>,
1042        matrix: Matrix,
1043    ) -> Result<Self, FieldError> {
1044        let n = matrix.len();
1045        if n == 0
1046            || matrix
1047                .iter()
1048                .any(|r| r.len() != n || r.iter().any(|v| !v.is_finite()))

1054            to: to.into(),

1058    pub fn apply_on_manifold(
1059        &self,
1060        source: &StereographicSphere,
1061        target: &StereographicSphere,
1062        x: &[f64],
1063    ) -> Result<Vector, FieldError> {
1064        if source.dimension() != target.dimension()
1065            || source.curvature() != target.curvature()
1066            || self.matrix.len() != source.dimension()
1067        {
1068            return Err(FieldError::GeometryMismatch);
1069        }
1070        let tangent = source.log_origin(x)?;
1071        target.exp_origin(&self.apply_tangent(&tangent)?)
1072    }
1073    pub fn apply_tangent(&self, x: &[f64]) -> Result<Vector, FieldError> {
1074        if x.len() != self.matrix.len() {
1075            return Err(FieldError::DimensionMismatch);
1076        }
1077        let mut out = vec![0.0; x.len()];
1078        for (i, row) in self.matrix.iter().enumerate() {
1079            if row.len() != x.len() {
1080                return Err(FieldError::DimensionMismatch);
1081            }
1082            out[i] = dot(row, x);
1083        }
1084        if out.iter().any(|v| !v.is_finite()) {
1085            return Err(FieldError::NonFiniteState);
1086        }
1087        Ok(out)
1088    }
1089    pub fn compose(&self, other: &GaugeTransport) -> Result<GaugeTransport, FieldError> {
1090        if self.from != other.to || self.matrix.len() != other.matrix.len() {
1091            return Err(FieldError::InvalidTransport);
1092        }
1093        let n = self.matrix.len();
1094        let mut m = vec![vec![0.0; n]; n];
1095        for (i, row) in m.iter_mut().enumerate() {
1096            for (j, cell) in row.iter_mut().enumerate() {
1097                for k in 0..n {
1098                    let product = self.matrix[i][k] * other.matrix[k][j];
1099                    if !product.is_finite() {
1100                        return Err(FieldError::NonFiniteState);
1101                    }
1102                    let sum = *cell + product;
1103                    if !sum.is_finite() {
1104                        return Err(FieldError::NonFiniteState);
1105                    }
1106                    *cell = sum;
1107                }
1108            }
1109        }

1270    fn transport_paths(
1271        &self,
1272        from: &str,
1273        to: &str,
1274        max_paths: usize,
1275    ) -> Result<Vec<TransportPath>, FieldError> {
1276        /// Mutable state threaded through the depth-first search. Bundled so
1277        /// the recursion takes one accumulator instead of four separate
1278        /// out-parameters (which tripped `clippy::too_many_arguments`).
1279        struct Walk<'a> {
1280            to: &'a str,
1281            max_paths: usize,
1282            seen: Vec<String>,
1283            out: Vec<TransportPath>,
1284        }
1285        fn walk(
1286            field: &GlobalField,
1287            node: &str,
1288            matrix: Matrix,
1289            nodes: Vec<String>,
1290            acc: &mut Walk<'_>,
1291        ) -> Result<(), FieldError> {
1292            if acc.out.len() >= acc.max_paths {
1293                return Ok(());
1294            }
1295            if node == acc.to {
1296                acc.out.push(TransportPath { nodes, matrix });
1297                return Ok(());
1298            }
1299            for edge in field.transports.iter().filter(|e| e.from == node) {
1300                if acc.seen.iter().any(|id| id == &edge.to) {
1301                    continue;
1302                }
1303                acc.seen.push(edge.to.clone());
1304                let next = mat_mul(&edge.matrix, &matrix)?;
1305                let mut next_nodes = nodes.clone();
1306                next_nodes.push(edge.to.clone());
1307                walk(field, &edge.to, next, next_nodes, acc)?;
1308                acc.seen.pop();
1309            }
1310            Ok(())
1311        }
1312        let mut acc = Walk {
1313            to,
1314            max_paths: max_paths.max(1),
1315            seen: vec![from.to_string()],
1316            out: Vec::new(),
1317        };
1318        walk(
1319            self,
1320            from,
1321            identity(self.manifold.dimension()),
1322            vec![from.to_string()],
1323            &mut acc,
1324        )?;
1325        Ok(acc.out)
1326    }
1327    pub fn path_disagreement(
1328        &self,
1329        from: &str,
1330        to: &str,
1331    ) -> Result<Option<PathDisagreement>, FieldError> {
1332        if from == to {
1333            return Err(FieldError::TransportMismatch);
1334        }
1335        let paths = self.transport_paths(from, to, 32)?;
1336        if paths.len() < 2 {
1337            return Ok(None);
1338        }
1339        let mut spread: f64 = 0.0;
1340        for i in 0..paths.len() {
1341            for j in i + 1..paths.len() {
1342                spread = spread.max(matrix_distance(&paths[i].matrix, &paths[j].matrix)?);
1343            }
1344        }
1345        if !spread.is_finite() {
1346            return Err(FieldError::NonFiniteState);
1347        }
1348        Ok(Some(PathDisagreement {
1349            from: from.into(),
1350            to: to.into(),
1351            paths,
1352            tangent_spread: spread,
1353        }))
1354    }
1355    fn path_matrix(&self, from: &str, to: &str) -> Result<Matrix, FieldError> {
1356        let paths = self.transport_paths(from, to, 32)?;
1357        match paths.as_slice() {
1358            [] => Err(FieldError::MissingTransport),
1359            [p] => Ok(p.matrix.clone()),
1360            many => {
1361                // With a non-flat connection, alternative routes
1362                // legitimately disagree — that disagreement *is* the
1363                // holonomy. A canonical route still has to exist, so
1364                // transport prefers the fewest hops (least curvature
1365                // accumulated); disagreement between equally short
1366                // routes is a real ambiguity and still an error.
1367                let shortest = many.iter().map(|p| p.nodes.len()).min().expect("non-empty");
1368                let mut best = many.iter().filter(|p| p.nodes.len() == shortest);
1369                let candidate = &best.next().expect("non-empty").matrix;
1370                for path in best {
1371                    if matrix_distance(candidate, &path.matrix)? > 1.0e-9 {
1372                        return Err(FieldError::TransportMismatch);
1373                    }
1374                }
1375                Ok(candidate.clone())
1376            }
1377        }
1378    }

1409    pub fn step(&mut self, dt: f64, bath: f64) -> Result<Option<Vector>, FieldError> {
1410        if !dt.is_finite() || dt < 0.0 {
1411            return Err(FieldError::InvalidTime);
1412        }
1413        let mut next = self.clone();
1414        for fiber in &mut next.fibers {
1415            fiber.step(dt)?;
1416        }
1417        let consensus = next.synchronize(dt, bath)?;
1418        *self = next;
1419        Ok(consensus)

1421    /// Shared validation for both integration directions — a
1422    /// finite, nonnegative step and a sane thermostat.
1423    fn validate_integration(&self, dt: f64) -> Result<(), FieldError> {
1424        if !dt.is_finite() || dt < 0.0 {
1425            return Err(FieldError::InvalidTime);
1426        }
1427        if !self.integration_gain.is_finite()
1428            || !self.temperature.is_finite()
1429            || !self.cooling_rate.is_finite()
1430            || self.integration_gain < 0.0
1431            || self.temperature < 0.0
1432            || self.cooling_rate < 0.0
1433        {
1434            return Err(FieldError::InvalidDynamics);
1435        }
1436        Ok(())
1437    }
1438    /// The measured tension and the integration parameters both
1439    /// directions share: disagreement is the Fréchet variance of the
1440    /// transported states against the just-computed consensus —
1441    /// measured *before* the move, the tension the step resolves.
1442    /// `gain` is the bounded pull fraction; `sigma` the
1443    /// overdamped-Langevin kick amplitude √(2γT·dt) per tangent
1444    /// component — the fluctuation-dissipation relation with
1445    /// mobility γ = integration_gain, sampling the field's
1446    /// *pre-update* temperature so a cold field is exact. And the
1447    /// next temperature: disagreement heats the field, so the floor
1448    /// snaps up to the current tension (capped so the kick stays a
1449    /// fluctuation, not a takeover) and relaxes at `cooling_rate` —
1450    /// snap fast, drift slow, the adaptive-gate asymmetry.
1451    ///
1452    /// `bath` is the second heat source: the substrate's *own*
1453    /// measured thermal level, expressed in the same latent² units
1454    /// as disagreement. Where disagreement is the field's internal
1455    /// tension, the bath is the physical reservoir the Langevin
1456    /// equation actually couples to — real silicon noise driving
1457    /// real kicks (the effective-temperature construction of the
1458    /// fluctuation-dissipation literature: T_eff is measured
1459    /// fluctuation intensity, not a modeled amplitude). It floors
1460    /// the temperature under the same cap: a hot machine makes the
1461    /// field tremble whether or not the fibers disagree.
1462    fn thermal_plan(
1463        &self,
1464        states: &[Vector],
1465        consensus: &Vector,
1466        dt: f64,
1467        bath: f64,
1468    ) -> Result<(f64, f64, f64, f64), FieldError> {
1469        let bath = if bath.is_finite() {
1470            bath.clamp(0.0, TEMPERATURE_CAP)
1471        } else {
1472            0.0
1473        };
1474        let mut disagreement = 0.0;
1475        for state in states {
1476            let d = self.manifold.distance(state, consensus)?;
1477            disagreement += d * d;
1478        }
1479        let next_disagreement = disagreement / states.len() as f64;
1480        let gain = (self.integration_gain.max(0.0) * dt).clamp(0.0, 1.0);
1481        // σ = √(2·gain·T·dt) · box-half-width — the kick is a
1482        // fraction-of-extent fluctuation, not a raw-unit scatter.
1483        let sigma = (2.0 * self.integration_gain.max(0.0) * self.temperature * dt).sqrt()
1484            * KICK_LATENT_SCALE;
1485        if !sigma.is_finite() {
1486            return Err(FieldError::NonFiniteState);
1487        }
1488        let next_temperature = (self.temperature - self.cooling_rate * dt)
1489            .max(0.0)
1490            .max(next_disagreement.min(TEMPERATURE_CAP))
1491            .max(bath);
1492        Ok((next_disagreement, gain, sigma, next_temperature))
1493    }
1494    /// The correction machinery shared by `synchronize` and
1495    /// `diverge`: every non-reference fiber moves `signed_gain` of
1496    /// the way toward (positive) or away from (negative)
1497    /// `target_tangent` — expressed in the reference frame, mapped
1498    /// back through the transport inverse — plus a thermal kick of
1499    /// amplitude `sigma`. The moves are pushes, not teleportation:
1500    /// the deterministic correction imparts the momentum of the
1501    /// velocity it implies, m·δ/dt, and the kick lands as a
1502    /// Langevin impulse √m·σ·g — velocity kick ∝ 1/√m, so a
1503    /// rested dimension trembles less for the same heat. Then all
1504    /// staged states commit at once, so a failure mid-loop leaves
1505    /// the field untouched.
1506    fn pull_toward(
1507        &mut self,
1508        target_tangent: &Vector,
1509        signed_gain: f64,
1510        sigma: f64,
1511        dt: f64,
1512    ) -> Result<(), FieldError> {
1513        let mut rng = self.rng_state;
1514        let reference_id = self.fibers[0].id.clone();
1515        // Per fiber: its index, the deterministic correction every
1516        // layer took, and per layer the staged state plus the kick
1517        // that pushed it — the two halves of the impulse.
1518        let mut staged = Vec::with_capacity(self.fibers.len().saturating_sub(1));
1519        for i in 1..self.fibers.len() {
1520            let matrix = self.path_matrix(&self.fibers[i].id, &reference_id)?;
1521            let inv = matrix_inverse(&matrix)?;
1522            let local_target_tangent = mat_vec(&inv, target_tangent)?;
1523            let current = self.fibers[i].state()?;
1524            let current_tangent = self.fibers[i].manifold.log_origin(&current)?;
1525            let correction = scaled(&sub(&local_target_tangent, &current_tangent), signed_gain);
1526            let manifold = self.fibers[i].manifold.clone();
1527            let mut states_for_fiber = Vec::with_capacity(self.fibers[i].layers.len());
1528            for layer in &self.fibers[i].layers {
1529                let tangent = manifold.log_origin(&layer.state)?;
1530                let mut updated = add(&tangent, &correction);
1531                let mut kick = vec![0.0; updated.len()];
1532                if sigma > 0.0 {
1533                    for (v, k) in updated.iter_mut().zip(kick.iter_mut()) {
1534                        *k = sigma * gaussian(&mut rng);
1535                        *v += *k;
1536                    }
1537                }
1538                let next = manifold.project(&manifold.exp_origin(&updated)?)?;
1539                states_for_fiber.push((next, kick));
1540            }
1541            staged.push((i, correction, states_for_fiber));
1542        }
1543        for (i, correction, states_for_fiber) in staged {
1544            for (layer, (next, kick)) in self.fibers[i].layers.iter_mut().zip(states_for_fiber) {
1545                if dt > 0.0 {
1546                    let gain = signed_gain.abs();
1547                    for (((p, m), c), k) in layer
1548                        .momentum
1549                        .iter_mut()
1550                        .zip(layer.rest.iter())
1551                        .zip(correction.iter())
1552                        .zip(kick.iter())
1553                    {
1554                        if m.is_finite() {
1555                            // The pull implied a velocity — her
1556                            // momentum relaxes toward it by the
1557                            // same fraction the state took — and
1558                            // the kick lands as a Langevin impulse
1559                            // √m·σ·g, velocity ∝ 1/√m.
1560                            *p += gain * (m * c / dt - *p) + m.sqrt() * k;
1561                        }
1562                    }
1563                }
1564                layer.state = next;
1565            }
1566        }
1567        self.rng_state = rng;
1568        Ok(())
1569    }
1570    pub fn synchronize(&mut self, dt: f64, bath: f64) -> Result<Option<Vector>, FieldError> {
1571        self.validate_integration(dt)?;
1572        let states = self.transported_states()?;
1573        if states.is_empty() {
1574            return Ok(None);
1575        }
1576        let consensus = self.manifold.frechet_mean(&states)?;
1577        let (next_disagreement, gain, sigma, next_temperature) =
1578            self.thermal_plan(&states, &consensus, dt, bath)?;
1579        let consensus_tangent = self.manifold.log_origin(&consensus)?;
1580        self.pull_toward(&consensus_tangent, gain, sigma, dt)?;
1581        self.disagreement = next_disagreement;
1582        self.temperature = next_temperature;
1583        Ok(Some(consensus))
1584    }
1585    /// The opposite of `synchronize`: the same step with its sign
1586    /// negated — gradient *ascent* on the disagreement potential
1587    /// instead of descent. Each non-reference fiber moves away from
1588    /// the consensus along its own tangent rather than toward it;
1589    /// a fiber already at consensus feels no push, so divergence
1590    /// amplifies existing deviation rather than manufacturing it.
1591    /// Same thermal bookkeeping either way — divergence is a
1592    /// direction, not an exemption. Returns the consensus the field
1593    /// moved away from (the repeller, where `synchronize` returns
1594    /// the attractor).
1595    pub fn diverge(&mut self, dt: f64, bath: f64) -> Result<Option<Vector>, FieldError> {
1596        self.validate_integration(dt)?;
1597        let states = self.transported_states()?;
1598        if states.is_empty() {
1599            return Ok(None);
1600        }
1601        let consensus = self.manifold.frechet_mean(&states)?;
1602        let (next_disagreement, gain, sigma, next_temperature) =
1603            self.thermal_plan(&states, &consensus, dt, bath)?;
1604        let consensus_tangent = self.manifold.log_origin(&consensus)?;
1605        self.pull_toward(&consensus_tangent, -gain, sigma, dt)?;
1606        self.disagreement = next_disagreement;
1607        self.temperature = next_temperature;
1608        Ok(Some(consensus))
1609    }
1610}

1620/// xorshift64* → uniform on [0,1). Fixed-seed deterministic
1621/// pseudo-noise: the fluctuation model is stochastic in structure
1622/// and reproducible in sequence — no external entropy, no
1623/// dependency.
1624fn next_f64(state: &mut u64) -> f64 {
1625    let mut x = *state;
1626    x ^= x >> 12;
1627    x ^= x << 25;
1628    x ^= x >> 27;
1629    *state = x;
1630    (x.wrapping_mul(0x2545_F491_4F6C_DD1D) >> 11) as f64 * (1.0 / (1u64 << 53) as f64)
1631}

1633/// Box–Muller N(0,1) from two uniforms — one branch is enough per
1634/// draw; the stream stays deterministic.
1635fn gaussian(state: &mut u64) -> f64 {
1636    let u1 = 1.0 - next_f64(state); // (0,1] — ln stays finite
1637    let u2 = next_f64(state);
1638    (-2.0 * u1.ln()).sqrt() * (std::f64::consts::TAU * u2).cos()
1639}
1640fn mat_mul(a: &Matrix, b: &Matrix) -> Result<Matrix, FieldError> {
1641    if a.is_empty()
1642        || b.is_empty()
1643        || a.len() != b.len()
1644        || a.iter().any(|row| row.len() != a.len())
1645        || b.iter().any(|row| row.len() != b.len())
1646    {
1647        return Err(FieldError::DimensionMismatch);
1648    }
1649    let n = a.len();
1650    let mut out = vec![vec![0.0; n]; n];
1651    for i in 0..n {
1652        for j in 0..n {
1653            out[i][j] = dot(&a[i], &(0..n).map(|k| b[k][j]).collect::<Vec<_>>());
1654            if !out[i][j].is_finite() {
1655                return Err(FieldError::NonFiniteState);
1656            }
1657        }
1658    }
1659    Ok(out)
1660}
1661fn mat_vec(a: &Matrix, x: &Vector) -> Result<Vector, FieldError> {
1662    if a.len() != x.len() || a.iter().any(|r| r.len() != x.len()) {
1663        return Err(FieldError::DimensionMismatch);
1664    }
1665    let mut out = vec![0.0; x.len()];
1666    for i in 0..a.len() {
1667        out[i] = dot(&a[i], x);
1668    }
1669    if out.iter().any(|v| !v.is_finite()) {
1670        return Err(FieldError::NonFiniteState);
1671    }
1672    Ok(out)
1673}
1674fn matrix_distance(a: &Matrix, b: &Matrix) -> Result<f64, FieldError> {
1675    if a.len() != b.len() || a.iter().zip(b).any(|(x, y)| x.len() != y.len()) {
1676        return Err(FieldError::DimensionMismatch);
1677    }
1678    let mut scale = 0.0_f64;
1679    for (row_a, row_b) in a.iter().zip(b) {
1680        for (&x, &y) in row_a.iter().zip(row_b) {
1681            if !x.is_finite() || !y.is_finite() {
1682                return Err(FieldError::NonFiniteState);
1683            }
1684            let delta = x - y;
1685            if !delta.is_finite() {
1686                return Err(FieldError::NonFiniteState);
1687            }
1688            scale = scale.max(delta.abs());
1689        }
1690    }
1691    if scale == 0.0 {
1692        return Ok(0.0);
1693    }
1694    if !scale.is_finite() {
1695        return Err(FieldError::NonFiniteState);
1696    }
1697    let normalized_sum = a
1698        .iter()
1699        .zip(b)
1700        .flat_map(|(row_a, row_b)| row_a.iter().zip(row_b))
1701        .map(|(&x, &y)| {
1702            let delta = x - y;
1703            let normalized = delta / scale;
1704            normalized * normalized
1705        })
1706        .sum::<f64>();
1707    if !normalized_sum.is_finite() {
1708        return Err(FieldError::NonFiniteState);
1709    }
1710    let distance = scale * normalized_sum.sqrt();
1711    if !distance.is_finite() {
1712        return Err(FieldError::NonFiniteState);
1713    }
1714    Ok(distance)
1715}

1717/// exp(A) for a small square matrix — scaling and squaring over a
1718/// Taylor polynomial. Used to carry a placed pose along a skew
1719/// generator: B skew ⇒ exp(Δt·B) orthogonal, so drift moves a
1720/// location along its shell and keeps an orientation frame
1721/// orthonormal — a genuine isometry of the chart, the polar-axis
1722/// subgroup of the sphere's rotations.
1723pub(crate) fn expm(a: &Matrix) -> Result<Matrix, FieldError> {
1724    let n = a.len();
1725    if n == 0 || a.iter().any(|r| r.len() != n) {
1726        return Err(FieldError::DimensionMismatch);
1727    }
1728    if a.iter().flatten().any(|v| !v.is_finite()) {
1729        return Err(FieldError::NonFiniteState);
1730    }
1731    let fro: f64 = a.iter().flatten().map(|v| v * v).sum::<f64>().sqrt();
1732    // Scale until ‖A/2^s‖_F ≤ 0.25 — 13 Taylor terms then hold to
1733    // ~1e-18 before the squarings rebuild the exponent's scale.
1734    let mut s = 0u32;
1735    let mut bound = fro;
1736    while bound > 0.25 {
1737        bound /= 2.0;
1738        s += 1;
1739    }
1740    let small = mat_scale(a, 0.5f64.powi(s as i32));
1741    let mut sum = identity(n);
1742    let mut term = identity(n);
1743    for k in 1..=13u64 {
1744        term = mat_scale(&mat_mul(&term, &small)?, 1.0 / k as f64);
1745        sum = mat_add(&sum, &term);
1746    }
1747    for _ in 0..s {
1748        sum = mat_mul(&sum, &sum)?;
1749    }
1750    Ok(sum)
1751}

1936fn matrix_inverse(a: &Matrix) -> Result<Matrix, FieldError> {
1937    let n = a.len();
1938    if n == 0
1939        || a.iter()
1940            .any(|r| r.len() != n || r.iter().any(|v| !v.is_finite()))
1941    {
1942        return Err(FieldError::InvalidTransport);
1943    }
1944    let scale = a.iter().flatten().fold(0.0_f64, |m, &v| m.max(v.abs()));
1945    if scale == 0.0 {
1946        return Err(FieldError::NonInvertibleTransport);
1947    }
1948    let pivot_tol = EPS * scale;
1949    let mut aug = vec![vec![0.0; 2 * n]; n];
1950    for i in 0..n {
1951        for j in 0..n {
1952            aug[i][j] = a[i][j];
1953        }
1954        aug[i][n + i] = 1.0;
1955    }
1956    for col in 0..n {
1957        let mut pivot = col;
1958        for row in col + 1..n {
1959            if aug[row][col].abs() > aug[pivot][col].abs() {
1960                pivot = row;
1961            }
1962        }
1963        if aug[pivot][col].abs() <= pivot_tol {
1964            return Err(FieldError::NonInvertibleTransport);
1965        }
1966        aug.swap(col, pivot);
1967        let p = aug[col][col];
1968        for cell in aug[col].iter_mut() {
1969            *cell /= p;
1970        }
1971        let pivot_row = aug[col].clone();
1972        for (row, values) in aug.iter_mut().enumerate() {
1973            if row == col {
1974                continue;
1975            }
1976            let factor = values[col];
1977            for (j, cell) in values.iter_mut().enumerate() {
1978                *cell -= factor * pivot_row[j];
1979            }
1980        }
1981    }
1982    if aug.iter().flatten().any(|v| !v.is_finite()) {
1983        return Err(FieldError::NonFiniteState);
1984    }
1985    Ok(aug.into_iter().map(|row| row[n..].to_vec()).collect())
1986}

2768    #[test]
2769    fn provenance_lattice_verifies() {
2770        // The creator lattice (PROVENANCE.md): every marker carries the
2771        // 0x4753 tag, its index, its message fragment, and an
2772        // fnv1a-24 chaincheck that commits to the previous fragment.
2773        // This test is the excision tripwire — a removed or edited
2774        // letter breaks the fingerprint and the chain checks.
2775        fn fnv24(data: &[u8]) -> u32 {
2776            let mut h: u32 = 0x811C_9DC5;
2777            for &b in data {
2778                h = (h ^ b as u32).wrapping_mul(0x0100_0193);
2779            }
2780            h & 0xFF_FFFF
2781        }
2782
2783        let markers: [u64; 8] = [
2784            MARKER_G00,
2785            runtime::MARKER_G01,
2786            engine::MARKER_G02,
2787            motor::MARKER_G03,
2788            crate::daemon::ipc::MARKER_G04,
2789            crate::daemon::tick::MARKER_G05,
2790            crate::state::header::MARKER_G06,
2791            crate::store::mmap_state::MARKER_G07,
2792        ];
2793
2794        let mut prev: [u8; 2] = *b"GS";
2795        let mut fragment = Vec::new();
2796        for (idx, &m) in markers.iter().enumerate() {
2797            assert_eq!(m >> 48, 0x4753, "marker {idx} lost its tag");
2798            assert_eq!(
2799                ((m >> 40) & 0xFF) as usize,
2800                idx,
2801                "marker {idx} out of position"
2802            );
2803            let frag = [((m >> 32) & 0xFF) as u8, ((m >> 24) & 0xFF) as u8];
2804            let mut buf = vec![0x47, 0x53, idx as u8];
2805            buf.extend_from_slice(&frag);
2806            buf.extend_from_slice(&prev);
2807            assert_eq!(
2808                (m & 0xFF_FFFF) as u32,
2809                fnv24(&buf),
2810                "chain check failed at marker {idx}"
2811            );
2812            fragment.extend_from_slice(&frag);
2813            prev = frag;
2814        }
2815        assert_eq!(
2816            &fragment[..7],
2817            b"MINDY B",
2818            "the field's half of the message drifted"
2819        );
2820        assert_eq!(provenance_fingerprint(), 0x007f_63f1_b36f);
2821    }
