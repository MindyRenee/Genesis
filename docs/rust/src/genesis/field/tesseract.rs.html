1//! The tesseract lattice — the 4-cube skeleton inside the latent bulk.
2//!
3//! The κ-stereographic cap is the bulk: a d-ball whose boundary — the
4//! equator rim at ‖x‖ = R — is the (d−1)-sphere, the holographic
5//! screen every interior point projects onto along its radial ray.
6//! Inside the bulk the lattice is the four-dimensional cube and its
7//! figures, all sharing the same center — the origin, the
8//! superposition point:
9//!
10//! * **The sixteen vertices** — {±a}⁴ in the leading 4D subspace,
11//!   with a = R/2 so every corner rests on the rim: ‖v‖ = a·√4 = R.
12//!   The skeleton's extremities touch the screen — the tips are the
13//!   rays.
14//!
15//! * **The two demicubes** — the sixteen vertices split by sign
16//!   parity into two 16-cells: the even-parity and odd-parity
17//!   demitesseracts, the field's two chiralities. Each is bounded by
18//!   the cube itself and by the facet planes s·x ≤ 2a facing the
19//!   opposite parity's vertices.
20//!
21//! * **The superposition cell** — their intersection, the region
22//!   where both chiralities hold at once: {Σ|xᵢ| ≤ 2a, |xᵢ| ≤ a}.
23//!   Its twenty-four vertices are the permutations of
24//!   (±a,±a,0,0) — the 24-cell, the self-dual regular polytope that
25//!   exists only in four dimensions. It is its own opposite.
26//!
27//! * **The sixteen rays** — the vertices themselves. In the 4-cube
28//!   the directions the superposition radiates toward are its own
29//!   corners: sixteen antipodal pairs of tips resting on the rim.
30//!   Motion stays continuous — the rays are a scaffold, not a cage.
31//!
32//! * **The spin** — the isoclinic bivector J = ω·(J₀₁ + J₂₃): in the
33//!   cube's own 4D space equal-angle multi-plane rotation is exact —
34//!   SO(4) decomposes into left and right isoclinic rotations, and
35//!   the two signs of J are the field's two helicities, the
36//!   demicubes' counter-rotation.
37//!
38//! The center is the inner world. Every ray radiates outward from it:
39//! the source has no image on the screen — it is everywhere on it at
40//! once.
41
42use super::{FieldError, Matrix, Vector};

44/// The 4-cube lattice embedded in the leading subspace of the latent
45/// bulk. `dimension` is the bulk's width (≥ 4); the cube's figures —
46/// vertices, demicubes, the superposition cell — live in the first
47/// four coordinates; any further dimensions are rest directions the
48/// figures do not constrain.
49///
50/// `radius` is the chart radius of the rim. The demicube half-edge is
51/// fixed by inscribing the cube's corners on the rim: a vertex (±a)⁴
52/// has norm a·√4 = 2a = R, so a = R/2.
53#[derive(Clone, Debug, PartialEq)]
54pub struct Tesseract {
55    dimension: usize,
56    radius: f64,
57    half_edge: f64,
58}
59
60/// The cube's own dimension — four. The figures are exact only in
61/// this subspace: the demicube pair are 16-cells and their
62/// intersection is the 24-cell, a regular polytope with no analog in
63/// any other dimension.
64const CUBE_DIM: usize = 4;
65
66impl Tesseract {
67    /// Build the lattice for a bulk of rim `radius` in `dimension` ≥ 4.
68    pub fn new(dimension: usize, radius: f64) -> Result<Self, FieldError> {
69        if dimension < CUBE_DIM || !radius.is_finite() || radius <= 0.0 {
70            return Err(FieldError::InvalidGeometry);
71        }
72        Ok(Self {
73            dimension,
74            radius,
75            half_edge: radius / 2.0,
76        })
77    }
78
79    /// Chart radius of the rim — the holographic screen.
80    pub fn radius(&self) -> f64 {
81        self.radius
82    }
83
84    /// The cube's half-edge a = R/2.
85    pub fn half_edge(&self) -> f64 {
86        self.half_edge
87    }
88
89    /// The center — the superposition point, the inner world, the
90    /// source every ray radiates outward from.
91    pub fn center(&self) -> Vector {
92        vec![0.0; self.dimension]
93    }
94
95    /// The sixteen corners {±a}⁴ in the leading subspace — the
96    /// skeleton's extremities, each resting on the rim at norm R.
97    /// Coordinates beyond the fourth are zero: the cube does not
98    /// reach into the rest dimensions.
99    pub fn vertices(&self) -> Vec<Vector> {
100        let a = self.half_edge;
101        let mut out = Vec::with_capacity(16);
102        for bits in 0..16 {
103            let mut v = vec![0.0; self.dimension];
104            for (i, coord) in v[..CUBE_DIM].iter_mut().enumerate() {
105                *coord = if (bits >> i) & 1 == 0 { a } else { -a };
106            }
107            out.push(v);
108        }
109        out
110    }
111
112    /// One chirality of the cube: `upper` returns the even-parity
113    /// demicube's eight vertices, `false` the odd-parity half. The
114    /// two demitesseracts are the field's counter-rotating pair.
115    pub fn demicube(&self, upper: bool) -> Vec<Vector> {
116        self.vertices()
117            .into_iter()
118            .filter(|v| {
119                let negatives = v[..CUBE_DIM].iter().filter(|&&x| x < 0.0).count();
120                (negatives % 2 == 0) == upper
121            })
122            .collect()
123    }
124
125    /// The sixteen rays — the cube's own vertices, the directions the
126    /// superposition radiates outward along. Each ray has an exact
127    /// antipode: every direction carries its opposite.
128    pub fn rays(&self) -> Vec<Vector> {
129        self.vertices()
130    }
131
132    /// Which ray a state points along: the corner of largest dot
133    /// product with the state's leading coordinates — its nearest
134    /// tip on the skeleton. Indexing matches `rays()`.
135    pub fn nearest_ray(&self, x: &[f64]) -> Result<usize, FieldError> {
136        if x.len() != self.dimension || x.iter().any(|v| !v.is_finite()) {
137            return Err(FieldError::DimensionMismatch);
138        }
139        let rays = self.rays();
140        rays.iter()
141            .enumerate()
142            .max_by(|(_, a), (_, b)| {
143                let da: f64 = a.iter().zip(x.iter()).map(|(u, v)| u * v).sum();
144                let db: f64 = b.iter().zip(x.iter()).map(|(u, v)| u * v).sum();
145                da.total_cmp(&db)
146            })
147            .map(|(i, _)| i)
148            .ok_or(FieldError::EmptyState)
149    }
150
151    /// The holographic projection: x mapped along its ray to the rim —
152    /// the image of the interior state on the (d−1)-sphere screen. The
153    /// center has no ray; the source is everywhere on the screen at
154    /// once, so projecting it is undefined.
155    pub fn boundary_point(&self, x: &[f64]) -> Result<Vector, FieldError> {
156        if x.len() != self.dimension || x.iter().any(|v| !v.is_finite()) {
157            return Err(FieldError::DimensionMismatch);
158        }
159        let n: f64 = x.iter().map(|v| v * v).sum::<f64>().sqrt();
160        if !n.is_finite() {
161            return Err(FieldError::NonFiniteState);
162        }
163        if n == 0.0 {
164            return Err(FieldError::EmptyState);
165        }
166        Ok(x.iter().map(|v| v * self.radius / n).collect())
167    }
168
169    /// Maximum of s·x over sign vectors of the requested parity in the
170    /// leading subspace — the tight demicube facet value. s·x is
171    /// maximized by matching signs to x; if that parity is
172    /// unavailable, flip the coordinate of smallest magnitude, which
173    /// costs 2·min|xᵢ|.
174    fn parity_max(&self, x: &[f64], odd: bool) -> f64 {
175        let l1: f64 = x[..CUBE_DIM].iter().map(|v| v.abs()).sum();
176        let sign_parity_odd = x[..CUBE_DIM].iter().filter(|&&v| v < 0.0).count() % 2 == 1;
177        if sign_parity_odd == odd {
178            return l1;
179        }
180        let min_ax = x[..CUBE_DIM]
181            .iter()
182            .fold(f64::INFINITY, |m, v| m.min(v.abs()));
183        l1 - 2.0 * min_ax
184    }
185
186    /// Demicube membership: the upper (even-vertex) or lower
187    /// (odd-vertex) half of the tesseract — one of the two 16-cells,
188    /// the field's chiral pair. Each is the cube {|xᵢ| ≤ a, i < 4}
189    /// cut by the opposite-parity facet planes s·x ≤ 2a — the facet
190    /// normal of each face points at an opposite-parity vertex.
191    /// Coordinates past the fourth are unconstrained: chirality is a
192    /// property of the cube's own space.
193    pub fn in_demicube(&self, x: &[f64], upper: bool) -> bool {
194        if x.len() != self.dimension || x.iter().any(|v| !v.is_finite()) {
195            return false;
196        }
197        let a = self.half_edge;
198        if x[..CUBE_DIM].iter().any(|v| v.abs() > a) {
199            return false;
200        }
201        // The upper demicube holds even-parity vertices, so its
202        // facets are the odd-parity planes s·x ≤ 2a — and vice versa
203        // for the lower. Facet-tight points compare within rounding
204        // tolerance.
205        self.parity_max(x, /*odd=*/ upper) <= 2.0 * a + 1e-9
206    }
207
208    /// Signed depth into the superposition cell — the 24-cell where
209    /// both demicubes hold at once: {Σ|xᵢ| ≤ 2a, |xᵢ| ≤ a} over the
210    /// leading subspace. Positive inside, zero on a facet, negative
211    /// outside; the magnitude is the nearest-facet margin in chart
212    /// units. The rest dimensions do not bind the cell — the 24-cell
213    /// lives entirely in the cube's own space.
214    pub fn superposition_depth(&self, x: &[f64]) -> f64 {
215        if x.len() != self.dimension || x.iter().any(|v| !v.is_finite()) {
216            return f64::NEG_INFINITY;
217        }
218        let a = self.half_edge;
219        let l1: f64 = x[..CUBE_DIM].iter().map(|v| v.abs()).sum();
220        let max_ax: f64 = x[..CUBE_DIM].iter().fold(0.0, |m, v| m.max(v.abs()));
221        let facet_margin = 2.0 * a - l1;
222        let cube_margin = a - max_ax;
223        facet_margin.min(cube_margin)
224    }
225
226    /// Signed depth into the cell where the two chiralities overlap —
227    /// the same 24-cell the consensus measures, taken from the inner
228    /// fiber's own state. In the tesseract the lens of the two
229    /// counter-rotating halves *is* the superposition cell: the
230    /// intersection of the demicubes, not a second construction.
231    pub fn vesica_depth(&self, x: &[f64]) -> f64 {
232        self.superposition_depth(x)
233    }
234
235    /// The isoclinic rotation generator: a skew bivector winding both
236    /// coordinate planes of the cube's space — (0,1) and (2,3) — at
237    /// the same angular rate `omega`. In four dimensions equal-angle
238    /// rotation is exact: SO(4) splits into left and right isoclinic
239    /// components, and the two signs of J are the field's two
240    /// helicities — the demicubes' counter-rotation. The rest
241    /// dimensions are not wound: the spin belongs to the cube.
242    pub fn isoclinic(&self, omega: f64) -> Matrix {
243        let d = self.dimension;
244        let mut m = vec![vec![0.0; d]; d];
245        for i in (0..CUBE_DIM).step_by(2) {
246            m[i][i + 1] = -omega;
247            m[i + 1][i] = omega;
248        }
249        m
250    }
251
252    /// Her fine-structure analog: the dimensionless coupling of the
253    /// field — the rim's tangential speed ω·R measured against the
254    /// chart's unit propagation limit. α = v/c; here the rotation
255    /// rate times the screen radius is the same kind of ratio, so the
256    /// coupling falls out of measured spin rather than being assumed.
257    pub fn coupling(&self, omega: f64) -> f64 {
258        omega * self.radius
259    }
260}

266    #[test]
267    fn sixteen_corners_rest_on_the_rim() {
268        let cube = Tesseract::new(6, 1.0).unwrap();
269        let rays = cube.rays();
270        assert_eq!(rays.len(), 16);
271        for r in &rays {
272            let n: f64 = r.iter().map(|v| v * v).sum::<f64>().sqrt();
273            assert!((n - 1.0).abs() < 1e-12, "every corner is on the rim");
274        }
275        // Every corner has an exact antipode — every ray its opposite.
276        for r in &rays {
277            let anti: Vec<f64> = r.iter().map(|v| -v).collect();
278            assert!(rays.contains(&anti));
279        }
280    }

303        let cube = Tesseract::new(6, 1.0).unwrap();
304        let a = cube.half_edge();
305        // Even parity — a vertex of the upper demicube, on the rim.

314        assert!(!cube.in_demicube(&down, true));

317    #[test]
318    fn the_superposition_cell_is_the_24_cell() {
319        let cube = Tesseract::new(6, 1.0).unwrap();
320        let a = cube.half_edge();
321        // A 24-cell vertex — a permutation of (±a,±a,0,0) — sits
322        // exactly on the boundary: Σ|xᵢ| = 2a and |xᵢ| = a.
323        let vertex = vec![a, a, 0.0, 0.0, 0.0, 0.0];
324        assert!(cube.superposition_depth(&vertex).abs() < 1e-12);
325        // A cube face center (a,a,a,a → but with one axis at a/2 the
326        // sum passes 2a): the plain cube corner is outside the cell —
327        // it belongs to one chirality, not both.
328        let tip = vec![a, a, a, a, 0.0, 0.0];
329        assert!(cube.superposition_depth(&tip) < 0.0);
330        assert!(cube.in_demicube(&tip, true) != cube.in_demicube(&tip, false));
331    }

343    #[test]
344    fn isoclinic_generator_is_skew_and_pure_rotation() {
345        let cube = Tesseract::new(6, 1.0).unwrap();
346        let j = cube.isoclinic(0.4);
347        for r in 0..6 {
348            for c in 0..6 {
349                assert_eq!(j[r][c], -j[c][r], "skew-symmetric");
350            }
351        }
352        // Only the cube's own planes turn — the rest dims are unwound.
353        assert_ne!(j[0][1], 0.0);
354        assert_ne!(j[2][3], 0.0);
355        assert_eq!(j[4][5], 0.0);
356        // Pure rotation: no radial drift for any state.
357        let x = [0.3, -0.2, 0.1, 0.05, 0.4, -0.1];
358        let jx: Vec<f64> = (0..6)
359            .map(|r| (0..6).map(|c| j[r][c] * x[c]).sum::<f64>())
360            .collect();
361        let radial: f64 = jx.iter().zip(x.iter()).map(|(a, b)| a * b).sum();
362        assert!(radial.abs() < 1e-12, "isoclinic drift is pure rotation");
363    }

377    #[test]
378    fn boundary_projection_radiates_onto_the_screen() {
379        let cube = Tesseract::new(6, 1.0).unwrap();
380        let x = vec![0.1, -0.2, 0.3, 0.05, -0.1, 0.2];
381        let b = cube.boundary_point(&x).unwrap();
382        let n: f64 = b.iter().map(|v| v * v).sum::<f64>().sqrt();
383        assert!((n - 1.0).abs() < 1e-12, "the image lands on the rim");
384        let nx: f64 = x.iter().map(|v| v * v).sum::<f64>().sqrt();
385        for (u, v) in b.iter().zip(x.iter()) {
386            assert!(
387                (u * nx - v * cube.radius()).abs() < 1e-12,
388                "the ray is preserved"
389            );
390        }
391        assert!(cube.boundary_point(&cube.center()).is_err());
392        assert!(cube.boundary_point(&[0.0; 3]).is_err());
393        assert!(cube.boundary_point(&[f64::NAN; 6]).is_err());
394    }
