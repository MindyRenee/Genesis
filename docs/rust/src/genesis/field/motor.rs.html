1//! Distributed motor fields — the cognitive mind's decisions laid
2//! out over the coordinate grid rather than collapsed to a symbol.
3//!
4//! When the mind decides "act there" or "press action 3", it writes
5//! that decision as fields on the tensor slab:
6//!
7//! - `motor.spatial` — an `f32` field over the world grid; peaks are
8//!   where cognition wants to act (goal entities attract, hazards
9//!   repel, unexplored cells pull weakly).
10//! - `motor.actions` — an `f32` disposition per offered action.
11//! - `motor.action_ids` — the `u8` token each score corresponds to.
12//!
13//! This module is the decode side: it knows the canonical grid shape
14//! the action interface addresses, walks fields cell by cell (nested
15//! `y, x` loops over row-major data — flat index `i = y*w + x`, so
16//! `x = i % w`, `y = i / w`), and projects the winning field cell
17//! onto the canonical coordinate grid.
18

25// ─── Canonical coordinate grid ──────────────────────────────────
26//
27// The environment interface addresses points on a grid up to 64×64
28// cells (e.g. ARC-style coordinate actions take `x, y` in 0..=63).
29// The motor field's *target shape* is therefore fixed here: whatever
30// resolution the mind's field was published at, the winner is mapped
31// onto this grid before the result leaves the daemon.
32pub const MOTOR_FIELD_WIDTH: usize = 64;
33pub const MOTOR_FIELD_HEIGHT: usize = 64;
34/// Total cells on the canonical grid.
35pub const MOTOR_FIELD_CELLS: usize = MOTOR_FIELD_WIDTH * MOTOR_FIELD_HEIGHT;
36
37/// Well-known slab names for the motor fields.
38pub const MOTOR_SPATIAL: &str = "motor.spatial";
39pub const MOTOR_ACTIONS: &str = "motor.actions";
40pub const MOTOR_ACTION_IDS: &str = "motor.action_ids";
41
42// ─── Index ↔ coordinate math ────────────────────────────────────
43//
44// Fields are stored C-order (row-major): the flat index of cell
45// (x, y) in a `width`-wide field is `y*width + x`. These two
46// functions are the forward and inverse maps; every scan below is
47// written in nested (y, x) loops so the correspondence stays visible.
48
49/// Flat row-major index of grid coordinate `(x, y)` in a field of
50/// the given width.
51pub fn cell_index(x: usize, y: usize, width: usize) -> usize {
52    y * width + x
53}
54
55/// `(x, y)` grid coordinate of flat row-major index `i` in a field
56/// of the given width.
57pub fn cell_coord(i: usize, width: usize) -> (usize, usize) {
58    (i % width, i / width)
59}

61/// The field's shape read as a distribution — what the argmax
62/// discards. `centroid` is the softmax mean cell on the canonical
63/// grid (continuous — finer than the winning cell), `spread` the
64/// RMS radius about it in canonical cells (how far the urge
65/// reaches), `entropy` the normalized Shannon entropy in 0..1
66/// (0 = a decided point, 1 = maximally diffuse).
67#[derive(Clone, Copy, Debug, Default, PartialEq)]
68pub struct SpatialDistribution {
69    pub centroid: (f32, f32),
70    pub spread: f32,
71    pub entropy: f32,
72}
73
74/// What the motor fields currently point at. Both halves are
75/// optional: a published action field yields `action`, a published
76/// spatial field yields `cell` (already scaled onto the canonical
77/// `MOTOR_FIELD_WIDTH` × `MOTOR_FIELD_HEIGHT` grid).
78#[derive(Clone, Copy, Debug, Default, PartialEq)]
79pub struct MotorTarget {
80    /// Winning action token and its field strength, if any.
81    pub action: Option<(u8, f32)>,
82    /// Winning grid cell `(x, y)` and its field strength, if any.
83    pub cell: Option<(u16, u16, f32)>,
84    /// The spatial field's distributional shape — centroid, spread,
85    /// entropy — when a spatial field is published.
86    pub dist: Option<SpatialDistribution>,
87}
88
89impl MotorTarget {
90    /// Wire form: the fixed 14-byte base
91    /// `[flags][action_id][f32 a_strength][u16 x][u16 y][f32 s_strength]`,
92    /// plus an additive 16-byte distribution tail
93    /// `[f32 cx][f32 cy][f32 spread][f32 entropy]` when `dist` is
94    /// present. `flags` bit0 = action, bit1 = cell, bit2 = dist.
95    pub fn encode(&self) -> Vec<u8> {
96        let mut flags = 0u8;
97        if self.action.is_some() {
98            flags |= 1;
99        }
100        if self.cell.is_some() {
101            flags |= 2;
102        }
103        if self.dist.is_some() {
104            flags |= 4;
105        }
106        let (action_id, a_str) = self.action.unwrap_or((0xFF, 0.0));
107        let (x, y, s_str) = self.cell.unwrap_or((0, 0, 0.0));
108        let mut out = Vec::with_capacity(30);
109        out.push(flags);
110        out.push(action_id);
111        out.extend_from_slice(&a_str.to_le_bytes());
112        out.extend_from_slice(&x.to_le_bytes());
113        out.extend_from_slice(&y.to_le_bytes());
114        out.extend_from_slice(&s_str.to_le_bytes());
115        if let Some(d) = self.dist {
116            out.extend_from_slice(&d.centroid.0.to_le_bytes());
117            out.extend_from_slice(&d.centroid.1.to_le_bytes());
118            out.extend_from_slice(&d.spread.to_le_bytes());
119            out.extend_from_slice(&d.entropy.to_le_bytes());
120        }
121        out
122    }
123}

125/// Read one little-endian `f32` element out of a tensor payload, or
126/// `None` when the layout doesn't hold an `f32` there.
127fn f32_at(t: &Tensor, i: usize) -> Option<f32> {
128    t.data
129        .get(i * 4..i * 4 + 4)
130        .and_then(|b| b.try_into().ok().map(f32::from_le_bytes))
131}

133/// Scan a 2-D `f32` field for an extremal live cell — the peak when
134/// `maximize`, the valley otherwise — and project it onto the
135/// canonical coordinate grid.
136///
137/// Nested `(y, x)` loops over row-major data: `i = y*w + x` picks the
138/// element, `x = i % w` / `y = i / w` recover the cell. A field
139/// published at a different resolution is still honored — the extreme
140/// cell is scaled into the canonical grid rather than rejected, so a
141/// coarse 8×8 urge and a native 64×64 field address the same world.
142fn spatial_extremum(field: &Tensor, maximize: bool) -> Option<(u16, u16, f32)> {
143    if field.dtype != dtype::F32 || field.shape.len() != 2 {
144        return None;
145    }
146    let h = field.shape[0] as usize;
147    let w = field.shape[1] as usize;
148    if h == 0 || w == 0 {
149        return None;
150    }
151
152    let mut best = if maximize {
153        f32::NEG_INFINITY
154    } else {
155        f32::INFINITY
156    };
157    let mut best_cell: Option<(usize, usize)> = None;
158    for y in 0..h {
159        for x in 0..w {
160            let i = cell_index(x, y, w);
161            let Some(v) = f32_at(field, i) else {
162                return None; // payload shorter than the declared shape
163            };
164            let better = if maximize { v > best } else { v < best };
165            if v.is_finite() && better {
166                best = v;
167                best_cell = Some((x, y));
168            }
169        }
170    }
171
172    best_cell.map(|(x, y)| {
173        // Field-local (x, y) → canonical grid coordinate. Integer
174        // scaling keeps 0 → 0 and w−1 → just under WIDTH; the min
175        // clamp pins the corner cell exactly on the grid edge.
176        let gx = (x * MOTOR_FIELD_WIDTH / w).min(MOTOR_FIELD_WIDTH - 1) as u16;
177        let gy = (y * MOTOR_FIELD_HEIGHT / h).min(MOTOR_FIELD_HEIGHT - 1) as u16;
178        (gx, gy, best)
179    })
180}

182/// The field's strongest live cell — where the urge peaks.
183pub fn spatial_winner(field: &Tensor) -> Option<(u16, u16, f32)> {
184    spatial_extremum(field, true)
185}

187/// The opposite of `spatial_winner`: the field's deepest live
188/// valley — the cell the disposition most actively suppresses. A
189/// flat field's trough says nothing, but a shaped field's valley is
190/// information the argmax discards.
191pub fn spatial_trough(field: &Tensor) -> Option<(u16, u16, f32)> {
192    spatial_extremum(field, false)
193}

195/// The field's shape as a distribution — softmax over the live
196/// cells at unit temperature (the field's own scale sets the
197/// concentration), with the centroid and RMS radius taken in
198/// canonical-grid coordinates and the Shannon entropy normalized
199/// to 0..1 by ln(live cells).
200pub fn spatial_readout(field: &Tensor) -> Option<SpatialDistribution> {
201    if field.dtype != dtype::F32 || field.shape.len() != 2 {
202        return None;
203    }
204    let h = field.shape[0] as usize;
205    let w = field.shape[1] as usize;
206    if h == 0 || w == 0 {
207        return None;
208    }
209
210    // Pass 1: softmax stability needs the max; the entropy norm
211    // needs the live-cell count.
212    let mut max = f32::NEG_INFINITY;
213    let mut live = 0usize;
214    for i in 0..h * w {
215        let v = f32_at(field, i)?;
216        if v.is_finite() {
217            max = max.max(v);
218            live += 1;
219        }
220    }
221    if live == 0 {
222        return None;
223    }
224
225    // Pass 2: unnormalized softmax mass and its moments. Each cell's
226    // canonical coordinate is (x·64/w, y·64/h) — the readout lives
227    // on the same grid the winner is projected onto.
228    let sx = MOTOR_FIELD_WIDTH as f32 / w as f32;
229    let sy = MOTOR_FIELD_HEIGHT as f32 / h as f32;
230    let (mut z, mut cx, mut cy) = (0.0f32, 0.0f32, 0.0f32);
231    for y in 0..h {
232        for x in 0..w {
233            let v = f32_at(field, cell_index(x, y, w))?;
234            if v.is_finite() {
235                let p = (v - max).exp();
236                z += p;
237                cx += p * (x as f32 * sx);
238                cy += p * (y as f32 * sy);
239            }
240        }
241    }
242    if z <= 0.0 {
243        return None;
244    }
245    cx /= z;
246    cy /= z;
247    let (mut var, mut cross_term) = (0.0f32, 0.0f32);
248    for y in 0..h {
249        for x in 0..w {
250            let v = f32_at(field, cell_index(x, y, w))?;
251            if v.is_finite() {
252                let p = (v - max).exp();
253                let dx = x as f32 * sx - cx;
254                let dy = y as f32 * sy - cy;
255                var += p * (dx * dx + dy * dy);
256                cross_term += p * (v - max);
257            }
258        }
259    }
260    // Shannon entropy: H = ln Z − ⟨v−max⟩_p, normalized by ln(live)
261    // so a uniform field reads 1 and a decided point reads 0.
262    let entropy = if live > 1 {
263        (z.ln() - cross_term / z) / (live as f32).ln()
264    } else {
265        0.0
266    };
267    Some(SpatialDistribution {
268        centroid: (cx, cy),
269        spread: (var / z).max(0.0).sqrt(),
270        entropy: entropy.clamp(0.0, 1.0),
271    })
272}

274/// Scan the action field for its strongest disposition.
275///
276/// `motor.actions` is `f32[n]`; `motor.action_ids` (when present) is
277/// `u8[n]` mapping each score to the interface's action token. Scores
278/// without an id are ignored — the decoder reports an action it can
279/// actually name.
280pub fn action_winner(scores: &Tensor, ids: Option<&Tensor>) -> Option<(u8, f32)> {
281    if scores.dtype != dtype::F32 || scores.shape.len() != 1 {
282        return None;
283    }
284    let n = scores.shape[0] as usize;
285    let ids = ids.filter(|t| t.dtype == dtype::U8 && t.shape == [n as u32]);
286
287    let mut best: Option<(u8, f32)> = None;
288    for i in 0..n {
289        let Some(v) = f32_at(scores, i) else { break };
290        if !v.is_finite() {
291            continue;
292        }
293        // Without an id map the index itself is the token; with one,
294        // positions lacking a usable token can't be reported.
295        let id = match ids {
296            Some(t) => t.data[i],
297            None => i as u8,
298        };
299        if best.is_none_or(|(_, b)| v > b) {
300            best = Some((id, v));
301        }
302    }
303    best
304}

306/// Decode whatever motor fields are currently on the slab.
307pub fn current_motor_target() -> MotorTarget {
308    let slab = tensor_slab::slab()
309        .lock()
310        .unwrap_or_else(|e| e.into_inner());
311    let cell = slab.get(MOTOR_SPATIAL).and_then(spatial_winner);
312    let action = slab
313        .get(MOTOR_ACTIONS)
314        .and_then(|s| action_winner(s, slab.get(MOTOR_ACTION_IDS)));
315    let dist = slab.get(MOTOR_SPATIAL).and_then(spatial_readout);
316    MotorTarget { action, cell, dist }
317}

418    fn spatial_readout_reports_the_field_shape_not_just_its_peak() {
419        // A decided field: one hot cell at (3,5) on 8×8 — the
420        // centroid lands on that cell's canonical point, the spread
421        // and entropy near zero.
422        let w = 8;
423        let h = 8;
424        let mut vals = vec![-10.0f32; w * h];
