25/// Observation coupling timescale (seconds). Each fiber moves a
26/// fraction `1 − e^(−dt/τ)` along the geodesic toward its observed
27/// projection per call — a first-order manifold filter: the
28/// observation informs the field's own state instead of
29/// overwriting it.
30const OBSERVATION_TAU_SECONDS: f64 = 0.5;

32/// Per-fiber emanation times (seconds). Between observations each
33/// fiber's engine radiates its state *outward* from the
34/// superposition at the center — negative damping, the source the
35/// user specified — wound by the Metatron lattice's isoclinic
36/// counter-rotation: the two demicubes of the 6D merkaba spin in
37/// opposite chiralities (the field's two helicities). The isotropic
38/// saturation term catches the outflow at a limit cycle on the rim
39/// r* = √(α/s) = 1 — the S⁵ boundary, the holographic screen the
40/// radiation lands on. Rates carry the old homeostatic τ's: the
41/// same physiological timescales, driving emanation instead of
42/// decay. The observation stays the authority — the drift only
43/// fills the gaps it leaves.
44const BODY_RELAX_TAU_S: f64 = 120.0;
45const PREDICTION_RELAX_TAU_S: f64 = 40.0;
46const AGENCY_RELAX_TAU_S: f64 = 60.0;

83    pub fn new() -> Result<Self, FieldError> {
84        let manifold = StereographicSphere::new(LATENT_DIMENSION, 1.0)?;
85        let mut field = GlobalField::new(manifold.clone());
86        // κ = 1 → the rim — the enclosing S⁵ screen — sits at chart
87        // radius 1, and the merkaba's tips are inscribed on it.
88        let lattice = MetatronCube::new(LATENT_DIMENSION, 1.0)?;
89
90        // Each fiber rides one demicube of the merkaba: the sign is
91        // its chirality — prediction ascends (+), body and agency
92        // descend (−). The rates counter-balance exactly (1/40 up =
93        // 1/120 + 1/60 down): the structure counter-rotates with zero
94        // net spin while the center holds still. Negative damping
95        // radiates the state outward from the superposition; the
96        // saturation bound turns the outflow back at the rim — the
97        // limit cycle at r* = √((1/τ)/(1/τ)) = 1 is the screen.
98        for (id, relax_tau, chirality) in [
99            ("body", BODY_RELAX_TAU_S, -1.0),
100            ("prediction", PREDICTION_RELAX_TAU_S, 1.0),
101            ("agency", AGENCY_RELAX_TAU_S, -1.0),
102        ] {
103            let rate = 1.0 / relax_tau;
104            let layer = SubsystemLayer::new(format!("{id}:latent"), vec![0.0; LATENT_DIMENSION]);
105            let engine = SystemEngine::zero(1, LATENT_DIMENSION)
106                .with_damping(vec![-rate])
107                .expect("finite damping always validates")
108                .with_saturation(vec![rate])
109                .expect("positive finite saturation always validates")
110                .with_spiral(vec![lattice.isoclinic(chirality * rate)])
111                .expect("the isoclinic bivector is skew");
112            field.add_fiber(SystemFiber::with_engine(
113                id,
114                manifold.clone(),
115                vec![layer],
116                engine,
117            )?)?;
118        }
119
120        for (from, to, matrix) in [
121            // The afferent edge is the non-flat one: somatic
122            // coordinates are *read in* a rotated inferential frame
123            // (the generative model's axes are not the raw body
124            // axes), so the triangle carries real holonomy. The
125            // reverse edge carries the inverse rotation, keeping
126            // the pair's own two-way transport consistent.
127            (
128                "body",
129                "prediction",
130                rotation(LATENT_DIMENSION, 0, 1, AFFERENT_FRAME_OFFSET),
131            ),
132            ("prediction", "agency", identity(LATENT_DIMENSION)),
133            ("agency", "body", identity(LATENT_DIMENSION)),
134            (
135                "prediction",
136                "body",
137                rotation(LATENT_DIMENSION, 0, 1, -AFFERENT_FRAME_OFFSET),
138            ),
139            ("agency", "prediction", identity(LATENT_DIMENSION)),
140            ("body", "agency", identity(LATENT_DIMENSION)),
141        ] {
142            field.add_transport(GaugeTransport::new(from, to, matrix)?)?;
143        }
144
145        Ok(Self {
146            field,
147            lattice,
148            spectrum: FieldSpectrum::new(),
149        })
150    }

189    /// Predict → correct → integrate, transactionally on a clone:
190    /// any failure leaves the live field untouched rather than
191    /// half-observed.
192    ///
193    /// `bath` is the substrate's measured thermal level in latent²
194    /// units (the caller derives it from the physical probe — the
195    /// silicon's own timing fluctuation). It floors the field
196    /// temperature alongside internal disagreement: the reservoir
197    /// the Langevin kick samples is real, not only modeled.
198    pub fn observe(
199        &mut self,
200        state: &GenesisCoreState,
201        dt: f64,
202        bath: f64,
203    ) -> Result<FieldIntegration, FieldError> {
204        if !dt.is_finite() || dt < 0.0 {
205            return Err(FieldError::InvalidTime);
206        }
207        let mut next = self.field.clone();
208
209        // Prediction: each fiber advances by its own law — the OU
210        // homeostatic drift of its engine, so the field carries a
211        // trajectory between observations rather than a frozen
212        // snapshot.
213        for fiber in &mut next.fibers {
214            fiber.step(dt)?;
215        }
216
217        // Correction: the observation is an offering, not a command.
218        // Each fiber's latent layer moves a bounded fraction along
219        // the geodesic toward the observed point — the field carries
220        // its own state forward and lets measurement perturb it.
221        let gain = 1.0 - (-dt / OBSERVATION_TAU_SECONDS).exp();
222        for (id, projection) in [
223            ("body", body_projection(state)),
224            ("prediction", prediction_projection(state)),
225            ("agency", agency_projection(state)),
226        ] {
227            let fiber = next
228                .fibers
229                .iter_mut()
230                .find(|fiber| fiber.id == id)
231                .ok_or(FieldError::MissingFiber)?;
232            let manifold = fiber.manifold.clone();
233            let observation = manifold.project(&projection)?;
234            let current = fiber.layers[0].state.clone();
235            fiber.layers[0].state = manifold.geodesic(&current, &observation, gain)?;
236        }
237
238        // Integration: while awake, the noradrenergic forebrain
239        // coupling pulls the fibers toward consensus — the bounded
240        // pull is the body's coherence mechanism. In REM the locus
241        // coeruleus is silent and the systems decorrelate, so the
242        // honest integration is ascent on the same disagreement
243        // potential — each fiber drifts by its own dynamics rather
244        // than being held to agreement. Same measured consensus and
245        // thermal bookkeeping either way; only the direction the
246        // field moves changes with the phase its own carrier runs on.
247        //
248        // The coupling integrates at the field's own substep rate —
249        // the same gain/sigma laws applied at finer dt, which is the
250        // more accurate quadrature of the identical dynamics, and it
251        // is what lets the spectrum resolve the trajectory's own
252        // frequencies rather than the tick's sample times.
253        let rem = state.zones.phase() == MentalPhase::REM;
254        let n = ((dt * SPECTRUM_STEP_HZ).round() as usize).clamp(1, 512);
255        let h = dt / n as f64;
256        let mut consensus = None;
257        for _ in 0..n {
258            consensus = if rem {
259                next.diverge(h, bath)?
260            } else {
261                next.synchronize(h, bath)?
262            };
263            self.spectrum.sample(&next, h);
264        }
265        let consensus = consensus.ok_or(FieldError::EmptyState)?;
266        self.spectrum.observe_consensus(&consensus, dt);
267        let integration = FieldIntegration {
268            boundary: self.lattice.boundary_point(&consensus).ok(),
269            superposition: self.lattice.superposition_depth(&consensus),
270            consensus,
271            disagreement: next.disagreement,
272            temperature: next.temperature,
273            fiber_count: next.fibers.len(),
274        };
275        self.field = next;
276        Ok(integration)
277    }

280/// Frame offset between the body's somatic axes and the prediction
281/// system's inferential axes — the one bend in the connection,

286/// Givens rotation in the (i, j) coordinate plane — the minimal
287/// non-identity transport: a fixed frame offset between two
288/// systems' latent bases, orthogonal so it preserves norms.
289fn rotation(n: usize, i: usize, j: usize, theta: f64) -> Vec<Vec<f64>> {
290    let mut m = identity(n);
291    let (c, s) = (theta.cos(), theta.sin());
292    m[i][i] = c;
293    m[j][j] = c;
294    m[i][j] = -s;
295    m[j][i] = s;
296    m
297}

299fn identity(n: usize) -> Vec<Vec<f64>> {
300    let mut matrix = vec![vec![0.0; n]; n];
301    for (i, row) in matrix.iter_mut().enumerate() {
302        row[i] = 1.0;
303    }
304    matrix
305}

307fn unit(value: f32, min: f32, max: f32) -> f64 {
308    if !value.is_finite() {
309        return 0.5;
310    }
311    ((value - min) / (max - min)).clamp(0.0, 1.0) as f64
312}

314/// Signed latent half-width. Normalized channels live in [0,1];
315/// the latent coordinate centers them on the origin at ±0.25 —
316/// neutral state is the sphere's pole, and the corner of the
317/// operating box (√6·0.25 ≈ 0.61) stays well inside the
318/// hemisphere cap, where the log/exp maps are accurate instead of
319/// being radially clamped onto the equator rim.
320const LATENT_HALF: f64 = 0.25;
321
322fn latent(unit: f64) -> f64 {
323    (unit - 0.5) * 2.0 * LATENT_HALF
324}

326fn unit_finite(value: f32) -> f64 {
327    crate::state::sanitize::finite_clamp(value, 0.0, 1.0) as f64
328}

330fn effective(state: &GenesisCoreState, id: NeurochemicalId) -> f32 {
331    let value = state.neurochemicals.effective(id);
332    if id == NeurochemicalId::Dopamine {
333        crate::state::sanitize::finite_or(value, 0.0)
334    } else {
335        value.clamp(0.0, 2.0)
336    }

351        latent(unit(s, 0.0, 2.0)),
352        latent(unit(b, 0.0, 2.0)),
353        latent(unit(c, 0.0, 2.0)),
354    ]
355}

594        assert_eq!(latent(0.5), 0.0); // neutral state is the sphere's pole
595        assert_eq!(latent(1.0), LATENT_HALF);
596    }

602        // box [−0.25, 0.25]^6. Its corner norm is √6·0.25 ≈ 0.61 —
603        // interior to the hemisphere cap, so project() must leave
604        // even corner projections untouched rather than clamping
605        // them onto the equator rim.

609            assert_eq!(manifold.project(&corner).unwrap(), corner);
610        }
611        let state = GenesisCoreState::new(1, 1_000);
612        for projection in [
613            body_projection(&state),
614            prediction_projection(&state),
615            agency_projection(&state),
616        ] {
617            assert!(projection.iter().all(|v| v.abs() <= LATENT_HALF));
618            assert_eq!(manifold.project(&projection).unwrap(), projection);
619        }
