1//! Ocular optics — the eye's front end before transduction.
2//!
3//! Three transmission modulators, each its own dynamics:
4//!
5//! - The **lids** (`Eyelid`): a voluntary orbicularis oculi held at a
6//!   commanded aperture, plus the blink — a brief palpebral closure
7//!   the brainstem schedules. The blink is a Gaussian dip in
8//!   transmission (~120 ms to close, ~240 ms total), not a step:
9//!   real closure sweeps the cornea and the retinal illuminance
10//!   follows the lid's area, not a switch.
11//! - The **pupil** (`Pupil`): the pretectal light reflex — log-
12//!   compressive constriction on retinal illuminance, so the
13//!   aperture tracks the log of the light, not the light. Steady
14//!   state solves `sphincter = dilator + k·log(L)`; integration is
15//!   first-order toward it at the iris muscle's own rate.
16//! - **Accommodation** folds in as defocus: the lens state reports a
17//!   blur radius so whatever samples the field can convolve at the
18//!   PSF the ciliary body actually set.
19
20/// Lid aperture: 1.0 fully open, 0.0 fully closed.
21///
22/// The voluntary set-point is the mind's (sleep drops it to ~0.15,
23/// waking opens it); the blink is a transient occlusion riding on
24/// top — the lid can never transmit more than its set-point allows.
25#[derive(Clone, Debug)]
26pub struct Eyelid {
27    /// Commanded aperture, [0, 1]. Voluntary — held until changed.
28    pub set_point: f64,
29    /// Scheduled blink interval in seconds.
30    pub blink_period: f64,
31    /// Phase within the blink cycle, [0, blink_period).
32    phase: f64,
33    /// Time within an active blink, seconds; `None` when open.
34    blink_t: Option<f64>,
35}
36
37/// One blink's Gaussian closure: total duration and depth.
38const BLINK_DURATION_S: f64 = 0.24;
39const BLINK_CENTER_S: f64 = 0.12;
40const BLINK_WIDTH_S: f64 = 0.05;
41const BLINK_DEPTH: f64 = 0.97;
42
43impl Eyelid {
44    pub fn new() -> Self {
45        Self {
46            set_point: 1.0,
47            blink_period: 4.0,
48            phase: 0.0,
49            blink_t: None,
50        }
51    }
52
53    /// Advance the lid. When the phase crosses the period a blink
54    /// starts; inside one, transmission follows a Gaussian dip.
55    pub fn step(&mut self, dt: f64) {
56        if let Some(t) = self.blink_t {
57            let t = t + dt;
58            self.blink_t = (t < BLINK_DURATION_S).then_some(t);
59        } else {
60            self.phase += dt;
61            if self.phase >= self.blink_period {
62                self.phase -= self.blink_period;
63                self.blink_t = Some(0.0);
64            }
65        }
66    }
67
68    /// Current transmission, [0, 1]: the set-point scaled down by any
69    /// active blink's Gaussian occlusion.
70    pub fn transmission(&self) -> f64 {
71        let open = self.set_point.clamp(0.0, 1.0);
72        match self.blink_t {
73            Some(t) => {
74                let dip =
75                    BLINK_DEPTH * (-0.5 * ((t - BLINK_CENTER_S) / BLINK_WIDTH_S).powi(2)).exp();
76                open * (1.0 - dip)
77            }
78            None => open,
79        }
80    }
81}
82
83impl Default for Eyelid {
84    fn default() -> Self {
85        Self::new()
86    }
87}

89/// The iris aperture in millimetres, driven by the light reflex.
90///
91/// The reflex is logarithmic (Weber–Fechner at the sphincter):
92/// the target diameter decreases with log luminance over the
93/// mesopic–photopic range, bounded by the anatomy (~2 mm fully
94/// constricted, ~8 mm fully dilated). A sympathetic component widens
95/// the floor under arousal — carried in as `ne_drive`.
96#[derive(Clone, Debug)]
97pub struct Pupil {
98    /// Diameter in mm.
99    pub diameter_mm: f64,
100}
101
102/// Anatomical bounds: maximally constricted to maximally dilated.
103const PUPIL_MIN_MM: f64 = 2.0;
104const PUPIL_MAX_MM: f64 = 8.0;
105/// Iris muscle rate constant — the reflex settles over ~300 ms.
106const PUPIL_RATE: f64 = 12.0;
107
108impl Pupil {
109    pub fn new() -> Self {
110        Self { diameter_mm: 5.0 }
111    }
112
113    /// Steady-state diameter for a mean luminance `lum` (≥0) and
114    /// sympathetic drive `ne` ∈ [0,1]. Log-compressive:
115    /// `d* = clamp(d_dark − k·log10(1+L), min, max)`, then the
116    /// dilator adds up to its own millimetre at full drive.
117    fn target(&self, lum: f64, ne: f64) -> f64 {
118        let photopic = PUPIL_MAX_MM - 2.2 * (1.0 + lum.max(0.0)).log10();
119        let dilated = photopic + ne.clamp(0.0, 1.0);
120        dilated.clamp(PUPIL_MIN_MM, PUPIL_MAX_MM)
121    }
122
123    /// First-order approach to the reflex target.
124    pub fn step(&mut self, mean_luminance: f64, ne_drive: f64, dt: f64) {
125        let target = self.target(mean_luminance, ne_drive);
126        self.diameter_mm += (target - self.diameter_mm) * (PUPIL_RATE * dt).min(1.0);
127        self.diameter_mm = self.diameter_mm.clamp(PUPIL_MIN_MM, PUPIL_MAX_MM);
128    }
129
130    /// Relative retinal irradiance multiplier: pupil *area* scales
131    /// the light admitted, so the gain is quadratic in diameter,
132    /// normalized so a 4 mm pupil transmits 1.
133    pub fn irradiance_gain(&self) -> f64 {
134        (self.diameter_mm / 4.0).powi(2)
135    }
136}
137
138impl Default for Pupil {
139    fn default() -> Self {
140        Self::new()
141    }
142}

144/// The whole ocular front end at an instant: lid transmission,
145/// pupil diameter, and the defocus the ciliary body dialed in.
146#[derive(Clone, Copy, Debug)]
147pub struct OcularState {
148    /// Lid transmission scalar ∈ [0,1].
149    pub lid: f64,
150    /// Pupil diameter, mm.
151    pub pupil_mm: f64,
152    /// Pupillary irradiance gain relative to a 4 mm pupil.
153    pub irradiance_gain: f64,
154    /// Accommodation defocus, Gaussian σ in field pixels (0 = sharp).
155    pub defocus_sigma: f64,
156}
