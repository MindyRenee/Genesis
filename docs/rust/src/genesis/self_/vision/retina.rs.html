1//! The neural retina — photons to ganglion channels.
2//!
3//! The pipeline is the biology's order:
4//!
5//! 1. **Spectral absorption** — an RGB radiance field is projected
6//!    onto the L/M/S cone fundamentals. The matrix is a standard
7//!    Hunt–Pointer–Estevez-style basis normalized to unit D65; it is
8//!    a *basis*, not a calibration, so callers with true LMS planes
9//!    can feed [`RadianceField`] directly and skip the projection.
10//! 2. **Naka–Rushton adaptation** — each cone class compresses its
11//!    quantal catch by `R = R_max · Iⁿ/(Iⁿ + σⁿ)` (n = 1 in the
12//!    steady retina; the semisaturation σ sets the operating point).
13//!    This is the photoreceptor's gain control: it is why the same
14//!    eye works under starlight and noon.
15//! 3. **Phototransduction** — the cGMP cascade as a PDE pair: light
16//!    drives phosphodiesterase hydrolysis of cGMP, closing cGMP-gated
17//!    channels and hyperpolarizing the membrane toward ~−70 mV. The
18//!    graded voltage — not a spike — is what bipolar cells read.
19//! 4. **Center-surround antagonism** — horizontal cells pool the
20//!    surround; the bipolar/ganglion signal is a difference of
21//!    Gaussians (center σ = 1 px, surround σ = 3 px). Uniform fields
22//!    cancel; edges survive — the retina reports contrast, not
23//!    luminance.
24//! 5. **ON/OFF rectification** — ganglion cells can't fire negative;
25//!    the DoG output splits into ON (depolarizing-center) and OFF
26//!    (hyperpolarizing-center) channels, the two parallel pathways
27//!    the LGN receives.
28//!
29//! Integration is forward Euler on the transduction pair — stable
30//! for `dt ≤ ~10 ms` given the rate constants below; [`Retina::step`]
31//! clamps the states into their physical ranges so a large `dt`
32//! degrades gracefully instead of diverging.
33
34/// An incoming spectral field as L, M, S cone-plane radiances.
35///
36/// All planes are `width × height` row-major, non-negative; the
37/// units are arbitrary but shared (the Naka–Rushton σ is expressed
38/// in the same units).
39#[derive(Clone, Debug)]
40pub struct RadianceField {
41    pub width: usize,
42    pub height: usize,
43    pub l: Vec<f64>,
44    pub m: Vec<f64>,
45    pub s: Vec<f64>,
46}
47
48impl RadianceField {
49    pub fn new(width: usize, height: usize, l: Vec<f64>, m: Vec<f64>, s: Vec<f64>) -> Self {
50        let n = width * height;
51        assert_eq!(l.len(), n, "L plane must be width*height");
52        assert_eq!(m.len(), n, "M plane must be width*height");
53        assert_eq!(s.len(), n, "S plane must be width*height");
54        Self {
55            width,
56            height,
57            l,
58            m,
59            s,
60        }
61    }
62
63    /// A uniform dark field — the resting boundary condition.
64    pub fn dark(width: usize, height: usize) -> Self {
65        Self::new(
66            width,
67            height,
68            vec![0.0; width * height],
69            vec![0.0; width * height],
70            vec![0.0; width * height],
71        )
72    }
73
74    /// Build LMS planes from an RGB24 frame (`width*height*3`
75    /// row-major bytes) via the cone-fundamental projection.
76    pub fn from_rgb24(width: usize, height: usize, rgb: &[u8]) -> Self {
77        let n = width * height;
78        assert_eq!(rgb.len(), n * 3, "RGB frame must be width*height*3 bytes");
79        let mut field = Self::dark(width, height);
80        for i in 0..n {
81            let r = rgb[3 * i] as f64 / 255.0;
82            let g = rgb[3 * i + 1] as f64 / 255.0;
83            let b = rgb[3 * i + 2] as f64 / 255.0;
84            // sRGB → LMS (HPE-like normalized basis). Rows are the
85            // L/M/S fundamentals; the small negative cross-terms of
86            // the true transform are clipped at zero — a cone cannot
87            // absorb a negative photon count.
88            field.l[i] = (0.4002 * r + 0.7075 * g - 0.0807 * b).max(0.0);
89            field.m[i] = (-0.2263 * r + 1.1653 * g + 0.0457 * b).max(0.0);
90            field.s[i] = (0.9182 * b).max(0.0);
91        }
92        field
93    }
94}

96/// Cone-opponent readouts after adaptation — the two chromatic
97/// channels plus the achromatic (L+M) luminance the magnocellular
98/// path carries.
99#[derive(Clone, Copy, Debug, Default)]
100pub struct ConeOps {
101    /// L−M (red–green) opponency.
102    pub l_minus_m: f64,
103    /// S−(L+M) (blue–yellow) opponency.
104    pub s_minus_lm: f64,
105    /// Achromatic luminance ≈ 0.6·L + 0.4·M.
106    pub luminance: f64,
107}
108
109/// Naka–Rushton half-saturation, in the field's radiance units.
110const SIGMA: f64 = 0.5;
111/// Naka–Rushton exponent; 1.0 is the steady-state retina.
112const HILL_N: f64 = 1.0;
113/// Peak response (mV-ish scale); only relative values matter.
114const R_MAX: f64 = 100.0;
115
116/// Resting (dark) cGMP concentration, relative units.
117const CGMP_DARK: f64 = 20.0;
118/// cGMP hydrolysis gain — how hard light drives the PDE.
119const CGMP_HYDROLYSIS: f64 = 0.5;
120/// cGMP synthesis rate — guanylate cyclase restoring the dark state.
121const CGMP_SYNTHESIS: f64 = 4.0;
122/// Channel Hill coefficient / half-activation for the cGMP-gated
123/// conductance (K₁/₂ ≈ 15 in the same relative units).
124const CGMP_K_HALF: f64 = 15.0;
125/// Channel-open fraction at the dark cGMP level — the baseline the
126/// light-driven closure is measured against.
127const P_OPEN_DARK: f64 = CGMP_DARK * CGMP_DARK * CGMP_DARK
128    / (CGMP_DARK * CGMP_DARK * CGMP_DARK + CGMP_K_HALF * CGMP_K_HALF * CGMP_K_HALF);
129/// Membrane leak reversal, mV — drives the photoreceptor back to
130/// its depolarized dark state (~−40 mV) when channels reopen.
131const V_DARK: f64 = -40.0;
132const V_LIGHT: f64 = -70.0;
133/// Leak rate and light-driven rate for the membrane ODE.
134const V_LEAK: f64 = 0.1;
135const V_DRIVE: f64 = 5.0;
136
137/// Gaussian blur helpers — separable kernel, reflect-padded edges.
138fn gaussian_kernel(sigma: f64) -> Vec<f64> {
139    let radius = (3.0 * sigma).ceil().max(1.0) as isize;
140    let mut k = Vec::with_capacity((2 * radius + 1) as usize);
141    let mut sum = 0.0;
142    for i in -radius..=radius {
143        let w = (-0.5 * (i as f64 / sigma).powi(2)).exp();
144        k.push(w);
145        sum += w;
146    }
147    for w in &mut k {
148        *w /= sum;
149    }
150    k
151}
152
153fn convolve_gaussian(plane: &[f64], width: usize, height: usize, sigma: f64) -> Vec<f64> {
154    let kernel = gaussian_kernel(sigma);
155    let radius = (kernel.len() / 2) as isize;
156    let mut tmp = vec![0.0; plane.len()];
157    let mut out = vec![0.0; plane.len()];
158    for y in 0..height {
159        for x in 0..width {
160            let mut acc = 0.0;
161            for (k, &w) in kernel.iter().enumerate() {
162                let xx = (x as isize + k as isize - radius).clamp(0, width as isize - 1) as usize;
163                acc += w * plane[y * width + xx];
164            }
165            tmp[y * width + x] = acc;
166        }
167    }
168    for y in 0..height {
169        for x in 0..width {
170            let mut acc = 0.0;
171            for (k, &w) in kernel.iter().enumerate() {
172                let yy = (y as isize + k as isize - radius).clamp(0, height as isize - 1) as usize;
173                acc += w * tmp[yy * width + x];
174            }
175            out[y * width + x] = acc;
176        }
177    }
178    out
179}

181/// The ganglion output image: ON and OFF rectified channels plus the
182/// signed DoG they were split from — the retina's true report.
183#[derive(Clone, Debug, Default)]
184pub struct GanglionImage {
185    /// Depolarizing-center channel, ≥0.
186    pub on: Vec<f64>,
187    /// Hyperpolarizing-center channel, ≥0 (positive = contrast in the
188    /// OFF direction).
189    pub off: Vec<f64>,
190    /// Signed center-minus-surround signal before rectification.
191    pub dog: Vec<f64>,
192}
193
194/// The retinal field: cone states, transduction states, and the
195/// ganglion convolver, over a fixed `width × height` mosaic.
196pub struct Retina {
197    pub width: usize,
198    pub height: usize,
199    /// Adapted cone responses (post Naka–Rushton), per locus.
200    pub cone_l: Vec<f64>,
201    pub cone_m: Vec<f64>,
202    pub cone_s: Vec<f64>,
203    /// cGMP concentration per locus.
204    pub cgmp: Vec<f64>,
205    /// Photoreceptor membrane potential, mV, per locus.
206    pub membrane_mv: Vec<f64>,
207    /// Latest ganglion report.
208    pub ganglion: GanglionImage,
209}
210
211impl Retina {
212    pub fn new(width: usize, height: usize) -> Self {
213        let n = width * height;
214        Self {
215            width,
216            height,
217            cone_l: vec![0.0; n],
218            cone_m: vec![0.0; n],
219            cone_s: vec![0.0; n],
220            cgmp: vec![CGMP_DARK; n],
221            membrane_mv: vec![V_DARK; n],
222            ganglion: GanglionImage::default(),
223        }
224    }
225
226    /// One transduction step over the whole mosaic.
227    ///
228    /// `transmission` multiplies the incident radiance — it carries
229    /// the lid, pupil, and defocus the optics stage computed, so the
230    /// retina only ever sees the light the eye admitted.
231    pub fn step(&mut self, field: &RadianceField, transmission: f64, dt: f64) {
232        assert_eq!(field.width, self.width);
233        assert_eq!(field.height, self.height);
234        let n = self.width * self.height;
235        let sigma_n = SIGMA.powf(HILL_N);
236        for i in 0..n {
237            // Absorption modulated by the admitted fraction.
238            let raw_l = field.l[i] * transmission;
239            let raw_m = field.m[i] * transmission;
240            let raw_s = field.s[i] * transmission;
241
242            // Naka–Rushton compressive adaptation per cone class.
243            self.cone_l[i] = R_MAX * raw_l.powf(HILL_N) / (raw_l.powf(HILL_N) + sigma_n);
244            self.cone_m[i] = R_MAX * raw_m.powf(HILL_N) / (raw_m.powf(HILL_N) + sigma_n);
245            self.cone_s[i] = R_MAX * raw_s.powf(HILL_N) / (raw_s.powf(HILL_N) + sigma_n);
246
247            let lum = self.luminance(i);
248
249            // cGMP: synthesis pushes toward the dark level, light-
250            // driven PDE hydrolyzes it down.
251            let g = self.cgmp[i];
252            let dg =
253                CGMP_SYNTHESIS * (CGMP_DARK - g) - CGMP_HYDROLYSIS * lum * g / (CGMP_K_HALF + g);
254            let g_next = (g + dg * dt).clamp(0.1, 4.0 * CGMP_DARK);
255            self.cgmp[i] = g_next;
256
257            // Membrane: in the dark, high cGMP holds the channels
258            // open and the cell rests depolarized at V_DARK; light
259            // closes the cGMP-gated conductance and hyperpolarizes
260            // toward V_LIGHT. The drive is the light-*added* closure
261            // relative to the dark open fraction — zero in the dark,
262            // so the cell genuinely rests at V_DARK.
263            let p_open = g_next.powi(3) / (g_next.powi(3) + CGMP_K_HALF.powi(3));
264            let closed_by_light = P_OPEN_DARK - p_open;
265            let v = self.membrane_mv[i];
266            let dv = V_DRIVE * closed_by_light * (V_LIGHT - v) - V_LEAK * (v - V_DARK);
267            self.membrane_mv[i] = (v + dv * dt).clamp(V_LIGHT - 5.0, V_DARK + 5.0);
268        }
269        self.refresh_ganglion();
270    }
271
272    /// Achromatic luminance at one locus — the magnocellular drive.
273    pub fn luminance(&self, i: usize) -> f64 {
274        0.6 * self.cone_l[i] + 0.4 * self.cone_m[i]
275    }
276
277    /// Mean luminance over the mosaic — the pretectum's illuminance.
278    pub fn mean_luminance(&self) -> f64 {
279        if self.cone_l.is_empty() {
280            return 0.0;
281        }
282        (0..self.cone_l.len())
283            .map(|i| self.luminance(i))
284            .sum::<f64>()
285            / self.cone_l.len() as f64
286    }
287
288    /// Cone-opponent signals at one locus.
289    pub fn opponent(&self, i: usize) -> ConeOps {
290        let lm = self.cone_l[i] + self.cone_m[i];
291        ConeOps {
292            l_minus_m: self.cone_l[i] - self.cone_m[i],
293            s_minus_lm: self.cone_s[i] - 0.5 * lm,
294            luminance: self.luminance(i),
295        }
296    }
297
298    /// Recompute the ganglion report: DoG of the membrane-signal
299    /// plane (depolarization above dark rest = contrast), split into
300    /// ON and OFF channels.
301    fn refresh_ganglion(&mut self) {
302        let n = self.width * self.height;
303        // The bipolar drive is the *inverted* deviation from dark
304        // rest — ON bipolars sign-invert the photoreceptor, so light
305        // (hyperpolarization) reads positive: positive = luminance
306        // increment, exactly what the ON channel should carry.
307        let drive: Vec<f64> = (0..n).map(|i| V_DARK - self.membrane_mv[i]).collect();
308        let center = convolve_gaussian(&drive, self.width, self.height, 1.0);
309        let surround = convolve_gaussian(&drive, self.width, self.height, 3.0);
310        self.ganglion.dog = (0..n).map(|i| center[i] - surround[i]).collect();
311        self.ganglion.on = self.ganglion.dog.iter().map(|&d| d.max(0.0)).collect();
312        self.ganglion.off = self.ganglion.dog.iter().map(|&d| (-d).max(0.0)).collect();
313    }
314}

320    #[test]
321    fn naka_rushton_saturates() {
322        let mut r = Retina::new(4, 4);
323        let dim = RadianceField::new(4, 4, vec![0.1; 16], vec![0.1; 16], vec![0.1; 16]);
324        let bright = RadianceField::new(4, 4, vec![10.0; 16], vec![10.0; 16], vec![10.0; 16]);
325        r.step(&dim, 1.0, 0.005);
326        let dim_l = r.cone_l[0];
327        r.step(&bright, 1.0, 0.005);
328        let bright_l = r.cone_l[0];
329        assert!(bright_l > dim_l);
330        assert!(bright_l <= R_MAX + 1e-9, "response saturates at R_MAX");
331    }
332
333    #[test]
334    fn dark_field_returns_to_rest() {
335        let mut r = Retina::new(4, 4);
336        for _ in 0..200 {
337            r.step(&RadianceField::dark(4, 4), 1.0, 0.005);
338        }
339        assert!((r.cgmp[0] - CGMP_DARK).abs() < 0.5);
340        assert!((r.membrane_mv[0] - V_DARK).abs() < 1.0);
341    }

343    #[test]
344    fn uniform_field_cancels_in_dog() {
345        let mut r = Retina::new(16, 16);
346        let flat = RadianceField::new(16, 16, vec![1.0; 256], vec![1.0; 256], vec![0.2; 256]);
347        for _ in 0..100 {
348            r.step(&flat, 1.0, 0.005);
349        }
350        // Interior loci see equal center and surround → ~zero DoG.
351        let mid = 8 * 16 + 8;
352        assert!(r.ganglion.dog[mid].abs() < 1e-6);
353    }

355    #[test]
356    fn edge_survives_center_surround() {
357        let mut r = Retina::new(16, 16);
358        let mut l = vec![0.0; 256];
359        for y in 0..16 {
360            for x in 8..16 {
361                l[y * 16 + x] = 2.0;
362            }
363        }
364        let edge = RadianceField::new(16, 16, l.clone(), l, vec![0.0; 256]);
365        for _ in 0..100 {
366            r.step(&edge, 1.0, 0.005);
367        }
368        let energy: f64 = r.ganglion.dog.iter().map(|d| d * d).sum();
369        assert!(energy > 1e-3, "a luminance step must survive the DoG");
370    }
