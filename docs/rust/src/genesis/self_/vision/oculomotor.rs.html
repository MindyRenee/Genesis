1//! Oculomotor — the gaze plant. Seeing is a motor act.
2//!
3//! The eye is never a passive sensor: fixation is held against the
4//! plant's inertia, and a saccade is a ballistic lunge at a salience
5//! target. The plant here is a critically damped spring in the
6//! field's pixel plane — the standard model of the ocular plant's
7//! dominant dynamics (the muscle's overdamped viscoelasticity is why
8//! saccades land without oscillating).
9//!
10//! The target comes from the cortical map: the activity-weighted
11//! centroid is the salience peak, and the fovea — the field center —
12//! chases it. This is the free-energy account of gaze: the eye moves
13//! to put its high-acuity tissue where the world is most uncertain.
14
15/// A 2-D ocular plant: position (gaze point in field pixels) and
16/// velocity, driven toward a target by a critically damped spring.
17///
18/// With ζ = 1 the response is the fastest non-oscillating one —
19/// matching the saccade's own ~30–80 ms ballistic profile at the
20/// natural ω below.
21#[derive(Clone, Debug)]
22pub struct GazePlant {
23    /// Fixation point in field-pixel coordinates.
24    pub x: f64,
25    pub y: f64,
26    /// Angular velocity proxy, pixels/second.
27    pub vx: f64,
28    pub vy: f64,
29}
30
31/// Natural frequency of the plant. ω = 30 rad/s gives a ~70 ms
32/// saccade to a mid-field target — inside the biological range.
33const PLANT_OMEGA: f64 = 30.0;
34/// Maximum eccentricity as a fraction of field half-extent — the
35/// eye cannot roll its fovea past the orbit's edge.
36const MAX_ECCENTRIC: f64 = 0.95;
37
38impl GazePlant {
39    /// Start fixated on field center.
40    pub fn new(width: usize, height: usize) -> Self {
41        Self {
42            x: width as f64 / 2.0,
43            y: height as f64 / 2.0,
44            vx: 0.0,
45            vy: 0.0,
46        }
47    }
48
49    /// Drive the plant toward `(tx, ty)` for `dt` seconds — exact
50    /// integration of the critically damped spring:
51    /// `x(t) = t + (d0 + (v0 + ω·d0)·t)·e^(−ωt)`, applied per axis.
52    /// The exact update keeps stability at any `dt` — a motor plant
53    /// must never overshoot into a numerical oscillation a real
54    /// muscle couldn't produce.
55    pub fn saccade_to(&mut self, tx: f64, ty: f64, dt: f64) {
56        let w = PLANT_OMEGA;
57        let e = (-w * dt).exp();
58        for (pos, vel, target) in [
59            (&mut self.x, &mut self.vx, tx),
60            (&mut self.y, &mut self.vy, ty),
61        ] {
62            let d0 = *pos - target;
63            *pos = target + (d0 + (*vel + w * d0) * dt) * e;
64            *vel = (*vel - w * (*vel + w * d0) * dt) * e;
65        }
66    }
67
68    /// Sample a radiance field around the current fixation — the
69    /// eye movement's visual consequence made continuous: bilinear
70    /// interpolation at `locus + gaze_offset`, where the offset is
71    /// fixation minus field center. Sampling the world through the
72    /// moved fovea *is* looking.
73    pub fn sample(&self, plane: &[f64], width: usize, height: usize) -> Vec<f64> {
74        let ox = (self.x - width as f64 / 2.0).clamp(
75            -MAX_ECCENTRIC * width as f64 / 2.0,
76            MAX_ECCENTRIC * width as f64 / 2.0,
77        );
78        let oy = (self.y - height as f64 / 2.0).clamp(
79            -MAX_ECCENTRIC * height as f64 / 2.0,
80            MAX_ECCENTRIC * height as f64 / 2.0,
81        );
82        let mut out = vec![0.0; width * height];
83        for y in 0..height {
84            for x in 0..width {
85                let sx = (x as f64 + ox).clamp(0.0, width as f64 - 1.0);
86                let sy = (y as f64 + oy).clamp(0.0, height as f64 - 1.0);
87                let x0 = sx.floor() as usize;
88                let y0 = sy.floor() as usize;
89                let x1 = (x0 + 1).min(width - 1);
90                let y1 = (y0 + 1).min(height - 1);
91                let fx = sx - x0 as f64;
92                let fy = sy - y0 as f64;
93                out[y * width + x] = plane[y0 * width + x0] * (1.0 - fx) * (1.0 - fy)
94                    + plane[y0 * width + x1] * fx * (1.0 - fy)
95                    + plane[y1 * width + x0] * (1.0 - fx) * fy
96                    + plane[y1 * width + x1] * fx * fy;
97            }
98        }
99        out
100    }
101}

103#[cfg(test)]
104mod tests {
105    use super::*;
106
107    #[test]
108    fn saccade_lands_without_oscillation() {
109        let mut g = GazePlant::new(32, 32);
110        let target = (24.0, 8.0);
111        for _ in 0..200 {
112            g.saccade_to(target.0, target.1, 0.005);
113        }
114        assert!((g.x - target.0).abs() < 1e-6 && (g.y - target.1).abs() < 1e-6);
115        assert!(g.vx.abs() < 1e-6 && g.vy.abs() < 1e-6);
116    }
117
118    #[test]
119    fn gaze_shift_moves_the_fovea() {
120        // A point source off-center: after fixating it, sampling must
121        // bring it toward field center.
122        let mut plane = vec![0.0; 32 * 32];
123        plane[8 * 32 + 24] = 1.0;
124        let mut g = GazePlant::new(32, 32);
125        g.x = 24.0;
126        g.y = 8.0;
127        let sampled = g.sample(&plane, 32, 32);
128        // Field center should now hold the source's intensity.
129        assert!(sampled[16 * 32 + 16] > 0.9);
130    }
