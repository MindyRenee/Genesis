1//! Basal forebrain — the cholinergic output equation and its modulators.
2//!
3//! ════════════════════════════════════════════════════════════════════
4//! ANATOMY
5//! ════════════════════════════════════════════════════════════════════
6//!
7//! The basal forebrain lies ventral to the striatum and anterior to the
8//! hypothalamus. It holds the major cholinergic projection to the cortex
9//! and hippocampus: Ch1/Ch2 (medial septum, vertical diagonal band)
10//! supply the hippocampal complex, Ch4 (nucleus basalis of Meynert)
11//! supplies neocortex and amygdala. It is also a sleep-wake structure:
12//! three intermingled populations — wake/REM-active cholinergic,
13//! parvalbumin GABAergic, and glutamatergic neurons organized in a
14//! Glu→Ch→PV excitatory hierarchy — are all inhibited by somatostatin+
15//! GABAergic sleep-promoting neurons (Xu et al., Nat. Neurosci. 2015),
16//! and it is the best-characterized site where the homeostatic sleep
17//! signal acts: extracellular adenosine rises there with time awake
18//! (Porkka-Heiskanen et al., Science 1997) and A1-receptor activation
19//! inhibits the region's wake-active neurons (Strecker et al.,
20//! J. Neurosci. 2000; Basheer et al., 2004). The sleep signal literally
21//! silences the arousal output this same region produces.
22//!
23//! ════════════════════════════════════════════════════════════════════
24//! THE OUTPUT EQUATION
25//! ════════════════════════════════════════════════════════════════════
26//!
27//! The region's release level is a function of the sleep-wake network's
28//! populations plus two modulators:
29//!
30//!   target = ACH_BASE
31//!          + ACH_FROM_LC · lc · (1 − A1) · D(α)
32//!          + ACH_FROM_REM · rem
33//!
34//! - The linear map of `lc` (wake) and `rem` (REM-on) reproduces the
35//!   microdialysis steady states — NREM 0.34, waking 0.58, REM 0.77
36//!   (Vazquez & Baghdoyan 2001) — see `state::neurochemical` for the
37//!   calibrated coefficients.
38//! - `A1` is adenosine-A1-receptor occupancy on the *wake-active*
39//!   cholinergic population: the somnogenic signal attenuates only the
40//!   LC-driven component, so the calibrated sleep profiles (which have
41//!   no LC drive) are preserved exactly while waking cortical ACh sags
42//!   as pressure builds.
43//! - `D(α)` is the expected-uncertainty drive (Yu & Dayan, Neuron 2005):
44//!   acetylcholine reports expected uncertainty — the brain's estimate
45//!   of how unreliable its own predictions are — so the wake component
46//!   is scaled up when confidence is low and down when it is high.
47//!
48//! The efferent edge is also here: [`sensory_precision_gain`] is the
49//! cholinergic gain on bottom-up prediction-error precision — ACh
50//! optimizes the expected precision of sensory channels by gain on
51//! supragranular pyramidal cells (Feldman & Friston, Front. Hum.
52//! Neurosci. 2010; Moran et al., J. Neurosci. 2013), implemented in
53//! `daemon::active_inference` as a divisor on observation variance.
54//!
55//! ════════════════════════════════════════════════════════════════════
56//! WHAT THIS MODULE IS NOT
57//! ════════════════════════════════════════════════════════════════════
58//!
59//! Process S itself (adenosine accumulation and glymphatic clearance),
60//! the AChE excess decay, the coupling matrix rows, and the network
61//! release of the *other* chemicals live in `state::neurochemical`;
62//! the populations that drive `lc`/`rem` live in `state::sleep_network`.
63//! This module owns only the basal forebrain's own math — what reads
64//! those signals and what they do to its output.
65
66use crate::state::sanitize::finite_clamp;
67
68// ─── Release-map coefficients ────────────────────────────────────────
69// Release levels as linear functions of the network populations. The
70// coefficients are fit to the network's measured stage means (wake,
71// NREM, REM) and to the release data they are meant to reproduce; the
72// fit and its tolerances are in tests/network_calibration.rs.
73//
74// Acetylcholine rises with LC (wake) and REM-on: it is high in waking
75// and REM and lowest in NREM (Vazquez & Baghdoyan 2001).
76/// Basal (state-independent) term of the release map.
77pub const ACH_BASE: f32 = 0.2895;
78/// Weight of the LC (wake) population's drive on cholinergic release.
79pub const ACH_FROM_LC: f32 = 0.3017;
80/// Weight of the REM-on population's drive on cholinergic release.
81pub const ACH_FROM_REM: f32 = 0.4512;
82
83// ─── A1 adenosine inhibition ─────────────────────────────────────────
84//
85// A1 is the high-affinity, Gi-coupled adenosine receptor on the basal
86// forebrain's wake-active neurons — the site where Process S acts on
87// this region's output. Occupancy follows the same n=2 Hill form the
88// coupling matrix uses for receptor gating.
89
90/// Effective adenosine level for half-maximal A1 occupancy.
91///
92/// A1 is high-affinity (submicromolar); on this model's normalized
93/// scale, half occupancy sits well below the Drowsy gate so meaningful
94/// suppression begins as pressure builds past the resting baseline.
95pub const A1_EC50: f32 = 0.35;
96/// Maximum fraction of the LC-driven release component that A1
97/// occupancy can suppress. Capped at one half: A1 activation inhibits
98/// firing of the wake-active cholinergic neurons but does not silence
99/// the population outright (Strecker et al., 2000).
100pub const A1_MAX_INHIBITION: f32 = 0.5;
101/// The resting extracellular adenosine level — the chemical's baseline
102/// and Process S's lower asymptote. Suppression is measured relative
103/// to occupancy at this level so rest is exactly neutral.
104const ADENOSINE_REST: f32 = 0.20;
105/// A1 occupancy at the resting level — the zero point of
106/// [`a1_suppression`]. Const-evaluable from the Hill form.
107const A1_OCCUPANCY_AT_REST: f32 =
108    ADENOSINE_REST * ADENOSINE_REST / (A1_EC50 * A1_EC50 + ADENOSINE_REST * ADENOSINE_REST);
109
110/// A1-receptor occupancy: Hill equation, n = 2.
111///
112///   occ = adn² / (EC50² + adn²)
113pub fn a1_occupancy(adenosine: f32) -> f32 {
114    let adn = adenosine.max(0.0);
115    let adn_sq = adn * adn;
116    adn_sq / (A1_EC50 * A1_EC50 + adn_sq)
117}
118
119/// Suppression of the wake-active cholinergic component, [0, A1_MAX].
120///
121/// Occupancy is re-referenced to its value at the resting level and
122/// normalized to [0, 1] over the working range, so the suppression is
123/// exactly zero at rest and approaches `A1_MAX_INHIBITION` as
124/// occupancy saturates toward one.
125pub fn a1_suppression(adenosine: f32) -> f32 {
126    let rel = (a1_occupancy(adenosine) - A1_OCCUPANCY_AT_REST) / (1.0 - A1_OCCUPANCY_AT_REST);
127    A1_MAX_INHIBITION * finite_clamp(rel, 0.0, 1.0)
128}
129
130// ─── Expected uncertainty (Yu & Dayan 2005) ──────────────────────────
131//
132// Acetylcholine reports expected uncertainty — in the Bayesian task
133// model, α_t = 1 − max_z P[z_t = z | D_t], the complement of the
134// posterior mass on the most likely context. The body's live
135// confidence estimate is the inference engine's tracked precision
136// (itself an inverse-error-variance target), so the reduced form is
137// α = 1 − precision: a well-predicted body reports little expected
138// uncertainty; sustained surprise reports a lot. When a regime
139// posterior over the chemical state is ever run live
140// (`state::phase_partition`), α is the same object one level up:
141// 1 − max belief.
142
143/// Expected uncertainty at the well-predicted operating point — the
144/// reference where the uncertainty drive is exactly neutral. Resting
145/// precision sits at ~0.9 (the precision tracker's resting target),
146/// so resting α ≈ 0.1.
147pub const UNCERTAINTY_REFERENCE: f32 = 0.1;
148/// Gain of the expected-uncertainty drive on the wake component.
149/// 0.15 moves the LC-driven term by at most ~±13% across the whole α
150/// range — attention modulates the state profile, it does not set it.
151pub const UNCERTAINTY_GAIN: f32 = 0.15;
152
153/// Map tracked inference precision onto the expected-uncertainty
154/// signal the cholinergic equation consumes: α = 1 − precision,
155/// clamped to [0, 1].
156pub fn expected_uncertainty(precision: f32) -> f32 {
157    finite_clamp(1.0 - precision, 0.0, 1.0)
158}
159
160/// The multiplicative drive expected uncertainty applies to the
161/// wake-active cholinergic component: 1 + gain·(α − α_ref).
162///
163/// High expected uncertainty raises cortical ACh — bottom-up evidence
164/// should dominate when priors are unreliable (Yu & Dayan 2005);
165/// low expected uncertainty relaxes it — a well-predicted world can
166/// lean on its priors.
167pub fn uncertainty_drive(alpha: f32) -> f32 {
168    1.0 + UNCERTAINTY_GAIN * (alpha - UNCERTAINTY_REFERENCE)
169}
170
171// ─── The release equation ────────────────────────────────────────────
172
173/// The wake-active component of cholinergic release: LC drive
174/// attenuated by A1 occupancy and scaled by the uncertainty drive.
175///
176/// Adenosine and α both act only here: A1 receptors sit on the
177/// wake-active population (not the REM-on one), and expected
178/// uncertainty is a waking-attention signal — so neither modulator
179/// can perturb the calibrated NREM floor or the REM rebound.
180pub fn wake_component(lc: f32, adenosine: f32, alpha: f32) -> f32 {
181    ACH_FROM_LC * lc * (1.0 - a1_suppression(adenosine)) * uncertainty_drive(alpha)
182}
183
184/// The basal forebrain's cholinergic release target under the
185/// sleep-wake network's population state.
186///
187///   target = ACH_BASE
188///          + ACH_FROM_LC · lc · (1 − A1(adn)) · D(α)
189///          + ACH_FROM_REM · rem
190///
191/// At resting adenosine and reference uncertainty this reduces to the
192/// calibrated linear map: NREM 0.34, waking 0.58, REM 0.77.
193pub fn release_target(lc: f32, rem: f32, adenosine: f32, alpha: f32) -> f32 {
194    finite_clamp(
195        ACH_BASE + wake_component(lc, adenosine, alpha) + ACH_FROM_REM * rem,
196        0.0,
197        1.0,
198    )
199}
200
201/// The cholinergic set-point when no sleep-wake network is running —
202/// the fallback path where the level homeostats instead of tracking a
203/// release target. The same two modulators act on the whole set-point
204/// (with no network there is no wake/REM decomposition).
205pub fn wake_setpoint(baseline: f32, adenosine: f32, alpha: f32) -> f32 {
206    finite_clamp(
207        baseline * (1.0 - a1_suppression(adenosine)) * uncertainty_drive(alpha),
208        0.05,
209        0.95,
210    )
211}
212
213// ─── Cholinergic sensory-precision gain ──────────────────────────────
214//
215// The efferent half of the Yu & Dayan claim: acetylcholine optimizes
216// the precision of bottom-up prediction errors — implemented in
217// `daemon::active_inference` as a divisor on the Kalman observation
218// variance, the gain of the units reporting prediction error
219// (Moran et al., 2013: cholinesterase inhibition altered sensory PE
220// responses through supragranular pyramidal gain).
221
222/// Gain of ACh deviation-from-baseline on sensory precision.
223pub const ACH_PRECISION_GAIN: f32 = 0.5;
224/// Bounds on the precision multiplier. Attention shifts the weight of
225/// evidence; it does not open or close the channel.
226pub const SENSORY_PRECISION_GAIN_MIN: f32 = 0.7;
227pub const SENSORY_PRECISION_GAIN_MAX: f32 = 1.4;
228
229/// The multiplier on sensory (observation) precision for a given
230/// effective acetylcholine level:
231///
232///   Π_ach = clamp(1 + k·(ach_eff − ach_baseline), min, max)
233///
234/// Anchored to the chemical's genetic baseline so tonic cholinergic
235/// tone is exactly neutral — elevated ACh raises the weight the
236/// generative model gives its observations; depressed ACh lowers it.
237pub fn sensory_precision_gain(ach_eff: f32) -> f32 {
238    let ach_baseline =
239        crate::state::neurochemical::NeurochemicalId::Acetylcholine.default_baseline();
240    finite_clamp(
241        1.0 + ACH_PRECISION_GAIN * (ach_eff - ach_baseline),
242        SENSORY_PRECISION_GAIN_MIN,
243        SENSORY_PRECISION_GAIN_MAX,
244    )
245}

247#[cfg(test)]
248mod tests {
249    use super::*;

251    #[test]
252    fn a1_occupancy_is_saturating_hill() {
253        // Zero at zero, EC50 at half, saturating toward one.
254        assert_eq!(a1_occupancy(0.0), 0.0);
255        assert!((a1_occupancy(A1_EC50) - 0.5).abs() < 1e-6);
256        assert!(a1_occupancy(0.95) < 1.0 && a1_occupancy(0.95) > 0.85);
257    }
258
259    #[test]
260    fn a1_suppression_is_zero_at_rest_and_capped() {
261        assert_eq!(a1_suppression(ADENOSINE_REST), 0.0);
262        assert!(a1_suppression(0.0) == 0.0);
263        let deep = a1_suppression(0.95);
264        assert!(deep > 0.4 && deep <= A1_MAX_INHIBITION);
265        // The somnogenic signal bites through the Drowsy gate.
266        assert!(a1_suppression(0.5) > 0.2);
267    }
268
269    #[test]
270    fn release_target_recovers_the_calibrated_profile_at_rest() {
271        // At resting adenosine and reference uncertainty the equation
272        // reduces to the fit linear map (tests/network_calibration.rs
273        // stage means).
274        let alpha = UNCERTAINTY_REFERENCE;
275        let nrem = release_target(0.151, 0.011, ADENOSINE_REST, alpha);
276        let wake = release_target(0.963, 0.0, ADENOSINE_REST, alpha);
277        let rem = release_target(0.133, 0.976, ADENOSINE_REST, alpha);
278        assert!((nrem - 0.34).abs() < 0.02, "NREM {nrem}");
279        assert!((wake - 0.58).abs() < 0.02, "wake {wake}");
280        assert!((rem - 0.77).abs() < 0.02, "REM {rem}");
281    }
282
283    #[test]
284    fn release_target_sags_only_on_the_wake_side() {
285        // Sleep pressure attenuates the LC-driven component; NREM and
286        // REM profiles (no LC drive) are untouched.
287        let alpha = UNCERTAINTY_REFERENCE;
288        let rested_wake = release_target(0.963, 0.0, ADENOSINE_REST, alpha);
289        let pressed_wake = release_target(0.963, 0.0, 0.9, alpha);
290        assert!(
291            pressed_wake < rested_wake - 0.1,
292            "wake under pressure {pressed_wake} vs rested {rested_wake}"
293        );
294        let nrem_rested = release_target(0.151, 0.011, ADENOSINE_REST, alpha);
295        let nrem_pressed = release_target(0.151, 0.011, 0.9, alpha);
296        assert!(
297            (nrem_rested - nrem_pressed).abs() < 0.02,
298            "NREM must not move: {nrem_rested} vs {nrem_pressed}"
299        );
300        let rem_pressed = release_target(0.133, 0.976, 0.9, alpha);
301        assert!(
302            rem_pressed > pressed_wake,
303            "REM rebound outlives pressure: {rem_pressed}"
304        );
305    }
306
307    #[test]
308    fn uncertainty_drive_is_neutral_at_reference() {
309        assert!((uncertainty_drive(UNCERTAINTY_REFERENCE) - 1.0).abs() < 1e-6);
310        // Confidence raises it one way, surprise the other.
311        assert!(uncertainty_drive(0.0) < 1.0);
312        assert!(uncertainty_drive(0.9) > 1.0);
313    }
314
315    #[test]
316    fn expected_uncertainty_complements_precision() {
317        assert!((expected_uncertainty(0.9) - 0.1).abs() < 1e-6);
318        assert_eq!(expected_uncertainty(1.5), 0.0);
319        assert_eq!(expected_uncertainty(-0.5), 1.0);
320    }
321
322    #[test]
323    fn precision_gain_anchors_at_tonic_level() {
324        let ach_baseline =
325            crate::state::neurochemical::NeurochemicalId::Acetylcholine.default_baseline();
326        assert!((sensory_precision_gain(ach_baseline) - 1.0).abs() < 1e-6);
327        assert!(sensory_precision_gain(0.9) > 1.0);
328        assert!(sensory_precision_gain(0.0) < 1.0);
329        assert!(sensory_precision_gain(0.0) >= SENSORY_PRECISION_GAIN_MIN);
330    }
