1//! Active zones — what Genesis is cognitively attending to vs what its
2//! subcognitive is processing in the background, plus its emergent
3//! mental phase.
4//!
5//! ## Modeling note
6//!
7//! The `NREM` and `REM` phases below are **computational states**
8//! assigned by neurochemical thresholds, not clinical sleep stages
9//! scored from EEG/EOG/EMG polysomnography. They are control modes —
10//! labels for regions of the neurochemical state space that gate
11//! background processing — not sleep architecture in the
12//! electrophysiological sense. Real NREM comprises N1–N3 stages with
13//! distinct slow-wave and spindle signatures, and real REM involves
14//! muscle atonia, rapid eye movements, and pontine-geniculate-
15//! occipital waves. The names are borrowed as analogues because the
16//! neurochemical profiles (low ACh + high GABA for NREM; high ACh +
17//! aminergic silence for REM) are inspired by those states, but the
18//! assignment is threshold-based, not signal-based.
19//!
20//! # Two independent axes
21//!
22//! Genesis's state has two orthogonal axes:
23//!
24//! 1. **Task zone** ([`CognitiveZone`]) — what it's *doing*. This is
25//!    manually set when a task starts (e.g., it enters `Coding` when
26//!    it begins writing code). It's the "what" of cognition.
27//!
28//! 2. **Mental phase** ([`MentalPhase`]) — what mental state it's *in*.
29//!    This is not manually set — it *emerges* from the neurochemical
30//!    dynamics. It doesn't choose to be in "flow" — flow emerges when
31//!    dopamine, ACh, and NE are high while cortisol is low. It's the
32//!    "how" of cognition.
33//!
34//! These are independent: it can be coding while stressed, or coding
35//! while in flow. The task is the same; the mental state changes how
36//! it performs it.
37
38/// Maximum number of subcognitive task IDs tracked in the state struct.
39pub const MAX_SUBCOGNITIVE_TASKS: usize = 8;

45/// What Genesis is cognitively focused on right now.
46///
47/// This is a single enum (not flags) because cognitive attention is
48/// serial — it can only be in one zone at a time.
49#[repr(u8)]
50#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
51pub enum CognitiveZone {
52    /// No active task — resting, waiting for input.
53    Idle = 0,
54    /// Interacting with the user.
55    Conversation = 1,
56    /// Writing or modifying code.
57    Coding = 2,
58    /// Reading, analysing, or searching code or data.
59    Analysis = 3,
60    /// Ingesting new information (documentation, web, files).
61    Learning = 4,
62    /// Introspective processing — reviewing own state or memories.
63    Reflection = 5,
64    /// Handling an error or recovering from a failure.
65    ErrorRecovery = 6,
66    /// Deliberating about next steps, weighing options.
67    Planning = 7,
68    /// Deep consolidation — "sleep" mode. Cognitive mind is offline;
69    /// the subcognitive runs full-throttle consolidation and dreaming.
70    Sleeping = 8,
71}
72
73impl CognitiveZone {
74    /// Returns a short kebab-case label for this zone.
75    pub const fn label(self) -> &'static str {
76        match self {
77            Self::Idle => "idle",
78            Self::Conversation => "conversation",
79            Self::Coding => "coding",
80            Self::Analysis => "analysis",
81            Self::Learning => "learning",
82            Self::Reflection => "reflection",
83            Self::ErrorRecovery => "error-recovery",
84            Self::Planning => "planning",
85            Self::Sleeping => "sleeping",
86        }
87    }
88
89    /// Whether this zone requires the cognitive mind to be active.
90    pub const fn is_cognitive(self) -> bool {
91        !matches!(self, Self::Idle | Self::Sleeping)
92    }
93
94    /// Convert a u8 zone index back to the enum.
95    pub const fn from_u8(v: u8) -> Self {
96        match v {
97            0 => Self::Idle,
98            1 => Self::Conversation,
99            2 => Self::Coding,
100            3 => Self::Analysis,
101            4 => Self::Learning,
102            5 => Self::Reflection,
103            6 => Self::ErrorRecovery,
104            7 => Self::Planning,
105            8 => Self::Sleeping,
106            _ => Self::Idle,
107        }
108    }
109}

115/// Genesis's emergent mental phase — its qualitative state of mind.
116///
117/// This is NOT manually set. It emerges from the coupled neurochemical
118/// dynamics. The neurochemical system computes it in
119/// [`NeurochemicalVector::compute_phase`](super::neurochemical::NeurochemicalVector).
120///
121/// The phases form a landscape of mental states. Transitions between
122/// them are phase transitions in the dynamical systems sense — small
123/// neurochemical changes can flip the system from one phase to another
124/// when near a boundary.
125#[repr(u8)]
126#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
127pub enum MentalPhase {
128    /// Default cognitive state — engaged but not in any extreme.
129    Active = 0,
130    /// High arousal, vigilant — NE and histamine dominant, GABA low.
131    Alert = 1,
132    /// Optimal performance state — high dopamine, high ACh, moderate NE,
133    /// low cortisol. The "in the zone" state.
134    Flow = 2,
135    /// Stress response — high cortisol, high NE, low serotonin.
136    /// Impaired memory encoding, heightened vigilance, narrowed focus.
137    Stress = 3,
138    /// Sleep pressure building — high adenosine, low arousal.
139    /// Cognitive performance declining, approaching sleep.
140    Drowsy = 4,
141    /// Non-REM sleep — low acetylcholine, high GABA, slow-wave dominant.
142    /// This is where glymphatic clearance and synaptic downscaling happen.
143    /// Adenosine is very high, histamine is very low, and the brain is
144    /// in a synchronized slow-wave state.
145    NREM = 5,
146    /// REM sleep — high acetylcholine, low NE/histamine (noradrenergic
147    /// silence), theta-dominant. This is where dreaming and emotional
148    /// processing happen. The cholinergic system is active while the
149    /// aminergic systems (NE, histamine, serotonin) are silenced.
150    REM = 6,
151    /// System overload — cortisol, NE, and global tone all very high.
152    /// Cognitive function impaired across the board. The system needs
153    /// to shed load or rest.
154    Overwhelmed = 7,
155}
156
157impl MentalPhase {
158    /// Returns a short kebab-case label for this phase.
159    pub const fn label(self) -> &'static str {
160        match self {
161            Self::Active => "active",
162            Self::Alert => "alert",
163            Self::Flow => "flow",
164            Self::Stress => "stress",
165            Self::Drowsy => "drowsy",
166            Self::NREM => "nrem",
167            Self::REM => "rem",
168            Self::Overwhelmed => "overwhelmed",
169        }
170    }
171
172    /// Whether this phase allows cognitive task execution.
173    /// When false, the subcognitive is the primary actor.
174    pub const fn allows_cognitive_task(self) -> bool {
175        !matches!(self, Self::NREM | Self::REM | Self::Overwhelmed)
176    }
177
178    /// Whether this phase is a "peak performance" state.
179    pub const fn is_peak(self) -> bool {
180        matches!(self, Self::Flow | Self::Alert)
181    }
182
183    /// Whether this phase is a distress state.
184    pub const fn is_distress(self) -> bool {
185        matches!(self, Self::Stress | Self::Overwhelmed)
186    }
187
188    /// Convert a u8 phase index back to the enum.
189    pub const fn from_u8(v: u8) -> Self {
190        match v {
191            0 => Self::Active,
192            1 => Self::Alert,
193            2 => Self::Flow,
194            3 => Self::Stress,
195            4 => Self::Drowsy,
196            5 => Self::NREM,
197            6 => Self::REM,
198            7 => Self::Overwhelmed,
199            _ => Self::Active,
200        }
201    }
202}

267/// The active zone state — cognitive task, emergent mental phase, and
268/// subcognitive background processes.
269///
270/// ## Layout (96 bytes — unchanged from v1, padding repurposed)
271/// ```text
272/// offset  field                    type         notes
273/// ------  -----                    ----         -----
274///   0     cognitive_zone           u8           CognitiveZone (task)
275///   1     emergent_phase           u8           MentalPhase (emergent)
276///   2     zone_override_active     u8           1 = task zone is manual
277///   3     subcognitive_task_count  u8           active bg task count
278///   4     subcognitive_flags       u32          bitfield of bg processes
279///   8     cognitive_task_id        u64          active cognitive task
280///  16     subcognitive_task_ids    [u64;8]      bg task IDs (0 = unused)
281///  80     zone_entered_at          u64          ms timestamp
282///  88     zone_duration_ms         u64          duration in current zone
283/// ```
284#[repr(C)]
285#[derive(Clone, Copy, Debug)]
286pub struct ActiveZones {
287    /// The cognitive task zone — what it's doing.
288    pub cognitive_zone: u8,
289    /// The emergent mental phase — what state it's in. Computed by
290    /// the neurochemical system, not manually set.
291    pub emergent_phase: u8,
292    /// Whether the cognitive task zone is being manually overridden
293    /// (e.g., user started a conversation, forcing Conversation zone
294    /// regardless of emergent phase). When 0, the task zone should
295    /// align with the emergent phase when possible.
296    pub zone_override_active: u8,
297    /// Number of active subcognitive background tasks.
298    pub subcognitive_task_count: u8,
299    /// Bitfield of active subcognitive background processes.
300    pub subcognitive_flags: u32,
301    /// ID of the active cognitive task (0 = none).
302    pub cognitive_task_id: u64,
303    /// IDs of subcognitive background tasks (0 = unused slot).
304    pub subcognitive_task_ids: [u64; MAX_SUBCOGNITIVE_TASKS],
305    /// Milliseconds-timestamp when the current zone was entered.
306    pub zone_entered_at: u64,
307    /// Duration (ms) spent in the previous zone before the last transition.
308    pub zone_duration_ms: u64,
309}
310
311impl ActiveZones {
312    /// Initialise in an idle state with no background tasks.
313    pub fn new(now_ms: u64) -> Self {
314        Self {
315            cognitive_zone: CognitiveZone::Idle as u8,
316            emergent_phase: MentalPhase::Active as u8,
317            zone_override_active: 0,
318            subcognitive_task_count: 0,
319            subcognitive_flags: 0,
320            cognitive_task_id: 0,
321            subcognitive_task_ids: [0; MAX_SUBCOGNITIVE_TASKS],
322            zone_entered_at: now_ms,
323            zone_duration_ms: 0,
324        }
325    }
326
327    // ─── Task zone ───────────────────────────────────────────────
328
329    /// Get the cognitive task zone as the enum type.
330    pub fn zone(&self) -> CognitiveZone {
331        CognitiveZone::from_u8(self.cognitive_zone)
332    }
333
334    /// Transition to a new cognitive task zone, recording the timestamp.
335    /// Sets `zone_override_active` since this is a manual transition.
336    pub fn transition_to(&mut self, zone: CognitiveZone, now_ms: u64) {
337        let prev_entered = self.zone_entered_at;
338        self.zone_duration_ms = now_ms.saturating_sub(prev_entered);
339        self.cognitive_zone = zone as u8;
340        self.zone_entered_at = now_ms;
341        self.zone_override_active = 1;
342    }
343
344    /// Clear the manual override, allowing the task zone to align with
345    /// the emergent phase.
346    pub fn clear_override(&mut self) {
347        self.zone_override_active = 0;
348    }
349
350    // ─── Mental phase ────────────────────────────────────────────
351
352    /// Get the emergent mental phase as the enum type.
353    pub fn phase(&self) -> MentalPhase {
354        MentalPhase::from_u8(self.emergent_phase)
355    }
356
357    /// Update the emergent phase from the neurochemical system.
358    /// This is called by the dynamics engine after each tick — the
359    /// phase is not manually set, it's propagated from neurochemistry.
360    pub fn sync_phase(&mut self, phase: MentalPhase) {
361        self.emergent_phase = phase as u8;
362    }
363
364    // ─── Subcognitive flags ──────────────────────────────────────
365
366    /// Returns `true` if all bits in `flag` are set in the subcognitive flags.
367    pub fn has_flag(&self, flag: u32) -> bool {
368        self.subcognitive_flags & flag == flag
369    }
370
371    /// Set the given subcognitive flag bit(s).
372    pub fn set_flag(&mut self, flag: u32) {
373        self.subcognitive_flags |= flag;
374    }
375
376    /// Clear the given subcognitive flag bit(s).
377    pub fn clear_flag(&mut self, flag: u32) {
378        self.subcognitive_flags &= !flag;
379    }
380
381    // ─── Subcognitive tasks ──────────────────────────────────────
382
383    /// Register a subcognitive background task. Returns `false` if
384    /// `task_id` is 0, already registered, or no free slot is available.
385    pub fn register_task(&mut self, task_id: u64) -> bool {
386        if task_id == 0 {
387            return false;
388        }
389        // Guard against duplicate registration — re-registering an
390        // existing task_id would insert a duplicate and inflate the
391        // count, making it inconsistent with deregister_task (which
392        // only removes the first match).
393        if self.subcognitive_task_ids.contains(&task_id) {
394            return false;
395        }
396        if let Some(slot) = self.subcognitive_task_ids.iter_mut().find(|id| **id == 0) {
397            *slot = task_id;
398            self.subcognitive_task_count = self.subcognitive_task_count.saturating_add(1);
399            true
400        } else {
401            false
402        }
403    }
404
405    /// Deregister a subcognitive background task by ID (no-op if not found).
406    pub fn deregister_task(&mut self, task_id: u64) {
407        // 0 is the "unused slot" sentinel, and `register_task` rejects
408        // it. Without this guard the search below matches the first
409        // *empty* slot, zeroes it (a no-op) and still decrements the
410        // count — permanently desynchronising `subcognitive_task_count`
411        // from the number of occupied slots, with no repair path.
412        if task_id == 0 {
413            return;
414        }
415        if let Some(slot) = self
416            .subcognitive_task_ids
417            .iter_mut()
418            .find(|id| **id == task_id)
419        {
420            *slot = 0;
421            self.subcognitive_task_count = self.subcognitive_task_count.saturating_sub(1);
422        }
423    }
424}

426// ─────────────────────────────────────────────────────────────────
427//  Transition telemetry
428// ─────────────────────────────────────────────────────────────────
429//
430// Phase transitions are a stated contribution, but nothing measured
431// them: `zone_duration_ms` holds exactly one sample — the last
432// transition's duration, overwritten every time — and there was no
433// count at all. So neither how often Genesis changes phase nor how
434// long it spends in each was answerable, and a mind that oscillated
435// between two phases looked identical to one that settled.
436//
437// This accumulates transition counts and time-in-phase outside the
438// mmap'd state. `ActiveZones` is exactly 96 bytes, filling its slot
439// between `memory` at 2568 and `checksum` at 3224 with no reserved
440// tail, so adding fields here would shift every later offset and force
441// a layout migration — a large, risky change for two counters. Keeping
442// the telemetry process-local avoids that, at the cost that it resets
443// on restart; the counts are a rate, and the zone history is
444// re-accumulated from the first transition after startup.
445//
446// Served by its own additive command (GET_ZONE_TRANSITIONS) for the
447// same reason as the sensor mask: the BodyState layouts are pinned by
448// desc_len arithmetic, so a new field is a new layout and a protocol
449// bump. A `u32` count plus a `u64` total per tracked state — twelve
450// bytes each — rides behind a count byte and changes nothing existing.
451
452/// How many distinct zones/phases are tracked. Sized to the wider of
453/// the two enums: `CognitiveZone` runs to `Sleeping = 8`, so nine slots
454/// are needed; `MentalPhase` tops out at `Overwhelmed = 7` and leaves
455/// index 8 unused. An index beyond this is ignored rather than
456/// panicking, since a new variant must not be able to fault the
457/// telemetry path.
458pub const TRACKED_STATES: usize = 9;
459
460/// Transition counts and accumulated dwell time, per state index.
461///
462/// `counts[i]` is how many times state `i` was entered; `ms[i]` the
463/// total milliseconds spent there. Both are monotonic for the life of
464/// the process.
465#[derive(Clone, Copy, Debug, Default)]
466pub struct TransitionTally {
467    /// Times each state index was entered.
468    pub counts: [u32; TRACKED_STATES],
469    /// Milliseconds accumulated in each state index.
470    pub ms: [u64; TRACKED_STATES],
471    /// Milliseconds spent in the state entered most recently, so a
472    /// caller can see the current dwell without waiting for the next
473    /// transition to close it out.
474    pub current_since_ms: u64,
475    /// Index of the most recently entered state.
476    pub current: u8,
477}
478
479impl TransitionTally {
480    /// Record a transition into `next` at `now_ms`, closing out the
481    /// time accumulated in the previous state.
482    ///
483    /// Safe to call with the same state twice: that is not a
484    /// transition, and counting it would inflate the rate exactly when
485    /// the system is most stable.
486    pub fn record(&mut self, next: u8, now_ms: u64) -> bool {
487        let idx = next as usize;
488        if idx >= TRACKED_STATES {
489            return false;
490        }
491        if next == self.current && self.counts[idx] > 0 {
492            // Same state: keep accumulating, count nothing.
493            return false;
494        }
495        if self.counts[self.current as usize] > 0 {
496            self.ms[self.current as usize] = self.ms[self.current as usize]
497                .saturating_add(now_ms.saturating_sub(self.current_since_ms));
498        }
499        self.counts[idx] = self.counts[idx].saturating_add(1);
500        self.current = next;
501        self.current_since_ms = now_ms;
502        true
503    }
504
505    /// Total time in `idx` including the in-progress dwell.
506    pub fn total_ms(&self, idx: u8, now_ms: u64) -> u64 {
507        let i = idx as usize;
508        if i >= TRACKED_STATES {
509            return 0;
510        }
511        let mut total = self.ms[i];
512        if self.current == idx && self.counts[i] > 0 {
513            total = total.saturating_add(now_ms.saturating_sub(self.current_since_ms));
514        }
515        total
516    }
517
518    /// Total transitions across all states.
519    pub fn total_transitions(&self) -> u64 {
520        self.counts.iter().map(|c| *c as u64).sum()
521    }
522}
