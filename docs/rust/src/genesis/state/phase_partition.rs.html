1//! Learned phase model — mental phases as latent regimes inferred from
2//! the neurochemical trajectory, not boundaries drawn by hand.
3//!
4//! A Gaussian hidden Markov model over the effective levels of the
5//! eighteen neurochemicals. Each hidden state is a regime with its own
6//! mean and diagonal variance; the transition matrix says how regimes
7//! persist and how they switch. Nothing about dwell time or the number
8//! of regimes is set by hand:
9//!
10//! - **Persistence** comes from the transition matrix. A fixed Dirichlet
11//!   pseudo-count on self-transitions keeps regimes from chattering. It
12//!   is a finite-state stand-in for the sticky HDP-HMM of Fox et al.
13//!   (2008), which is nonparametric and sampled; it is not that model.
14//! - **The number of regimes** is chosen by BIC over `1..=MAX_PHASES`,
15//!   each fit from several random starts by Baum–Welch. This is known to
16//!   overselect when the trajectory is a continuous drift and the emission
17//!   model is misspecified, which is what the 48-hour measurement showed.
18//!   Cross-validated likelihood, which Celeux & Durand (2008) recommend
19//!   for this situation, is not yet implemented.
20//! - **Online labelling** is the forward filter: the belief over
21//!   regimes after each tick, and the phase is its argmax.
22//!
23//! Fitting uses scaled forward–backward, so long trajectories do not
24//! underflow. A refit aligns the new regimes to the old ones by mean,
25//! so a phase keeps its index across refits when the regime it names
26//! still exists.
27//!
28//! Persisted as a checksummed sidecar (see [`PhaseModel::to_bytes`]).
29//! The core state keeps only the current index, in `emergent_phase`,
30//! so the state schema is unchanged.
31

37/// Upper bound on regimes. The `emergent_phase` byte indexes them.
38pub const MAX_PHASES: usize = 8;
39
40/// BIC needs at least this many samples per regime to be trusted.
41const MIN_SAMPLES_PER_REGIME: usize = 40;
42/// Floor on each regime's per-dimension variance, so a regime cannot
43/// collapse onto a single sample.
44const VAR_FLOOR: f64 = 1e-4;
45/// Pseudo-count added to each self-transition in the M-step.
46const STICKY_KAPPA: f64 = 5.0;
47const EM_ITERS: usize = 100;
48const EM_TOL: f64 = 1e-7;
49const RESTARTS: u64 = 3;
50
51const MAGIC: [u8; 4] = *b"GPHM";
52const FORMAT_VERSION: u32 = 2;

54/// One sample: the effective level of each neurochemical.
55pub type Point = [f32; D];
56type Vector = [f64; D];

68/// A fitted Gaussian HMM plus the online filter state.
69#[derive(Clone, Debug, PartialEq)]
70pub struct PhaseModel {
71    initial: Vec<f64>,
72    transition: Vec<Vec<f64>>,
73    mean: Vec<Vector>,
74    var: Vec<Vector>,
75    belief: Vec<f64>,
76    current: usize,
77}
78
79impl PhaseModel {
80    /// Number of regimes.
81    pub fn k(&self) -> usize {
82        self.initial.len()
83    }
84
85    /// The current phase index.
86    pub fn current(&self) -> usize {
87        self.current
88    }
89
90    /// Mean of regime `j` (its typical neurochemical profile).
91    pub fn mean(&self, j: usize) -> Option<&Vector> {
92        self.mean.get(j)
93    }
94
95    /// Transition probability from regime `i` to `j`.
96    pub fn transition(&self, i: usize, j: usize) -> f64 {
97        self.transition[i][j]
98    }
99
100    /// Fold one sample into the filter and return the current phase.
101    ///
102    /// A non-finite sample is ignored outright: it would poison the
103    /// belief for every later tick.
104    pub fn observe(&mut self, x: &Point) -> usize {
105        if !x.iter().all(|v| v.is_finite()) {
106            return self.current;
107        }
108        self.belief = self.advance(&self.belief, &to_vector(x));
109        self.current = argmax(&self.belief);
110        self.current
111    }
112
113    /// One forward-filter step: predict through the transition matrix,
114    /// weight by how well each regime explains `x`, normalize in log
115    /// space.
116    fn advance(&self, belief: &[f64], x: &Vector) -> Vec<f64> {
117        let k = self.k();
118        let logw: Vec<f64> = (0..k)
119            .map(|j| {
120                let pred: f64 = (0..k).map(|i| belief[i] * self.transition[i][j]).sum();
121                pred.max(f64::MIN_POSITIVE).ln() + log_gaussian(&self.mean[j], &self.var[j], x)
122            })
123            .collect();
124        normalize_log(&logw)
125    }
126
127    /// Filter a whole sequence from the initial distribution.
128    fn filtered_belief(&self, xs: &[Vector]) -> Vec<f64> {
129        xs.iter()
130            .fold(self.initial.clone(), |b, x| self.advance(&b, x))
131    }
132
133    /// Refit to a window of recent samples. Keeps the old phase indices
134    /// where a regime still exists. Returns false when there is not
135    /// enough data to fit, in which case the model is unchanged.
136    pub fn refit(&mut self, samples: &[Point]) -> bool {
137        let Some(mut fresh) = fit(samples) else {
138            return false;
139        };
140        align_labels(&mut fresh, self);
141        let xs = finite_vectors(samples);
142        fresh.belief = fresh.filtered_belief(&xs);
143        fresh.current = argmax(&fresh.belief);
144        *self = fresh;
145        true
146    }
147
148    /// Serialize as
149    /// `MAGIC | version | k | current | initial | transition | mean | var | belief | crc32`.
150    pub fn to_bytes(&self) -> Vec<u8> {
151        let k = self.k();
152        let mut out = Vec::with_capacity(16 + 8 * (k + k * k + 3 * k * D) + 4);
153        out.extend_from_slice(&MAGIC);
154        out.extend_from_slice(&FORMAT_VERSION.to_le_bytes());
155        out.extend_from_slice(&(k as u32).to_le_bytes());
156        out.extend_from_slice(&(self.current as u32).to_le_bytes());
157        let flat = self
158            .initial
159            .iter()
160            .chain(self.transition.iter().flatten())
161            .chain(self.mean.iter().flatten())
162            .chain(self.var.iter().flatten())
163            .chain(self.belief.iter());
164        for v in flat {
165            out.extend_from_slice(&v.to_le_bytes());
166        }
167        let crc = crc32_ieee(&out);
168        out.extend_from_slice(&crc.to_le_bytes());
169        out
170    }

172    /// Parse a sidecar, rejecting anything that fails the checksum or
173    /// describes a model that cannot exist.
174    pub fn from_bytes(bytes: &[u8]) -> Result<Self, PartitionError> {
175        let (body, tail) = bytes
176            .split_last_chunk::<4>()
177            .ok_or(PartitionError::Truncated)?;
178        if body.get(..4) != Some(&MAGIC[..]) {
179            return Err(PartitionError::BadMagic);
180        }
181        if u32::from_le_bytes(take(body, 4)?) != FORMAT_VERSION {
182            return Err(PartitionError::BadVersion);
183        }
184        if u32::from_le_bytes(*tail) != crc32_ieee(body) {
185            return Err(PartitionError::BadChecksum);
186        }
187        let mut r = Reader { buf: body, at: 8 };
188        let k = r.u32()? as usize;
189        let current = r.u32()? as usize;
190        if !(1..=MAX_PHASES).contains(&k) || current >= k {
191            return Err(PartitionError::Inconsistent);
192        }
193        let initial = r.vec(k)?;
194        let transition = (0..k).map(|_| r.vec(k)).collect::<Result<Vec<_>, _>>()?;
195        let mean = (0..k).map(|_| r.vector()).collect::<Result<Vec<_>, _>>()?;
196        let var = (0..k).map(|_| r.vector()).collect::<Result<Vec<_>, _>>()?;
197        let belief = r.vec(k)?;
198        if r.at != body.len() {
199            return Err(PartitionError::Inconsistent);
200        }
201        let model = Self {
202            initial,
203            transition,
204            mean,
205            var,
206            belief,
207            current,
208        };
209        if model.is_valid() {
210            Ok(model)
211        } else {
212            Err(PartitionError::Inconsistent)
213        }
214    }
215
216    fn is_valid(&self) -> bool {
217        let stochastic = |row: &[f64]| {
218            row.iter().all(|v| v.is_finite() && *v >= 0.0)
219                && (row.iter().sum::<f64>() - 1.0).abs() < 1e-6
220        };
221        stochastic(&self.initial)
222            && stochastic(&self.belief)
223            && self.transition.iter().all(|r| stochastic(r))
224            && self.mean.iter().flatten().all(|v| v.is_finite())
225            && self
226                .var
227                .iter()
228                .flatten()
229                .all(|v| v.is_finite() && *v >= VAR_FLOOR)
230    }
231}

233/// Fit a regime model to `samples`, choosing the regime count by BIC.
234/// Returns `None` when there is not enough data for even one regime.
235pub fn fit(samples: &[Point]) -> Option<PhaseModel> {
236    let xs = finite_vectors(samples);
237    let n = xs.len();
238    let mut best: Option<(f64, PhaseModel)> = None;
239    for k in 1..=MAX_PHASES {
240        if n < MIN_SAMPLES_PER_REGIME * k {
241            break;
242        }
243        let (model, ll) = (0..RESTARTS)
244            .map(|r| em(&xs, k, seed(k, r)))
245            .max_by(|a, b| a.1.total_cmp(&b.1))?;
246        let params = (k - 1) + k * (k - 1) + 2 * k * D;
247        let bic = -2.0 * ll + params as f64 * (n as f64).ln();
248        if best.as_ref().is_none_or(|(b, _)| bic < *b) {
249            best = Some((bic, model));
250        }
251    }
252    best.map(|(_, m)| m)
253}

255/// Baum–Welch from one random start. Returns the fitted model and its
256/// log-likelihood.
257fn em(xs: &[Vector], k: usize, seed: u64) -> (PhaseModel, f64) {
258    let mut rng = Rng::new(seed);
259    let global = global_variance(xs);
260    let mut means: Vec<Vector> = Vec::with_capacity(k);
261    while means.len() < k {
262        let c = xs[rng.below(xs.len())];
263        if !means.contains(&c) {
264            means.push(c);
265        }
266    }
267    let off = if k > 1 { 0.1 / (k - 1) as f64 } else { 0.0 };
268    let mut model = PhaseModel {
269        initial: vec![1.0 / k as f64; k],
270        transition: (0..k)
271            .map(|i| {
272                (0..k)
273                    .map(|j| {
274                        if i == j {
275                            1.0 - off * (k - 1) as f64
276                        } else {
277                            off
278                        }
279                    })
280                    .collect()
281            })
282            .collect(),
283        mean: means,
284        var: vec![global; k],
285        belief: vec![1.0 / k as f64; k],
286        current: 0,
287    };
288    let mut prev = f64::NEG_INFINITY;
289    let mut ll = f64::NEG_INFINITY;
290    for _ in 0..EM_ITERS {
291        let logb: Vec<Vec<f64>> = xs
292            .iter()
293            .map(|x| {
294                (0..k)
295                    .map(|j| log_gaussian(&model.mean[j], &model.var[j], x))
296                    .collect()
297            })
298            .collect();
299        let (e_ll, gamma, xi) = forward_backward(&model.initial, &model.transition, &logb);
300        ll = e_ll;
301        if (ll - prev).abs() < EM_TOL * (1.0 + ll.abs()) {
302            break;
303        }
304        prev = ll;
305        m_step(&mut model, xs, &gamma, &xi);
306    }
307    model.belief = model.initial.clone();
308    model.current = argmax(&model.belief);
309    (model, ll)
310}

312/// Scaled forward–backward. Returns the log-likelihood, the per-tick
313/// regime posteriors `gamma`, and the expected transition counts `xi`.
314///
315/// Each tick's emissions are shifted by their maximum before
316/// exponentiating, and the forward pass renormalizes every step; the
317/// log-likelihood accumulates the shifts and scales exactly, so nothing
318/// underflows however long the trajectory.
319fn forward_backward(
320    initial: &[f64],
321    trans: &[Vec<f64>],
322    logb: &[Vec<f64>],
323) -> (f64, Vec<Vec<f64>>, Vec<Vec<f64>>) {
324    let n = logb.len();
325    let k = initial.len();
326    let mut ll = 0.0;
327    let mut b = vec![vec![0.0; k]; n];
328    for t in 0..n {
329        let m = logb[t].iter().copied().fold(f64::NEG_INFINITY, f64::max);
330        ll += m;
331        for j in 0..k {
332            b[t][j] = (logb[t][j] - m).exp();
333        }
334    }
335    let mut alpha = vec![vec![0.0; k]; n];
336    let mut c = vec![0.0; n];
337    for j in 0..k {
338        alpha[0][j] = initial[j] * b[0][j];
339    }
340    c[0] = alpha[0].iter().sum::<f64>().max(f64::MIN_POSITIVE);
341    alpha[0].iter_mut().for_each(|v| *v /= c[0]);
342    for t in 1..n {
343        for j in 0..k {
344            let pred: f64 = (0..k).map(|i| alpha[t - 1][i] * trans[i][j]).sum();
345            alpha[t][j] = pred * b[t][j];
346        }
347        c[t] = alpha[t].iter().sum::<f64>().max(f64::MIN_POSITIVE);
348        alpha[t].iter_mut().for_each(|v| *v /= c[t]);
349    }
350    ll += c.iter().map(|v| v.ln()).sum::<f64>();
351
352    let mut beta = vec![vec![1.0; k]; n];
353    for t in (0..n.saturating_sub(1)).rev() {
354        for i in 0..k {
355            beta[t][i] = (0..k)
356                .map(|j| trans[i][j] * b[t + 1][j] * beta[t + 1][j])
357                .sum::<f64>()
358                / c[t + 1];
359        }
360    }
361
362    let gamma: Vec<Vec<f64>> = (0..n)
363        .map(|t| normalize(&(0..k).map(|i| alpha[t][i] * beta[t][i]).collect::<Vec<_>>()))
364        .collect();
365    let mut xi = vec![vec![0.0; k]; k];
366    for t in 0..n.saturating_sub(1) {
367        let mut pair = vec![vec![0.0; k]; k];
368        for i in 0..k {
369            for j in 0..k {
370                pair[i][j] = alpha[t][i] * trans[i][j] * b[t + 1][j] * beta[t + 1][j];
371            }
372        }
373        let total: f64 = pair.iter().flatten().sum::<f64>().max(f64::MIN_POSITIVE);
374        for i in 0..k {
375            for j in 0..k {
376                xi[i][j] += pair[i][j] / total;
377            }
378        }
379    }
380    (ll, gamma, xi)
381}

383/// Re-estimate parameters from the posteriors, with the sticky prior
384/// on self-transitions.
385fn m_step(model: &mut PhaseModel, xs: &[Vector], gamma: &[Vec<f64>], xi: &[Vec<f64>]) {
386    let k = model.k();
387    model.initial = gamma[0].clone();
388    for (i, xi_row) in xi.iter().enumerate() {
389        let mut row: Vec<f64> = xi_row
390            .iter()
391            .enumerate()
392            .map(|(j, x)| x + if i == j { STICKY_KAPPA } else { 0.0 })
393            .collect();
394        let s: f64 = row.iter().sum();
395        row.iter_mut().for_each(|v| *v /= s);
396        model.transition[i] = row;
397    }
398    for j in 0..k {
399        let g: f64 = gamma.iter().map(|gt| gt[j]).sum::<f64>().max(1e-12);
400        let mut mean = [0.0; D];
401        for (x, gt) in xs.iter().zip(gamma) {
402            for d in 0..D {
403                mean[d] += gt[j] * x[d];
404            }
405        }
406        mean.iter_mut().for_each(|v| *v /= g);
407        let mut var = [0.0; D];
408        for (x, gt) in xs.iter().zip(gamma) {
409            for d in 0..D {
410                let e = x[d] - mean[d];
411                var[d] += gt[j] * e * e;
412            }
413        }
414        var.iter_mut().for_each(|v| *v = (*v / g).max(VAR_FLOOR));
415        model.mean[j] = mean;
416        model.var[j] = var;
417    }
418}

420/// Reorder `new`'s regimes so each one takes the index of the nearest
421/// unclaimed regime in `old`. Regimes with no counterpart are appended.
422fn align_labels(new: &mut PhaseModel, old: &PhaseModel) {
423    let k = new.k();
424    let mut used = vec![false; k];
425    let mut order = Vec::with_capacity(k);
426    for old_mean in old.mean.iter().take(old.k().min(k)) {
427        let best = (0..k)
428            .filter(|&j| !used[j])
429            .min_by(|&a, &b| {
430                sq_dist(&new.mean[a], old_mean).total_cmp(&sq_dist(&new.mean[b], old_mean))
431            })
432            .expect("an unclaimed regime exists while o < k");
433        used[best] = true;
434        order.push(best);
435    }
436    order.extend((0..k).filter(|&j| !used[j]));
437    new.initial = order.iter().map(|&p| new.initial[p]).collect();
438    new.transition = order
439        .iter()
440        .map(|&p| order.iter().map(|&q| new.transition[p][q]).collect())
441        .collect();
442    new.mean = order.iter().map(|&p| new.mean[p]).collect();
443    new.var = order.iter().map(|&p| new.var[p]).collect();
444}

446fn log_gaussian(mean: &Vector, var: &Vector, x: &Vector) -> f64 {
447    (0..D)
448        .map(|d| {
449            let e = x[d] - mean[d];
450            -0.5 * ((2.0 * PI * var[d]).ln() + e * e / var[d])
451        })
452        .sum()
453}
454
455fn normalize_log(logw: &[f64]) -> Vec<f64> {
456    let m = logw.iter().copied().fold(f64::NEG_INFINITY, f64::max);
457    normalize(&logw.iter().map(|l| (l - m).exp()).collect::<Vec<_>>())
458}

460fn normalize(v: &[f64]) -> Vec<f64> {
461    let s: f64 = v.iter().sum::<f64>().max(f64::MIN_POSITIVE);
462    v.iter().map(|x| x / s).collect()
463}

465fn argmax(v: &[f64]) -> usize {
466    v.iter()
467        .enumerate()
468        .max_by(|a, b| a.1.total_cmp(b.1))
469        .map_or(0, |(i, _)| i)
470}
471
472fn sq_dist(a: &Vector, b: &Vector) -> f64 {
473    a.iter().zip(b).map(|(x, y)| (x - y) * (x - y)).sum()
474}

480fn finite_vectors(samples: &[Point]) -> Vec<Vector> {
481    samples

486}

488fn global_variance(xs: &[Vector]) -> Vector {
489    let n = xs.len().max(1) as f64;
490    let mut mean = [0.0; D];
491    for x in xs {
492        for d in 0..D {
493            mean[d] += x[d] / n;
494        }
495    }
496    let mut var = [0.0; D];
497    for x in xs {
498        for d in 0..D {
499            var[d] += (x[d] - mean[d]).powi(2) / n;
500        }
501    }
502    var.map(|v| v.max(VAR_FLOOR))
503}

505fn seed(k: usize, restart: u64) -> u64 {

509fn take<const N: usize>(body: &[u8], at: usize) -> Result<[u8; N], PartitionError> {
510    body.get(at..at + N)
511        .and_then(|s| s.try_into().ok())
512        .ok_or(PartitionError::Truncated)
513}
514
515struct Reader<'a> {
516    buf: &'a [u8],
517    at: usize,
518}
519
520impl Reader<'_> {
521    fn array<const N: usize>(&mut self) -> Result<[u8; N], PartitionError> {
522        let v = take::<N>(self.buf, self.at)?;
523        self.at += N;
524        Ok(v)
525    }
526    fn u32(&mut self) -> Result<u32, PartitionError> {
527        Ok(u32::from_le_bytes(self.array()?))
528    }
529    fn f64(&mut self) -> Result<f64, PartitionError> {
530        Ok(f64::from_le_bytes(self.array()?))
531    }
532    fn vec(&mut self, n: usize) -> Result<Vec<f64>, PartitionError> {
533        (0..n).map(|_| self.f64()).collect()
534    }
535    fn vector(&mut self) -> Result<Vector, PartitionError> {
536        let mut v = [0.0; D];
537        for x in &mut v {
538            *x = self.f64()?;
539        }
540        Ok(v)
541    }
542}

544/// Small deterministic generator for start points. Fits must be
545/// reproducible: the same trajectory always yields the same model.
546struct Rng(u64);
547
548impl Rng {
549    fn new(seed: u64) -> Self {
550        Self(seed.wrapping_mul(0x9E37_79B9_7F4A_7C15) | 1)
551    }
552    fn next_u64(&mut self) -> u64 {
553        self.0 ^= self.0 << 13;
554        self.0 ^= self.0 >> 7;
555        self.0 ^= self.0 << 17;
556        self.0
557    }
558    fn below(&mut self, n: usize) -> usize {
559        (self.next_u64() % n as u64) as usize
560    }
561}

580    /// Two-regime sticky sequence: regimes at 0.2 and 0.8, switching
581    /// with probability `1 - stay` per tick. Noise is drawn
582    /// independently per dimension, as sampling noise is.

584        let mut rng = Rng::new(seed);
585        let mut regime = 0usize;

595                }
